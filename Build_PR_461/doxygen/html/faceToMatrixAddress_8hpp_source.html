15/* @class FaceToMatrixAddress
16 * @brief class storing the mapping between mesh faces and target matrix sparsity pattern
17 *
18 * Based on given computational mesh this class stores a mapping for a consistent iteration
19 * procedure for matrices which share the same sparsity pattern.
20 *
21 * This class implements the finite volume 3/5/7 pt stencil specific generation
22 * of sparsity patterns from a given unstructured mesh
23 *
24 */
25template<typename IndexType = localIdx, typename MeshType = UnstructuredMesh>
26class FaceToMatrixAddress

28
29 // clang-format off
30 // NOTE The following data member store a simple mapping from face ids to offsets in the
31 // corresponding rows.
32 // I.e. Assume the following row [ . a_18 . a_20 d a_18 . a_20 . ]
33 // where a_i corresponds to a value on a face i
34 // this yields:
35 // ownerOffset[18] = 0
36 // ownerOffset[20] = 1
37 // and
38 // neighbourOffset[18] = 4
39 // neighbourOffset[20] = 5
40 // clang-format on
