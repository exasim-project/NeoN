28la::SolverStats iterativeSolveImpl(
29 Expression<typename VectorType::ElementType>& exp,
30 la::LinearSystem<
31 typename VectorType::ElementType,

39)

70/* @brief solve an expression
71 *
72 * @param exp - Expression which is to be solved/updated.
73 * @param solution - Solution field, where the solution will be 'written to'.
74 * @param t - the time at the start of the time step.
75 * @param dt - time step for the temporal integration
76 * @param fvSchemes - Dictionary containing spatial operator and time integration properties
77 * @param fvSolution - Dictionary containing linear solver properties
78 * @param p - A chainable functor that performs manipulations on the assembled system
79 */
80template<typename VectorType, typename IndexType>
81la::SolverStats solve(
82 Expression<typename VectorType::ElementType, IndexType>& exp,
83 VectorType& solution,
84 scalar t,
85 scalar dt,
86 const Dictionary& fvSchemes,
87 const Dictionary& fvSolution,
88 std::vector<const PostAssemblyBase<typename VectorType::ElementType, IndexType>*> p = {}
89)
90{
91 if (exp.temporalOperators().size() == 0 && exp.spatialOperators().size() == 0)
92 {
93 NF_ERROR_EXIT("No temporal or implicit terms to solve.");
94 }
95 exp.read(fvSchemes);
96 auto integrator = timeIntegration::TimeIntegration<VectorType>(
97 fvSchemes.subDict("timeIntegration"), fvSolution
98 );
99
100 if (exp.temporalOperators().size() > 0 && integrator.explicitIntegration())
101 {
102 // integrate equations in time
103 integrator.solve(exp, solution, t, dt);
104 return {{.numIter = -1, .initResNorm = 0, .finalResNorm = 0, .solveTime = 0}};
105 }
106 else
107 {
108 return detail::iterativeSolveImpl(exp, solution, t, dt, fvSolution, p);
109 }
110}
