16/* @class FaceToMatrixAddress
17 * @brief class storing the mapping between mesh faces and target matrix sparsity pattern
18 *
19 * Based on given computational mesh this class stores a mapping for a consistent iteration
20 * procedure for matrices which share the same sparsity pattern.
21 *
22 * This class implements the finite volume 3/5/7 pt stencil specific generation
23 * of sparsity patterns from a given unstructured mesh
24 *
25 */
26template<typename IndexType = localIdx, typename MeshType = UnstructuredMesh>
27class FaceToMatrixAddress
28{
29
30 // clang-format off
31 // NOTE The following data members store a simple mapping from face ids to offsets within
32 // the corresponding row of the CSR values array.
33 //
34 // For an internal face f with owner P and neighbour N (P < N by construction):
35 // ownerOffset[f] = offset within row P for column N → A[P, N] (upper triangular)
36 // neighbourOffset[f] = offset within row N for column P → A[N, P] (lower triangular)
37 //
38 // Example row layout for a cell with two lower entries and one upper entry:
39 // Row P: [ A[P,j0] A[P,j1] | A[P,P] | A[P,N] ]
40 // lower (j<P) diag upper (N>P)
41 // ownerOffset[f] = 3 (position of A[P,N] within row P)
42 // neighbourOffset[f] = 0 (position of A[N,P] within row N)
43 // clang-format on
44 Array<uint8_t>
45 ownerOffset_;
46

50 Array<uint8_t> diagOffset_;

140 /* @brief Returns the flat values-array index of the diagonal entry for cell celli.
141 * diagIdx(celli) = rowOffs[celli] + diagOffset[celli]
142 */
143 KOKKOS_INLINE_FUNCTION localIdx diagIdx(localIdx celli) const
144 {
145 return rowOffsV_[celli] + diagOffsetV_[celli];
146 }
147
148 /* @brief Returns the flat values-array index of the upper-triangular entry A[own, nei].
149 *
150 * By construction own < nei for every internal face, so the column index nei is greater
151 * than the row index own — this entry lies in the upper triangle.
152 *
153 * upperIdx(own, f) = rowOffs[own] + ownerOffset[f]
154 */
155 KOKKOS_INLINE_FUNCTION localIdx upperIdx(localIdx own, localIdx faceIdx) const
156 {
157 return rowOffsV_[own] + ownerOffsetV_[faceIdx];
158 }
159
160 /* @brief Returns the flat values-array index of the lower-triangular entry A[nei, own].
161 *
162 * By construction nei > own for every internal face, so the column index own is less
163 * than the row index nei — this entry lies in the lower triangle.
164 *
165 * lowerIdx(nei, f) = rowOffs[nei] + neighbourOffset[f]
166 */
167 KOKKOS_INLINE_FUNCTION localIdx lowerIdx(localIdx nei, localIdx faceIdx) const
168 {
169 return rowOffsV_[nei] + neighbourOffsetV_[faceIdx];
170 }

183inline Vector<localIdx> computeRowToDiagonalMap(
184 const Vector<localIdx>& rows, std::shared_ptr<const FaceToMatrixAddress<>> matIt
185)
186{
187
188 const auto exec = rows.exec();
189 Vector<localIdx> ret(exec, rows.size());
190 auto retV = ret.view();
191 const auto rowsV = rows.view();
192 const auto [rowOffs, diagOffs] =
193 views(matIt->sparsityPattern()->rowOffs(), matIt->diagOffset());
194
195 parallelFor(
196 exec,
197 {0, ret.size()},
198 KOKKOS_LAMBDA(const localIdx i) {
199 localIdx cell = rowsV[i];
200 auto rowStart = rowOffs[cell];
201 retV[i] = rowStart + diagOffs[cell];
202 },
203 "computeRowToDiagonalMap"
204 );
205
206 return ret;
207}
