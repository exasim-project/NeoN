17struct FaceToMatrixView
18{
19
20 // TODO check performance
21 /* @brief Returns the flat values-array index of the diagonal entry for cell celli.
22 * diagIdx(celli) = rowOffs[celli] + diagOffset[celli]
23 */
24 KOKKOS_INLINE_FUNCTION localIdx diagIdx(localIdx celli) const
25 {
26 return rowOffs[celli] + diagOffset[celli];
27 }
28
29 /* @brief Returns the flat values-array index of the upper-triangular entry A[own, nei].
30 *
31 * By construction own < nei for every internal face, so the column index nei is greater
32 * than the row index own — this entry lies in the upper triangle.
33 *
34 * upperIdx(own, f) = rowOffs[own] + ownerOffset[f]
35 */
36 KOKKOS_INLINE_FUNCTION localIdx upperIdx(localIdx own, localIdx faceIdx) const
37 {
38 return rowOffs[own] + ownerOffset[faceIdx];
39 }
40
41 /* @brief Returns the flat values-array index of the lower-triangular entry A[nei, own].
42 *
43 * By construction nei > own for every internal face, so the column index own is less
44 * than the row index nei — this entry lies in the lower triangle.
45 *
46 * lowerIdx(nei, f) = rowOffs[nei] + neighbourOffset[f]
47 */
48 KOKKOS_INLINE_FUNCTION localIdx lowerIdx(localIdx nei, localIdx faceIdx) const
49 {
50 return rowOffs[nei] + neighbourOffset[faceIdx];
51 }
52
53 // corresponding views
54 View<const uint8_t> ownerOffset;
55
56 View<const uint8_t> neighbourOffset;
57
58 View<const uint8_t> diagOffset;
59
60 // row offsets borrowed from the owning Matrix's sparsity pattern
61 View<const localIdx> rowOffs;
62};

103/* @class FaceToMatrixAddress
104 * @brief Stores the mapping between mesh faces and matrix row offsets.
105 *
106 * Based on a given computational mesh this class stores a mapping for a consistent iteration
107 * procedure for matrices which share the same sparsity pattern.
108 *
109 * This class implements the finite volume 3/5/7 pt stencil specific generation
110 * of face-to-matrix offset mappings. The sparsity pattern itself is owned by
111 * the Matrix, not by this class; this class only borrows a view of the row offsets.
112 *
113 */
114class FaceToMatrixAddress : public NeoN::SupportsCopyTo<FaceToMatrixAddress>
115{
116 void validate() const;
117
118 // clang-format off
119 // NOTE The following data members store a simple mapping from face ids to offsets within
120 // the corresponding row of the CSR values array.
121 //
122 // For an internal face f with owner P and neighbour N (P < N by construction):
123 // ownerOffset[f] = offset within row P for column N → A[P, N] (upper triangular)
124 // neighbourOffset[f] = offset within row N for column P → A[N, P] (lower triangular)
125 //
126 // Example row layout for a cell with two lower entries and one upper entry:
127 // Row P: [ A[P,j0] A[P,j1] | A[P,P] | A[P,N] ]
128 // lower (j<P) diag upper (N>P)
129 // ownerOffset[f] = 3 (position of A[P,N] within row P)
130 // neighbourOffset[f] = 0 (position of A[N,P] within row N)
131 // clang-format on
132 Array<uint8_t>
133 ownerOffset_;
134
135 Array<uint8_t> neighbourOffset_;
137
138 Array<uint8_t> diagOffset_;
139
140public:
141
142 /* @brief constructor
143 *
144 * @param ownerOffset face-to-upper-offset mapping for the owner row
145 * @param neighbourOffset face-to-lower-offset mapping for the neighbour row
146 * @param diagOffset cell-to-diagonal-offset mapping
147 */
148 FaceToMatrixAddress(
149 Array<uint8_t> ownerOffset, Array<uint8_t> neighbourOffset, Array<uint8_t> diagOffset
150 );
