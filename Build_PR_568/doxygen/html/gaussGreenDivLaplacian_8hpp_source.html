98 // Format-generic overload, mirroring DivOperator/LaplacianOperator's
99 // implicitOperation<SystemMatrixType> -- the non-template overloads above still win for the
100 // CSR default, so existing DSL callers are unaffected; ELL callers deduce SystemMatrixType
101 // from the ls argument. Header-inline (not gaussGreenDivLaplacian.cpp): OperatorModel<T>'s
102 // vtable forces this to instantiate in whatever TU constructs a SpatialOperator from a
103 // GaussGreenDivLaplacian -- e.g. DivLapOptimizer::optimize(), itself header-only -- so a
104 // declare-in-header/define-in-cpp split would leave it undefined there (the same failure mode
105 // DdtOperator hit before its equivalent methods were made header-inline).
106 template<typename SystemMatrixType>
107 void implicitOperation(la::LinearSystem<ValueType, ValueType, SystemMatrixType>& ls) const

140 // Shared by the CSR entries (implicitOperation(la::LinearSystem<ValueType>&) and the
141 // segregated form, both defined in gaussGreenDivLaplacian.cpp) and the ELL entry above, so
142 // the assembly sequence (internal faces, non-orthogonal correction, physical/proc boundaries,
143 // deferred div correction) has one source of truth instead of being duplicated per format.
144 // Header-inline for the same reason as implicitOperation<SystemMatrixType> above -- it must
145 // be instantiable wherever that public template is. Cell-based assembly stays CSR-only
146 // (computeDivLaplacianIntCellBasedImpl is CSR-hardcoded); a CellBasedIterator on a non-CSR
147 // system is rejected rather than silently falling back to face-based assembly.
148 template<typename AssemblyType, typename SystemMatrixType>
149 void implicitOperationImpl(la::LinearSystem<AssemblyType, ValueType, SystemMatrixType>& ls
150 ) const
151 {
152 if constexpr (std::is_same_v<SystemMatrixType, la::CSRMatrix<AssemblyType, localIdx>>)
153 {
154 if (auto* cellIter =
155 dynamic_cast<la::CellBasedIterator*>(ls.getMeshIterator()->get().get()))
156 {
157 if (!cellIter->getCellBasedData())
158 {
159 cellIter->setComputeCellBasedData(
160 this->getVector().mesh(), ls.matrix().sparsity(), ls.faceToMatrixAddress()
161 );
162 }
163 computeDivLaplacianIntCellBasedImpl(
164 ls,
165 this->getVector(),
166 flux_,
167 gamma_,
168 *divSurfaceInterpolation_,
169 *faceNormalGradient_,
170 coeffA_,
171 coeffB_
172 );
173 }
174 else
175 {
176 computeDivLaplacianIntImpl(
177 ls,
178 this->getVector(),
179 flux_,
180 gamma_,
181 *divSurfaceInterpolation_,
182 *faceNormalGradient_,
183 coeffA_,
184 coeffB_
185 );
186 }
187 }
188 else
189 {
190 NF_ASSERT(
191 dynamic_cast<la::CellBasedIterator*>(ls.getMeshIterator()->get().get()) == nullptr,
192 "Cell-based iteration is not implemented for ELL assembly"
193 );
194 computeDivLaplacianIntImpl(
195 ls,
196 this->getVector(),
197 flux_,
198 gamma_,
199 *divSurfaceInterpolation_,
200 *faceNormalGradient_,
201 coeffA_,
202 coeffB_
203 );
204 }
205 computeLaplacianNonOrthCorrImpl(
206 ls, gamma_, this->getVector(), coeffB_, *faceNormalGradient_
207 );
208 computeDivLaplacianBoundImpl(
209 ls,
210 this->getVector(),
211 flux_,
212 gamma_,
213 *divSurfaceInterpolation_,
214 *faceNormalGradient_,
215 coeffA_,
216 coeffB_
217 );
218 computeDivLaplacianProcBoundImpl(
219 ls,
220 this->getVector(),
221 flux_,
222 gamma_,
223 *divSurfaceInterpolation_,
224 *faceNormalGradient_,
225 coeffA_,
226 coeffB_
227 );
228
229 // Deferred correction for a corrected div scheme (e.g. linearUpwind): the fused matrix
230 // uses upwind div weights, the gradient correction is added explicitly to the rhs.
231 if (divSurfaceInterpolation_->corrected())
232 {
233 const auto& mesh = this->getVector().mesh();
234 SurfaceField<ValueType> correction(
235 this->getVector().exec(),
236 "divCorrection",
237 mesh,
238 createCalculatedBCs<SurfaceBoundary<ValueType>>(mesh)
239 );
240 divSurfaceInterpolation_->correction(flux_, this->getVector(), correction);
241 addDivCorrectionToRhs(ls, flux_, correction, coeffA_);
242 }
243 }
244};
