45/* @brief Concept satisfied when T can assemble its temporal contribution into a
46 * LinearSystem whose matrix coefficients are scalar while the RHS holds T's
47 * field value type (segregated vector-solve form).
48 */
49template<typename T>
50concept HasTemporalImplicitOperatorScalarMtx = requires(T t) {
51 {
52 t.implicitOperation(
53 std::declval<la::LinearSystem<scalar, typename T::VectorValueType>&>(),
54 std::declval<NeoN::scalar>(),
55 std::declval<NeoN::scalar>()
56 )
57 } -> std::same_as<void>;
58};

60/* @brief Concept satisfied when T can assemble its temporal contribution into a native
61 * ELL-backed LinearSystem (scalar matrix, scalar rhs). Matches
62 * DdtOperator::implicitOperation<SystemMatrixType>'s template shape directly.
63 */
64template<typename T>
65concept HasTemporalImplicitOperatorELL = requires(T t) {
66 {
67 t.implicitOperation(
68 std::declval<la::LinearSystem<scalar, scalar, la::ELLMatrix<scalar, localIdx>>&>(),
69 std::declval<NeoN::scalar>(),
70 std::declval<NeoN::scalar>()
71 )
72 } -> std::same_as<void>;
73};
74
75/* @brief Concept satisfied when T can assemble its temporal contribution into a native
76 * ELL-backed LinearSystem in the segregated vector-solve form (scalar matrix,
77 * T::VectorValueType rhs). Only meaningful when VectorValueType != scalar -- for scalar T
78 * this collides with HasTemporalImplicitOperatorELL (same declval type), which is why the
79 * dispatch below only ever calls this branch when ValueType != scalar (mirrors
80 * HasTemporalImplicitOperatorScalarMtx's CSR counterpart).
81 */
82template<typename T>
83concept HasTemporalImplicitOperatorScalarMtxELL = requires(T t) {
84 {
85 t.implicitOperation(
86 std::declval<la::LinearSystem<
87 scalar,
88 typename T::VectorValueType,
89 la::ELLMatrix<scalar, localIdx>>&>(),
90 std::declval<NeoN::scalar>(),
91 std::declval<NeoN::scalar>()
92 )
93 } -> std::same_as<void>;
94};

96template<typename T>
97concept HasTemporalOperator =
98 HasTemporalExplicitOperator<T> || HasTemporalImplicitOperator<T>
99 || HasTemporalImplicitOperatorScalarMtx<T> || HasTemporalImplicitOperatorELL<T>
100 || HasTemporalImplicitOperatorScalarMtxELL<T>;

102/* @class TemporalOperator
103 * @brief A class to represent a TemporalOperator in NeoNs DSL
104 *
105 * The design here is based on the type erasure design pattern
106 * see https://www.youtube.com/watch?v=4eeESJQk-mw
107 *
108 * Motivation for using type erasure is that concrete implementation
109 * of TemporalOperator e.g ddt, d2dt2, etc can be stored in a vector of
110 * TemporalOperator
111 *
112 * @ingroup dsl
113 */
114template<typename ValueType>
115class TemporalOperator
116{
117public:
118
119 using VectorValueType = ValueType;
120
121 template<HasTemporalOperator T>
122 TemporalOperator(T cls) : model_(std::make_unique<TemporalOperatorModel<T>>(std::move(cls)))
123 {}
124
125 TemporalOperator(const TemporalOperator& eqnOperator) : model_ {eqnOperator.model_->clone()} {}
126
127 TemporalOperator(TemporalOperator&& eqnOperator) : model_ {std::move(eqnOperator.model_)} {}
128
129 TemporalOperator& operator=(const TemporalOperator& eqnOperator)
130 {
131 model_ = eqnOperator.model_->clone();
132 return *this;
133 }
134
135 void explicitOperation(Vector<ValueType>& source, scalar t, scalar dt) const
136 {
137 model_->explicitOperation(source, t, dt);
138 }
139
140 void implicitOperation(la::LinearSystem<ValueType>& ls, scalar t, scalar dt) const
141 {
142 model_->implicitOperation(ls, t, dt);
143 }
144
145 /* @brief Implicit temporal assembly into a scalar-matrix / ValueType-rhs linear system
146 * (segregated vector-solve form). Disabled when ValueType == scalar to avoid
147 * colliding with the same-type overload above.
148 */
149 void implicitOperation(la::LinearSystem<scalar, ValueType>& ls, scalar t, scalar dt) const
150 requires(!std::is_same_v<ValueType, scalar>)
151 {
152 model_->implicitOperationScalarMtx(ls, t, dt);
153 }
154
155 /* @brief Implicit temporal assembly into a native ELL-backed (scalar matrix, scalar rhs)
156 * linear system. A fixed, concrete parameter type distinct from both overloads
157 * above, so this coexists without ambiguity regardless of ValueType.
158 */
159 void implicitOperation(
160 la::LinearSystem<scalar, scalar, la::ELLMatrix<scalar, localIdx>>& ls, scalar t, scalar dt
161 ) const
162 {
163 model_->implicitOperationELL(ls, t, dt);
164 }
165
166 /* @brief Implicit temporal assembly into a native ELL-backed LinearSystem in the segregated
167 * vector-solve form (scalar matrix, ValueType rhs). Disabled when ValueType == scalar
168 * to avoid colliding with the same-type ELL overload above (identical declval type).
169 */
170 void implicitOperation(
171 la::LinearSystem<scalar, ValueType, la::ELLMatrix<scalar, localIdx>>& ls,
172 scalar t,
173 scalar dt
174 ) const
175 requires(!std::is_same_v<ValueType, scalar>)
176 {
177 model_->implicitOperationScalarMtxELL(ls, t, dt);
178 }
179
180 /* returns the fundamental type of an operator, ie explicit, implicit */
181 Operator::Type getType() const { return model_->getType(); }
182
183 std::string getName() const { return model_->getName(); }
184
185 Coeff& getCoefficient() { return model_->getCoefficient(); }
186
187 Coeff getCoefficient() const { return model_->getCoefficient(); }
188
189 /* @brief Given an input this function reads required properties */
190 void read(const Input& input) { model_->read(input); }
191
192 /* @brief Get the executor */
193 const Executor& exec() const { return model_->exec(); }
194
195 /* @brief Get the ddtScheme */
196 NeoN::finiteVolume::cellCentred::DdtScheme ddtScheme() const { return model_->ddtScheme(); }

200 /* @brief Base class defining the concept of a term. This effectively
201 * defines what functions need to be implemented by a concrete Operator implementation
202 * */
203 struct TemporalOperatorConcept
204 {
205 virtual ~TemporalOperatorConcept() = default;
206
207 virtual void explicitOperation(Vector<ValueType>& source, scalar t, scalar dt) = 0;
208
209 virtual void implicitOperation(la::LinearSystem<ValueType>& ls, scalar t, scalar dt) = 0;
210
211 /* @brief Temporal assembly into LinearSystem<scalar, ValueType> for the
212 * scalar-matrix / ValueType-rhs (segregated vector-solve) form.
213 * Concrete operators that don't support this form leave it as a no-op.
214 */
215 virtual void implicitOperationScalarMtx(
216 la::LinearSystem<scalar, ValueType>& ls, scalar t, scalar dt
217 ) = 0;

232 * don't support this yet throw, same as implicitOperationELL /
233 * implicitOperationScalarMtx.
234 */
235 virtual void implicitOperationScalarMtxELL(
236 la::LinearSystem<scalar, ValueType, la::ELLMatrix<scalar, localIdx>>& ls,
237 scalar t,

241 /* @brief Given an input this function reads required properties */
242 virtual void read(const Input& input) = 0;
243
244 /* returns the name of the operator */
245 virtual std::string getName() const = 0;
246
247 /* returns the fundamental type of an operator, ie explicit, implicit, temporal */
248 virtual Operator::Type getType() const = 0;

257 virtual const Executor& exec() const = 0;

269 // Templated derived class to implement the type-specific behavior
270 template<typename ConcreteTemporalOperatorType>
271 struct TemporalOperatorModel : TemporalOperatorConcept
272 {
273 /* @brief build with concrete TemporalOperator */
274 TemporalOperatorModel(ConcreteTemporalOperatorType concreteOp)
275 : concreteOp_(std::move(concreteOp))
276 {}
277
278 /* returns the name of the operator */
279 std::string getName() const override { return concreteOp_.getName(); }
280
281 virtual void explicitOperation(Vector<ValueType>& source, scalar t, scalar dt) override
282 {
283 if constexpr (HasTemporalExplicitOperator<ConcreteTemporalOperatorType>)
284 {
285 concreteOp_.explicitOperation(source, t, dt);
286 }
287 }
288
289 virtual void
290 implicitOperation(la::LinearSystem<ValueType>& ls, scalar t, scalar dt) override
291 {
292 if constexpr (HasTemporalImplicitOperator<ConcreteTemporalOperatorType>)
293 {
294 concreteOp_.implicitOperation(ls, t, dt);
295 }
296 }
297
298 virtual void implicitOperationScalarMtx(
299 la::LinearSystem<scalar, ValueType>& ls, scalar t, scalar dt
300 ) override
301 {
302 if constexpr (HasTemporalImplicitOperatorScalarMtx<ConcreteTemporalOperatorType>)
303 {
304 concreteOp_.implicitOperation(ls, t, dt);
305 }
306 else
307 {
308 // Reached only for an implicit temporal operator that lacks the scalar-matrix
309 // (segregated vector-solve) overload. Silently skipping it would drop its
310 // contribution (e.g. an implicit ddt term) and yield a wrong system, so fail
311 // fast instead.
312 NF_ERROR_EXIT(
313 "Temporal operator '" << getName()
314 << "' does not support scalar-matrix (segregated) "
315 "assembly."
316 );
317 }
318 }
319
320 virtual void implicitOperationELL(
321 la::LinearSystem<scalar, scalar, la::ELLMatrix<scalar, localIdx>>& ls,
322 scalar t,
323 scalar dt
324 ) override
325 {
326 if constexpr (HasTemporalImplicitOperatorELL<ConcreteTemporalOperatorType>)
327 {
328 concreteOp_.implicitOperation(ls, t, dt);
329 }
330 else
331 {
332 NF_ERROR_EXIT(
333 "Temporal operator '" << getName() << "' does not support ELL assembly."
334 );
335 }
336 }
337
338 virtual void implicitOperationScalarMtxELL(
339 la::LinearSystem<scalar, ValueType, la::ELLMatrix<scalar, localIdx>>& ls,
340 scalar t,
341 scalar dt
342 ) override
343 {
344 if constexpr (HasTemporalImplicitOperatorScalarMtxELL<ConcreteTemporalOperatorType>)
345 {
346 concreteOp_.implicitOperation(ls, t, dt);
347 }
348 else
349 {
350 NF_ERROR_EXIT(
351 "Temporal operator '" << getName()
352 << "' does not support scalar-matrix (segregated) "
353 "ELL assembly."
354 );
355 }
356 }
357
358 /* @brief Given an input this function reads required coeffs */
359 virtual void read(const Input& input) override { concreteOp_.read(input); }
360
361 /* returns the fundamental type of an operator, ie explicit, implicit, temporal */
362 Operator::Type getType() const override { return concreteOp_.getType(); }
363
364 /* @brief Get the executor */
365 const Executor& exec() const override { return concreteOp_.exec(); }
366
367 /* @brief get the associated coefficient for this term */
368 virtual Coeff& getCoefficient() override { return concreteOp_.getCoefficient(); }
369
370 /* @brief get the associated coefficient for this term */
371 virtual Coeff getCoefficient() const override { return concreteOp_.getCoefficient(); }
372
373 /* @brief return the ddtScheme read by the ddtOperator */
374 NeoN::finiteVolume::cellCentred::DdtScheme ddtScheme() const override
375 {
376 if constexpr (requires { concreteOp_.scheme(); })
377 {
378 return concreteOp_.scheme();
379 }
380 else
381 {
382 return NeoN::finiteVolume::cellCentred::DdtScheme::None;
383 }
384 }
385 // The Prototype Design Pattern
386 std::unique_ptr<TemporalOperatorConcept> clone() const override
387 {
388 return std::make_unique<TemporalOperatorModel>(*this);
389 }
390
391 ConcreteTemporalOperatorType concreteOp_;
392 };

394 std::unique_ptr<TemporalOperatorConcept> model_;
