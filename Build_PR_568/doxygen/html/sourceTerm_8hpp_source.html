41 // Format-generic Sp assembly, defined here (not sourceTerm.cpp) so callers can
42 // instantiate it for any SystemMatrixType, e.g. ELLMatrix. Overloads (not replaces) the
43 // non-template implicitOperation() below -- for the CSR default, that exact non-template
44 // match wins over this template per normal overload resolution, so DSL callers are
45 // unaffected; ELL (or any other format) callers deduce SystemMatrixType here directly.
46 template<typename SystemMatrixType>
47 void implicitOperation(la::LinearSystem<ValueType, ValueType, SystemMatrixType>& ls) const
48 {
49 if (!spCoeff_)
50 {
51 NF_ERROR_EXIT("Not implemented");
52 }
53 // Sp implicit: diagonal += scaling * spCoeff * volume
54 const auto operatorScaling = this->getCoefficient();
55 const auto vol = spCoeff_->mesh().cellVolumes().view();
56 const auto [coeff] = views(spCoeff_->internalVector());
57 auto values = ls.matrix().values().view();
58 const auto ma = ls.matrix().faceToMatrixView();
59
60 NeoN::parallelFor(
61 ls.exec(),
62 {0, coeff.size()},
63 NEON_LAMBDA(const localIdx celli) {
64 values[ma.diagIdx(celli)] +=
65 operatorScaling[celli] * coeff[celli] * vol[celli] * one<ValueType>();
66 },
67 "Sp::implicitOperation"
68 );
69 }
