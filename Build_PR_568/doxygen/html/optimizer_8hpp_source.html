54template<typename ExpressionType>
55class DivLapOptimizer : public Optimizer<ExpressionType>
56{
57public:
58
59 ~DivLapOptimizer() {}
60
61 ExpressionType optimize(const ExpressionType& expr) const override
62 {
63 ExpressionType out(expr);
64
65 // early return if not both operators are present
66 if (!(expr.template hasOperator<Operator::Type::Implicit>("DivOperator")
67 && expr.template hasOperator<Operator::Type::Implicit>("LaplacianOperator")))
68 {
69 return out;
70 }
71
72 using ValueType = typename ExpressionType::ExpressionValueType;
73 auto divOperator =
74 out.template getOperator<SpatialOperator<ValueType>, Operator::Type::Implicit>(
75 "DivOperator"
76 )
77 .getConfig();
78 auto lapOperator =
79 out.template getOperator<SpatialOperator<ValueType>, Operator::Type::Implicit>(
80 "LaplacianOperator"
81 )
82 .getConfig();
83
84 dsl::SpatialOperator<ValueType> divLapOperator =
85 finiteVolume::cellCentred::GaussGreenDivLaplacian<ValueType>(
86 expr.exec(), divOperator, lapOperator
87 );
88 out.addOperator(divLapOperator);
89
90 out.template dropOperator<Operator::Type::Implicit>("DivOperator");
91 out.template dropOperator<Operator::Type::Implicit>("LaplacianOperator");
92
93 return out;
94 }
95};
