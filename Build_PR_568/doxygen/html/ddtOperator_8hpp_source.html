42 // Format-generic implicit temporal assembly. Defined here (not ddtOperator.cpp) so any TU
43 // can instantiate it for any SystemMatrixType without needing an explicit instantiation --
44 // same reasoning as SourceTerm::implicitOperation<SystemMatrixType>. SystemMatrixType is
45 // deduced from ls's own type, so the existing CSR call site (TemporalOperator's type
46 // erasure) is unaffected.
47 template<typename SystemMatrixType>
48 void implicitOperation(
49 la::LinearSystem<ValueType, ValueType, SystemMatrixType>& ls, scalar t, scalar dt
50 ) const
51 {
52 const int level = oldTimeLevel(this->field_);
53
54 if (scheme_ == DdtScheme::BDF1)
55 {
56 bdf1Kernel(ls, t, dt);
57 }
58 else if (level < 2)
59 {
60 bdf1Kernel(ls, t, dt); // startup step
61 }
62 else
63 {
64 bdf2Kernel(ls, t, dt);
65 }
66 }

68 template<typename SystemMatrixType>
69 void bdf1Kernel(la::LinearSystem<ValueType, ValueType, SystemMatrixType>& ls, scalar, scalar dt)
70 const
71 {
72 const auto vol = this->getVector().mesh().cellVolumes().view();
73 const auto operatorScaling = this->getCoefficient();
74 const auto oldVector = oldTime(this->field_).internalVector().view();
75 auto [rhs, values] = views(ls.rhs(), ls.matrix().values());
76 const auto ma = ls.matrix().faceToMatrixView();
77
78 const scalar a0a1 = 1.0 / dt;
79
80 parallelFor(
81 ls.exec(),
82 {0, oldVector.size()},
83 NEON_LAMBDA(const localIdx celli) {
84 const auto commonCoef = operatorScaling[celli] * vol[celli];
85 values[ma.diagIdx(celli)] += commonCoef * a0a1 * one<ValueType>();
86 rhs[celli] += commonCoef * a0a1 * oldVector[celli];
87 },
88 "ddtOperator::implicitOperation<BDF1>"
89 );
90 }
91
92 template<typename SystemMatrixType>
93 void bdf2Kernel(la::LinearSystem<ValueType, ValueType, SystemMatrixType>& ls, scalar, scalar dt)
94 const
95 {
96 const auto vol = this->getVector().mesh().cellVolumes().view();
97 const auto operatorScaling = this->getCoefficient();
98 auto& old = oldTime(this->field_);
99 auto& oldOld = oldTime(old);
100 const auto [oldVector, oldOldVector] = views(old.internalVector(), oldOld.internalVector());
101 auto [rhs, values] = views(ls.rhs(), ls.matrix().values());
102
103 const auto ma = ls.matrix().faceToMatrixView();
104
105 const scalar a0 = 1.5 / dt;
106 const scalar a1 = 2.0 / dt;
107 const scalar a2 = -0.5 / dt;
108
109 parallelFor(
110 ls.exec(),
111 {0, oldVector.size()},
112 NEON_LAMBDA(const localIdx celli) {
113 const auto commonCoef = operatorScaling[celli] * vol[celli];
114 values[ma.diagIdx(celli)] += commonCoef * a0 * one<ValueType>();
115 rhs[celli] +=
116 commonCoef * a1 * oldVector[celli] + commonCoef * a2 * oldOldVector[celli];
117 },
118 "ddtOperator::implicitOperation<BDF2>"
119 );
120 }

122 // Segregated counterpart of bdf1Kernel above -- same structure, but the diagonal coefficient
123 // is a scalar shared across every rhs component instead of a ValueType. Header-inline and
124 // SystemMatrixType-generic for the same reason as bdf1Kernel.
125 template<typename SystemMatrixType>
126 void bdf1KernelScalarMtx(
127 la::LinearSystem<scalar, ValueType, SystemMatrixType>& ls, scalar, scalar dt
128 ) const
129 {
130 const auto vol = this->getVector().mesh().cellVolumes().view();
131 const auto operatorScaling = this->getCoefficient();
132 const auto oldVector = oldTime(this->field_).internalVector().view();
133 auto [rhs, values] = views(ls.rhs(), ls.matrix().values());
134 const auto ma = ls.matrix().faceToMatrixView();
135
136 const scalar a0a1 = 1.0 / dt;
137
138 parallelFor(
139 ls.exec(),
140 {0, oldVector.size()},
141 NEON_LAMBDA(const localIdx celli) {
142 const auto commonCoef = operatorScaling[celli] * vol[celli];
143 values[ma.diagIdx(celli)] += commonCoef * a0a1;
144 rhs[celli] += commonCoef * a0a1 * oldVector[celli];

180 /* @brief Implicit temporal assembly into a scalar-matrix / ValueType-rhs linear system
181 * (segregated vector-solve form). Only present when ValueType != scalar; for scalar
182 * fields the same-type overload above already covers LinearSystem<scalar, scalar>.
183 * The scalar diagonal entry scales every rhs component equally. Format-generic over
184 * SystemMatrixType (CSR or ELL) and header-inline, same reasoning as
185 * implicitOperation<SystemMatrixType> above. Guarded on ValueType (the class's own
186 * parameter, not a separate member-template one), so unlike the old declare-in-header
187 * /define-in-cpp version, SystemMatrixType is the only substitutable parameter and is
188 * always deduced from ls -- no risk of an explicit template argument silently picking
189 * an unrelated, never-defined specialization.
190 */
191 template<typename SystemMatrixType>
192 requires(!std::is_same_v<ValueType, scalar>)
193 void implicitOperation(
194 la::LinearSystem<scalar, ValueType, SystemMatrixType>& ls, scalar t, scalar dt
195 ) const
196 {
197 const int level = oldTimeLevel(this->field_);
198
199 if (scheme_ == DdtScheme::BDF1)
200 {
201 bdf1KernelScalarMtx(ls, t, dt);
202 }
203 else if (level < 2)
204 {
205 bdf1KernelScalarMtx(ls, t, dt); // startup step
206 }
207 else
208 {
209 bdf2KernelScalarMtx(ls, t, dt);
210 }
211 }
