312/*@brief helper function that creates a zero initialised linear system based on a given mesh
313 */
314template<
315 typename ValueType,
316 typename RHSValueType = ValueType,
317 typename SystemMatrixType = CSRMatrix<ValueType, localIdx>,
318 typename BoundaryMatrixType = COOMatrix<ValueType, localIdx>>
319LinearSystem<ValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType> createEmptyLinearSystem(
320 const UnstructuredMesh& mesh,
321 std::shared_ptr<MeshIterationStrategy> strategy = std::make_shared<FaceBasedIterator>()
322)
323{
324 auto [sp, mi] =
325 createSparsityPatternFaceToMatrixAddress<typename SystemMatrixType::MatrixSparsityType>(mesh
326 );
327 auto bSp =
328 createBoundarySparsityPattern<typename BoundaryMatrixType::MatrixSparsityType>(mesh, *mi);
329 const auto exec = sp->exec();
330 const auto nCells = static_cast<localIdx>(mesh.nCells());
331 const auto nProcFaces = static_cast<localIdx>(mesh.nProcBoundaryFaces());
332 using IndexType = typename BoundaryMatrixType::MatrixSparsityType::SparsityIndexType;
333
334 Vector<IndexType> offDiagColIdxs(exec, nProcFaces, 0);
335 Vector<IndexType> offDiagRowIdxs(exec, nProcFaces, 0);
336
337#ifdef NF_WITH_MPI_SUPPORT
338 auto commPattern = computeCommunicationPattern(mesh);
339 if (nProcFaces > 0)
340 {
341 const localIdx nBoundaryFaces = static_cast<localIdx>(mesh.nBoundaryFaces());
342 const auto faceOwnersH = mesh.boundaryMesh().faceOwners().copyToHost();
343 const auto faceOwnersV = faceOwnersH.view();
344 Vector<IndexType> rowH(SerialExecutor {}, nProcFaces, 0);
345 Vector<IndexType> colH(SerialExecutor {}, nProcFaces, 0);
346 auto rowHV = rowH.view();
347 auto colHV = colH.view();
348 for (localIdx i = 0; i < nProcFaces; ++i)
349 {
350 // Store the local row index directly. The global offset used to be added here and
351 // subtracted again on the Ginkgo side; keeping the rows local avoids that round-trip.
352 // The column index stays global (it identifies a remote cell) and is consumed by
353 // Ginkgo's distributed index_map.
354 rowHV[i] = faceOwnersV[nBoundaryFaces + i];
355 colHV[i] = static_cast<IndexType>(commPattern.recvIdx[static_cast<std::size_t>(i)]);
356 }
357 // offDiagRowSortPerm[j] = proc-face index whose row/col belongs at sorted position j.
358 // Already computed (and stored) in BoundaryMesh; reuse it here to avoid re-sorting.
359 const auto& offDiagRowSortPerm = mesh.boundaryMesh().getRowSortPerm();
360 {
361 std::vector<IndexType> sortedRow(static_cast<std::size_t>(nProcFaces));
362 std::vector<IndexType> sortedCol(static_cast<std::size_t>(nProcFaces));
363 for (localIdx j = 0; j < nProcFaces; ++j)
364 {
365 auto src = offDiagRowSortPerm[static_cast<std::size_t>(j)];
366 sortedRow[static_cast<std::size_t>(j)] = rowHV[src];
367 sortedCol[static_cast<std::size_t>(j)] = colHV[src];
368 }
369 offDiagRowIdxs = Vector<IndexType>(exec, std::move(sortedRow));
370 offDiagColIdxs = Vector<IndexType>(exec, std::move(sortedCol));
371 }
372 commPattern.offDiagRowSortPerm = std::move(offDiagRowSortPerm);
373 }
374#endif
375
376 auto offDiagSp = std::make_shared<const typename BoundaryMatrixType::MatrixSparsityType>(
377 std::move(offDiagColIdxs), std::move(offDiagRowIdxs), Dimensions {nCells, nCells}
378 );
379
380 LinearSystem<ValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType> ls {
381 SystemMatrixType(Vector<ValueType>(sp->exec(), sp->nnz(), zero<ValueType>()), sp, mi),
382 Vector<RHSValueType>(sp->exec(), sp->rows(), zero<RHSValueType>()),
383 BoundaryMatrixType(Vector<ValueType>(exec, nProcFaces, zero<ValueType>()), offDiagSp),
384 BoundaryMatrixType(Vector<ValueType>(bSp->exec(), bSp->nnz(), zero<ValueType>()), bSp),
385 Vector<RHSValueType>(bSp->exec(), bSp->nnz(), zero<RHSValueType>()),
386 strategy
387 };
388
389#ifdef NF_WITH_MPI_SUPPORT
390 ls.commPattern() = std::move(commPattern);
391#endif
392
393 return ls;
394}

402template<
403 typename MatrixValueType,
404 typename RHSValueType,
405 typename SystemMatrixType,
406 typename BoundaryMatrixType>
407inline la::LinearSystem<MatrixValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType>
408removeBoundaryContributions(
409 const la::LinearSystem<MatrixValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType>&
410 lsIn
411)
412{
413 auto ls =
414 la::LinearSystem<MatrixValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType>(lsIn);
415 auto lsView = ls.view();
416 auto& matrix = lsView.matrix;
417 auto& rhs = lsView.rhs;
418 auto& bMatrix = lsView.boundaryMatrix;
419 auto& bRhs = lsView.boundaryRhs;
420
421 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
422
423 parallelFor(
424 ls.exec(),
425 {0, bMatrix.values.size()},
426 NEON_LAMBDA(const localIdx facei) {
427 const auto celli = bMatrix.sparsity.rowOffs[facei]; // cell index stored in rowOffs
428 Kokkos::atomic_add(&matrix.values[ma.diagIdx(celli)], bMatrix.values[facei]);
429 Kokkos::atomic_add(&rhs[celli], bRhs[facei]);
430 },
431 "removeBoundaryContributions"
432 );
433
434 return ls;
435}
