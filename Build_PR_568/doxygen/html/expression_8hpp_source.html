65template<typename ValueType, typename IndexType = localIdx>
66class SetReference : public PostAssemblyBase<ValueType, IndexType>
67{
68public:
69
70 SetReference(localIdx refCell, ValueType refValue) : refCell_(refCell), refValue_(refValue) {}
71
72 void operator()(la::LinearSystem<ValueType, ValueType, la::CSRMatrix<ValueType, IndexType>>& ls
73 ) const override
74 {
75#ifdef NF_WITH_MPI_SUPPORT
76 // For distributed systems, only the rank owning refCell applies the constraint.
77 // For non-distributed systems (each rank holds a full copy), every rank applies it.
78 if (!ls.commPattern().sendCounts.empty())
79 {
80 mpi::Environment mpiEnv;
81 if (mpiEnv.isInitialized() && mpiEnv.rank() != 0) return;
82 }
83#endif
84 auto lsView = ls.view();
85 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
86 auto refVal = refValue_;
87 auto refCell = refCell_;
88 parallelFor(
89 ls.exec(),
90 {refCell, refCell + 1},
91 NEON_LAMBDA(const localIdx celli) {
92 auto dIdx = ma.diagIdx(celli);
93 auto diagVal = lsView.matrix.values[dIdx];
94 lsView.rhs[celli] += diagVal * refVal;
95 lsView.matrix.values[dIdx] += diagVal;
96 },
97 "SetReference"
98 );
99 }
100
104 void applyScalarMatrix(la::LinearSystem<
105 scalar,
106 ValueType,
107 la::CSRMatrix<scalar, IndexType>,
108 la::COOMatrix<scalar, IndexType>>& ls) const override
109 {
110#ifdef NF_WITH_MPI_SUPPORT
111 // For distributed systems, only the rank owning refCell applies the constraint.
112 // For non-distributed systems (each rank holds a full copy), every rank applies it.
113 if (!ls.commPattern().sendCounts.empty())
114 {
115 mpi::Environment mpiEnv;
116 if (mpiEnv.isInitialized() && mpiEnv.rank() != 0) return;
117 }
118#endif
119 auto lsView = ls.view();
120 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
121 auto refVal = refValue_;
122 auto refCell = refCell_;
123 parallelFor(
124 ls.exec(),
125 {refCell, refCell + 1},
126 NEON_LAMBDA(const localIdx celli) {
127 auto dIdx = ma.diagIdx(celli);
128 auto diagVal = lsView.matrix.values[dIdx];
129 lsView.rhs[celli] += diagVal * refVal;
130 lsView.matrix.values[dIdx] += diagVal;
131 },
132 "SetReference"
133 );
134 }

162template<typename ValueType, typename IndexType = localIdx>
163class FixedValueConstraints : public PostAssemblyBase<ValueType, IndexType>
164{
165 static_assert(
166 std::is_same_v<ValueType, scalar>,
167 "FixedValueConstraints only supports scalar fields. "
168 "For non-scalar fields implement applyScalarMatrix()."
169 );
170
171public:
172
173 FixedValueConstraints(View<const scalar> mask, View<const ValueType> values, localIdx nCells)
174 : mask_(mask), values_(values), nCells_(nCells)
175 {}
176
177 void operator()(la::LinearSystem<ValueType, ValueType, la::CSRMatrix<ValueType, IndexType>>& ls
178 ) const override
179 {
180 auto lsView = ls.view();
181 const auto rowOffs = ls.matrix().sparsity()->rowOffs().view();
182 const auto colIdxs = ls.matrix().sparsity()->colIdxs().view();
183 auto matrixValues = lsView.matrix.values;
184 auto rhs = lsView.rhs;
185 auto mask = mask_;
186 auto vals = values_;
187 // Sweep every row: pinned rows drop their off-diagonals; non-pinned rows that couple into
188 // a pinned column absorb the term into their rhs and zero the coefficient. Parallelising
189 // over rows avoids atomics since each row owns its own rhs entry and off-diagonal slots.
190 parallelFor(
191 ls.exec(),
192 {0, nCells_},
193 NEON_LAMBDA(const localIdx row) {
194 const bool rowPinned = mask[row] != scalar(0);
195 ValueType diagVal = zero<ValueType>();
196 for (auto o = rowOffs[row]; o < rowOffs[row + 1]; ++o)
197 {
198 const auto col = colIdxs[o];
199 if (col == row)
200 {
201 diagVal = matrixValues[o];
202 continue;
203 }
204 if (rowPinned)
205 {
206 // pinned cell's own row: decouple it entirely
207 matrixValues[o] = zero<ValueType>();
208 }
209 else if (mask[col] != scalar(0))
210 {
211 // column cut: absorb coupling into rhs, zero the coefficient
212 rhs[row] -= matrixValues[o] * vals[col];
213 matrixValues[o] = zero<ValueType>();
214 }
215 }
216 if (rowPinned)
217 {
218 rhs[row] = diagVal * vals[row];
219 }
220 },
221 "FixedValueConstraints"
222 );
223 // Zero offDiagonalMatrix entries for pinned rows so that proc-boundary
224 // couplings do not contribute to the residual of constrained cells.
225 auto& offDiag = ls.offDiagonalMatrix();
226 const localIdx nnz = offDiag.nNonZeros();
227 if (nnz > 0)
228 {
229 const auto offRowIdxs = offDiag.sparsity()->rowIdxs().view();
230 auto offValues = offDiag.values().view();
231 parallelFor(
232 ls.exec(),
233 {0, nnz},
234 NEON_LAMBDA(const localIdx i) {
235 if (mask[offRowIdxs[i]] != scalar(0)) offValues[i] = zero<ValueType>();
236 },
237 "FixedValueConstraints::offDiag"
238 );
239 }
240 }
241
242private:
243
244 View<const scalar> mask_;
245 View<const ValueType> values_;
246 localIdx nCells_;
247};

364 /*@brief subtract explicit source terms from the linear system rhs, scaled by cell volumes */
365 template<typename AssemblyType = ValueType>
366 void assembleExplicitSource(
367 la::LinearSystem<AssemblyType, ValueType>& ls, const UnstructuredMesh& mesh
368 ) const
369 {
370 auto expTmp = explicitOperation(static_cast<localIdx>(mesh.nCells()));
371 auto [vol, expSource, rhs] = views(mesh.cellVolumes(), expTmp, ls.rhs());
372 parallelFor(
373 ls.exec(),
374 {0, static_cast<localIdx>(rhs.size())},
375 NEON_LAMBDA(const localIdx i) { rhs[i] -= expSource[i] * vol[i]; }
376 );
377 }

414 /* @brief assemble into a given linear system (implicit operators only, no explicit sources)
415 *
416 * @param ps post-assembly functors applied to the system after assembly
417 */
418 template<typename AssemblyType = ValueType>
419 void assemble(
420 scalar t,
421 scalar dt,
422 la::LinearSystem<AssemblyType, ValueType>& ls,
423 std::vector<const PostAssemblyBase<ValueType, IndexType>*> ps = {}
424 ) const
425 {
426 assembleSpatialOperator(ls); // add spatial operator
427 assembleTemporalOperator(ls, t, dt); // add temporal operators
428
429 // Post-assembly functors apply on the same-type form via operator(); the segregated
430 // scalar-matrix / ValueType-rhs form dispatches to applyScalarMatrix instead.
431 if constexpr (std::is_same_v<AssemblyType, ValueType>)
432 {
433 for (const auto* p : ps)
434 {
435 (*p)(ls);
436 }
437 }
438 else if constexpr (std::is_same_v<AssemblyType, scalar>)
439 {
440 for (const auto* p : ps)
441 {
442 p->applyScalarMatrix(ls);
443 }
444 }
445 }

467 template<typename OperatorType, Operator::Type Type>
468 bool hasOperatorOfType(const std::string& name) const
469 {
470 auto opType = Type;
471 auto matchNameAndType = [name, opType](const auto& op)
472 { return op.getName() == name && op.getType() == opType; };
473 if constexpr (std::is_same_v<OperatorType, SpatialOperator<ValueType>>)
474 {
475 return std::ranges::any_of(spatialOperators_, matchNameAndType);
476 }
477 else if constexpr (std::is_same_v<OperatorType, TemporalOperator<ValueType>>)
478 {
479 return std::ranges::any_of(temporalOperators_, matchNameAndType);
480 }
481 return false;
482 }
483
485 template<Operator::Type Type>
486 bool hasOperator(const std::string& name) const
487 {
488 return hasOperatorOfType<SpatialOperator<ValueType>, Type>(name)
489 || hasOperatorOfType<TemporalOperator<ValueType>, Type>(name);
490 }
491
493 template<typename OperatorType, Operator::Type Type>
494 OperatorType& getOperator(const std::string& name)
495 {
496 if (!hasOperatorOfType<OperatorType, Type>(name))
497 {
498 throw std::runtime_error {"No operator with given name and type found"};
499 }
500 auto opType = Type;
501 auto matchNameAndType = [name, opType](const auto& op)
502 { return op.getName() == name && op.getType() == opType; };
503 if constexpr (std::is_same_v<OperatorType, SpatialOperator<ValueType>>)
504 {
505 return *std::ranges::find_if(spatialOperators_, matchNameAndType);
506 }
507 else if constexpr (std::is_same_v<OperatorType, TemporalOperator<ValueType>>)
508 {
509 return *std::ranges::find_if(temporalOperators_, matchNameAndType);
510 }
511 throw std::runtime_error {"Unknown operator type"};
512 // should never be reached, shut up compiler warning
513 return spatialOperators_[0];
514 }
515
517 template<Operator::Type Type>
518 void dropOperator(const std::string& name)
519 {
520 if (!hasOperator<Type>(name))
521 {
522 throw std::runtime_error {"No operator with given name and type found"};
523 }
524 auto opType = Type;
525 auto matchNameAndType = [name, opType](const auto& op)
526 { return op.getName() == name && op.getType() == opType; };
527 if (hasOperatorOfType<SpatialOperator<ValueType>, Type>(name))
528 {
529 std::erase_if(spatialOperators_, matchNameAndType);
530 }
531 else
532 {
533 std::erase_if(temporalOperators_, matchNameAndType);
534 }
535 }
