237template<typename ValueType>
238void communicateBoundaryData(
239 const CommunicationPattern& commPattern,
240 const std::vector<std::pair<localIdx, localIdx>> procPatchOffset,
241 const std::vector<int>& targetRanks,
242 Vector<ValueType>& boundaryData
243)
244{
245 auto mpiEnv = commPattern.env;
246 auto commRanks = mpiEnv.sizeRank();
247 auto sendSize = commPattern.sendCounts[commRanks];
248
249 // For each mesh-order patch, point sdispls[targetRank] at that patch's
250 // start offset in boundaryData. Ranks not in targetRanks keep the default
251 // 0 displacement (sendCounts[r] == 0 means MPI reads no bytes from there).
252 NF_ASSERT(
253 procPatchOffset.size() == targetRanks.size(),
254 "procPatchOffset and targetRanks must have the same length"
255 );
256 auto sdispls = std::vector<int>(commRanks, 0);
257 for (std::size_t p = 0; p < procPatchOffset.size(); ++p)
258 {
259 sdispls[targetRanks[p]] = static_cast<int>(procPatchOffset[p].first);
260 }
261
262 // MPI-01 fix: derive per-rank recv counts via MPI_Alltoall on send counts.
263 auto recvCountsVec = std::vector<int>(commRanks, 0);
264 {
265 auto sendCountsInt = std::vector<int>(commRanks, 0);
266 for (int r = 0; r < commRanks; ++r)
267 sendCountsInt[r] = static_cast<int>(commPattern.sendCounts[r]);
268 MPI_Alltoall(
269 sendCountsInt.data(), 1, MPI_INT, recvCountsVec.data(), 1, MPI_INT, mpiEnv.comm()
270 );
271 }
272 auto rdispls = std::vector<int>(commRanks, 0);
273 for (int r = 1; r < commRanks; ++r)
274 rdispls[r] = rdispls[r - 1] + recvCountsVec[r - 1];
275 int totalRecv = (commRanks > 0) ? rdispls.back() + recvCountsVec.back() : 0;
276
277 // MPI-03 fix: size recvBuffer to actual recv count, not full boundary size.
278 auto recvBuffer = Vector<ValueType>(boundaryData.exec(), static_cast<localIdx>(totalRecv));
279
280 // Flush any pending GPU kernels writing to boundaryData before MPI reads it.
281 // Kernels launched by per-BC correctBoundaryCondition() in the caller are async
282 // on the GPU executor; without this fence MPI would send pre-kernel device data.
283 fence(boundaryData.exec());
284
285 MPI_Alltoallv(
286 boundaryData.data(),
287 commPattern.sendCounts.data(),
288 sdispls.data(),
289 mpi::getType<ValueType>(),
290 recvBuffer.data(),
291 recvCountsVec.data(),
292 rdispls.data(),
293 mpi::getType<ValueType>(),
294 mpiEnv.comm()
295 );
296
297 auto exec = boundaryData.exec();
298 auto outV = boundaryData.view();
299 const auto inV = recvBuffer.view();
300
301 // Build a per-patch descriptor: [outStart, outEnd, inStart] so the unpack
302 // kernel knows both where to write (outStart..outEnd in boundaryData) and
303 // where to read (inStart in recvBuffer, which is rdispls[targetRank]).
304 // Mirror to a device-resident Vector — capturing host data in a GPU lambda
305 // dereferences host memory on device and trips Kokkos bounds checks.
306 auto offsetHost = std::vector<localIdx>(3 * procPatchOffset.size());
307 for (std::size_t p = 0; p < procPatchOffset.size(); p++)
308 {
309 offsetHost[3 * p] = procPatchOffset[p].first;
310 offsetHost[3 * p + 1] = procPatchOffset[p].second;
311 offsetHost[3 * p + 2] = static_cast<localIdx>(rdispls[targetRanks[p]]);
312 }
313 auto offsetVec = Vector<localIdx>(exec, offsetHost);
314 const auto offsetV = offsetVec.view();
315
316 parallelFor(
317 exec,
318 {0, procPatchOffset.size()},
319 NEON_LAMBDA(const localIdx p) {
320 const auto outStart = offsetV[3 * p];
321 const auto outEnd = offsetV[3 * p + 1];
322 const auto inStart = offsetV[3 * p + 2];
323 for (localIdx j = 0; j < outEnd - outStart; j++)
324 {
325 outV[outStart + j] = inV[inStart + j];
326 }
327 },
328 "copyMap"
329 );
330
331 // Ensure the unpack kernel completes before callers continue to use boundaryData.
332 fence(exec);
333}

335// Specialization for Vec3
336template<>
337inline void communicateBoundaryData(
338 const CommunicationPattern& commPattern,
339 const std::vector<std::pair<localIdx, localIdx>> procPatchOffset,
340 const std::vector<int>& targetRanks,
341 Vector<Vec3>& boundaryData
342)
343{
344 auto mpiEnv = commPattern.env;
345 auto commRanks = mpiEnv.sizeRank();
346 auto sendSize = commPattern.sendCounts[commRanks];
347
348 // compute send displacements. MPI_Alltoallv requires `int` count/displacement
349 // arrays — using std::vector<int> here matches the MPI signature regardless of
350 // how localIdx is configured at build time.
351 std::vector<int> sendCounts(commPattern.sendCounts.size(), 0);
352 for (int i = 0; i < sendCounts.size(); i++)
353 {
354 sendCounts[i] = commPattern.sendCounts[i] * 3;
355 }
356
357 // MPI-01 fix: derive per-rank recv counts from the 3x-multiplied send counts.
358 auto recvCountsVec = std::vector<int>(commRanks, 0);
359 MPI_Alltoall(sendCounts.data(), 1, MPI_INT, recvCountsVec.data(), 1, MPI_INT, mpiEnv.comm());
360 auto rdispls = std::vector<int>(commRanks, 0);
361 for (int r = 1; r < commRanks; ++r)
362 rdispls[r] = rdispls[r - 1] + recvCountsVec[r - 1];
363 int totalRecv3 = (commRanks > 0) ? rdispls.back() + recvCountsVec.back() : 0;
364
365 // Per-rank displacement using the mesh-order target rank for each patch.
366 // See the scalar overload above for rationale.
367 NF_ASSERT(
368 procPatchOffset.size() == targetRanks.size(),
369 "procPatchOffset and targetRanks must have the same length"
370 );
371 auto sdispls = std::vector<int>(commRanks, 0);
372 for (std::size_t p = 0; p < procPatchOffset.size(); ++p)
373 {
374 sdispls[targetRanks[p]] = 3 * static_cast<int>(procPatchOffset[p].first);
375 }
376
377 auto exec = boundaryData.exec();
378 auto boundaryDataSize = boundaryData.size();
379 auto sendBuffer = Vector<NeoN::scalar>(boundaryData.exec(), 3 * boundaryData.size());
380 auto sendBufferV = sendBuffer.view();
381 auto boundaryDataV = boundaryData.view();
382
383 parallelFor(
384 exec,
385 {0, boundaryData.size()},
386 NEON_LAMBDA(const localIdx i) {
387 sendBufferV[3 * i + 0] = boundaryDataV[i][0];
388 sendBufferV[3 * i + 1] = boundaryDataV[i][1];
389 sendBufferV[3 * i + 2] = boundaryDataV[i][2];
390 },
391 "copyMap"
392 );
393
394 // MPI-03 fix: size to actual recv scalar count.
395 auto recvBuffer = Vector<NeoN::scalar>(boundaryData.exec(), static_cast<localIdx>(totalRecv3));
396 auto recvBufferV = recvBuffer.view();
397
398 // Flush the pack kernel above (and any caller-launched BC kernels) before MPI
399 // reads sendBuffer. parallelFor is async on GPU.
400 fence(exec);
401
402 MPI_Alltoallv(
403 sendBuffer.data(),
404 sendCounts.data(),
405 sdispls.data(),
406 mpi::getType<scalar>(),
407 recvBuffer.data(),
408 recvCountsVec.data(),
409 rdispls.data(),
410 mpi::getType<scalar>(),
411 mpiEnv.comm()
412 );
413
414 const auto inV = recvBuffer.view();
415 auto outV = boundaryData.view();
416
417 // Build a per-patch descriptor: [outStart, outEnd, inStart (in scalar units)]
418 // mirroring rdispls[targetRank] so the unpack kernel reads from the correct
419 // position in the compacted recvBuffer. Mirror to device — see scalar overload.
420 auto offsetHost = std::vector<localIdx>(3 * procPatchOffset.size());
421 for (std::size_t p = 0; p < procPatchOffset.size(); p++)
422 {
423 offsetHost[3 * p] = procPatchOffset[p].first;
424 offsetHost[3 * p + 1] = procPatchOffset[p].second;
425 offsetHost[3 * p + 2] = static_cast<localIdx>(rdispls[targetRanks[p]]);
426 }
427 auto offsetVec = Vector<localIdx>(exec, offsetHost);
428 const auto offsetV = offsetVec.view();
429
430 parallelFor(
431 exec,
432 {0, procPatchOffset.size()},
433 NEON_LAMBDA(const localIdx p) {
434 const auto outStart = offsetV[3 * p];
435 const auto outEnd = offsetV[3 * p + 1];
436 const auto inStart = offsetV[3 * p + 2]; // scalar offset in recvBuffer
437 for (localIdx j = 0; j < outEnd - outStart; j++)
438 {
439 outV[outStart + j][0] = inV[inStart + 3 * j + 0];
440 outV[outStart + j][1] = inV[inStart + 3 * j + 1];
441 outV[outStart + j][2] = inV[inStart + 3 * j + 2];
442 }
443 },
444 "copyMap"
445 );
446
447 // Ensure the unpack kernel completes before callers continue to use boundaryData.
448 fence(exec);
449}
