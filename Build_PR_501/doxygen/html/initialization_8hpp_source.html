18inline void initialize(int argc, char* argv[])
19{
20 mpi::Environment mpiEnv;
21
22 // Bind each MPI rank to a distinct GPU on multi-GPU nodes.
23 //
24 // Without explicit binding every rank defaults to device 0, leaving all
25 // other GPUs idle and causing CUDA/HIP context contention on GPU 0.
26 // We use a shared-memory sub-communicator (MPI_COMM_TYPE_SHARED) to obtain
27 // the local rank within the node, then map:
28 // device_id = local_rank % num_devices
29 //
30 // Skipped when:
31 // - MPI is not initialised (serial run or pre-MPI test harness), OR
32 // - only one device is visible (single-GPU node or CPU-only build) —
33 // Kokkos::num_devices() returns 0 for CPU-only backends.
34 // - KOKKOS_DEVICE_ID or --kokkos-device-id is already set by the user,
35 // in which case Kokkos::initialize(argc, argv) below will honour it.
36#if defined(NF_WITH_MPI_SUPPORT)
37 if (mpiEnv.isInitialized() && Kokkos::num_devices() > 1
38 && std::getenv("KOKKOS_DEVICE_ID") == nullptr)
39 {
40 MPI_Comm nodeComm;
41 MPI_Comm_split_type(
42 mpiEnv.comm(),
43 MPI_COMM_TYPE_SHARED,
44 static_cast<int>(mpiEnv.rank()),
45 MPI_INFO_NULL,
46 &nodeComm
47 );
48 int localRank = 0;
49 MPI_Comm_rank(nodeComm, &localRank);
50 MPI_Comm_free(&nodeComm);
51
52 Kokkos::InitializationSettings settings;
53 settings.set_device_id(localRank % Kokkos::num_devices());
54 Kokkos::initialize(settings);
55 }
56 else
57#endif
58 {
59 Kokkos::initialize(argc, argv);
60 }
61
62 cpptrace::register_terminate_handler();
63 Logging::setNeonDefaultPattern(mpiEnv);
64}
