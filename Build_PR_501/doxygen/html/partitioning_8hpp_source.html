14template<typename FieldType, typename MeshType, typename BcType>
15FieldType partitionVolField(
16 FieldType field, const MeshType& mesh, BcType bcs, NeoN::mpi::Environment mpiEnviron
17)
18{
19 localIdx localCells = mesh.nCells();
20 localIdx firstCell = 0;
21 localIdx lastCell = localCells;
22
23 if (mpiEnviron.rank() == 0)
24 {
25 lastCell = localCells;
26 }
27 if (mpiEnviron.rank() >= 1 && mpiEnviron.rank() != mpiEnviron.sizeRank() - 1)
28 {
29 firstCell = localCells;
30 lastCell = localCells + localCells;
31 }
32
33 if (mpiEnviron.rank() == mpiEnviron.sizeRank() - 1)
34 {
35 firstCell = localCells + localCells;
36 lastCell = localCells + localCells + localCells;
37 }
38
39 auto internalVector = take(field.internalVector(), firstCell, lastCell);
40
41 return {field.exec(), field.name + "Part", mesh, internalVector, bcs};
42}

105template<typename FieldType, typename MeshType, typename BcType>
106FieldType partitionSurfaceField(
107 FieldType field,
108 MeshType& mesh,
109 BcType bcs,
110 NeoN::mpi::Environment mpiEnviron,
111 bool flip = false
112)
113{
114 auto exec = mesh.exec();
115 localIdx localCells = mesh.nCells();
116 localIdx localFaces = mesh.nInternalFaces();
117 localIdx firstFace = 0;
118 localIdx lastFace = localFaces;
119
120 localIdx firstBoundaryFace = 0;
121 localIdx secondBoundaryFace = 0;
122
123 scalar signLeft = 1.0;
124 scalar signRight = 1.0;
125
126 // N-rank generic formulas (replaces the rank==0/1/2 if/else chain).
127 // Derivation and R=2/3/4 hand-verification: see
128 // .planning/phases/02-linear-system-correctness/02-02-SUMMARY.md
129 // .planning/phases/02-linear-system-correctness/02-RESEARCH.md (NRANK-01 section).
130 // 3-rank oracle (src/NeoN/test/distributed/partitioning.cpp:218-230) preserved.
131 const localIdx r = mpiEnviron.rank();
132 const localIdx R = mpiEnviron.sizeRank();
133
134 firstFace = r * (localFaces + 1);
135
136 if (r == 0)
137 {
138 // First rank: left boundary = global left domain boundary; right boundary = right
139 // proc-face.
140 firstBoundaryFace = R * (localFaces + 1) - 1;
141 secondBoundaryFace = localFaces;
142 if (flip)
143 {
144 signRight = -1.0;
145 }
146 }
147 else if (r == R - 1)
148 {
149 // Last rank: left boundary = left proc-face; right boundary = global right domain boundary.
150 firstBoundaryFace = R * (localFaces + 1);
151 secondBoundaryFace = (R - 1) * (localFaces + 1) - 1;
152 if (flip)
153 {
154 signRight = -1.0;
155 }
156 }
157 else
158 {
159 // Middle rank: both boundaries are proc-faces.
160 firstBoundaryFace = r * (localFaces + 1) - 1;
161 secondBoundaryFace = r * (localFaces + 1) + localFaces;
162 if (flip)
163 {
164 signLeft = -1.0;
165 }
166 }
167
168 lastFace = firstFace + localFaces + 2; // two extra faces for boundary + 1
169
170 // surface field contains all faces
171 FieldType ret = {field.exec(), field.name + "Part", mesh, bcs};
172
173 NF_ASSERT(lastFace - firstFace == mesh.nTotalFaces(), "different size");
174
175 // NOTE last two values are boundaries and are overwritten next
176 auto internalVector = take(field.internalVector(), firstFace, lastFace);
177 // value lastFace and lastFace+1 are incorrect
178 // value lastFace is the left boundary
179 // and lastFace + 1 is at the new boundary so it should be lastFace
180
181 auto outV = internalVector.view();
182 auto inV = field.internalVector().view();
183
184 // set first boundary face
185 NeoN::parallelFor(
186 // lastface
187 exec,
188 {0, 1},
189 NEON_LAMBDA(const localIdx i) { outV[localFaces] = signLeft * inV[firstBoundaryFace]; },
190 "copyMap"
191 );
192
193 // set second boundary face
194 NeoN::parallelFor(
195 exec,
196 {0, 1},
197 NEON_LAMBDA(const localIdx i) {
198 outV[localFaces + 1] = signRight * inV[secondBoundaryFace];
199 },
200 "copyMap"
201 );
202
203
204 NF_ASSERT(ret.internalVector().size() == internalVector.size(), "different size");
205 ret.internalVector() = internalVector;
206 return ret;
207}
