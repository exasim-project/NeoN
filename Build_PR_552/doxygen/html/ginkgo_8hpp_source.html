162inline std::optional<L1ResidualControl> readL1ResidualControl(const Dictionary& cfg)
163{
164 const std::string flag = "l1ScaledResidual";
165 if (!cfg.contains(flag))
166 {
167 return std::nullopt;
168 }
169 // A boolean read from a dictionary file is stored as a word/string, not a bool;
170 // accept the common representations rather than assuming a single type.
171 bool enabled = false;
172 if (cfg.isType<bool>(flag))
173 {
174 enabled = cfg.get<bool>(flag);
175 }
176 else if (cfg.isType<int>(flag))
177 {
178 enabled = cfg.get<int>(flag) != 0;
179 }
180 else if (cfg.isType<std::string>(flag))
181 {
182 const std::string v = cfg.get<std::string>(flag);
183 enabled = (v == "true" || v == "yes" || v == "on" || v == "1");
184 }
185 if (!enabled)
186 {
187 return std::nullopt;
188 }
189
190 L1ResidualControl control {0.0, 0.0, 1000, 0, 1};
191
192 // criteria entries may be stored as int, label or scalar depending on source
193 auto readScalar = [](const Dictionary& d, const std::string& key, scalar fallback)
194 {
195 if (!d.contains(key)) return fallback;
196 if (d.isType<int>(key)) return scalar(d.get<int>(key));
197 if (d.isType<label>(key)) return scalar(d.get<label>(key));
198 return d.get<scalar>(key);
199 };
200 auto readInt = [](const Dictionary& d, const std::string& key, localIdx fallback)
201 {
202 if (!d.contains(key)) return fallback;
203 if (d.isType<int>(key)) return localIdx(d.get<int>(key));
204 if (d.isType<scalar>(key)) return localIdx(d.get<scalar>(key));
205 return d.get<localIdx>(key);
206 };
207
208 if (cfg.contains("criteria"))
209 {
210 const Dictionary& criteria = cfg.subDict("criteria");
211 control.tolerance = readScalar(criteria, "absolute_residual_norm", control.tolerance);
212 control.relTol = readScalar(criteria, "initial_residual_norm", control.relTol);
213 control.maxIter = readInt(criteria, "iteration", control.maxIter);
214 }
215 control.minIter = readInt(cfg, "minIter", control.minIter);
216 control.checkFrequency = readInt(cfg, "checkFrequency", control.checkFrequency);
217
218 return control;
219}
