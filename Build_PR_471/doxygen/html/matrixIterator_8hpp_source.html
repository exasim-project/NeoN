121 // Function to pre-compute cell-based connectivity
122 std::shared_ptr<CellBasedData> computeCellBasedData(
123 const UnstructuredMesh& mesh,
124 std::shared_ptr<const la::CsrSparsityPattern<localIdx>> sparsityPattern,
125 const std::shared_ptr<const la::FaceToMatrixAddress> faceToMatrixAddress
126 )
127 {
128 // Count faces per cell (owner and neighbour contributions)
129 auto exec = mesh.exec();
130 auto cellToFaceStencil = finiteVolume::cellCentred::CellToFaceStencil(mesh);
131 auto cellFaces = cellToFaceStencil.computeInternalStencil();
132 auto owner = mesh.faceOwners();
133 auto neighbour = mesh.faceNeighbors();
134
135 // Pre-compute face neighbour, sign, and matrix column indices
136 const auto totalFaceConnections = cellFaces.size();
137 Vector<localIdx> faceNeighbour(exec, totalFaceConnections);
138 Vector<scalar> faceSign(exec, totalFaceConnections);
139 Vector<localIdx> matrixColumnIdx(exec, totalFaceConnections);
140
141 const auto [diagOffs, ownOffs, neiOffs] = views(
142 faceToMatrixAddress->diagOffset(),
143 faceToMatrixAddress->ownerOffset(),
144 faceToMatrixAddress->neighbourOffset()
145 );
146
147 const auto [ownerV, neighbourV] = views(owner, neighbour);
148 const auto [matrixRowOffs] = views(sparsityPattern->rowOffs());
149
150 auto [faceNeighbourV, faceSignV, matrixColumnIdxV] =
151 views(faceNeighbour, faceSign, matrixColumnIdx);
152
153 auto [cellFacesValues, cellFacesSegments] = cellFaces.views();
154
155 parallelFor(
156 exec,
157 {0, mesh.nCells()},
158 NEON_LAMBDA(const localIdx celli) {
159 const auto faces = cellFacesSegments[celli + 1] - cellFacesSegments[celli];
160 const auto startIdx = cellFacesSegments[celli];
161
162 for (localIdx i = 0; i < faces; ++i)
163 {
164 const auto faceIdx = cellFacesValues[startIdx + i];
165 const auto own = ownerV[faceIdx];
166 const auto nei = neighbourV[faceIdx];
167
168 const auto isOwner = (own == celli);
169 const auto neiCell = isOwner ? nei : own;
170
171 faceNeighbourV[startIdx + i] = neiCell;
172 faceSignV[startIdx + i] = isOwner ? 1.0 : -1.0;
173
174 // Compute matrix column index
175 const auto rowStart = matrixRowOffs[celli];
176 if (isOwner)
177 {
178 matrixColumnIdxV[startIdx + i] = rowStart + ownOffs[faceIdx];
179 }
180 else
181 {
182 matrixColumnIdxV[startIdx + i] = rowStart + neiOffs[faceIdx];
183 }
184 }
185 },
186 "computeFaceData"
187 );
188
189 return std::make_shared<CellBasedData>(
190 mesh.nCells(), cellFaces, faceNeighbour, faceSign, matrixColumnIdx
191 );
192 }
193
194 std::shared_ptr<CellBasedData> cellBasedIteratorData_;
195};
