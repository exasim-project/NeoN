105 template<typename valueType>
106 void startComm(Vector<valueType>& field, const std::string& commName)
107 {
108 NF_DEBUG_ASSERT(
109 !CommBuffer_.contains(commName) || (!CommBuffer_[commName]),
110 "There is already an ongoing communication for key " << commName << "."
111 );
112
113 CommBuffer_[commName] = findDuplexBuffer();
114 if (!CommBuffer_[commName])
115 {
116 CommBuffer_[commName] = createNewDuplexBuffer();
117 }
118
119 CommBuffer_[commName]->initComm<valueType>(commName);
120 for (size_t rank = 0; rank < mpiEnviron_.sizeRank(); ++rank)
121 {
122 auto rankBuffer = CommBuffer_[commName]->getSend<valueType>(rank);
123 for (size_t data = 0; data < sendMap_[rank].size(); ++data)
124 rankBuffer[data] =
125 field.view()[static_cast<localIdx>(sendMap_[rank][data].local_idx)];
126 }
127 CommBuffer_[commName]->startComm();
128 }

143 template<typename valueType>
144 void finaliseComm(Vector<valueType>& field, std::string commName)
145 {
146 NF_DEBUG_ASSERT(
147 CommBuffer_.contains(commName) && CommBuffer_[commName],
148 "No communication associated with key: " << commName
149 );
150
151 CommBuffer_[commName]->waitComplete();
152 for (size_t rank = 0; rank < mpiEnviron_.sizeRank(); ++rank)
153 {
154 auto rankBuffer = CommBuffer_[commName]->getReceive<valueType>(rank);
155 for (size_t data = 0; data < receiveMap_[rank].size(); ++data)
156 field.view()[static_cast<localIdx>(receiveMap_[rank][data].local_idx)] =
157 rankBuffer[data];
158 }
159 CommBuffer_[commName]->finaliseComm();
160 CommBuffer_[commName] = nullptr;
161 }
