173 void communicate(CommunicationPattern& commPattern)
174 {
175 auto mpiEnv = commPattern.env;
176 int commRanks = mpiEnv.sizeRank();
177
178 auto boundaryMatrixMap = Vector<localIdx>(exec(), commPattern.boundaryMapVector);
179
180 // 1. copy bValues which need to be communicated into sendBuffer
181 auto commSize = commPattern.sendCounts[mpiEnv.sizeRank()];
182 auto commBuffer = Vector<ValueType>(exec(), commSize);
183 auto recvBuffer = Vector<ValueType>(exec(), commSize);
184
185 NF_PING();
186 // TODO with the right displacements boundary values could be taken
187 // directly without copy to a sendbuffer first. however the recv is still needed
188 //
189 // copy map needs to be on the device for filling the consecutive
190 // sendBuffer but for sending with mpi data is on the host
191 auto copyMap = Vector<localIdx>(exec(), commPattern.commIdx);
192 std::cout << __FILE__ << ":" << __LINE__ << " commSize " << commSize << " copyMap.size "
193 << copyMap.size() << "\n";
194 copy(boundaryMatrix_.values(), copyMap, commBuffer);
195 NF_PING();
196
197 // zero out processor boundary values
198 // set(0.0, copyMap, boundaryMatrix_.values());
199
200 // TODO compute using scan
201 auto sdispls = std::vector<int>(commRanks, 0);
202 for (int i = 1; i < sdispls.size(); i++)
203 {
204 auto prev = sdispls[i - 1];
205 sdispls[i] = commPattern.sendCounts[i - 1] + prev;
206 }
207
208 MPI_Alltoallv(
209 commBuffer.data(),
210 commPattern.sendCounts.data(),
211 sdispls.data(),
212 mpi::getType<ValueType>(),
213 recvBuffer.data(),
214 commPattern.sendCounts.data(),
215 sdispls.data(),
216 mpi::getType<ValueType>(),
217 mpiEnv.comm()
218 );
219
220 // 3. apply received values to corresponding matrix
221 // add diagonal contributions
222 sub(recvBuffer, boundaryMatrixMap, matrix_.values());
223 }
