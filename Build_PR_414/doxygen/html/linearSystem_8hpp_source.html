195 void communicate(CommunicationPattern& commPattern)
196 {
197 auto mpiEnv = commPattern.env;
198 int commRanks = mpiEnv.sizeRank();
199
200 auto boundaryMatrixMap = Vector<localIdx>(exec(), commPattern.boundaryMapVector);
201
202 // 1. copy bValues which need to be communicated into sendBuffer
203 auto commSize = commPattern.sendCounts[mpiEnv.sizeRank()];
204 auto recvBuffer = Vector<ValueType>(exec(), commSize);
205
206 // TODO compute using scan
207 auto sdispls = std::vector<int>(commRanks, 0);
208 for (int i = 1; i < sdispls.size(); i++)
209 {
210 auto prev = sdispls[i - 1];
211 sdispls[i] = commPattern.sendCounts[i - 1] + prev;
212 }
213
214 MPI_Alltoallv(
215 nonLocalMatrix_.values().data(),
216 commPattern.sendCounts.data(),
217 sdispls.data(),
218 mpi::getType<ValueType>(),
219 recvBuffer.data(),
220 commPattern.sendCounts.data(),
221 sdispls.data(),
222 mpi::getType<ValueType>(),
223 mpiEnv.comm()
224 );
225
226 // 3. apply received values to corresponding matrix
227 // add diagonal contributions
228 // sub(recvBuffer, boundaryMatrixMap, matrix_.values());
229 }
