181 void communicate(CommunicationPattern& commPattern)
182 {
183 auto mpiEnv = commPattern.env;
184 int commRanks = mpiEnv.sizeRank();
185
186 // auto boundaryMatrixMap = Vector<localIdx>(exec(), commPattern.boundaryMapVector);
187 auto ftma = matrix_.faceToMatrixAddress();
188 auto nsp = ftma->nonLocalSparsityPattern();
189 auto rowToDiagonalMap = la::computeRowToDiagonalMap(nsp->rowOffs(), ftma);
190
191 // 1. copy bValues which need to be communicated into sendBuffer
192 auto commSize = commPattern.sendCounts[mpiEnv.sizeRank()];
193 auto recvBuffer = Vector<ValueType>(exec(), commSize);
194
195 // TODO compute using scan
196 auto sdispls = std::vector<int>(commRanks, 0);
197 for (int i = 1; i < sdispls.size(); i++)
198 {
199 auto prev = sdispls[i - 1];
200 sdispls[i] = commPattern.sendCounts[i - 1] + prev;
201 }
202
203 MPI_Alltoallv(
204 nonLocalMatrix_.values().data(),
205 commPattern.sendCounts.data(),
206 sdispls.data(),
207 mpi::getType<ValueType>(),
208 recvBuffer.data(),
209 commPattern.sendCounts.data(),
210 sdispls.data(),
211 mpi::getType<ValueType>(),
212 mpiEnv.comm()
213 );
214
215 std::cout << __FILE__ << ":" << __LINE__ << " rank " << mpiEnv.rank() << " recvBuffer "
216 << recvBuffer.view()[0] << " matrixValue " << matrix_.values().view()[9] << "\n";
217
218 // 3. apply received values to corresponding matrix
219 // add diagonal contributions
220 add(recvBuffer, rowToDiagonalMap, matrix_.values());
221 std::cout << __FILE__ << ":" << __LINE__ << " rank " << mpiEnv.rank() << " recvBuffer "
222 << recvBuffer.view()[0] << " matrixValue " << matrix_.values().view()[9] << "\n";
223 }

315/*@brief helper function that creates a zero initialised linear system based on a given mesh
316 */
317template<
318 typename ValueType,
319 typename SystemMatrixType = CSRMatrix<ValueType, localIdx>,
320 typename BoundaryMatrixType = COOMatrix<ValueType, localIdx>>
321LinearSystem<ValueType, SystemMatrixType, BoundaryMatrixType> createEmptyLinearSystem(
322 const UnstructuredMesh& mesh,
323 std::shared_ptr<MeshIterationStrategy> strategy = std::make_shared<FaceBasedIterator>()
324)
325{
326 auto [sp, mi] =
327 createSparsityPatternFaceToMatrixAddress<typename SystemMatrixType::MatrixSparsityType>(mesh
328 );
329 auto bSp =
330 createBoundarySparsityPattern<typename BoundaryMatrixType::MatrixSparsityType>(mesh, *mi);
331 const auto exec = sp->exec();
332 const auto nCells = static_cast<localIdx>(mesh.nCells());
333 const auto nProcFaces = static_cast<localIdx>(mesh.nProcBoundaryFaces());
334 using IndexType = typename BoundaryMatrixType::MatrixSparsityType::SparsityIndexType;
335
336 Vector<IndexType> offDiagColIdxs(exec, nProcFaces, 0);
337 Vector<IndexType> offDiagRowIdxs(exec, nProcFaces, 0);
338
339#ifdef NF_WITH_MPI_SUPPORT
340 auto commPattern = computeCommunicationPattern(mesh);
341 if (nProcFaces > 0)
342 {
343 const localIdx nBoundaryFaces = static_cast<localIdx>(mesh.nBoundaryFaces());
344 const auto faceOwnersH = mesh.boundaryMesh().faceOwners().copyToHost();
345 const auto faceOwnersV = faceOwnersH.view();
346 Vector<IndexType> rowH(SerialExecutor {}, nProcFaces, 0);
347 Vector<IndexType> colH(SerialExecutor {}, nProcFaces, 0);
348 auto rowHV = rowH.view();
349 auto colHV = colH.view();
350 const auto globalOffset = static_cast<IndexType>(mesh.globalOffset());
351 for (localIdx i = 0; i < nProcFaces; ++i)
352 {
353 rowHV[i] = faceOwnersV[nBoundaryFaces + i] + globalOffset;
354 colHV[i] = static_cast<IndexType>(commPattern.recvIdx[static_cast<std::size_t>(i)]);
355 }
356 offDiagRowIdxs = rowH.copyToExecutor(exec);
357 offDiagColIdxs = colH.copyToExecutor(exec);
358 }
359#endif
360
361 auto offDiagSp = std::make_shared<const typename BoundaryMatrixType::MatrixSparsityType>(
362 std::move(offDiagColIdxs), std::move(offDiagRowIdxs), Dimensions {nCells, nCells}
363 );
364}
365
366 LinearSystem<ValueType, SystemMatrixType, BoundaryMatrixType> ls {
367 SystemMatrixType(Vector<ValueType>(sp->exec(), sp->nnz(), zero<ValueType>()), sp, mi),
368 Vector<ValueType>(sp->exec(), sp->rows(), zero<ValueType>()),
369 BoundaryMatrixType(Vector<ValueType>(exec, nProcFaces, zero<ValueType>()), offDiagSp),
370 BoundaryMatrixType(Vector<ValueType>(bSp->exec(), bSp->nnz(), zero<ValueType>()), bSp),
371 Vector<ValueType>(bSp->exec(), bSp->nnz(), zero<ValueType>()),
372 strategy
373 };
374
375#ifdef NF_WITH_MPI_SUPPORT
376 ls.commPattern() = std::move(commPattern);
377#endif
378
379 return ls;
380}

384template<typename ValueType>
385inline la::LinearSystem<ValueType>
386removeBoundaryContributions(const la::LinearSystem<ValueType>& lsIn)
387{
388 auto ls = la::LinearSystem<ValueType>(lsIn);
389 auto lsView = ls.view();
390 auto& matrix = lsView.matrix;
391 auto& rhs = lsView.rhs;
392 auto& bMatrix = lsView.boundaryMatrix;
393 auto& bRhs = lsView.boundaryRhs;
394
395 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
396
397 // Boundary matrix
398 parallelFor(
399 ls.exec(),
400 {0, bMatrix.values.size()},
401 NEON_LAMBDA(const localIdx facei) {
402 const auto celli = bMatrix.sparsity.rowOffs[facei]; // cell index stored in rowOffs
403 Kokkos::atomic_add(&matrix.values[ma.diagIdx(celli)], bMatrix.values[facei]);
404 Kokkos::atomic_add(&rhs[celli], bRhs[facei]);
405 },
406 "removeBoundaryContributions"
407 );
408
409 return ls;
410}

412inline void scaledInvDiagNegLUx(
413 const la::LinearSystem<Vec3>& ls, // In,
414 const Vector<Vec3>& a,
415 const Vector<Vec3>& aBound,
416 const UnstructuredMesh& mesh,
417 Vector<scalar>& rAU,
418 Vector<Vec3>& out
419)
420{
421 // auto ls = removeBoundaryContributions(lsIn);
422 auto& mtx = ls.matrix();
423
424 auto& vol = mesh.cellVolumes();
425 auto& bMesh = mesh.boundaryMesh();
426 NF_ASSERT(mtx.nRows() == a.size(), "Dimension mismatch");
427 NF_ASSERT(mtx.nRows() == out.size(), "Dimension mismatch");
428
429 const auto [rowOffsV, colIdxV, matrixV, rAUV, volV, aV, aBoundV, bV] = views(
430 mtx.sparsity()->rowOffs(),
431 ls.matrix().sparsity()->colIdxs(),
432 mtx.values(),
433 rAU,
434 vol,
435 a,
436 aBound,
437 ls.rhs()
438 );
439 const auto [nonLocalMtx, nonLocalRows] =
440 views(ls.nonLocalMatrix().values(), ls.nonLocalMatrix().rowOffs());
441 auto outV = out.view();
442
443 auto procFacesStart = bMesh.nBoundaryFaces();
444
445 // localIdx curNonLocalRow = 0;
446 parallelFor(
447 mtx.exec(),
448 {0, mtx.nRows()},
449 NEON_LAMBDA(const localIdx rowi) {
450 // local mtx first
451 outV[rowi] = zero<Vec3>();
452 for (auto i = rowOffsV[rowi]; i < rowOffsV[rowi + 1]; i++)
453 {
454 auto colI = colIdxV[i];
455 if (rowi == colI)
456 {
457 rAUV[rowi] = volV[rowi] / matrixV[i][0];
458 }
459 else
460 {
461 outV[rowi] -= matrixV[i] * aV[colI];
462 }
463 }
464
465 // check
466 // FIXME this scans everytime all boundary values
467 localIdx curNonLocalRow = 0;
468 auto nonLocalVal = zero<Vec3>();
469 for (auto i = 0; i < nonLocalRows.size(); i++)
470 {
471 if (nonLocalRows[i] == rowi)
472 {
473 nonLocalVal = nonLocalMtx[i];
474 curNonLocalRow = i;
475 // FIXME is the sign correct?
476 outV[rowi] -= nonLocalVal * aBoundV[procFacesStart + curNonLocalRow];
477 // curNonLocalRow = 0;
478 // break;
479 }
480 }
481
482 outV[rowi] += bV[rowi];
483 outV[rowi] *= rAUV[rowi] / volV[rowi];
484 }
485 );
486}
