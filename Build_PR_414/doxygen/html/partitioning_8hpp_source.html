14template<typename FieldType, typename MeshType, typename BcType>
15FieldType partitionVolField(
16 FieldType field, const MeshType& mesh, BcType bcs, NeoN::mpi::Environment mpiEnviron
17)
18{
19 localIdx localCells = mesh.nCells();
20 localIdx firstCell = 0;
21 localIdx lastCell = localCells;
22
23 if (mpiEnviron.rank() == 0)
24 {
25 lastCell = localCells;
26 }
27 if (mpiEnviron.rank() >= 1 && mpiEnviron.rank() != mpiEnviron.sizeRank() - 1)
28 {
29 firstCell = localCells;
30 lastCell = localCells + localCells;
31 }
32
33 if (mpiEnviron.rank() == mpiEnviron.sizeRank() - 1)
34 {
35 firstCell = localCells + localCells;
36 lastCell = localCells + localCells + localCells;
37 }
38
39 auto internalVector = take(field.internalVector(), {firstCell, lastCell});
40
41 return {field.exec(), field.name + "Part", mesh, internalVector, bcs};
42}

46template<typename FieldType, typename MeshType, typename BcType>
47FieldType partitionSurfaceField(
48 FieldType field,
49 MeshType& mesh,
50 BcType bcs,
51 NeoN::mpi::Environment mpiEnviron,
52 bool flip = false
53)
54{
55 auto exec = mesh.exec();
56 localIdx localCells = mesh.nCells(); // 4
57 localIdx localFaces = mesh.nInternalFaces(); // 3
58 localIdx firstFace = 0;
59 localIdx lastFace = localFaces;
60
61 //
62 localIdx firstBoundaryFace = 0;
63 localIdx secondBoundaryFace = 0;
64
65 scalar signLeft = 1.0;
66 scalar signRight = 1.0;
67
68 // FIXME this only works for 3 ranks
69 if (mpiEnviron.rank() == 0)
70 {
71 // first boundary face is the left boundary remaining local
72 firstBoundaryFace = 3 * localFaces + 2; // 11
73 secondBoundaryFace = localFaces; // 3
74
75 if (flip)
76 {
77 // signLeft = -1.0;
78 signRight = -1.0;
79 }
80 }
81 if (mpiEnviron.rank() == 1)
82 {
83 firstFace = localFaces + 1; // 4 last face rank 0
84
85 firstBoundaryFace = localFaces; // should be 3
86 secondBoundaryFace = firstBoundaryFace + localCells; // should 3 + 4
87
88 // new face has different direction compared to unpartitioned case
89 // signRight = -1.0;
90 if (flip)
91 {
92 signLeft = -1.0;
93 // signRight = -1.0;
94 }
95 }
96 if (mpiEnviron.rank() == 2)
97 {
98 firstFace = localCells + localCells; // 8 last face rank 1
99
100 // first boundary face is the right boundary remaining local
101 // firstBoundaryFace = 2 * localFaces + 1; // 7
102 firstBoundaryFace = 3 * localFaces + 3; // 11
103 secondBoundaryFace = 2 * localFaces + 1; // firstBoundaryFace + localCells + 1; // 12
104
105 // NOTE FIXME signRight is flipped since
106 // new face has different direction compared to unpartitioned case
107 if (flip)
108 {
109 // signLeft = -1.0;
110 signRight = -1.0;
111 }
112 }
113
114 lastFace = firstFace + localFaces + 2; // two extra faces for boundary + 1
115
116 // surface field contains all faces
117 FieldType ret = {field.exec(), field.name + "Part", mesh, bcs};
118
119 NF_ASSERT(lastFace - firstFace == mesh.nTotalFaces(), "different size");
120
121 // NOTE last two values are boundaries and are overwritten next
122 auto internalVector = take(field.internalVector(), {firstFace, lastFace});
123 // value lastFace and lastFace+1 are incorrect
124 // value lastFace is the left boundary
125 // and lastFace + 1 is at the new boundary so it should be lastFace
126
127 auto outV = internalVector.view();
128 auto inV = field.internalVector().view();
129 auto inBV = field.boundaryData().value().view();
130 const localIdx nGlobalInternalFaces = static_cast<localIdx>(field.internalVector().size());
131
132 // firstBoundaryFace may point beyond internalVector into the global boundary data
133 const bool firstIsPhysical = firstBoundaryFace >= nGlobalInternalFaces;
134 const localIdx firstBoundaryPatchIdx = firstBoundaryFace - nGlobalInternalFaces;
135
136 // set first boundary face
137 NeoN::parallelFor(
138 exec,
139 {0, 1},
140 NEON_LAMBDA(const localIdx i) {
141 outV[localFaces] =
142 signLeft * (firstIsPhysical ? inBV[firstBoundaryPatchIdx] : inV[firstBoundaryFace]);
143 },
144 "copyMap"
145 );
146
147 // set second boundary face (always an internal face in the global mesh)
148 NeoN::parallelFor(
149 exec,
150 {0, 1},
151 NEON_LAMBDA(const localIdx i) {
152 outV[localFaces + 1] = signRight * inV[secondBoundaryFace];
153 },
154 "copyMap"
155 );
156
157
158 ret.internalVector() = internalVector;
159
160 // Copy boundary face values (regular + proc) into boundaryData
161 const auto nif = mesh.nInternalFaces();
162 const auto totalBoundary = mesh.nBoundaryFaces() + mesh.nProcBoundaryFaces();
163 auto bDataV = ret.boundaryData().value().view();
164 NeoN::parallelFor(
165 exec,
166 {0, totalBoundary},
167 NEON_LAMBDA(const localIdx bfi) { bDataV[bfi] = outV[nif + bfi]; },
168 "partitionSurfaceField::copyBoundaryData"
169 );
170
171 return ret;
172}
