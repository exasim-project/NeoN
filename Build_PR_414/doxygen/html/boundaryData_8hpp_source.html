225 void communicate(std::pair<localIdx, localIdx> range, int neighborRank)
226 {
227 const auto [rangeStart, rangeEnd] = range;
228 const localIdx patchSize = rangeEnd - rangeStart;
229
230 mpi::Environment mpiEnv;
231 CommBuffer buf;
232 buf.rangeStart = rangeStart;
233 buf.patchSize = patchSize;
234
235 const auto byteCount =
236 static_cast<mpi_label_t>(patchSize) * static_cast<mpi_label_t>(sizeof(ValueType));
237 const auto neighborRankLabel = static_cast<mpi_label_t>(neighborRank);
238
239 const bool useGpuPath = mpiEnv.gpuAwareMpi() && std::holds_alternative<GPUExecutor>(exec_);
240
241 MPI_Request sendReq, recvReq;
242 if (useGpuPath)
243 {
244 buf.deviceRecvBuf = Vector<ValueType>(exec_, patchSize, ValueType {});
245 mpi::isend<char>(
246 reinterpret_cast<const char*>(value_.data() + rangeStart),
247 byteCount,
248 neighborRankLabel,
249 0,
250 mpiEnv.comm(),
251 &sendReq
252 );
253 mpi::irecv<char>(
254 reinterpret_cast<char*>(buf.deviceRecvBuf->data()),
255 byteCount,
256 neighborRankLabel,
257 0,
258 mpiEnv.comm(),
259 &recvReq
260 );
261 }
262 else
263 {
264 auto valH = value_.copyToHost();
265 buf.sendBuf.resize(static_cast<std::size_t>(patchSize));
266 buf.recvBuf.resize(static_cast<std::size_t>(patchSize));
267 for (localIdx k = 0; k < patchSize; k++)
268 buf.sendBuf[static_cast<std::size_t>(k)] = valH.view()[rangeStart + k];
269 mpi::isend<char>(
270 reinterpret_cast<const char*>(buf.sendBuf.data()),
271 byteCount,
272 neighborRankLabel,
273 0,
274 mpiEnv.comm(),
275 &sendReq
276 );
277 mpi::irecv<char>(
278 reinterpret_cast<char*>(buf.recvBuf.data()),
279 byteCount,
280 neighborRankLabel,
281 0,
282 mpiEnv.comm(),
283 &recvReq
284 );
285 }
286 communicating_ = true;
287 requests_.push_back(sendReq);
288 requests_.push_back(recvReq);
289 commBuffers_.push_back(std::move(buf));
290 }

292 bool isComplete() const
293 {
294 if (requests_.empty() || !communicating_) return true;
295 for (auto& req : requests_)
296 {
297 if (!mpi::test(&req)) return false;
298 }
299 communicating_ = false;
300 return true;
301 }

306 void waitAll() const
307 {
308#ifdef NF_WITH_MPI_SUPPORT
309 if (requests_.empty() || !communicating_) return;
310 while (!isComplete())
311 {
312 }
313 mpi::Environment mpiEnv;
314 const bool useGpuPath = mpiEnv.gpuAwareMpi() && std::holds_alternative<GPUExecutor>(exec_);
315 if (useGpuPath)
316 {
317 for (const auto& buf : commBuffers_)
318 {
319 auto srcView = buf.deviceRecvBuf->view();
320 auto dstView = value_.view();
321 const localIdx start = buf.rangeStart;
322 parallelFor(
323 exec_,
324 {0, buf.patchSize},
325 KOKKOS_LAMBDA(const localIdx k) { dstView[start + k] = srcView[k]; }
326 );
327 }
328 }
329 else
330 {
331 auto valH = value_.copyToHost();
332 for (const auto& buf : commBuffers_)
333 {
334 for (localIdx k = 0; k < buf.patchSize; k++)
335 valH.view()[buf.rangeStart + k] = buf.recvBuf[static_cast<std::size_t>(k)];
336 }
337 value_ = valH.copyToExecutor(exec_);
338 }
339 requests_.clear();
340 communicating_ = false;
341 commBuffers_.clear();
342#endif
343 }

386template<typename ValueType>
387void communicateBoundaryData(
388 const CommunicationPattern& commPattern,
389 const std::vector<std::pair<localIdx, localIdx>> procPatchOffset,
390 Vector<ValueType>& boundaryData
391)
392{
393 auto mpiEnv = commPattern.env;
394 auto commRanks = mpiEnv.sizeRank();
395 auto sendSize = commPattern.sendCounts[commRanks];
396
397 // compute send displacements
398 auto sdispls = std::vector<int>(commRanks, 0);
399 int j = 0;
400 for (int i = 0; i < sdispls.size(); i++)
401 {
402 // check if rank communicates data
403 if (commPattern.sendCounts[i] > 0)
404 {
405 // send displ patch start
406 auto [start, end] = procPatchOffset[j];
407 sdispls[i] = start;
408 j++;
409 }
410 }
411
412 // FIXME recvBuffer does not need to same size as boundaryData
413 auto recvBuffer = Vector<ValueType>(boundaryData.exec(), boundaryData.size());
414 MPI_Alltoallv(
415 boundaryData.data(),
416 commPattern.sendCounts.data(),
417 sdispls.data(),
418 mpi::getType<ValueType>(),
419 recvBuffer.data(),
420 commPattern.sendCounts.data(),
421 sdispls.data(),
422 mpi::getType<ValueType>(),
423 mpiEnv.comm()
424 );
425
426 // FIXME TODO this might not be needed communication happens directly into boundaryData
427 // auto copyIdx = Vector<localIdx>(boundaryData.exec(), procPatchOffset);
428 auto exec = boundaryData.exec();
429 auto outV = boundaryData.view();
430 // const auto idxV = copyIdx.view();
431 const auto inV = recvBuffer.view();
432
433 parallelFor(
434 exec,
435 {0, procPatchOffset.size()},
436 NEON_LAMBDA(const localIdx p) {
437 auto [start, end] = procPatchOffset[p];
438 for (auto i = start; i < end; i++)
439 {
440 outV[i] = inV[i];
441 }
442 },
443 "copyMap"
444 );
445}

448template<>
449inline void communicateBoundaryData(
450 const CommunicationPattern& commPattern,
451 const std::vector<std::pair<localIdx, localIdx>> procPatchOffset,
452 Vector<Vec3>& boundaryData
453)
454{
455 auto mpiEnv = commPattern.env;
456 auto commRanks = mpiEnv.sizeRank();
457 auto sendSize = commPattern.sendCounts[commRanks];
458
459 // compute send displacements
460 std::vector<localIdx> sendCounts(commPattern.sendCounts.size(), 0);
461 for (int i = 0; i < sendCounts.size(); i++)
462 {
463 sendCounts[i] = commPattern.sendCounts[i] * 3;
464 }
465
466 auto sdispls = std::vector<int>(commRanks, 0);
467 int j = 0;
468 for (int i = 0; i < sdispls.size(); i++)
469 {
470 // check if rank communicates data
471 if (sendCounts[i] > 0)
472 {
473 // send displ patch start
474 auto [start, end] = procPatchOffset[j];
475 sdispls[i] = 3 * start;
476 j++;
477 }
478 }
479
480 // FIXME recvBuffer does not need to be same size as boundaryData
481 auto exec = boundaryData.exec();
482 auto boundaryDataSize = boundaryData.size();
483 auto sendBuffer = Vector<NeoN::scalar>(boundaryData.exec(), 3 * boundaryData.size());
484 auto sendBufferV = sendBuffer.view();
485 auto boundaryDataV = boundaryData.view();
486
487 parallelFor(
488 exec,
489 {0, boundaryData.size()},
490 NEON_LAMBDA(const localIdx i) {
491 sendBufferV[3 * i + 0] = boundaryDataV[i][0];
492 sendBufferV[3 * i + 1] = boundaryDataV[i][1];
493 sendBufferV[3 * i + 2] = boundaryDataV[i][2];
494 },
495 "copyMap"
496 );
497
498 auto recvBuffer = Vector<NeoN::scalar>(boundaryData.exec(), 3 * boundaryData.size());
499 auto recvBufferV = recvBuffer.view();
500 MPI_Alltoallv(
501 sendBuffer.data(),
502 sendCounts.data(),
503 sdispls.data(),
504 mpi::getType<scalar>(),
505 recvBuffer.data(),
506 sendCounts.data(),
507 sdispls.data(),
508 mpi::getType<scalar>(),
509 mpiEnv.comm()
510 );
511
512 const auto inV = recvBuffer.view();
513 auto outV = boundaryData.view();
514 parallelFor(
515 exec,
516 {0, procPatchOffset.size()},
517 NEON_LAMBDA(const localIdx p) {
518 auto [start, end] = procPatchOffset[p];
519 for (auto i = start; i < end; i++)
520 {
521 outV[i][0] = inV[3 * i + 0];
522 outV[i][1] = inV[3 * i + 1];
523 outV[i][2] = inV[3 * i + 2];
524 }
525 },
526 "copyMap"
527 );
528}
