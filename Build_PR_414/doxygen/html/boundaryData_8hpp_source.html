223template<typename ValueType>
224void communicateBoundaryData(
225 const CommunicationPattern& commPattern,
226 const std::vector<std::pair<localIdx, localIdx>> procPatchOffset,
227 Vector<ValueType>& boundaryData
228)
229{
230 auto mpiEnv = commPattern.env;
231 auto commRanks = mpiEnv.sizeRank();
232 auto sendSize = commPattern.sendCounts[commRanks];
233
234 // compute send displacements
235 auto sdispls = std::vector<int>(commRanks, 0);
236 int j = 0;
237 for (int i = 0; i < sdispls.size(); i++)
238 {
239 // check if rank communicates data
240 if (commPattern.sendCounts[i] > 0)
241 {
242 // send displ patch start
243 auto [start, end] = procPatchOffset[j];
244 sdispls[i] = start;
245 j++;
246 }
247 }
248
249 // FIXME recvBuffer does not need to same size as boundaryData
250 auto recvBuffer = Vector<ValueType>(boundaryData.exec(), boundaryData.size());
251 MPI_Alltoallv(
252 boundaryData.data(),
253 commPattern.sendCounts.data(),
254 sdispls.data(),
255 mpi::getType<ValueType>(),
256 recvBuffer.data(),
257 commPattern.sendCounts.data(),
258 sdispls.data(),
259 mpi::getType<ValueType>(),
260 mpiEnv.comm()
261 );
262
263 // FIXME TODO this might not be needed communication happens directly into boundaryData
264 // auto copyIdx = Vector<localIdx>(boundaryData.exec(), procPatchOffset);
265 auto exec = boundaryData.exec();
266 auto outV = boundaryData.view();
267 // const auto idxV = copyIdx.view();
268 const auto inV = recvBuffer.view();
269
270 parallelFor(
271 exec,
272 {0, procPatchOffset.size()},
273 NEON_LAMBDA(const localIdx p) {
274 auto [start, end] = procPatchOffset[p];
275 for (auto i = start; i < end; i++)
276 {
277 outV[i] = inV[i];
278 }
279 },
280 "copyMap"
281 );
282}

286inline void communicateBoundaryData(
287 const CommunicationPattern& commPattern,
288 const std::vector<std::pair<localIdx, localIdx>> procPatchOffset,
289 Vector<Vec3>& boundaryData
290)
291{
292 auto mpiEnv = commPattern.env;
293 auto commRanks = mpiEnv.sizeRank();
294 auto sendSize = commPattern.sendCounts[commRanks];
295
296 // compute send displacements
297 std::vector<localIdx> sendCounts(commPattern.sendCounts.size(), 0);
298 for (int i = 0; i < sendCounts.size(); i++)
299 {
300 sendCounts[i] = commPattern.sendCounts[i] * 3;
301 }
302
303 auto sdispls = std::vector<int>(commRanks, 0);
304 int j = 0;
305 for (int i = 0; i < sdispls.size(); i++)
306 {
307 // check if rank communicates data
308 if (sendCounts[i] > 0)
309 {
310 // send displ patch start
311 auto [start, end] = procPatchOffset[j];
312 sdispls[i] = 3 * start;
313 j++;
314 }
315 }
316
317 // FIXME recvBuffer does not need to be same size as boundaryData
318 auto exec = boundaryData.exec();
319 auto boundaryDataSize = boundaryData.size();
320 auto sendBuffer = Vector<NeoN::scalar>(boundaryData.exec(), 3 * boundaryData.size());
321 auto sendBufferV = sendBuffer.view();
322 auto boundaryDataV = boundaryData.view();
323
324 parallelFor(
325 exec,
326 {0, boundaryData.size()},
327 NEON_LAMBDA(const localIdx i) {
328 sendBufferV[3 * i + 0] = boundaryDataV[i][0];
329 sendBufferV[3 * i + 1] = boundaryDataV[i][1];
330 sendBufferV[3 * i + 2] = boundaryDataV[i][2];
331 },
332 "copyMap"
333 );
334
335 auto recvBuffer = Vector<NeoN::scalar>(boundaryData.exec(), 3 * boundaryData.size());
336 auto recvBufferV = recvBuffer.view();
337 MPI_Alltoallv(
338 sendBuffer.data(),
339 sendCounts.data(),
340 sdispls.data(),
341 mpi::getType<scalar>(),
342 recvBuffer.data(),
343 sendCounts.data(),
344 sdispls.data(),
345 mpi::getType<scalar>(),
346 mpiEnv.comm()
347 );
348
349 const auto inV = recvBuffer.view();
350 auto outV = boundaryData.view();
351 parallelFor(
352 exec,
353 {0, procPatchOffset.size()},
354 NEON_LAMBDA(const localIdx p) {
355 auto [start, end] = procPatchOffset[p];
356 for (auto i = start; i < end; i++)
357 {
358 outV[i][0] = inV[3 * i + 0];
359 outV[i][1] = inV[3 * i + 1];
360 outV[i][2] = inV[3 * i + 2];
361 }
362 },
363 "copyMap"
364 );
365}
