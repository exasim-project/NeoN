221template<typename ValueType>
222void communicateBoundaryData(
223 const CommunicationPattern& commPattern,
224 const std::vector<localIdx> procPatchOffset,
225 Vector<ValueType>& boundaryData
226)
227{
228 // // 2. exchange
229 auto mpiEnv = commPattern.env;
230 auto commRanks = mpiEnv.sizeRank();
231 auto sendSize = commPattern.sendCounts[commRanks];
232
233 // compute send displacements
234 auto sdispls = std::vector<int>(commRanks, 0);
235 int j = 0;
236 for (int i = 0; i < sdispls.size(); i++)
237 {
238 // check if rank communicates data
239 if (commPattern.sendCounts[i] > 0)
240 {
241 // send displ patch start
242 sdispls[i] = procPatchOffset[j];
243 j++;
244 }
245 }
246
247 // FIXME recvBuffer does not need to same size as boundaryData
248 auto recvBuffer = Vector<ValueType>(boundaryData.exec(), boundaryData.size());
249 auto copyIdx = Vector<localIdx>(boundaryData.exec(), procPatchOffset);
250 MPI_Alltoallv(
251 boundaryData.data(),
252 commPattern.sendCounts.data(),
253 sdispls.data(),
254 mpi::getType<ValueType>(),
255 recvBuffer.data(),
256 commPattern.sendCounts.data(),
257 sdispls.data(),
258 mpi::getType<ValueType>(),
259 mpiEnv.comm()
260 );
261
262 // FIXME TODO this might not be needed communication happens directly into boundaryData
263 auto exec = boundaryData.exec();
264 auto outV = boundaryData.view();
265 const auto idxV = copyIdx.view();
266 const auto inV = recvBuffer.view();
267
268 parallelFor(
269 exec,
270 {0, idxV.size()},
271 NEON_LAMBDA(const localIdx i) { outV[idxV[i]] = inV[idxV[i]]; },
272 "copyMap"
273 );
274}
