18/* @brief blended central/upwind weights of the limitedLinear scheme.
19 *
20 * w_f = limiter * w_CD + (1 - limiter) * w_upwind, with the limiter built from the TVD ratio r of
21 * the upwind-cell gradient to the face gradient: limiter = max(min(twoByk * r, 1), 0), where
22 * twoByk = 2/k: k = 1 limits most strongly (fully bounded), smaller k relaxes the blend towards
23 * central differencing.
24 */
25void computeLimitedLinearWeights(
26 const VolumeField<scalar>& src,
27 const SurfaceField<scalar>& flux,
28 const SurfaceField<scalar>& cdWeights,
29 const SurfaceField<Vec3>& faceDeltaOwner,
30 const SurfaceField<Vec3>& faceDeltaNeighbour,
31 scalar twoByk,
32 SurfaceField<scalar>& weights
33);

35/* @brief limitedLinear surface interpolation.
36 *
37 * Registered for scalar fields only. Limiting a vector field is conventionally done on
38 * magSqr(phi), which needs a derived scalar field and its gradient; that variant is not
39 * implemented here, so a Vec3 field must use another scheme.
40 */
41template<typename ValueType>
42class LimitedLinear :
43 public SurfaceInterpolationFactory<ValueType>::template Register<LimitedLinear<ValueType>>
44{
45 using Base =
46 SurfaceInterpolationFactory<ValueType>::template Register<LimitedLinear<ValueType>>;
47
48public:
49
50 LimitedLinear(const Executor& exec, const UnstructuredMesh& mesh, Input input)
51 : Base(exec, mesh), geometryScheme_(GeometryScheme::readOrCreate(mesh)),
52 twoByk_(readTwoByk(input))
53 {
54 // Opt in to the per-face cell-to-face offsets while the mesh centres are still alive; the
55 // geometry scheme frees them on the first read of any cached geometry field.
56 geometryScheme_->ensureFaceDeltas();
57 };

118 // Scheme spec "limitedLinear <k>": the factory consumed the name, so only k remains. k is
119 // written as an integer (e.g. 1) or a fractional value (e.g. 0.2). It is mandatory: a missing
120 // or non-numeric coefficient is a spec error and is rejected rather than silently replaced by a
121 // default that would change the scheme's behaviour.
122 static scalar readTwoByk(Input& input)
123 {
124 scalar k = 0;
125 if (std::holds_alternative<NeoN::TokenList>(input))
126 {
127 auto& tokens = std::get<NeoN::TokenList>(input);
128 if (tokens.peekIs<scalar>())
129 {
130 k = tokens.next<scalar>();
131 }
132 else if (tokens.peekIs<label>())
133 {
134 k = static_cast<scalar>(tokens.next<label>());
135 }
136 else
137 {
138 NF_THROW("limitedLinear requires a numeric coefficient: 'limitedLinear <k>'");
139 }
140 }
141 else
142 {
143 const auto& dict = std::get<NeoN::Dictionary>(input);
144 if (!dict.contains("limitedLinearCoeff"))
145 {
146 NF_THROW("limitedLinear requires the dictionary entry 'limitedLinearCoeff'");
147 }
148 k = dict.get<scalar>("limitedLinearCoeff");
149 }
150
151 if (!(k >= scalar(0) && k <= scalar(1)))
152 {
153 NF_THROW(
154 std::string("limitedLinear coefficient must be in [0, 1], got ") + std::to_string(k)
155 );
156 }
157 return scalar(2) / Kokkos::max(k, ROOTVSMALL);
158 }
159
160 const std::shared_ptr<GeometryScheme> geometryScheme_;
161 scalar twoByk_;
162};
