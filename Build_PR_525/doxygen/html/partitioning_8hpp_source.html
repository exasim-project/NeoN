18template<typename BoundaryType>
19auto setProcessorBoundaryHelper(
20 UnstructuredMesh& mesh, const std::vector<BoundaryType>& bcs, NeoN::mpi::Environment mpiEnviron
21)
22{
23 const auto rank = mpiEnviron.rank();
24 const auto ranks = mpiEnviron.sizeRank();
25 const bool isMiddle = (rank > 0 && rank < ranks - 1);
26 auto ret = std::vector<BoundaryType> {};
27 for (int i = 0; i < static_cast<int>(bcs.size()); i++)
28 {
29 // boundary index 1 is always the proc boundary for first/last ranks;
30 // middle ranks have all boundaries as proc.
31 const bool isProc = (i == 1) || isMiddle;
32 std::string boundaryType = isProc ? "processor" : "calculated";
33 auto patchDict = Dictionary({{"type", boundaryType}});
34 ret.emplace_back(mesh, patchDict, i);
35 }
36 return ret;
37}

45template<typename ValueType>
46Vector<ValueType> partitionInternalVector(
47 const Vector<ValueType>& globalVector,
48 const UnstructuredMesh& localMesh,
49 NeoN::mpi::Environment mpiEnviron
50)
51{
52 const localIdx localSize = static_cast<localIdx>(globalVector.size()) / mpiEnviron.sizeRank();
53 const localIdx firstIdx = mpiEnviron.rank() * localMesh.nCells();
54 return take(globalVector, {firstIdx, firstIdx + localSize});
55}

58template<typename FieldType>
59FieldType
60oneDPartitionField(FieldType field, UnstructuredMesh& mesh, NeoN::mpi::Environment mpiEnviron)
61{
62 auto internalVector = partitionInternalVector(field.internalVector(), mesh, mpiEnviron);
63 auto bcsPart = setProcessorBoundaryHelper(mesh, field.boundaryConditions(), mpiEnviron);
64 FieldType ret(field.exec(), field.name + "Part", mesh, bcsPart);
65 ret.internalVector() = internalVector;
66
67 const auto nProcBoundaryFaces = mesh.nProcBoundaryFaces();
68 if (nProcBoundaryFaces > 0)
69 {
70 const localIdx localSize =
71 static_cast<localIdx>(field.internalVector().size()) / mpiEnviron.sizeRank();
72 const localIdx firstIdx = mpiEnviron.rank() * mesh.nCells();
73 const auto nBoundaryFaces = mesh.nBoundaryFaces();
74 auto weightsH = mesh.boundaryMesh().weights().copyToHost();
75 auto globalInternalH = field.internalVector().copyToHost();
76 auto retBdH = ret.boundaryData().value().copyToHost();
77 const auto weightsHV = weightsH.view();
78 const auto globalInternalHV = globalInternalH.view();
79 auto retBdHV = retBdH.view();
80 for (localIdx procFacei = 0; procFacei < nProcBoundaryFaces; procFacei++)
81 {
82 const auto bcfacei = nBoundaryFaces + procFacei;
83 // weight > 0: owner side (right-facing) — neighbour is at firstIdx + localSize
84 // weight < 0: non-owner side (left-facing) — neighbour is at firstIdx - 1
85 const localIdx gIdx = weightsHV[bcfacei] > 0 ? firstIdx + localSize : firstIdx - 1;
86 retBdHV[bcfacei] = globalInternalHV[gIdx];
87 }
88 ret.boundaryData().value() = retBdH.copyToExecutor(field.exec());
89 }
90
91 return ret;
92}

104template<typename FieldType, typename MeshType, typename BcType>
105FieldType partitionSurfaceField(
106 FieldType field,
107 MeshType& mesh,
108 BcType bcs,
109 NeoN::mpi::Environment mpiEnviron,
110 bool flip = false
111)
112{
113 auto exec = mesh.exec();
114 localIdx localCells = mesh.nCells();
115 localIdx localFaces = mesh.nInternalFaces();
116 localIdx firstFace = 0;
117
118 localIdx firstBoundaryFace = 0;
119 localIdx secondBoundaryFace = 0;
120
121 scalar signLeft = 1.0;
122 scalar signRight = 1.0;
123
124 // FIXME: hardcoded for 3 ranks
125 if (mpiEnviron.rank() == 0)
126 {
127 firstBoundaryFace = 3 * localFaces + 2;
128 secondBoundaryFace = localFaces;
129 if (flip) signRight = -1.0;
130 }
131 if (mpiEnviron.rank() == 1)
132 {
133 firstFace = localFaces + 1;
134 firstBoundaryFace = localFaces;
135 secondBoundaryFace = firstBoundaryFace + localCells;
136 if (flip) signLeft = -1.0;
137 }
138 if (mpiEnviron.rank() == 2)
139 {
140 firstFace = localCells + localCells;
141 firstBoundaryFace = 3 * localFaces + 3;
142 secondBoundaryFace = 2 * localFaces + 1;
143 if (flip) signRight = -1.0;
144 }
145
146 localIdx lastFace = firstFace + localFaces + 2;
147
148 FieldType ret = {field.exec(), field.name + "Part", mesh, bcs};
149
150 NF_ASSERT(lastFace - firstFace == mesh.nTotalFaces(), "different size");
151
152 auto internalVector = take(field.internalVector(), {firstFace, lastFace});
153
154 auto outV = internalVector.view();
155 auto inV = field.internalVector().view();
156 auto inBV = field.boundaryData().value().view();
157 const localIdx nGlobalInternalFaces = static_cast<localIdx>(field.internalVector().size());
158
159 const bool firstIsPhysical = firstBoundaryFace >= nGlobalInternalFaces;
160 const localIdx firstBoundaryPatchIdx = firstBoundaryFace - nGlobalInternalFaces;
161
162 NeoN::parallelFor(
163 exec,
164 {0, 1},
165 NEON_LAMBDA(const localIdx i) {
166 outV[localFaces] =
167 signLeft * (firstIsPhysical ? inBV[firstBoundaryPatchIdx] : inV[firstBoundaryFace]);
168 },
169 "partitionSurfaceField::firstBoundaryFace"
170 );
171
172 NeoN::parallelFor(
173 exec,
174 {0, 1},
175 NEON_LAMBDA(const localIdx i) {
176 outV[localFaces + 1] = signRight * inV[secondBoundaryFace];
177 },
178 "partitionSurfaceField::secondBoundaryFace"
179 );
180
181 ret.internalVector() = internalVector;
182
183 const auto nif = mesh.nInternalFaces();
184 const auto totalBoundary = mesh.nBoundaryFaces() + mesh.nProcBoundaryFaces();
185 auto bDataV = ret.boundaryData().value().view();
186 NeoN::parallelFor(
187 exec,
188 {0, totalBoundary},
189 NEON_LAMBDA(const localIdx bfi) { bDataV[bfi] = outV[nif + bfi]; },
190 "partitionSurfaceField::copyBoundaryData"
191 );
192
193 return ret;
194}
