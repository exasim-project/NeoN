54/* @brief The Kokkos execution space a NeoN executor runs its kernels on. */

58/* @brief Default-constructed instance of the executor's Kokkos execution space. */

65/* @brief Unmanaged Kokkos view over memory owned by the given executor. */
66template<typename ExecutorType, typename ValueType>
67auto createKokkosView(const ExecutorType&, ValueType* ptr, size_t size)
68{
69 using Memory = typename KokkosSpaces<std::remove_cvref_t<ExecutorType>>::memory;
