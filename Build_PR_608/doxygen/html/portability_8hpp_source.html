7/* @file portability.hpp
8 * @brief Kokkos-free function annotations shared by device and host-only translation units.
9 *
10 * NeoN compiles in two halves (see doc/splitCompilation.rst):
11 * - device TUs include Kokkos, may launch kernels (NEON_LAMBDA, parallelFor) and are built with
12 * the device toolchain at C++20;
13 * - host-only TUs never see a Kokkos header and may be built with a different host compiler and
14 * a newer standard (NeoN_HOST_CXX_STANDARD). They are compiled with NEON_HOST_ONLY_TU=1.
15 *
16 * Headers shared by both halves must therefore not include Kokkos. Functions that have to be
17 * callable from kernels use NEON_INLINE_FUNCTION from this header instead of
18 * KOKKOS_INLINE_FUNCTION. The expansion matches Kokkos' own for the CUDA and HIP backends; for all
19 * other backends (Serial, OpenMP, Threads, SYCL) Kokkos expands it to plain `inline` as well.
20 *
21 * The annotation differs between a CUDA/HIP TU and a host-only TU. That affects neither layout nor
22 * mangling of the annotated function, and it is the same arrangement Eigen and Thrust rely on for
23 * mixed .cpp/.cu builds.
24 */
25

40/* @brief Abort the program from host or device code.
41 *
42 * Replacement for Kokkos::abort in headers that must stay Kokkos-free. On the device the message
43 * is printed and the kernel traps; on the host it is printed to stderr before std::abort.
44 */
45NEON_INLINE_FUNCTION void abort(const char* message)
46{
47#if defined(__CUDA_ARCH__)
48 printf("NeoN abort: %s\n", message);
49 __trap();
50#elif defined(__HIP_DEVICE_COMPILE__) || defined(__SYCL_DEVICE_ONLY__)
51 (void)message;
52 __builtin_trap();
53#else
54 std::fprintf(stderr, "NeoN abort: %s\n", message);
55 std::abort();
56#endif
57}

59} // namespace NeoN::detail
