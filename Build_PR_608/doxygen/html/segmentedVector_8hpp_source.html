21/* @brief Prefix-sum kernel of segmentsFromIntervals.
22 *
23 * A named functor rather than a NEON_LAMBDA: under nvcc a NEON_LAMBDA is an
24 * extended lambda whose host-side callable is registered per translation unit,
25 * and for an enclosing *function template* the registration belongs to a comdat
26 * group the linker may drop while keeping the body. The surviving copy then
27 * calls a null pointer — every segmentsFromIntervals<localIdx> call inside
28 * libNeoN segfaults (e.g. CellToFaceStencil::computeInternalStencil, the
29 * stencil CellLimitedGrad builds). A functor carries no such registration.
30 */
31template<typename IndexType>
32struct IntervalOffsetScan
33{
34 using value_type = IndexType;
35
36 View<const IndexType> intervals;
37 View<IndexType> offsets;
38
39 NEON_INLINE_FUNCTION
40 void operator()(const localIdx i, IndexType& update, const bool final) const
41 {
42 update += intervals[i];
43 if (final)
44 {
45 // offsets is a view, thus [] takes unsigned idx
46 offsets[i] = update;
47 }
48 }
49};
50
51} // namespace detail

65template<typename IndexType>
66IndexType segmentsFromIntervals(const Vector<IndexType>& intervals, Vector<IndexType>& offsets)
67{
68 IndexType finalValue = 0;
69 const auto inView = intervals.view();
70 // skip the first element of the offsets
71 // assumed to be zero
72 auto offsView = offsets.view().subview(1);
73 // NOTE avoid compiler warning by static_casting to localIdx since offsView
74 // is a View
75 NF_ASSERT_EQUAL(inView.size(), offsView.size());
76 NeoN::parallelScan(
77 intervals.exec(),
78 {0, offsView.size()},
79 detail::IntervalOffsetScan<IndexType> {inView, offsView},
80 finalValue
81 );
82 return finalValue;
83}

176 /*
177 * @brief Create a segmented vector from intervals.
178 * @param intervals The intervals to create the segmented vector from.
179 * @note The intervals are the lengths of each segment
180 */
181 SegmentedVector(const Vector<IndexType>& intervals)
182 : values_(intervals.exec(), 0),
183 segments_(intervals.exec(), intervals.size() + 1, IndexType(0))

185 IndexType valueSize = segmentsFromIntervals(intervals, segments_);
