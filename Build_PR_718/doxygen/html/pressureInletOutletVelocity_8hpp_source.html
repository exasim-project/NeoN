24// Shared kernel behind PressureInletOutletVelocity, the velocity counterpart to a
25// pressure-driven outlet (OpenFOAM's pressureInletOutletVelocityFvPatchVectorField, a
26// directionMixedFvPatchVectorField). Literal OpenFOAM blends with a TENSOR-valued fraction
27// valueFraction() = neg(phi) * (I - n (x) n)
28// i.e. the tangential components are forced to a fixed tangentialVelocity on inflow and
29// zero-gradient on outflow, while the normal component is *always* left to zero-gradient
30// extrapolation (the pressure equation, not this BC, sets the normal velocity). NeoN's
31// boundaryData() valueFraction is a single scalar shared by all 3 components of a Vec3 field
32// (see e.g. gaussGreenDiv.cpp/gaussGreenLaplacian.cpp, which multiply a whole Vec3 refValue by
33// one scalar fraction), so that tensor cannot be represented directly.
34//
35// It does not need to be: the tensor blend is reproduced exactly by making refValue itself carry
36// the normal-direction extrapolation, recomputed every call from the CURRENT owner-cell value
37// (the same "recompute from live field data" pattern already used by e.g. totalPressure):
38// n = faceNormal / faceArea (unit outward normal)
39// refValue = n*(n & ownerValue) + (tangentialVelocity - n*(n & tangentialVelocity))
40// valueFraction = neg(phi) (plain scalar, as inletOutlet)
41// refGrad = 0
42// value = valueFraction*refValue + (1 - valueFraction)*ownerValue
43// On inflow (valueFraction = 1): value = n*(n & ownerValue) + tangentialVelocity_projected,
44// identical to literal's tensor-blend inflow value. On outflow (valueFraction = 0): value =
45// ownerValue, identical to literal's outflow value (full zero-gradient on all 3 components). This
46// holds for any U, not just a flow aligned with a coordinate axis -- no approximation, and no
47// change to NeoN's scalar valueFraction machinery.
48//
49// When no flux is available (hasFlux == false) every face is treated as outflow, degenerating to
50// a pure zeroGradient BC -- the same fallback inletOutlet uses.
51inline void setPressureInletOutletVelocityValue(
52 Field<NeoN::Vec3>& domainVector,
53 const UnstructuredMesh& mesh,
54 std::pair<localIdx, localIdx> range,
55 NeoN::Vec3 tangentialVelocity,
56 View<const NeoN::scalar> boundaryFlux,
57 bool hasFlux
58)
59{
60 const auto internalV = domainVector.internalVector().view();
61
62 auto [faceNormalsV, faceAreasV, boundaryFaceOwnersV] = views(
63 mesh.boundaryMesh().faceNormals(),
64 mesh.boundaryMesh().faceAreas(),
65 mesh.boundaryMesh().faceOwners()
66 );
67
68 auto [refGradV, valueV, valueFractionV, refValueV] = views(
69 domainVector.boundaryData().refGrad(),
70 domainVector.boundaryData().value(),
71 domainVector.boundaryData().valueFraction(),
72 domainVector.boundaryData().refValue()
73 );
74
75 NeoN::parallelFor(
76 domainVector.exec(),
77 range,
78 NEON_LAMBDA(const localIdx i) {
79 const localIdx owner = boundaryFaceOwnersV[i];
80 const NeoN::Vec3 ownerValue = internalV[owner];
81 const NeoN::Vec3 n = faceNormalsV[i] / faceAreasV[i];
82
83 const NeoN::scalar valueFraction =
84 (hasFlux && boundaryFlux[i] < 0.0) ? NeoN::scalar(1) : NeoN::scalar(0);
85
86 const NeoN::Vec3 tangentialVelocityProjected =
87 tangentialVelocity - n * (n & tangentialVelocity);
88 const NeoN::Vec3 refValue = n * (n & ownerValue) + tangentialVelocityProjected;
89
90 refGradV[i] = NeoN::zero<NeoN::Vec3>();
91 refValueV[i] = refValue;
92 valueFractionV[i] = valueFraction;
93 valueV[i] = valueFraction * refValue + (NeoN::scalar(1) - valueFraction) * ownerValue;
94 },
95 "setPressureInletOutletVelocityValue"
96 );
97}

101// PressureInletOutletVelocity: velocity outlet counterpart to a pressure-driven boundary
102// (mirrors OpenFOAM's pressureInletOutletVelocity). See detail::setPressureInletOutletVelocityValue
103// for the exact formula and its exact equivalence to OpenFOAM's tensor-valued directionMixed
104// blend.
105//
106// Needs, via the context-aware correctBoundaryCondition(domainVector, ctx) overload, a
107// surface-scalar flux, key `phiName_` (dict key "phi", default "phi"). Without it, degrades to
108// plain zero-gradient (the same fallback the plain, context-less overload always uses).
109class PressureInletOutletVelocity :
110 public VolumeBoundaryFactory<NeoN::Vec3>::template Register<PressureInletOutletVelocity>
111{
112 using Base = VolumeBoundaryFactory<NeoN::Vec3>::template Register<PressureInletOutletVelocity>;
113
114public:
115
116 using Base::correctBoundaryCondition;
117
118 // assignable = true: recomputed every correctBoundaryConditions() call. fixesValue = false:
119 // this is a mixed BC (per-face valueFraction), not a global Dirichlet patch.
120 PressureInletOutletVelocity(
121 const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID
122 )
123 : Base(mesh, dict, patchID, {.assignable = true, .fixesValue = false}), mesh_(mesh),
124 tangentialVelocity_(
125 dict.contains("tangentialVelocity") ? dict.get<NeoN::Vec3>("tangentialVelocity")
126 : NeoN::Vec3 {}
127 ),
128 phiName_(dict.contains("phi") ? dict.get<std::string>("phi") : std::string("phi"))
129 {}
130
131 // No context => no flux available: behave as zero-gradient (all faces treated as outflow).
132 virtual void correctBoundaryCondition(Field<NeoN::Vec3>& domainVector) final
133 {
134 detail::setPressureInletOutletVelocityValue(
135 domainVector,
136 mesh_,
137 this->range(),
138 tangentialVelocity_,
139 View<const NeoN::scalar> {},
140 false
141 );
142 }
143
144 // Context-aware: read the face flux phi and switch inflow/outflow per face.
145 virtual void
146 correctBoundaryCondition(Field<NeoN::Vec3>& domainVector, const BoundaryContext& ctx) final
147 {
148 if (ctx.hasSurfaceScalar(phiName_))
149 {
150 const auto& phi = ctx.surfaceScalarField(phiName_);
151 detail::setPressureInletOutletVelocityValue(
152 domainVector,
153 mesh_,
154 this->range(),
155 tangentialVelocity_,
156 phi.boundaryData().value().view(),
157 true
158 );
159 }
160 else
161 {
162 correctBoundaryCondition(domainVector);
163 }
164 }
165
166 static std::string name() { return "pressureInletOutletVelocity"; }
167
168 std::string getName() const override { return name(); }
169
170 static std::string doc()
171 {
172 return "Velocity outlet paired with a pressure-driven boundary (OpenFOAM's "
173 "pressureInletOutletVelocity): on inflow faces (phi < 0) the tangential velocity "
174 "is set to tangentialVelocity (default zero) while the normal component is left "
175 "to zero-gradient; on outflow faces all 3 components are zero-gradient. Switched "
176 "per face by the sign of the face flux phi.";
177 }
178
179 static std::string schema() { return "none"; }
180
181 virtual std::unique_ptr<VolumeBoundaryFactory<NeoN::Vec3>> clone() const final
182 {
183 return std::make_unique<PressureInletOutletVelocity>(*this);
184 }
185
186private:
187
188 const UnstructuredMesh& mesh_;
189 NeoN::Vec3 tangentialVelocity_;
190 std::string phiName_;
191};
