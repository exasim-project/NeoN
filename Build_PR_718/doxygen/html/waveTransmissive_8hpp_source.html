27// Shared kernel behind WaveTransmissive, a non-reflecting compressible outflow BC. Mirrors
28// OpenFOAM's waveTransmissiveFvPatchField (itself a specialisation of advectiveFvPatchField,
29// Euler ddt only).
30//
31// Per boundary face, with the outward wave speed
32// w = max(0, Un + c), Un = (phi/rho)/magSf (normal velocity), c = sqrt(gamma/psi)
33// (psi = 1/(R*T), so gamma/psi = gamma*R*T = c^2 -- the local speed of sound) and
34// alpha = w*dt*deltaCoeffs
35// the patch is a mixed BC (refGrad = 0 throughout):
36// lInf > 0 (relax toward fieldInf over a finite length lInf, the usual non-reflecting case):
37// k = w*dt/lInf
38// refValue = (p_oldTime + k*fieldInf) / (1 + k)
39// valueFraction = (1 + k) / (1 + alpha + k)
40// lInf <= 0 (pure wave transmission, no far-field relaxation):
41// refValue = p_oldTime
42// valueFraction = 1 / (1 + alpha)
43// value = valueFraction*refValue + (1 - valueFraction)*ownerCellValue
44//
45// Only ever evaluated for scalar fields in practice (OpenFOAM templates it generically, but
46// every real case uses it on p) -- kept as a bare function over scalar views rather than
47// templated on ValueType to avoid an unused, unverified vector/tensor path.
48inline void setWaveTransmissiveValue(
49 Field<NeoN::scalar>& domainVector,
50 const UnstructuredMesh& mesh,
51 std::pair<localIdx, localIdx> range,
52 NeoN::scalar gamma,
53 NeoN::scalar fieldInf,
54 NeoN::scalar lInf,
55 NeoN::scalar dt,
56 View<const NeoN::scalar> boundaryPhi,
57 View<const NeoN::scalar> boundaryRho,
58 View<const NeoN::scalar> boundaryPsi,
59 View<const NeoN::scalar> boundaryPOld,
60 bool haveContext
61)
62{
63 const auto internalV = domainVector.internalVector().view();
64 const auto [faceAreasV, deltaCoeffsV, boundaryFaceOwnersV] = views(
65 mesh.boundaryMesh().faceAreas(),
66 mesh.boundaryMesh().deltaCoeffs(),
67 mesh.boundaryMesh().faceOwners()
68 );
69
70 auto [refGradV, valueV, valueFractionV, refValueV] = views(
71 domainVector.boundaryData().refGrad(),
72 domainVector.boundaryData().value(),
73 domainVector.boundaryData().valueFraction(),
74 domainVector.boundaryData().refValue()
75 );
76
77 NeoN::parallelFor(
78 domainVector.exec(),
79 range,
80 NEON_LAMBDA(const localIdx i) {
81 const localIdx owner = boundaryFaceOwnersV[i];
82 const NeoN::scalar ownerValue = internalV[owner];
83
84 refGradV[i] = NeoN::scalar(0);
85
86 if (!haveContext)
87 {
88 // Degenerate fallback with no phi/rho/psi/old-time/dt available: behave as a
89 // plain zero-gradient (owner-cell) extrapolation.
90 valueFractionV[i] = NeoN::scalar(0);
91 refValueV[i] = ownerValue;
92 valueV[i] = ownerValue;
93 return;
94 }
95
96 const NeoN::scalar Un = (boundaryPhi[i] / boundaryRho[i]) / faceAreasV[i];
97 const NeoN::scalar c = Kokkos::sqrt(gamma / boundaryPsi[i]);
98 const NeoN::scalar w = Kokkos::max(Un + c, NeoN::scalar(0));
99 const NeoN::scalar alpha = w * dt * deltaCoeffsV[i];
100
101 NeoN::scalar refValue;
102 NeoN::scalar valueFraction;
103 if (lInf > NeoN::scalar(0))
104 {
105 const NeoN::scalar k = w * dt / lInf;
106 refValue = (boundaryPOld[i] + k * fieldInf) / (NeoN::scalar(1) + k);
107 valueFraction = (NeoN::scalar(1) + k) / (NeoN::scalar(1) + alpha + k);
108 }
109 else
110 {
111 refValue = boundaryPOld[i];
112 valueFraction = NeoN::scalar(1) / (NeoN::scalar(1) + alpha);
113 }
114
115 refValueV[i] = refValue;
116 valueFractionV[i] = valueFraction;
117 valueV[i] = valueFraction * refValue + (NeoN::scalar(1) - valueFraction) * ownerValue;
118 },
119 "setWaveTransmissiveValue"
120 );
121}

125// WaveTransmissive: a non-reflecting compressible outflow BC, mirroring OpenFOAM's
126// waveTransmissive (built on advectiveFvPatchField). See detail::setWaveTransmissiveValue for
127// the exact formula and OpenFOAM source cross-reference.
128//
129// Needs, via the context-aware correctBoundaryCondition(domainVector, ctx) overload:
130// - a surface-scalar flux, key `phiName_` (dict key "phi", default "phi")
131// - a volume-scalar density, key `rhoName_` (dict key "rho", default "rho")
132// - a volume-scalar compressibility psi = 1/(R*T), key `psiName_` (dict key "psi", default
133// "psi")
134// - the field this BC is itself attached to, inserted under `fieldName_` (dict key "field",
135// default "p") -- needed for its oldTime() boundary value. NeoN's BC strategy objects are
136// not told which field they belong to, so the caller must additionally insert that same
137// field into the context under this name for this BC to find its own history.
138// - the current timestep, key "deltaT" (BoundaryContext::insertValue), via
139// BoundaryContext::hasValue/value rather than a field slot.
140// Any one missing degrades to a plain zero-gradient extrapolation (the same degeneration the
141// plain, context-less correctBoundaryCondition(domainVector) overload always uses), since a
142// partial wave-speed calculation would silently give the wrong far-field behaviour rather than
143// an honest fallback.
144class WaveTransmissive :
145 public VolumeBoundaryFactory<NeoN::scalar>::template Register<WaveTransmissive>
146{
147 using Base = VolumeBoundaryFactory<NeoN::scalar>::template Register<WaveTransmissive>;
148
149public:
150
151 using Base::correctBoundaryCondition;
152
153 // assignable = true: recomputed every correctBoundaryConditions() call. fixesValue =
154 // false: this is a mixed BC (per-face valueFraction), not a global Dirichlet patch.
155 WaveTransmissive(const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID)
156 : Base(mesh, dict, patchID, {.assignable = true, .fixesValue = false}), mesh_(mesh),
157 gamma_(dict.get<NeoN::scalar>("gamma")), // required: no physically-sane default
158 fieldInf_(
159 dict.contains("fieldInf") ? dict.get<NeoN::scalar>("fieldInf") : NeoN::scalar(0)
160 ),
161 lInf_(dict.contains("lInf") ? dict.get<NeoN::scalar>("lInf") : NeoN::scalar(-1)),
162 phiName_(dict.contains("phi") ? dict.get<std::string>("phi") : std::string("phi")),
163 rhoName_(dict.contains("rho") ? dict.get<std::string>("rho") : std::string("rho")),
164 psiName_(dict.contains("psi") ? dict.get<std::string>("psi") : std::string("psi")),
165 fieldName_(dict.contains("field") ? dict.get<std::string>("field") : std::string("p"))
166 {}
167
168 // No context => degenerate to zero-gradient (see detail kernel's haveContext == false path).
169 virtual void correctBoundaryCondition(Field<NeoN::scalar>& domainVector) final
170 {
171 detail::setWaveTransmissiveValue(
172 domainVector,
173 mesh_,
174 this->range(),
175 gamma_,
176 fieldInf_,
177 lInf_,
178 NeoN::scalar(0),
179 View<const NeoN::scalar> {},
180 View<const NeoN::scalar> {},
181 View<const NeoN::scalar> {},
182 View<const NeoN::scalar> {},
183 false
184 );
185 }
186
187 // Context-aware: the real non-reflecting wave-speed calculation.
188 virtual void
189 correctBoundaryCondition(Field<NeoN::scalar>& domainVector, const BoundaryContext& ctx) final
190 {
191 const bool haveContext = ctx.hasSurfaceScalar(phiName_) && ctx.hasScalar(rhoName_)
192 && ctx.hasScalar(psiName_) && ctx.hasScalar(fieldName_)
193 && ctx.hasValue("deltaT");
194
195 if (!haveContext)
196 {
197 correctBoundaryCondition(domainVector);
198 return;
199 }
200
201 const auto& phi = ctx.surfaceScalarField(phiName_);
202 const auto& rho = ctx.scalarFieldPtr(rhoName_);
203 const auto& psi = ctx.scalarFieldPtr(psiName_);
204 const auto& pOld = oldTime(ctx.scalarFieldPtr(fieldName_));
205 const NeoN::scalar dt = ctx.value("deltaT");
206
207 detail::setWaveTransmissiveValue(
208 domainVector,
209 mesh_,
210 this->range(),
211 gamma_,
212 fieldInf_,
213 lInf_,
214 dt,
215 phi.boundaryData().value().view(),
216 rho.boundaryData().value().view(),
217 psi.boundaryData().value().view(),
218 pOld.boundaryData().value().view(),
219 true
220 );
221 }

232 "rho, psi, the field's own old-time value, and the current deltaT in the "
233 "BoundaryContext; without them it degrades to zero-gradient.";
234 }
