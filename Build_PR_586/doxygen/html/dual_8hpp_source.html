78 v_ += r.v_;

83 KOKKOS_INLINE_FUNCTION Dual& operator-=(const Dual& r)

89
90 KOKKOS_INLINE_FUNCTION Dual& operator*=(const Dual& r)
91 {
92 for (int i = 0; i < NDeriv; ++i) d_[i] = d_[i] * r.v_ + v_ * r.d_[i];
93 v_ *= r.v_;
94 return *this;
95 }
96
97 KOKKOS_INLINE_FUNCTION Dual& operator/=(const Dual& r)
98 {
99 const ValueType invR = ValueType(1) / r.v_;
100 for (int i = 0; i < NDeriv; ++i) d_[i] = (d_[i] - v_ * invR * r.d_[i]) * invR;
101 v_ *= invR;
102 return *this;
103 }
104
105private:

107 ValueType v_;
108 ValueType d_[NDeriv];

145// --- comparison -------------------------------------------------------------
146// Comparisons act on the primal only. This is the standard convention and it is
147// also where forward-mode AD silently loses information: a branch taken on a
148// comparison is not differentiated, so limiters and upwind switches yield the
149// derivative of whichever branch was selected. See the differentiability audit
150// in doc/ad.md.
151

153 template<typename T, int N> \
154 KOKKOS_INLINE_FUNCTION bool operator OP(const Dual<T, N>& a, const Dual<T, N>& b) \
155 { \
156 return a.value() OP b.value(); \
157 }

171KOKKOS_INLINE_FUNCTION Dual<T, N> sqrt(const Dual<T, N>& a)
172{
173 using Kokkos::sqrt;
174 const T s = sqrt(a.value());
175 Dual<T, N> r(s);
176 const T c = T(0.5) / s;

178 return r;

202template<typename T, int N>
203KOKKOS_INLINE_FUNCTION Dual<T, N> mag(const Dual<T, N>& a)
204{
205 const T s = (a.value() > T(0)) ? T(1) : ((a.value() < T(0)) ? T(-1) : T(0));
206 Dual<T, N> r(s * a.value());
207 for (int i = 0; i < N; ++i) r.deriv(i) = s * a.deriv(i);
208 return r;
209}
