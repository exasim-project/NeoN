61 DualType declare(const std::string& name, ValueType value, ValueType scale = ValueType(0))
62 {
63 if (static_cast<int>(names_.size()) >= NDeriv)
64 {
65 throw std::runtime_error(
66 "DesignVariables: capacity " + std::to_string(NDeriv)

70 }
71 const int slot = static_cast<int>(names_.size());
72 names_.push_back(name);
73 values_.push_back(value);

75 ValueType s = scale;
76 if (s == ValueType(0))
77 {
78 const ValueType a = (value < ValueType(0)) ? -value : value;
79 s = (a > ValueType(0)) ? a : ValueType(1);
80 }
81 scales_.push_back(s);
82
83 return DualType(value, slot);
84 }
