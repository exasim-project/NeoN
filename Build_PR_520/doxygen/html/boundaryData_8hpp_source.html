214 void communicate(std::pair<localIdx, localIdx> range, int neighborRank)
215 {
216 const auto [rangeStart, rangeEnd] = range;
217 const localIdx patchSize = rangeEnd - rangeStart;
218
219 auto valH = value_.copyToHost();
220
221 CommBuffer buf;
222 buf.rangeStart = rangeStart;
223 buf.patchSize = patchSize;
224 buf.sendBuf.resize(static_cast<std::size_t>(patchSize));
225 buf.recvBuf.resize(static_cast<std::size_t>(patchSize));
226 for (localIdx k = 0; k < patchSize; k++)
227 buf.sendBuf[static_cast<std::size_t>(k)] = valH.view()[rangeStart + k];
228
229 mpi::Environment mpiEnv;
230 MPI_Request sendReq, recvReq;
231 MPI_Isend(
232 buf.sendBuf.data(),
233 static_cast<int>(patchSize) * static_cast<int>(sizeof(ValueType)),
234 MPI_BYTE,
235 neighborRank,
236 0,
237 mpiEnv.comm(),
238 &sendReq
239 );
240 MPI_Irecv(
241 buf.recvBuf.data(),
242 static_cast<int>(patchSize) * static_cast<int>(sizeof(ValueType)),
243 MPI_BYTE,
244 neighborRank,
245 0,
246 mpiEnv.comm(),
247 &recvReq
248 );
249 communicating_ = true;
250 requests_.push_back(sendReq);
251 requests_.push_back(recvReq);
252 commBuffers_.push_back(std::move(buf));
253 }
254
255 bool isComplete()
256 {
257 if (requests_.empty() || !communicating_) return true;
258 int flag = 0;
259 MPI_Testall(
260 static_cast<int>(requests_.size()), requests_.data(), &flag, MPI_STATUSES_IGNORE
261 );
262 auto complete = flag != 0;
263 if (complete)
264 {
265 communicating_ = false;
266 }
267 return complete;
268 }
269
270 void waitAll()
271 {
272 if (requests_.empty() || !communicating_) return;
273 MPI_Waitall(static_cast<int>(requests_.size()), requests_.data(), MPI_STATUSES_IGNORE);
274 auto valH = value_.copyToHost();
275 for (const auto& buf : commBuffers_)
276 {
277 for (localIdx k = 0; k < buf.patchSize; k++)
278 valH.view()[buf.rangeStart + k] = buf.recvBuf[static_cast<std::size_t>(k)];
279 }
280 value_ = valH.copyToExecutor(exec_);
281 requests_.clear();
282 communicating_ = false;
283 commBuffers_.clear();
284 }
