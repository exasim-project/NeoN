199#ifdef NF_WITH_MPI_SUPPORT
200 void communicate(std::pair<localIdx, localIdx> range, int neighborRank)
201 {
202 const auto [rangeStart, rangeEnd] = range;
203 const localIdx patchSize = rangeEnd - rangeStart;
204
205 auto valH = value_.copyToHost();
206 std::vector<ValueType> sendBuf(static_cast<std::size_t>(patchSize));
207 std::vector<ValueType> recvBuf(static_cast<std::size_t>(patchSize));
208 for (localIdx k = 0; k < patchSize; k++)
209 sendBuf[static_cast<std::size_t>(k)] = valH.view()[rangeStart + k];
210
211 mpi::Environment mpiEnv;
212 MPI_Sendrecv(
213 sendBuf.data(),
214 static_cast<int>(patchSize) * static_cast<int>(sizeof(ValueType)),
215 MPI_BYTE,
216 neighborRank,
217 0,
218 recvBuf.data(),
219 static_cast<int>(patchSize) * static_cast<int>(sizeof(ValueType)),
220 MPI_BYTE,
221 neighborRank,
222 0,
223 mpiEnv.comm(),
224 MPI_STATUS_IGNORE
225 );
226
227 for (localIdx k = 0; k < patchSize; k++)
228 valH.view()[rangeStart + k] = recvBuf[static_cast<std::size_t>(k)];
229
230 value_ = valH.copyToExecutor(exec_);
231 }
232#endif
