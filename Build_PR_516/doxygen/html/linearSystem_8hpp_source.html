205 typename BoundaryMatrixType = COOMatrix<ValueType, localIdx>>
206LinearSystem<ValueType, SystemMatrixType, BoundaryMatrixType> createEmptyLinearSystem(
207 const UnstructuredMesh& mesh,
208 std::shared_ptr<MeshIterationStrategy> strategy = std::make_shared<FaceBasedIterator>()
209)
210{
211 auto [sp, mi] =
212 createSparsityPatternFaceToMatrixAddress<typename SystemMatrixType::MatrixSparsityType>(mesh

215 createBoundarySparsityPattern<typename BoundaryMatrixType::MatrixSparsityType>(mesh, *mi);

217 SystemMatrixType(Vector<ValueType>(sp->exec(), sp->nnz(), zero<ValueType>()), sp, mi),

220 Vector<ValueType>(bSp->exec(), bSp->nnz(), zero<ValueType>()),
221 strategy
222 };

227template<typename ValueType>
228inline la::LinearSystem<ValueType>
229removeBoundaryContributions(const la::LinearSystem<ValueType>& lsIn)
230{
231 auto ls = la::LinearSystem<ValueType>(lsIn);
232 auto lsView = ls.view();
233 auto& matrix = lsView.matrix;
234 auto& rhs = lsView.rhs;
235 auto& bMatrix = lsView.boundaryMatrix;
236 auto& bRhs = lsView.boundaryRhs;
237
238 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
239
240 parallelFor(
241 ls.exec(),
242 {0, bMatrix.values.size()},
243 NEON_LAMBDA(const localIdx facei) {
244 const auto celli = bMatrix.sparsity.rowOffs[facei]; // cell index stored in rowOffs
245 Kokkos::atomic_add(&matrix.values[ma.diagIdx(celli)], bMatrix.values[facei]);
246 Kokkos::atomic_add(&rhs[celli], bRhs[facei]);
247 },
248 "removeBoundaryContributions"
249 );
250
251 return ls;
252}
