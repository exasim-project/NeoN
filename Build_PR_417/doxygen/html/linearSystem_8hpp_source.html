159template<typename ValueTypeIn, typename IndexTypeIn, typename ValueTypeOut, typename IndexTypeOut>
160LinearSystem<ValueTypeOut, IndexTypeOut>
161convertLinearSystem(const LinearSystem<ValueTypeIn, IndexTypeIn>& ls)
162{
163 auto exec = ls.exec();
164 Vector<ValueTypeOut> convertedRhs(exec, ls.rhs().data(), ls.rhs().size());
165 return {
166 convert<ValueTypeIn, IndexTypeIn, ValueTypeOut, IndexTypeOut>(exec, ls.view.matrix),
167 convertedRhs,
168 ls.sparsityPattern()
169 };

172/*@brief helper function that creates a zero initialised linear system based on given sparsity
173 * pattern
174 */
175template<typename ValueType, typename IndexType>
176LinearSystem<ValueType, IndexType> createEmptyLinearSystem(
177 const UnstructuredMesh& mesh,
178 std::shared_ptr<const SparsityPattern<IndexType>> sparsity,
179 std::shared_ptr<const SparsityPattern<IndexType>> boundarySparsity
180)
181{
182 const auto& exec = mesh.exec();
183 localIdx rows {sparsity->rows()};
184 localIdx nnzs {sparsity->nnz()};
185 localIdx nBoundaryFaces {mesh.boundaryMesh().faceCells().size()};
186
187 return {
188 CSRMatrix<ValueType, IndexType> {
189 Vector<ValueType>(exec, nnzs, zero<ValueType>()), sparsity
190 },
191 Vector<ValueType> {exec, rows, zero<ValueType>()},
192 CSRMatrix<ValueType, IndexType> {
193 Vector<ValueType>(exec, nBoundaryFaces, zero<ValueType>()), sparsity
194 },
195 Vector<ValueType> {exec, nBoundaryFaces, zero<ValueType>()},
196 };
197}
