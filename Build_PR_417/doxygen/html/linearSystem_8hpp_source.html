166/*@brief helper function that creates a zero initialised linear system based on given sparsity
167 * pattern
168 */
169template<typename ValueType, typename SparsityType>
170LinearSystem<ValueType, Matrix<ValueType, SparsityType>> createEmptyLinearSystem(
171 const UnstructuredMesh& mesh,
172 std::shared_ptr<const SparsityType> sparsity,
173 std::shared_ptr<const SparsityType> bsparsity
174)
175{
176 NF_ASSERT(mesh.nCells() == sparsity->rows(), "Inconsistent sparsity pattern");
177
178 const auto& exec = mesh.exec();
179 localIdx rows {sparsity->rows()};
180 localIdx nnzs {sparsity->nnz()};
181 localIdx nBoundaryFaces {mesh.boundaryMesh().faceCells().size()};
182
183 return {
184 {Vector<ValueType>(exec, nnzs, zero<ValueType>()), sparsity},
185 {exec, rows, zero<ValueType>()},
186 {Vector<ValueType>(exec, nBoundaryFaces, zero<ValueType>()), bsparsity},
187 {exec, bsparsity->rows(), zero<ValueType>()}
188 };
189}
