60template<typename ValueType>
61class Processor : public VolumeBoundaryFactory<ValueType>::template Register<Processor<ValueType>>
62{
63 using Base = VolumeBoundaryFactory<ValueType>::template Register<Processor<ValueType>>;
64
65public:
66
67 using ProcessorType = Processor<ValueType>;
68
69 Processor(const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID)
70 : Base(mesh, dict, patchID, {.assignable = true}),
71 nbrRank_(static_cast<int>(mesh.boundaryMesh().neighbourRank(
72 )[static_cast<size_t>(patchID)]))
73 {
74 // store the local cell indices adjacent to this processor patch
75 const auto faceCellsHost = mesh.boundaryMesh().faceCells().copyToHost();
76 const auto faceCellsView = faceCellsHost.view();
77 const localIdx start = this->start_;
78 const localIdx end = this->end_;
79 const auto patchSize = static_cast<std::size_t>(end - start);
80 faceOwnerCells_.resize(patchSize);
81 for (std::size_t i = 0; i < patchSize; ++i)
82 faceOwnerCells_[i] = faceCellsView[start + static_cast<localIdx>(i)];
83 }
84
85 virtual void correctBoundaryCondition([[maybe_unused]] Field<ValueType>& domainVector) final
86 {
87#ifdef NF_WITH_MPI_SUPPORT
88 if (nbrRank_ < 0) return;
89
90 const localIdx patchSize = static_cast<localIdx>(faceOwnerCells_.size());
91 if (patchSize == 0) return;
92
93 // pack: gather internal cell values for the faces of this patch into a device vector
94 Vector<localIdx> faceOwnerDev(domainVector.exec(), faceOwnerCells_.data(), patchSize);
95 Vector<ValueType> sendDev(domainVector.exec(), patchSize);
96 {
97 const auto intView = domainVector.internalVector().view();
98 const auto faceOwnerV = faceOwnerDev.view();
99 auto sendView = sendDev.view();
100 parallelFor(
101 domainVector.exec(),
102 {0, patchSize},
103 NEON_LAMBDA(const localIdx i) { sendView[i] = intView[faceOwnerV[i]]; }
104 );
105 }
106 fence(domainVector.exec());
107
108 // copy packed values to host for MPI
109 auto sendHost = sendDev.copyToHost();
110
111 std::vector<ValueType> recvBuf(static_cast<std::size_t>(patchSize));
112 mpi::Environment mpiEnv;
113 MPI_Sendrecv(
114 sendHost.data(),
115 patchSize * static_cast<int>(sizeof(ValueType)),
116 MPI_BYTE,
117 nbrRank_,
118 0,
119 recvBuf.data(),
120 patchSize * static_cast<int>(sizeof(ValueType)),
121 MPI_BYTE,
122 nbrRank_,
123 0,
124 mpiEnv.comm(),
125 MPI_STATUS_IGNORE
126 );
127
128 // copy received values to device and write into the boundary data patch range
129 Vector<ValueType> recvDev(domainVector.exec(), recvBuf.data(), patchSize);
130 {
131 const localIdx start = this->start_;
132 auto bDataView = domainVector.boundaryData().value().view();
133 const auto recvView = recvDev.view();
134 parallelFor(
135 domainVector.exec(),
136 {0, patchSize},
137 NEON_LAMBDA(const localIdx i) { bDataView[start + i] = recvView[i]; }
138 );
139 }
140 fence(domainVector.exec());
141#endif
142 }
