14/* @brief Map a temperature field's boundary data onto an energy field's boundary data
15 * through the constant-specific-heat relation `he = Cv * (T - Tref)`.
16 *
17 * Mirrors what OpenFOAM's basicThermo does internally for every solver that transports `he`
18 * rather than `T` directly (fixedEnergyFvPatchScalarField/gradientEnergyFvPatchScalarField,
19 * src/thermophysicalModels/basic/derivedFvPatchFields/): a solver that reads `0/T` BCs but
20 * transports `he` calls this once per outer iteration (after `T.correctBoundaryConditions()`,
21 * before assembling/solving the energy equation) to push the current `T` boundary state into
22 * `he`'s own boundary storage, then calls `he.correctBoundaryConditions()` so each patch's own
23 * BC type (FixedEnergy/GradientEnergy, see fixedEnergy.hpp/gradientEnergy.hpp) re-derives
24 * `he`'s boundary value from what was just written.
25 *
26 * `he = Cv*(T - Tref)` is affine, so the map is applied component-wise to both halves of the
27 * mixed-BC representation every NeoN boundary condition shares (refValue/refGrad/
28 * valueFraction, see e.g. fixedFluxPressure.hpp):
29 * - refValue (the Dirichlet/fixedValue-type target): he.refValue = Cv*(T.refValue - Tref)
30 * - refGrad (the Neumann/gradient-type target): he.refGrad = Cv*T.refGrad
31 * - valueFraction (the Dirichlet/Neumann blend): copied through unchanged -- the blend
32 * itself doesn't depend on which units are being transported.
33 * This correctly reproduces OpenFOAM's fixedEnergy BC exactly (a pure Dirichlet target) and
34 * its gradientEnergy BC exactly for constant, spatially-uniform Cv (where the general
35 * chain-rule gradient map collapses to this simple scaling); a spatially-varying or
36 * temperature-dependent Cv would need the per-face Cv as a Field rather than a scalar here --
37 * out of scope for this constant-Cv implementation.
38 *
39 * Internal field values are untouched entirely -- this operates on boundary data only (it
40 * never needed a whole Field, just the refValue/refGrad/valueFraction triple), so it takes
41 * BoundaryData directly: call as
42 * `mapEnergyBoundaryFromTemperature(T.boundaryData(), he.boundaryData(), Cv, Tref)`. Call
43 * `he.correctBoundaryConditions()` afterward to materialise the values.
44 */
45template<typename ValueType>
46void mapEnergyBoundaryFromTemperature(
47 const BoundaryData<ValueType>& temperature,
48 BoundaryData<ValueType>& energy,
49 scalar Cv,
50 scalar Tref
51)
52{
53 auto [heRefValue, heRefGrad, heValueFraction] =
54 views(energy.refValue(), energy.refGrad(), energy.valueFraction());
55 auto [TRefValue, TRefGrad, TValueFraction] =
56 views(temperature.refValue(), temperature.refGrad(), temperature.valueFraction());
57
58 NeoN::parallelFor(
59 energy.exec(),
60 {0, energy.refValue().size()},
61 NEON_LAMBDA(const localIdx i) {
62 heRefValue[i] = Cv * (TRefValue[i] - Tref);
63 heRefGrad[i] = Cv * TRefGrad[i];
64 heValueFraction[i] = TValueFraction[i];
65 },
66 "mapEnergyBoundaryFromTemperature"
67 );
68}
