144 [[nodiscard]] LinearSystem<ValueType, MatrixType> copyToHost() const
145 {
146 if (faceToMatrixAddress_ == nullptr)

149 matrix_.copyToHost(),

151 boundaryMatrix_.copyToHost(),
152 boundaryRhs_.copyToHost(),
153 {}
154 };
155 }
156 auto mi = std::make_shared<FaceToMatrixAddress<LinearSystemIndexType>>(
157 faceToMatrixAddress_->ownerOffset().copyToHost(),
158 faceToMatrixAddress_->neighbourOffset().copyToHost(),
159 faceToMatrixAddress_->diagOffset().copyToHost(),
160 std::make_shared<SparsityPattern<LinearSystemIndexType>>(
161 faceToMatrixAddress_->sparsityPattern()->copyToHost()
162 ),
163 std::make_shared<SparsityPattern<LinearSystemIndexType>>(
164 faceToMatrixAddress_->boundarySparsityPattern()->copyToHost()
165 )
166 );
167 return {
168 matrix_.copyToHost(),
169 rhs_.copyToHost(),
170 boundaryMatrix_.copyToHost(),
171 boundaryRhs_.copyToHost(),
172 mi
173 };
