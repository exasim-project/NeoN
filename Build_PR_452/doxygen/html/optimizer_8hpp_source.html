49 auto lapOperator =

53 .getConfig();
54
55 dsl::SpatialOperator<typename ExpressionType::ExpressionValueType> divLapOperator =
56 finiteVolume::cellCentred::GaussGreenDivLaplacian<
57 typename ExpressionType::ExpressionValueType>(
58 expr.exec(), divOperator, lapOperator
59 );
60 out.addOperator(divLapOperator);

63 out.dropOperator("LaplacianOperator");
