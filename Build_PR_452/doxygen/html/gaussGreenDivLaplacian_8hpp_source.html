81 void read(const Input& input)
82 {
83 const UnstructuredMesh& mesh = this->field_.mesh();
84 TokenList laplTokens;
85 TokenList divTokens;
86 if (std::holds_alternative<Dictionary>(input))
87 {
88 auto dict = std::get<Dictionary>(input);
89 std::string lapSchemeName = "laplacian(" + gamma_.name + "," + this->field_.name + ")";
90 std::string divSchemeName = "div(" + flux_.name + "," + this->getVector().name + ")";
91 laplTokens = dict.subDict("laplacianSchemes").get<NeoN::TokenList>(lapSchemeName);
92 divTokens = dict.subDict("divSchemes").get<NeoN::TokenList>(divSchemeName);
93 }
94 else
95 {
96 NF_ERROR_EXIT("only dictionary input supported");
97 }
98 laplTokens.remove(0);
99 divTokens.remove(0);
100 lapSurfaceInterpolation_ =
101 std::make_shared<SurfaceInterpolation<ValueType>>(this->exec(), mesh, laplTokens);
102 divSurfaceInterpolation_ =
103 std::make_shared<SurfaceInterpolation<ValueType>>(this->exec(), mesh, divTokens);
104 laplTokens.remove(0);
105 faceNormalGradient_ =
106 std::make_shared<FaceNormalGradient<ValueType>>(this->exec(), mesh, laplTokens);
107 }
