199 void communicate(CommunicationPattern& commPattern)
200 {
201 auto mpiEnv = commPattern.env;
202 int commRanks = mpiEnv.sizeRank();
203
204 // auto boundaryMatrixMap = Vector<localIdx>(exec(), commPattern.boundaryMapVector);
205 auto nsp = faceToMatrixAddress_->nonLocalSparsityPattern();
206 auto rowToDiagonalMap = la::computeRowToDiagonalMap(nsp->rowOffs(), faceToMatrixAddress_);
207
208 // 1. copy bValues which need to be communicated into sendBuffer
209 auto commSize = commPattern.sendCounts[mpiEnv.sizeRank()];
210 auto recvBuffer = Vector<ValueType>(exec(), commSize);
211
212 // TODO compute using scan
213 auto sdispls = std::vector<int>(commRanks, 0);
214 for (int i = 1; i < sdispls.size(); i++)
215 {
216 auto prev = sdispls[i - 1];
217 sdispls[i] = commPattern.sendCounts[i - 1] + prev;
218 }
219
220 MPI_Alltoallv(
221 nonLocalMatrix_.values().data(),
222 commPattern.sendCounts.data(),
223 sdispls.data(),
224 mpi::getType<ValueType>(),
225 recvBuffer.data(),
226 commPattern.sendCounts.data(),
227 sdispls.data(),
228 mpi::getType<ValueType>(),
229 mpiEnv.comm()
230 );
231
232 std::cout << __FILE__ << ":" << __LINE__ << " rank " << mpiEnv.rank() << " recvBuffer "
233 << recvBuffer.view()[0] << " matrixValue " << matrix_.values().view()[9] << "\n";
234
235 // 3. apply received values to corresponding matrix
236 // add diagonal contributions
237 add(recvBuffer, rowToDiagonalMap, matrix_.values());
238 std::cout << __FILE__ << ":" << __LINE__ << " rank " << mpiEnv.rank() << " recvBuffer "
239 << recvBuffer.view()[0] << " matrixValue " << matrix_.values().view()[9] << "\n";
240 }

343template<typename ValueType>
344inline la::LinearSystem<ValueType>
345removeBoundaryContributions(const la::LinearSystem<ValueType>& lsIn)
346{
347 auto ls = la::LinearSystem<ValueType>(lsIn);
348 const auto matIt = ls.faceToMatrixAddress();
349 auto lsView = ls.view();
350 auto& matrix = lsView.matrix;
351 auto& rhs = lsView.rhs;
352 auto& bMatrix = lsView.boundaryMatrix;
353 auto& bRhs = lsView.boundaryRhs;
354 auto& mtx = ls.matrix();
355 const auto [diagOffs, rowOffs] = views(matIt->diagOffset(), mtx.rowOffs());
356
357 // Boundary matrix
358 parallelFor(
359 ls.exec(),
360 {0, bMatrix.values.size()},
361 NEON_LAMBDA(const localIdx facei) {
362 const auto celli = bMatrix.sparsity.rowOffs[facei]; // cell index stored in rowOffs
363 Kokkos::atomic_add(
364 &matrix.values[rowOffs[celli] + diagOffs[celli]], bMatrix.values[facei]
365 );
366 Kokkos::atomic_sub(&rhs[celli], bRhs[facei]);
367 },
368 "addBoundaryContributions"
369 );
370
371 //
372 auto nonLocalMatrix = ls.nonLocalMatrix().view();
373 parallelFor(
374 ls.exec(),
375 {0, nonLocalMatrix.values.size()},
376 NEON_LAMBDA(const localIdx facei) {
377 const auto celli =
378 nonLocalMatrix.sparsity.rowOffs[facei]; // cell index stored in rowOffs
379 Kokkos::atomic_add(
380 &matrix.values[rowOffs[celli] + diagOffs[celli]], nonLocalMatrix.values[facei]
381 );

385 "addBoundaryContributions"
386 );
387
388 return ls;

391inline void scaledInvDiagNegLUx(
392 const la::LinearSystem<Vec3>& ls, // In,
393 const Vector<Vec3>& a,
394 const Vector<Vec3>& aBound,
395 const UnstructuredMesh& mesh,
396 Vector<scalar>& rAU,
397 Vector<Vec3>& out
398)
399{
400 // auto ls = removeBoundaryContributions(lsIn);
401 auto& mtx = ls.matrix();
402
403 auto& vol = mesh.cellVolumes();
404 auto& bMesh = mesh.boundaryMesh();
405 NF_ASSERT(mtx.nRows() == a.size(), "Dimension mismatch");
406 NF_ASSERT(mtx.nRows() == out.size(), "Dimension mismatch");
407
408 const auto [rowOffsV, colIdxV, matrixV, rAUV, volV, aV, aBoundV, bV] = views(
409 mtx.sparsity()->rowOffs(),
410 ls.matrix().sparsity()->colIdxs(),
411 mtx.values(),
412 rAU,
413 vol,
414 a,
415 aBound,
416 ls.rhs()
417 );
418 const auto [nonLocalMtx, nonLocalRows] =
419 views(ls.nonLocalMatrix().values(), ls.nonLocalMatrix().rowOffs());
420 auto outV = out.view();
421
422 auto procFacesStart = bMesh.nBoundaryFaces();
423
424 // localIdx curNonLocalRow = 0;
425 parallelFor(
426 mtx.exec(),
427 {0, mtx.nRows()},
428 NEON_LAMBDA(const localIdx rowi) {
429 // local mtx first
430 outV[rowi] = zero<Vec3>();
431 for (auto i = rowOffsV[rowi]; i < rowOffsV[rowi + 1]; i++)
432 {
433 auto colI = colIdxV[i];
434 if (rowi == colI)
435 {
436 rAUV[rowi] = volV[rowi] / matrixV[i][0];
437 }
438 else
439 {
440 outV[rowi] -= matrixV[i] * aV[colI];
441 }
442 }
443
444 // check
445 // FIXME this scans everytime all boundary values
446 localIdx curNonLocalRow = 0;
447 auto nonLocalVal = zero<Vec3>();
448 for (auto i = 0; i < nonLocalRows.size(); i++)
449 {
450 if (nonLocalRows[i] == rowi)
451 {
452 nonLocalVal = nonLocalMtx[i];
453 curNonLocalRow = i;
454 // FIXME is the sign correct?
455 outV[rowi] -= nonLocalVal * aBoundV[procFacesStart + curNonLocalRow];
456 // curNonLocalRow = 0;
457 // break;
458 }
459 }
460
461 outV[rowi] += bV[rowi];
462 outV[rowi] *= rAUV[rowi] / volV[rowi];
463 }
464 );
465}
