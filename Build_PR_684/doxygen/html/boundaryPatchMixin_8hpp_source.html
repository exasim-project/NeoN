72 /* @brief returns dict["type"], throws if it is missing or not registered in Factory */

75 {
76 if (!dict.contains("type"))
77 {
78 NF_THROW("boundary condition on patch " << patchID << " has no 'type' entry");
79 }
80 const auto& type = dict.get<std::string>("type");
81 if (!Factory::table().contains(type))
82 {
83 std::string valid;
84 for (const auto& entry : Factory::table())
85 {
86 valid += " " + entry.first;
87 }
88 NF_THROW(

92 }
