257/*@brief helper function that creates a zero initialised linear system based on a given mesh
258 */
259template<
260 typename ValueType,
261 typename SystemMatrixType = CSRMatrix<ValueType, localIdx>,
262 typename BoundaryMatrixType = COOMatrix<ValueType, localIdx>>
263LinearSystem<ValueType, SystemMatrixType, BoundaryMatrixType> createEmptyLinearSystem(
264 const UnstructuredMesh& mesh,
265 std::shared_ptr<MeshIterationStrategy> strategy = std::make_shared<FaceBasedIterator>()
266)
267{
268 auto [sp, mi] =
269 createSparsityPatternFaceToMatrixAddress<typename SystemMatrixType::MatrixSparsityType>(mesh
270 );
271 auto bSp =
272 createBoundarySparsityPattern<typename BoundaryMatrixType::MatrixSparsityType>(mesh, *mi);
273 const auto exec = sp->exec();
274 const auto nCells = static_cast<localIdx>(mesh.nCells());
275 const auto nProcFaces = static_cast<localIdx>(mesh.nProcBoundaryFaces());
276 using IndexType = typename BoundaryMatrixType::MatrixSparsityType::SparsityIndexType;
277
278 Vector<IndexType> offDiagColIdxs(exec, nProcFaces, 0);
279 Vector<IndexType> offDiagRowIdxs(exec, nProcFaces, 0);
280
281#ifdef NF_WITH_MPI_SUPPORT
282 auto commPattern = computeCommunicationPattern(mesh);
283 if (nProcFaces > 0)
284 {
285 const localIdx nBoundaryFaces = static_cast<localIdx>(mesh.nBoundaryFaces());
286 const auto faceOwnersH = mesh.boundaryMesh().faceOwners().copyToHost();
287 const auto faceOwnersV = faceOwnersH.view();
288 Vector<IndexType> rowH(SerialExecutor {}, nProcFaces, 0);
289 Vector<IndexType> colH(SerialExecutor {}, nProcFaces, 0);
290 auto rowHV = rowH.view();
291 auto colHV = colH.view();
292 for (localIdx i = 0; i < nProcFaces; ++i)
293 {
294 rowHV[i] = faceOwnersV[nBoundaryFaces + i];
295 colHV[i] = static_cast<IndexType>(commPattern.recvIdx[static_cast<std::size_t>(i)]);
296 }
297 offDiagRowIdxs = rowH.copyToExecutor(exec);
298 offDiagColIdxs = colH.copyToExecutor(exec);
299 }
300#endif
301
302 auto offDiagSp = std::make_shared<const typename BoundaryMatrixType::MatrixSparsityType>(
303 std::move(offDiagColIdxs), std::move(offDiagRowIdxs), Dimensions {nCells, nCells}
304 );
305
306 LinearSystem<ValueType, SystemMatrixType, BoundaryMatrixType> ls {
307 SystemMatrixType(Vector<ValueType>(sp->exec(), sp->nnz(), zero<ValueType>()), sp, mi),
308 Vector<ValueType>(sp->exec(), sp->rows(), zero<ValueType>()),
309 BoundaryMatrixType(Vector<ValueType>(exec, nProcFaces, zero<ValueType>()), offDiagSp),
310 BoundaryMatrixType(Vector<ValueType>(bSp->exec(), bSp->nnz(), zero<ValueType>()), bSp),
311 Vector<ValueType>(bSp->exec(), bSp->nnz(), zero<ValueType>()),
312 strategy
313 };
314
315#ifdef NF_WITH_MPI_SUPPORT
316 ls.commPattern() = std::move(commPattern);
317#endif
318
319 return ls;
320}

324template<typename ValueType>
325inline la::LinearSystem<ValueType>
326removeBoundaryContributions(const la::LinearSystem<ValueType>& lsIn)
327{
328 auto ls = la::LinearSystem<ValueType>(lsIn);
329 auto lsView = ls.view();
330 auto& matrix = lsView.matrix;
331 auto& rhs = lsView.rhs;
332 auto& bMatrix = lsView.boundaryMatrix;
333 auto& bRhs = lsView.boundaryRhs;
334
335 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
336
337 parallelFor(
338 ls.exec(),
339 {0, bMatrix.values.size()},
340 NEON_LAMBDA(const localIdx facei) {
341 const auto celli = bMatrix.sparsity.rowOffs[facei]; // cell index stored in rowOffs
342 Kokkos::atomic_add(&matrix.values[ma.diagIdx(celli)], bMatrix.values[facei]);
343 Kokkos::atomic_add(&rhs[celli], bRhs[facei]);
344 },
345 "removeBoundaryContributions"
346 );
347
348 return ls;
349}
