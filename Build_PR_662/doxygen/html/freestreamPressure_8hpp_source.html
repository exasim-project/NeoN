23// Kernel behind freestreamPressure, OpenFOAM's freestreamPressureFvPatchScalarField (a mixed BC
24// with refValue = freestreamValue, refGrad = 0), the complement of freestreamVelocity:
25//
26// valueFraction = 0.5 + 0.5 (Up . nf) / |Up|
27//
28// with Up the patch value of the velocity field. Outflow (Up . nf > 0) fixes the pressure to
29// freestreamValue (valueFraction -> 1), inflow (Up . nf < 0) is zero-gradient (valueFraction -> 0),
30// and the blend is smooth in between. Without a velocity (hasU == false) every face is
31// zero-gradient, as the other flow-direction BCs degrade without their context.
32inline void setFreestreamPressureValue(
33 Field<NeoN::scalar>& domainVector,
34 const UnstructuredMesh& mesh,
35 std::pair<localIdx, localIdx> range,
36 NeoN::scalar freestreamValue,
37 View<const Vec3> boundaryU,
38 bool hasU
39)
40{
41 const auto internalV = domainVector.internalVector().view();
42 auto [refGradV, valueV, valueFractionV, refValueV, boundaryFaceOwnersV, faceAreaVectorV] =
43 views(
44 domainVector.boundaryData().refGrad(),
45 domainVector.boundaryData().value(),
46 domainVector.boundaryData().valueFraction(),
47 domainVector.boundaryData().refValue(),
48 mesh.boundaryMesh().faceOwners(),
49 mesh.boundaryMesh().faceNormals()
50 );
51 NeoN::parallelFor(
52 domainVector.exec(),
53 range,
54 NEON_LAMBDA(const localIdx i) {
55 const localIdx owner = boundaryFaceOwnersV[i];
56 const NeoN::scalar internal = internalV[owner];
57
58 NeoN::scalar valueFraction = NeoN::scalar(0);
59 if (hasU)
60 {
61 const Vec3 up = boundaryU[i];
62 const Vec3 sf = faceAreaVectorV[i];
63 const NeoN::scalar magSf = Kokkos::max(NeoN::mag(sf), NeoN::scalar(1e-300));
64 const NeoN::scalar upDotNf =
65 (up[0] * sf[0] + up[1] * sf[1] + up[2] * sf[2]) / magSf;
66 valueFraction = NeoN::scalar(0.5)
67 + NeoN::scalar(0.5) * upDotNf
68 / Kokkos::max(NeoN::mag(up), NeoN::scalar(1e-300));
69 }
70
71 valueFractionV[i] = valueFraction;
72 refValueV[i] = freestreamValue;
73 refGradV[i] = NeoN::zero<NeoN::scalar>();
74 valueV[i] =
75 freestreamValue * valueFraction + internal * (NeoN::scalar(1) - valueFraction);
76 },
77 "setFreestreamPressureValue"
78 );
79}
80
81}

83// freestreamPressure: far-field pressure that blends the freestream value with zero-gradient
84// according to the direction of the velocity at the patch, mirroring OpenFOAM's
85// freestreamPressure (a mixedFvPatchScalarField without the supersonic option).
86//
87// The velocity is looked up by name (dict key "U", default "U") from the BoundaryContext of the
88// context-aware correctBoundaryCondition overload; the plain overload has none and degenerates to
89// zeroGradient.
90//
91// Dictionary: freestreamValue (scalar, required), U (optional field name).
92class FreestreamPressure :
93 public VolumeBoundaryFactory<NeoN::scalar>::template Register<FreestreamPressure>
94{
95 using Base = VolumeBoundaryFactory<NeoN::scalar>::template Register<FreestreamPressure>;
96
97public:
98
99 using Base::correctBoundaryCondition;
100
101 // assignable = false, as OpenFOAM's mixed patch fields (constrainHbyA then sets HbyA to the
102 // patch value). fixesValue = false and
103 // mixed = true, as for freestreamVelocity.
104 FreestreamPressure(const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID)
105 : Base(mesh, dict, patchID, {.assignable = false, .fixesValue = false, .mixed = true}),
106 mesh_(mesh), freestreamValue_(dict.get<NeoN::scalar>("freestreamValue")),
107 UName_(dict.contains("U") ? dict.get<std::string>("U") : std::string("U"))
108 {}
109
110 // No context => no velocity: behave as zero-gradient.
111 virtual void correctBoundaryCondition(Field<NeoN::scalar>& domainVector) final
112 {
113 detail::setFreestreamPressureValue(
114 domainVector, mesh_, this->range(), freestreamValue_, View<const Vec3> {}, false
115 );
116 }
117
118 virtual void
119 correctBoundaryCondition(Field<NeoN::scalar>& domainVector, const BoundaryContext& ctx) final
120 {
121 if (ctx.hasVector(UName_))
122 {
123 const auto& U = ctx.vectorFieldPtr(UName_);
124 detail::setFreestreamPressureValue(
125 domainVector,
126 mesh_,
127 this->range(),
128 freestreamValue_,
129 U.boundaryData().value().view(),
130 true
131 );
132 }
133 else
134 {
135 correctBoundaryCondition(domainVector);
136 }
137 }
138
139 static std::string name() { return "freestreamPressure"; }
140
141 std::string getName() const override { return name(); }
142
143 static std::string doc()
144 {
145 return "Far-field pressure (OpenFOAM's freestreamPressure): valueFraction = 0.5 + 0.5 "
146 "(U.nf)/|U| from the patch velocity, so outflow takes freestreamValue and inflow "
147 "is zero-gradient, blending smoothly in between.";
148 }
149
150 static std::string schema() { return "none"; }
151
152 virtual std::unique_ptr<VolumeBoundaryFactory<NeoN::scalar>> clone() const final
153 {
154 return std::make_unique<FreestreamPressure>(*this);
155 }
156
157private:
158
159 const UnstructuredMesh& mesh_;
160 NeoN::scalar freestreamValue_;
161 std::string UName_;
162};
