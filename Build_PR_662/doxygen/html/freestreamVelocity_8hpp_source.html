22// Kernel behind freestreamVelocity, OpenFOAM's freestreamVelocityFvPatchVectorField (a mixed BC
23// with refValue = freestreamValue, refGrad = 0):
24//
25// valueFraction = 0.5 - 0.5 (Up . nf) / |Up|
26//
27// where Up is the patch's OWN current value (the previous evaluation, as OpenFOAM reads *this in
28// updateCoeffs()). The blend is smooth between the two limits:
29// outflow (Up . nf > 0): valueFraction -> 0, zero-gradient
30// inflow (Up . nf < 0): valueFraction -> 1, fixed to freestreamValue
31// The evaluated face value is valueFraction*freestreamValue + (1 - valueFraction)*owner cell.
32//
33// A zero patch value would make the ratio 0/0. OpenFOAM starts the patch at freestreamValue, so
34// the direction test falls back to freestreamValue there.
35inline void setFreestreamVelocityValue(
36 Field<Vec3>& domainVector,
37 const UnstructuredMesh& mesh,
38 std::pair<localIdx, localIdx> range,
39 Vec3 freestreamValue
40)
41{
42 const auto internalV = domainVector.internalVector().view();
43 auto [refGradV, valueV, valueFractionV, refValueV, boundaryFaceOwnersV, faceAreaVectorV] =
44 views(
45 domainVector.boundaryData().refGrad(),
46 domainVector.boundaryData().value(),
47 domainVector.boundaryData().valueFraction(),
48 domainVector.boundaryData().refValue(),
49 mesh.boundaryMesh().faceOwners(),
50 mesh.boundaryMesh().faceNormals()
51 );
52 NeoN::parallelFor(
53 domainVector.exec(),
54 range,
55 NEON_LAMBDA(const localIdx i) {
56 const localIdx owner = boundaryFaceOwnersV[i];
57 const Vec3 internal = internalV[owner];
58
59 Vec3 up = valueV[i];
60 NeoN::scalar magUp = NeoN::mag(up);
61 if (!(magUp > NeoN::scalar(1e-300)))
62 {
63 up = freestreamValue;
64 magUp = NeoN::mag(up);
65 }
66
67 // boundaryMesh().faceNormals() is the face area vector Sf; nf = Sf/|Sf|
68 const Vec3 sf = faceAreaVectorV[i];
69 const NeoN::scalar magSf = Kokkos::max(NeoN::mag(sf), NeoN::scalar(1e-300));
70 const NeoN::scalar upDotNf = (up[0] * sf[0] + up[1] * sf[1] + up[2] * sf[2]) / magSf;
71
72 const NeoN::scalar valueFraction =
73 NeoN::scalar(0.5)
74 - NeoN::scalar(0.5) * upDotNf / Kokkos::max(magUp, NeoN::scalar(1e-300));
75
76 valueFractionV[i] = valueFraction;
77 refValueV[i] = freestreamValue;
78 refGradV[i] = NeoN::zero<Vec3>();
79 valueV[i] =
80 freestreamValue * valueFraction + internal * (NeoN::scalar(1) - valueFraction);
81 },
82 "setFreestreamVelocityValue"
83 );
84}
85
86}

88// freestreamVelocity: far-field velocity that blends the freestream value with zero-gradient
89// according to the direction of the patch velocity relative to the face normal. It mirrors
90// OpenFOAM's freestreamVelocity (a mixedFvPatchVectorField). Unlike inletOutlet it does not need
91// a flux: the direction comes from the patch's own previous value, so the plain and the
92// context-aware overload behave the same.
93//
94// Dictionary: freestreamValue (Vec3, required).
95class FreestreamVelocity :
96 public VolumeBoundaryFactory<NeoN::Vec3>::template Register<FreestreamVelocity>
97{
98 using Base = VolumeBoundaryFactory<NeoN::Vec3>::template Register<FreestreamVelocity>;
99
100public:
101
102 using Base::correctBoundaryCondition;
103
104 // assignable = false, as OpenFOAM's mixed patch fields (constrainHbyA then sets HbyA to the
105 // patch value). fixesValue = false and
106 // mixed = true: the per-face valueFraction drives the operators via the mixed-BC kernels, and
107 // OpenFOAM's mixed patches report fixesValue() = true (see BoundaryAttributes::mixed).
108 FreestreamVelocity(const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID)
109 : Base(mesh, dict, patchID, {.assignable = false, .fixesValue = false, .mixed = true}),
110 mesh_(mesh), freestreamValue_(dict.get<NeoN::Vec3>("freestreamValue"))
111 {}
112
113 virtual void correctBoundaryCondition(Field<NeoN::Vec3>& domainVector) final
114 {
115 detail::setFreestreamVelocityValue(domainVector, mesh_, this->range(), freestreamValue_);
116 }
117
118 virtual void
119 correctBoundaryCondition(Field<NeoN::Vec3>& domainVector, const BoundaryContext&) final
120 {
121 correctBoundaryCondition(domainVector);
122 }
123
124 static std::string name() { return "freestreamVelocity"; }
125
126 std::string getName() const override { return name(); }
127
128 static std::string doc()
129 {
130 return "Far-field velocity (OpenFOAM's freestreamVelocity): valueFraction = 0.5 - 0.5 "
131 "(U.nf)/|U| from the patch's own previous value, so inflow takes freestreamValue "
132 "and outflow is zero-gradient, blending smoothly in between.";
133 }
134
135 static std::string schema() { return "none"; }
136
137 virtual std::unique_ptr<VolumeBoundaryFactory<NeoN::Vec3>> clone() const final
138 {
139 return std::make_unique<FreestreamVelocity>(*this);
140 }

142private:
143
144 const UnstructuredMesh& mesh_;
145 NeoN::Vec3 freestreamValue_;
146};
