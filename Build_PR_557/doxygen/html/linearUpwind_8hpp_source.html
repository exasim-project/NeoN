46/* @brief computational kernel to perform a linearUpwind interpolation
47** from a source volume field to a surface field. The face value is the upwind
48** cell value plus a gradient-based correction:
49**
50** phi_f = phi_U + (Cf - C_U) & grad(phi)_U
51**
52** where U is the upwind cell (owner if faceFlux >= 0, neighbour otherwise),
53** Cf is the face centre and C_U the upwind cell centre. The gradient is computed
54** internally with a Gauss-Green scheme. This reduces to plain upwind where the
55** field is locally constant and is second-order accurate for smooth fields.
56**
57** @pre src.correctBoundaryConditions() must have been called by the caller so that
58** processor-boundary ghost values are up to date before the gradient is computed.
59** @param src the input field
60** @param flux the face flux determining the upwind direction
61** @param faceDeltaOwner Cf - C_owner per internal face (from the geometry scheme)
62** @param faceDeltaNeighbour Cf - C_neighbour per internal face (from the geometry scheme)
63** @param dst the target surface field
64*/
65template<typename ValueType>
66void computeLinearUpwindInterpolation(
67 const VolumeField<ValueType>& src,
68 const SurfaceField<scalar>& flux,
69 const SurfaceField<Vec3>& faceDeltaOwner,
70 const SurfaceField<Vec3>& faceDeltaNeighbour,
71 SurfaceField<ValueType>& dst
72);
73
74/* @brief computes only the gradient correction part of the linearUpwind face value,
75** `(Cf - C_U) & grad(phi)_U`, into dst (without the upwind cell value). Boundary faces are set to
76** zero, matching OpenFOAM which corrects only coupled patches. Used for the implicit deferred
77** correction (the explicit RHS source added by the divergence operator).
78**
79** @pre src.correctBoundaryConditions() must have been called by the caller so that
80** processor-boundary ghost values are up to date before the gradient is computed.
81*/
82template<typename ValueType>
83void computeLinearUpwindCorrection(
84 const VolumeField<ValueType>& src,
85 const SurfaceField<scalar>& flux,
86 const SurfaceField<Vec3>& faceDeltaOwner,
87 const SurfaceField<Vec3>& faceDeltaNeighbour,
88 SurfaceField<ValueType>& dst
89);

91template<typename ValueType>
92class LinearUpwind :
93 public SurfaceInterpolationFactory<ValueType>::template Register<LinearUpwind<ValueType>>
94{
95 using Base = SurfaceInterpolationFactory<ValueType>::template Register<LinearUpwind<ValueType>>;
96
97public:
98
99 LinearUpwind(const Executor& exec, const UnstructuredMesh& mesh, Input input)
100 : Base(exec, mesh), geometryScheme_(GeometryScheme::readOrCreate(mesh)),
101 gradSchemeName_(readGradSchemeName(input)) {};
102
103 static std::string name() { return "linearUpwind"; }
104
105 static std::string doc() { return "linearUpwind interpolation"; }
106
107 static std::string schema() { return "none"; }
108
109 void interpolate(
110 [[maybe_unused]] const VolumeField<ValueType>& src,
111 [[maybe_unused]] SurfaceField<ValueType>& dst
112 ) const override
113 {
114 NF_ERROR_EXIT("linearUpwind interpolation scheme requires a faceFlux");
115 }
116
117 void interpolate(
118 const SurfaceField<scalar>& flux,
119 const VolumeField<ValueType>& src,
120 SurfaceField<ValueType>& dst
121 ) const override
122 {
123 computeLinearUpwindInterpolation(
124 src, flux, geometryScheme_->faceDeltaOwner(), geometryScheme_->faceDeltaNeighbour(), dst
125 );
126 }

128 // linearUpwind shares upwind's implicit weights; the gradient correction is an explicit
129 // (deferred) term applied through interpolate() above, mirroring OpenFOAM's corrected scheme.
130 void weight(const VolumeField<ValueType>&, SurfaceField<scalar>&) const override
131 {
132 NF_ERROR_EXIT("linearUpwind interpolation scheme requires a faceFlux");
133 }
134
135 void weight(
136 const SurfaceField<scalar>& faceFlux,
137 const VolumeField<ValueType>& src,
138 SurfaceField<scalar>& weights
139 ) const override
140 {
141 computeUpwindInterpolationWeights(faceFlux, src, weights);
142 }
143
144 // linearUpwind carries an explicit gradient correction beyond the upwind weights.
145 bool corrected() const override { return true; }
146
147 void correction(
148 const SurfaceField<scalar>& faceFlux,
149 const VolumeField<ValueType>& src,
150 SurfaceField<ValueType>& corr
151 ) const override
152 {
153 computeLinearUpwindCorrection(
154 src,
155 faceFlux,
156 geometryScheme_->faceDeltaOwner(),
157 geometryScheme_->faceDeltaNeighbour(),
158 corr
159 );
160 }
161
162 InlineWeightKernel inlineWeightKernel(const SurfaceField<scalar>& /*flux*/) const override
163 {
164 return UpwindInlineKernel {};
165 }
166
167 std::unique_ptr<SurfaceInterpolationFactory<ValueType>> clone() const override
168 {
169 return std::make_unique<LinearUpwind>(*this);
170 }

172private:
173
174 // The OpenFOAM scheme spec is "linearUpwind <gradScheme>"; NeoN currently has a single
175 // (Gauss-Green) gradient scheme, so the name is read for spec-compatibility but not used to
176 // select a scheme. Returns a default when absent.
177 static std::string readGradSchemeName(Input& input)
178 {
179 if (std::holds_alternative<NeoN::TokenList>(input))
180 {
181 auto& tokens = std::get<NeoN::TokenList>(input);
182 if (tokens.peekIs<std::string>())
183 {
184 return tokens.next<std::string>();
185 }
186 }
187 return "Gauss";
188 }

190 const std::shared_ptr<GeometryScheme> geometryScheme_;
191 std::string gradSchemeName_;
192};
