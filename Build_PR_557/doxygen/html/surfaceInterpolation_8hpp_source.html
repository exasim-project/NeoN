37 static std::unique_ptr<SurfaceInterpolationFactory<ValueType>>
38 create(const Executor& exec, const UnstructuredMesh& uMesh, const Input& inputs)
39 {
40 // input is dictionary the key is "interpolation"
41 std::string key =
42 (std::holds_alternative<NeoN::Dictionary>(inputs))
43 ? std::get<NeoN::Dictionary>(inputs).get<std::string>("surfaceInterpolation")
44 : std::get<NeoN::TokenList>(inputs).next<std::string>();

46 SurfaceInterpolationFactory<ValueType>::keyExistsOrError(key);

74 /* @brief Whether this scheme adds an explicit (deferred) correction on top of its implicit
75 * weights, e.g. linearUpwind. For corrected schemes interpolate() returns the weighted value
76 * plus correction(); implicit assemblers add surfaceIntegrate(faceFlux*correction()) to the
77 * rhs.

79 virtual bool corrected() const { return false; }

81 /* @brief The explicit correction part of interpolate(), i.e. interpolate() minus the value
