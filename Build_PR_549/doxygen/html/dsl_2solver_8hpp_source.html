30template<typename VectorType, typename IndexType>
31la::SolverStats iterativeSolveImpl(
32 Expression<typename VectorType::ElementType>& exp,
33 la::LinearSystem<typename VectorType::ElementType>& ls,
34 VectorType& solution,
35 scalar t,
36 scalar dt,
37 const Dictionary& fvSchemes,
38 const Dictionary& fvSolution,
39 std::vector<const PostAssemblyBase<typename VectorType::ElementType, IndexType>*> ps = {}
40)
41{
42 auto optExp = optimize(exp);
43 optExp.read(fvSchemes);
44 optExp.assemble(t, dt, ls, ps);
45
46 // TODO move that to expression explicit operation or
47 // into functor ?
48 // subtract the explicit source term from the rhs
49 auto expTmp = optExp.explicitOperation(solution.mesh().nCells());
50 auto [vol, expSource, rhs] = views(solution.mesh().cellVolumes(), expTmp, ls.rhs());
51 parallelFor(
52 solution.exec(),
53 {0, rhs.size()},
54 NEON_LAMBDA(const localIdx i) { rhs[i] -= expSource[i] * vol[i]; }
55 );
56
57 auto solver = la::Solver(solution.exec(), fvSolution);
58 fence(solution.exec());
59
60 // Do some sanity checks before trying to solve
61 NF_ASSERT(ls.exec() == solution.exec(), "Executors are not the same");
62 return solver.solve(ls, solution.internalVector());
63}

83/* @brief solve an expression
84 *
85 * @param exp - Expression which is to be solved/updated.
86 * @param solution - Solution field, where the solution will be 'written to'.
87 * @param t - the time at the start of the time step.
88 * @param dt - time step for the temporal integration
89 * @param fvSchemes - Dictionary containing spatial operator and time integration properties
90 * @param fvSolution - Dictionary containing linear solver properties
91 * @param p - A chainable functor that performs manipulations on the assembled system
92 */
93template<typename VectorType, typename IndexType>
94la::SolverStats solve(
95 Expression<typename VectorType::ElementType, IndexType>& exp,
96 VectorType& solution,
97 scalar t,
98 scalar dt,
99 const Dictionary& fvSchemes,
100 const Dictionary& fvSolution,
101 std::vector<const PostAssemblyBase<typename VectorType::ElementType, IndexType>*> p = {}
102)
103{
104 if (exp.temporalOperators().size() == 0 && exp.spatialOperators().size() == 0)
105 {
106 NF_ERROR_EXIT("No temporal or implicit terms to solve.");
107 }
108 exp.read(fvSchemes);
109 auto integrator = timeIntegration::TimeIntegration<VectorType>(
110 fvSchemes.subDict("timeIntegration"), fvSolution
111 );
112
113 if (exp.temporalOperators().size() > 0 && integrator.explicitIntegration())
114 {
115 // integrate equations in time
116 integrator.solve(exp, solution, t, dt);
117 return {{.numIter = -1, .initResNorm = 0, .finalResNorm = 0, .solveTime = 0}};
118 }
119 else
120 {
121 return detail::iterativeSolveImpl(exp, solution, t, dt, fvSolution, p);
122 }
123}

128//
129// Componentwise helpers. NeoN's working tree does NOT provide the cmpt* /
130// componentMultiply family from the feat/underRelax reference, so the few
131// componentwise ops the kernel needs are defined inline here over Vec3[i]
132// using Kokkos::abs / Kokkos::max. zero<T>()/one<T>() already exist and are
133// reused as-is. copySign/componentCopySign are ported VERBATIM from the
134// reference (their definitions are sound; only their kernel call site was
135// dropped, which is the sign-drop regression this kernel must NOT reproduce).

138KOKKOS_INLINE_FUNCTION scalar copySign(const scalar mag, const scalar s)
139{
140 return (s >= 0) ? mag : -mag;
141}
142
144KOKKOS_INLINE_FUNCTION scalar componentCopySign(const scalar mag, const scalar s)
145{
146 return copySign(mag, s);
147}

150KOKKOS_INLINE_FUNCTION Vec3 componentCopySign(const Vec3& mag, const Vec3& s)
151{
152 return Vec3(copySign(mag[0], s[0]), copySign(mag[1], s[1]), copySign(mag[2], s[2]));
153}

179/* @brief Apply equation (matrix) under-relaxation to an assembled LinearSystem.
180 *
181 * OpenFOAM-parity path: NeoN bakes boundary contributions permanently into
182 * the CSR diagonal at assembly time, so the augmented diagonal
183 * `D_aug = matrix.values[diagIdx(cell)]` already contains the boundary diagonal.
184 * OpenFOAM `fvMatrix::relax(alpha)` divides the ENTIRE augmented diagonal (internal +
185 * boundary) by alpha: it adds the boundary internalCoeffs to the diagonal, clamps for
186 * dominance, then `D /= alpha` (the boundary is removed from the stored diag afterwards
187 * but re-added un-divided at solve, netting the same augmented value). This kernel
188 * therefore relaxes the augmented diagonal DIRECTLY — no internal-only reconstruction.
189 * Per cell:
190 *
191 * D_aug : matrix.values[diagIdx(cell)] (augmented, boundary-baked)
192 * D_dom : max(mag(D_aug), sumMagOffDiag[cell]) (dominance clamp on the augmented diag)
193 * D_relaxed : componentCopySign(D_dom / alpha, D_aug) (scale whole augmented diag by 1/alpha)
194 * write : matrix.values[diagIdx(cell)] = D_relaxed
195 * source : rhs[cell] += (D_relaxed - D_aug) * psi_prev[cell]
196 *
197 * History (boundary-diagonal relaxation fix, 2026-06-16): an earlier path reconstructed the
198 * internal-only diagonal (D_internal = D_aug + boundaryDiag), divided ONLY the internal part by
199 * alpha, and re-added the boundary UN-divided (D_relaxed = D_internal/alpha - boundaryDiag). That
200 * left the relaxed augmented diagonal too large by boundaryDiag*(1-alpha)/alpha at every
201 * boundary-adjacent cell, corrupting rAU = V/diag and HbyA and diverging the PIMPLE outer loop.
202 * The synthetic unit tests masked it (boundaryDiag identically zero) and the rAU oracle encoded
203 * the same wrong formula; it surfaced only as the full neoPimpleFoam-vs-pimpleFoam parity
204 * divergence.
205 *
206 * The source correction makes the relaxed system share the fixed point of the
207 * unrelaxed system (the correction cancels at the converged solution). `psi_prev`
208 * is the field value at solve entry (`solution.internalVector()` — no
209 * separate snapshot). The negative-diagonal sign convention (operators atomic_sub
210 * positive flux into the diagonal; `scaledInvDiagNegLUx` reads `V/diag[0]` raw,
211 * no mag()) is preserved componentwise via `componentCopySign` so `rAU`/`HbyA`
212 * stay correct for alpha < 1.
213 *
214 * @param ls The assembled LinearSystem (mutated in place on the augmented diagonal).
215 * @param solution The solution VolumeField — supplies psi_prev (internalVector) and the
216 * mesh (boundaryMesh().faceOwners() for the boundary-diagonal owner).
217 * @param alpha The under-relaxation factor. alpha <= 0 or alpha == 1 is a bitwise no-op.
218 */
219// Overload set is templated on the FULL LinearSystem parameter pack (mirroring
220// removeBoundaryContributions, linearSystem.hpp) rather than LinearSystem<ElementType>.
221// This matters because the real momentum system assembled by
222// the DSL is the SEGREGATED vector-solve form — a SCALAR matrix (MatrixValueType == scalar) with
223// a Vec3 RHS (RHSValueType == Vec3) — not LinearSystem<Vec3>. The diagonal/off-diagonal/boundary
224// algebra therefore runs in MatrixValueType (scalar here), while the source correction and field
225// read run in RHSValueType (Vec3 here); the cross term (dRelaxed - dAug) * field is
226// MatrixValueType * RHSValueType (scalar * Vec3 = Vec3), which the existing operator* already
227// supports. The synthetic NeoN unit tests use createEmptyLinearSystem<Vec3>(mesh) where
228// MatrixValueType == RHSValueType == Vec3, so both forms are now exercised.
229template<
230 typename VectorType,
231 typename MatrixValueType,
232 typename RHSValueType,
233 typename SystemMatrixType,
234 typename BoundaryMatrixType>
235void applyMatrixRelaxation(
236 la::LinearSystem<MatrixValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType>& ls,
237 const VectorType& solution,
238 scalar alpha
239)
240{
241 // alpha<=0 guard: bitwise no-op. MUST be first so neither
242 // matrix.values nor rhs is touched when no relaxation is requested.
243 if (alpha <= 0.0 || alpha == 1.0)
244 {
245 return;
246 }
247
248
249 const scalar invAlpha = 1.0 / alpha;
250
251 // View mechanics copied from removeBoundaryContributions (linearSystem.hpp) — the
252 // in-tree, sign-correct precedent for view + diagonal addressing.
253 auto lsView = ls.view();
254 auto& matrix = lsView.matrix;
255 auto& rhs = lsView.rhs;
256 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
257 const auto [rowOffs, colIdxs] = views(ls.matrix().rowOffs(), ls.matrix().colIdxs());
258 const auto field = solution.internalVector().view();
259
260 const localIdx nCells = field.size();
261
262 // FUSED single-pass relaxation (PERF): compute the off-diagonal magnitude sum inline and apply
263 // the dominance clamp + diagonal boost + source correction in ONE kernel, eliminating the
264 // per-call O(nCells) `sumOff` scratch allocation (and its zero-fill) and the second kernel
265 // launch. Each cell needs only its OWN row's off-diagonal sum — no cross-cell dependency — and
266 // relaxation only ever writes a cell's own diagonal (matrix.values[diagIdx]) and rhs[celli].
267 // The row walk skips the diagonal (colIdxs[idx] != celli), and a cell's diagonal write never
268 // aliases another cell's off-diagonal storage, so fusing is bitwise-identical to the prior two
269 // passes.
270 //
271 // Matches OpenFOAM fvMatrix::relax(alpha): OF adds the boundary internalCoeffs to the diagonal,
272 // clamps for dominance, then `D /= alpha` with the boundary contribution INCLUDED in the
273 // division (removed from the stored diag afterwards but re-added un-divided at solve via
274 // addBoundaryDiag) — netting a relaxed AUGMENTED diagonal of D_aug / alpha. NeoN bakes the
275 // boundary permanently into matrix.values[diagIdx], so the augmented diagonal is already in
276 // hand: we relax dAug DIRECTLY, with NO internal-only reconstruction. (The earlier
277 // reconstruct-internal / re-add-un-divided-boundary path left the relaxed diagonal too large by
278 // boundaryDiag*(1-alpha)/alpha at every boundary-adjacent cell, corrupting rAU/HbyA and
279 // diverging the PIMPLE outer loop — the boundary-diagonal relaxation bug.)
280 //
281 // The dominance clamp runs on the augmented diagonal vs the off-diagonal magnitude
282 // sum; componentCopySign preserves dAug's sign so scaledInvDiagNegLUx (rAU = V/diag,
283 // no mag()) stays correct for alpha < 1. The source correction rhs += (dRelaxed -
284 // dAug)*psi_prev makes the relaxed system share the unrelaxed fixed point and
285 // exactly preserves the residual at psi_prev. Off-diagonal sum + diagonal algebra are in
286 // MatrixValueType; the source-correction cross term (dRelaxed - dAug)*field is MatrixValueType
287 // * RHSValueType (scalar
288 // * Vec3 in the segregated momentum form).
289 parallelFor(
290 ls.exec(),
291 {0, nCells},
292 NEON_LAMBDA(const localIdx celli) {
293 // Off-diagonal magnitude sum — cell-based CSR row gather, NO atomics (perf lever).
294 auto sumOff = zero<MatrixValueType>();
295 for (localIdx idx = rowOffs[celli]; idx < rowOffs[celli + 1]; ++idx)
296 {
297 if (colIdxs[idx] != celli)
298 {
299 sumOff = sumOff + componentMag(matrix.values[idx]);
300 }
301 }
302 const auto diagIdx = ma.diagIdx(celli);
303 const auto dAug = matrix.values[diagIdx]; // augmented, boundary-baked
304 const auto dDom = componentMax(componentMag(dAug), sumOff); // dominance clamp
305 const auto dRelaxed =
306 componentCopySign(dDom * invAlpha, dAug); // scale whole augmented diag by 1/alpha
307 matrix.values[diagIdx] = dRelaxed;
308 rhs[celli] = rhs[celli] + (dRelaxed - dAug) * field[celli]; // source corr
309 },
310 "applyMatrixRelaxation"
311 );
312}

318/* @brief Apply explicit field under-relaxation to a solution field's internal vector.
319 *
320 * Computes the OpenFOAM `relax()` blend on the internal field only:
321 *
322 * psi[c] = prev[c] + alpha * (psi[c] - prev[c])
323 *
324 * `prev` is the caller-owned previous-iteration snapshot (e.g. from
325 * `fieldRelaxationSnapshot`, taken at the top of the outer corrector). This is a
326 * stateless free function mirroring `applyMatrixRelaxation`; NeoN stays
327 * OpenFOAM-agnostic and sees only the scalar `alpha`.
328 *
329 * Boundary scope: only the internal vector is blended. The caller is
330 * responsible for calling `solution.correctBoundaryConditions()` afterwards so the
331 * boundary values are re-derived from the boundary conditions. (OpenFOAM's `relax()`
332 * blends boundary values directly; both paths agree at the fixed point and for
333 * `fixedValue` patches.)
334 *
335 * `alpha <= 0` or `alpha == 1` is a bitwise no-op (the early return touches nothing).
336 * This is REQUIRED so the final outer iteration (`alpha == 1`) and an unset
337 * factor leave the field byte-for-byte unchanged, side-stepping the `alpha == 1` ULP
338 * round-trip trap of `prev + 1*(cur - prev)`.
339 *
340 * @param solution The solution VolumeField — its internal vector is blended in place.
341 * @param previous The previous-iteration snapshot of the internal vector (deep copy).
342 * @param alpha The under-relaxation factor. alpha <= 0 or alpha == 1 is a bitwise no-op.
343 */
344template<typename VectorType>
345void applyFieldRelaxation(
346 VectorType& solution, const Vector<typename VectorType::ElementType>& previous, scalar alpha
347)
348{
349 // Bitwise no-op guard: MUST be first so the internal vector is not touched when no
350 // relaxation is requested (final iteration / unset factor, alpha==1 ULP trap).
351 if (alpha <= 0.0 || alpha == 1.0)
352 {
353 return;
354 }
355
356 NF_ASSERT(solution.size() == previous.size(), "applyFieldRelaxation: field/prev size mismatch");
357
358 auto [current, prev] = views(solution.internalVector(), previous);
359 parallelFor(
360 solution.exec(),
361 {0, solution.size()},
362 NEON_LAMBDA(const localIdx celli) {
363 current[celli] = prev[celli] + alpha * (current[celli] - prev[celli]);
364 },
365 "applyFieldRelaxation"
366 );
367}

369/* @brief Snapshot a field's internal vector for use as the `previous` arg to
370 * `applyFieldRelaxation` (caller-owned prevIter primitive).
371 *
372 * Returns an independent on-executor deep copy of `field.internalVector()` via the
373 * `Vector` copy constructor — subsequent mutation of `field` does not affect the
374 * returned snapshot. Do NOT hand-roll a `parallelFor` copy; the copy ctor already
375 * performs an executor-correct deep copy.
376 */
377template<typename VectorType>
378[[nodiscard]] auto fieldRelaxationSnapshot(const VectorType& field)
379{
380 return Vector<typename VectorType::ElementType>(field.internalVector());
381}
