17// Per-iteration residual-norm history. The criteria here pass residual_norm = nullptr, so
18// the norm comes from the residual vector; device scalars stage through the host master.
19class ResidualHistoryLogger : public gko::log::Logger
20{
21public:
22
23 ResidualHistoryLogger() : gko::log::Logger(gko::log::Logger::iteration_complete_mask) {}
24
25 void clear() { history_.clear(); }
26
27 const std::vector<double>& history() const { return history_; }
28
29protected:
30
31 // The base declares three on_iteration_complete overloads and forwards new->old; overriding
32 // one hides the other two. Ginkgo resolves the call inside Logger's own scope, so the hiding
33 // is invisible to it -- this only silences the #611-D warning it provokes per TU.
34 using gko::log::Logger::on_iteration_complete;
35
36 void on_iteration_complete(
37 const gko::LinOp*,
38 const gko::LinOp*,
39 const gko::LinOp*,
40 const gko::size_type&,
41 const gko::LinOp* residual,
42 const gko::LinOp* residual_norm,
43 const gko::LinOp* implicit_sq_norm,
44 const gko::array<gko::stopping_status>*,
45 bool
46 ) const override
47 {
48 if (auto norm = dynamic_cast<const Dense*>(residual_norm))
49 {
50 history_.push_back(readScalar(norm));
51 }
52 else if (auto res = dynamic_cast<const Dense*>(residual))
53 {
54 auto exec = res->get_executor();
55 auto norm2 = Dense::create(exec, gko::dim<2> {1, 1});
56 res->compute_norm2(norm2);
57 history_.push_back(readScalar(norm2.get()));
58 }
59 else if (auto sq = dynamic_cast<const Dense*>(implicit_sq_norm))
60 {
61 history_.push_back(std::sqrt(std::abs(readScalar(sq))));
62 }
63 }
64
65private:
66
67 static double readScalar(const Dense* d)
68 {
69 auto exec = d->get_executor();
70 if (exec->get_master().get() != exec.get())
71 {
72 auto host = gko::clone(exec->get_master(), d);
73 return host->at(0, 0);
74 }
75 return d->at(0, 0);
76 }
77
78 mutable std::vector<double> history_;
79};
