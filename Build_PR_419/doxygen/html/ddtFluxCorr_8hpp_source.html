23KOKKOS_INLINE_FUNCTION
24scalar ddtFluxCorrLimiter(const scalar fluxMag, const scalar corrMag)
25{
26 constexpr scalar small = scalar(1e-30);
27 const scalar ratio = corrMag / (fluxMag + small);
28 return scalar(1) - Kokkos::min(ratio, scalar(1));
29}

31// ------------------------------------------------------------------
32// BDF1 kernel (also used for BDF2 startup)
33// ------------------------------------------------------------------
34inline void ddtFluxCorrBDF1Kernel(
35 const Executor& exec,
36 const UnstructuredMesh& mesh,
37 const SurfScalarField& flux0,
38 const SurfaceField<Vec3>& uf0,
39 SurfScalarField& fluxCorr,
40 scalar dt
41)
42{
43 auto [outV, flux0V, uf0V, SfV] = views(
44 fluxCorr.internalVector(), flux0.internalVector(), uf0.internalVector(), mesh.faceAreas()
45 );
46
47 const scalar a1 = scalar(1) / dt;
48 const size_t n = outV.size();
49
50 parallelFor(
51 exec,
52 {size_t(0), n},
53 KOKKOS_LAMBDA(const localIdx i) {
54 const auto d = (SfV[i] & uf0V[i]);
55 const auto corr = flux0V[i] - d;
56
57 const scalar limiter = ddtFluxCorrLimiter(mag(flux0V[i]), mag(corr));
58
59 outV[i] = limiter * a1 * corr;
60 },
61 "ddtFluxCorr::BDF1"
62 );
63}

66// BDF2 kernel
67// ------------------------------------------------------------------
68inline void ddtFluxCorrBDF2Kernel(
69 const Executor& exec,
70 const UnstructuredMesh& mesh,
71 const SurfScalarField& flux0,
72 const SurfScalarField& flux00,
73 const SurfaceField<Vec3>& uf0,
74 const SurfaceField<Vec3>& uf00,
75 SurfScalarField& fluxCorr,
76 scalar dt
77)
78{
79 auto [outV, flux0V, flux00V, uf0V, uf00V, SfV] = views(
80 fluxCorr.internalVector(),
81 flux0.internalVector(),
82 flux00.internalVector(),
83 uf0.internalVector(),
84 uf00.internalVector(),
85 mesh.faceAreas()
86 );
87
88 const scalar a1 = 2.0 / dt;
89 const scalar a2 = -0.5 / dt;
90 const size_t n = outV.size();
91
92 parallelFor(
93 exec,
94 {size_t(0), n},
95 KOKKOS_LAMBDA(const localIdx i) {
96 const auto d1 = (SfV[i] & uf0V[i]);
97 const auto corr1 = flux0V[i] - d1;
98
99 const auto d2 = (SfV[i] & uf00V[i]);
100 const auto corr2 = flux00V[i] - d2;
101
102 const scalar limiter1 = ddtFluxCorrLimiter(mag(flux0V[i]), mag(corr1));
103 const scalar limiter2 = ddtFluxCorrLimiter(mag(flux00V[i]), mag(corr2));
104
105 outV[i] = limiter1 * a1 * corr1 + limiter2 * a2 * corr2;
106 },
107 "ddtFluxCorr::BDF2"
108 );
109}

113inline SurfScalarField
114ddtFluxCorr(const VolVectorField& U, const SurfScalarField& phi, scalar dt, DdtScheme scheme)
115{
116 const auto& mesh = U.mesh();
117 const auto& exec = phi.exec();
118
119 // --- interpolation
120 SurfaceInterpolation<Vec3> interp(exec, mesh, TokenList({std::string("linear")}));
121
122 // --- boundary conditions consistent with U
123 auto surfaceBCs = createFluxCorrBCsFromU(mesh, U);
124
125 SurfScalarField fluxCorr(exec, "ddtFluxCorr", mesh, surfaceBCs);
126
127 const int level = oldTimeLevel(U);
128
129 // --- BDF1 / startup
130 const auto& U0 = oldTime(U);
131 const auto& phi0 = oldTime(phi);
132 auto uf0 = interp.interpolate(U0);
133
134 if (scheme == DdtScheme::BDF2 && level >= 2) // --- BDF2 contribution
135 {
136 const auto& U00 = oldTime(U0);
137 const auto& phi00 = oldTime(phi0);
138 auto uf00 = interp.interpolate(U00);
139
140 detail::ddtFluxCorrBDF2Kernel(exec, mesh, phi0, phi00, uf0, uf00, fluxCorr, dt);
141 }
142 else
143 {
144 detail::ddtFluxCorrBDF1Kernel(exec, mesh, phi0, uf0, fluxCorr, dt);
145 }
146
147 fluxCorr.correctBoundaryConditions();
148 return fluxCorr;
149}
