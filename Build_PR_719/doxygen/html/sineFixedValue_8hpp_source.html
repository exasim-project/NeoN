22// Shared kernel behind SineFixedValue, a time-varying Dirichlet BC mirroring OpenFOAM's
23// uniformFixedValue patch field when its Function1 is a sine (Foam::Function1Types::Sine, "sin"
24// form -- the default waveType). Per OpenFOAM's Sine::sinValue:
25// value(t) = amplitude * sin(2*pi * frequency * (t - t0)) * scale + level
26// (amplitude defaults to 1, t0 defaults to 0, matching Sine's own dict-read defaults). The whole
27// patch gets one uniform value every call (valueFraction = 1, refGrad = 0), recomputed from the
28// current simulation time -- mirroring uniformFixedValueFvPatchField::updateCoeffs(), which
29// re-evaluates its Function1 at db().time().timeOutputValue() on every call rather than caching
30// the value.
31//
32// When no time is available (haveContext == false, e.g. the plain correctBoundaryCondition path
33// with no BoundaryContext) the patch degrades to zero-gradient, the same fallback convention
34// every other context-aware NeoN BC uses on missing context.
35inline void setSineFixedValue(
36 Field<NeoN::scalar>& domainVector,
37 const UnstructuredMesh& mesh,
38 std::pair<localIdx, localIdx> range,
39 NeoN::scalar amplitude,
40 NeoN::scalar frequency,
41 NeoN::scalar scale,
42 NeoN::scalar level,
43 NeoN::scalar t0,
44 NeoN::scalar time,
45 bool haveContext
46)
47{
48 const auto internalV = domainVector.internalVector().view();
49 const auto boundaryFaceOwnersV = mesh.boundaryMesh().faceOwners().view();
50
51 auto [refGradV, valueV, valueFractionV, refValueV] = views(
52 domainVector.boundaryData().refGrad(),
53 domainVector.boundaryData().value(),
54 domainVector.boundaryData().valueFraction(),
55 domainVector.boundaryData().refValue()
56 );
57
58 NeoN::scalar uniformValue = NeoN::scalar(0);
59 if (haveContext)
60 {
61 constexpr NeoN::scalar twoPi = NeoN::scalar(2) * Kokkos::numbers::pi_v<NeoN::scalar>;
62 uniformValue = amplitude * Kokkos::sin(twoPi * frequency * (time - t0)) * scale + level;
63 }
64
65 NeoN::parallelFor(
66 domainVector.exec(),
67 range,
68 NEON_LAMBDA(const localIdx i) {
69 refGradV[i] = NeoN::scalar(0);
70
71 if (!haveContext)
72 {
73 const localIdx owner = boundaryFaceOwnersV[i];
74 const NeoN::scalar ownerValue = internalV[owner];
75 valueFractionV[i] = NeoN::scalar(0);
76 refValueV[i] = ownerValue;
77 valueV[i] = ownerValue;
78 return;
79 }
80
81 valueFractionV[i] = NeoN::scalar(1);
82 refValueV[i] = uniformValue;
83 valueV[i] = uniformValue;
84 },
85 "setSineFixedValue"
86 );
87}
88
89} // namespace detail

91// SineFixedValue: a time-varying Dirichlet BC mirroring OpenFOAM's uniformFixedValue patch field
92// configured with a sine Function1. See detail::setSineFixedValue for the exact formula.
93//
94// Needs, via the context-aware correctBoundaryCondition(domainVector, ctx) overload, the current
95// simulation time (BoundaryContext::insertValue, key "time"). Without it, degrades to a plain
96// zero-gradient extrapolation (the same fallback the plain, context-less overload always uses).
97class SineFixedValue : public VolumeBoundaryFactory<NeoN::scalar>::template Register<SineFixedValue>
98{
99 using Base = VolumeBoundaryFactory<NeoN::scalar>::template Register<SineFixedValue>;
100
101public:
102
103 using Base::correctBoundaryCondition;
104
105 // assignable = true: recomputed every correctBoundaryConditions() call. fixesValue = true:
106 // a genuine global Dirichlet patch, same as fixedValue (without a time in the context it
107 // degrades to zero-gradient for that call).
108 SineFixedValue(const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID)
109 : Base(mesh, dict, patchID, {.assignable = true, .fixesValue = true}), mesh_(mesh),
110 amplitude_(
111 dict.contains("amplitude") ? dict.get<NeoN::scalar>("amplitude") : NeoN::scalar(1)
112 ),
113 frequency_(dict.get<NeoN::scalar>("frequency")), // required: no physically-sane default
114 scale_(dict.get<NeoN::scalar>("scale")), // required: no physically-sane default
115 level_(dict.get<NeoN::scalar>("level")), // required: no physically-sane default
116 t0_(dict.contains("t0") ? dict.get<NeoN::scalar>("t0") : NeoN::scalar(0))
117 {}
118
119 // No context => no simulation time available: degrade to zero-gradient.
120 virtual void correctBoundaryCondition(Field<NeoN::scalar>& domainVector) final
121 {
122 detail::setSineFixedValue(
123 domainVector,
124 mesh_,
125 this->range(),
126 amplitude_,
127 frequency_,
128 scale_,
129 level_,
130 t0_,
131 NeoN::scalar(0),
132 false
133 );
134 }
135
136 // Context-aware: evaluate the sine at the current simulation time.
137 virtual void
138 correctBoundaryCondition(Field<NeoN::scalar>& domainVector, const BoundaryContext& ctx) final
139 {
140 if (!ctx.hasValue("time"))
141 {
142 correctBoundaryCondition(domainVector);
143 return;
144 }
145
146 detail::setSineFixedValue(
147 domainVector,
148 mesh_,
149 this->range(),
150 amplitude_,
151 frequency_,
152 scale_,
153 level_,
154 t0_,
155 ctx.value("time"),
156 true
157 );
158 }
159
160 static std::string name() { return "sineFixedValue"; }
161
162 std::string getName() const override { return name(); }
163
164 static std::string doc()
165 {
166 return "Time-varying uniform Dirichlet value (OpenFOAM's uniformFixedValue with a sine "
167 "Function1): value(t) = amplitude*sin(2*pi*frequency*(t - t0))*scale + level "
168 "(amplitude defaults to 1, t0 defaults to 0). Needs the current simulation time "
169 "in the BoundaryContext (key \"time\"); without it, degrades to zero-gradient.";
170 }
171
172 static std::string schema() { return "none"; }
173
174 virtual std::unique_ptr<VolumeBoundaryFactory<NeoN::scalar>> clone() const final
175 {
176 return std::make_unique<SineFixedValue>(*this);
177 }
178
179private:
180
181 const UnstructuredMesh& mesh_;
182 NeoN::scalar amplitude_;
183 NeoN::scalar frequency_;
184 NeoN::scalar scale_;
185 NeoN::scalar level_;
186 NeoN::scalar t0_;
187};
