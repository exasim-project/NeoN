46/* @brief whether JuNe implements every implicit operator of the expression */
47template<typename ValueType, typename IndexType>
48bool canAssemble(const Expression<ValueType, IndexType>& expr)
49{
50 bool hasSpatial = false;
51 for (const auto& op : expr.spatialOperators())
52 {
53 if (op.getType() != Operator::Type::Implicit) continue;
54 const auto name = op.getName();
55 const auto jlOp = op.juliaOP();
56 // JuNe only defines the upwind, linearUpwind and linear div schemes
57 const bool supportedDiv = name == "DivOperator"
58 && (jlOp.starts_with("Div{Float64, upwind{")
59 || jlOp.starts_with("Div{Float64, linearUpwind{")
60 || jlOp.starts_with("Div{Float64, linear{"));
61 if (name != "LaplacianOperator" && !supportedDiv) return false;
62 hasSpatial = true;
63 }
64 for (const auto& op : expr.temporalOperators())
65 {
66 if (op.getType() != Operator::Type::Implicit) continue;
67 // an empty string is steadyState, which assembles nothing; JuNe rewrites BDF2 to BDF1
68 const auto jlOp = op.juliaOP();
69 if (op.getName() != "DdtOperator" || jlOp.find("BDF2") != std::string::npos) return false;
70 }
71 // phi, and with it the mesh, comes from the spatial operators
72 return hasSpatial;
73}

75/* @brief the Julia expression of all implicit operators, built from the current scheme reads */
76template<typename ValueType, typename IndexType>
77std::string opString(const Expression<ValueType, IndexType>& expr)
78{
79 std::string ret;
80 auto append = [&ret](const auto& op)
81 {
82 if (op.getType() != Operator::Type::Implicit || op.juliaOP().empty()) return;
83 ret += ret.empty() ? op.juliaOP() : " + " + op.juliaOP();
84 };
85 for (const auto& op : expr.temporalOperators())
86 {
87 append(op);
88 }
89 for (const auto& op : expr.spatialOperators())
90 {
91 append(op);
92 }
93 return ret;
94}

110/* @brief returns the config entry of the first implicit spatial operator named opName */
111template<typename HoldType, typename ValueType, typename IndexType>
112const HoldType* findConfig(
113 const Expression<ValueType, IndexType>& expr, const std::string& opName, const std::string& key
114)
115{
116 for (const auto& op : expr.spatialOperators())
117 {
118 if (op.getType() == Operator::Type::Implicit && op.getName() == opName)
119 {
120 return &op.getConfig().template get<NeoN::detail::RefHolder<HoldType>>(key).c;
121 }
122 }
123 return nullptr;
124}

128/* @brief assemble all implicit operators of expr into ls with JuNe
129 *
130 * Only valid if canAssemble(expr). JuNe adds to the existing coefficients, so ls must be reset.
131 */
132template<typename ValueType, typename IndexType>
133void assemble(
134 const Expression<ValueType, IndexType>& expr,
135 la::LinearSystem<ValueType, ValueType>& ls,
136 scalar dt,
137 Strategy strategy = Strategy::FromIterator
138)
139{
140 namespace fvcc = NeoN::finiteVolume::cellCentred;
141
142 const auto* divPhi = detail::findConfig<fvcc::VolumeField<ValueType>>(expr, "DivOperator", "field");
143 const auto* lapPhi =
144 detail::findConfig<fvcc::VolumeField<ValueType>>(expr, "LaplacianOperator", "field");
145 NF_ASSERT(divPhi || lapPhi, "JuNe assembly needs a div or laplacian operator");
146 const auto& phi = divPhi ? *divPhi : *lapPhi;
147 const auto& mesh = phi.mesh();
148
149 // JuNe skips the inputs of operators that are not in the op string, so absent operators
150 // get empty arrays
151 const auto empty = Vector<scalar>(ls.exec(), 0);
152 const auto emptyOld = Vector<ValueType>(ls.exec(), 0);
153 const auto* faceFlux =
154 detail::findConfig<fvcc::SurfaceField<scalar>>(expr, "DivOperator", "flux");
155 const auto* gamma =
156 detail::findConfig<fvcc::SurfaceField<scalar>>(expr, "LaplacianOperator", "gamma");
157 const auto fusedOPString = opString(expr);
158 const bool hasDdt = fusedOPString.starts_with("Ddt{");
159
160 const auto ifaceFluxV = faceFlux ? faceFlux->internalVector().juliaPtr() : empty.juliaPtr();
161 const auto bfaceFluxV =
162 faceFlux ? faceFlux->boundaryData().value().juliaPtr() : empty.juliaPtr();
163 const auto iGamma = gamma ? gamma->internalVector().juliaPtr() : empty.juliaPtr();
164 const auto bGamma = gamma ? gamma->boundaryData().value().juliaPtr() : empty.juliaPtr();
165 const auto oldVector =
166 hasDdt ? fvcc::oldTime(phi).internalVector().juliaPtr() : emptyOld.juliaPtr();
167
168 const auto nInternalFaces = mesh.nInternalFaces();
169 const auto nTotalFaces = mesh.nTotalFaces();
170 const auto nCells = mesh.nCells();
171 bool useGPU = std::holds_alternative<GPUExecutor>(mesh.exec());
172
173 // the cached geometry-scheme coefficients: the mesh centres they are computed from are
174 // released after the first geometry read
175 const auto& nonOrthDeltaCoeffs =
176 fvcc::GeometryScheme::readOrCreate(mesh)->nonOrthDeltaCoeffs();
177 const auto ideltaCoeffs = nonOrthDeltaCoeffs.internalVector().juliaPtr();
178 const auto bdeltaCoeffs = nonOrthDeltaCoeffs.boundaryData().value().juliaPtr();
179
180 const auto magFaceAreas = mesh.faceAreas().juliaPtr();
181 const auto boundaryFaceOwners = mesh.boundaryMesh().faceOwners().juliaPtr();
182 const auto vol = mesh.cellVolumes().juliaPtr();
183 auto refGrad = phi.boundaryData().refGrad().juliaPtr();
184 auto valueFraction = phi.boundaryData().valueFraction().juliaPtr();
185 auto refValue = phi.boundaryData().refValue().juliaPtr();
186 auto rhsPtr = ls.rhs().juliaPtr();
187 auto bRhs = ls.boundaryRhs().juliaPtr();
188
189 auto iterator = std::dynamic_pointer_cast<la::CellBasedIterator>(ls.getMeshIterator()->get());
190 if (strategy == Strategy::FromIterator)
191 {
192 strategy = iterator ? Strategy::Cell : Strategy::Face;
193 }
194
195 jl_module_t* mod = (jl_module_t*)jl_eval_string("JuNe");
196
197 if (strategy == Strategy::Cell)
198 {
199 NF_ASSERT(iterator, "JuNe cell-based assembly needs a CellBasedIterator");
200 const auto matIt = ls.faceToMatrixAddress();
201 if (!iterator->getCellBasedData())
202 {
203 iterator->setComputeCellBasedData(mesh, ls.matrix().sparsity(), matIt);
204 }
205 auto cellBasedData = iterator->getCellBasedData();
206
207 auto [cellFacesValues, cellFacesSegments] = cellBasedData->cellFaces.juliaPtrs();
208 auto faceSignV = cellBasedData->faceSign.juliaPtr();
209 auto matrixColumnIdxV = cellBasedData->matrixColumnIdx.juliaPtr();
210 const auto diagOffs = matIt->diagOffset().juliaPtr();
211 const auto rowOffs = ls.matrix().sparsity()->rowOffs().juliaPtr();
212 const auto jowners = mesh.faceOwners().juliaPtr();
213 auto valPtr = ls.matrix().juliaPtr();
214 auto bValues = ls.boundaryMatrix().juliaPtr();
215
216 size_t nbinputs = useGPU ? 29 : 27;
217 jl_value_t* args[29];
218 args[0] = jl_box_int32(iterator->size());
219 args[1] = (jl_value_t*)jowners;
220 args[2] = (jl_value_t*)cellFacesSegments;
221 args[3] = (jl_value_t*)diagOffs;
222 args[4] = (jl_value_t*)rowOffs;
223 args[5] = (jl_value_t*)cellFacesValues;
224 args[6] = (jl_value_t*)faceSignV;
225 args[7] = (jl_value_t*)valPtr;
226 args[8] = jl_cstr_to_string(fusedOPString.c_str());
227 args[9] = (jl_value_t*)ifaceFluxV;
228 args[10] = (jl_value_t*)bfaceFluxV;
229 args[11] = (jl_value_t*)iGamma;
230 args[12] = (jl_value_t*)bGamma;
231 args[13] = (jl_value_t*)ideltaCoeffs;
232 args[14] = (jl_value_t*)bdeltaCoeffs;
233 args[15] = (jl_value_t*)matrixColumnIdxV;
234 args[16] = (jl_value_t*)magFaceAreas;
235 args[17] = (jl_value_t*)valueFraction;
236 args[18] = (jl_value_t*)boundaryFaceOwners;
237 args[19] = (jl_value_t*)bValues;
238 args[20] = (jl_value_t*)bRhs;
239 args[21] = (jl_value_t*)vol;
240 args[22] = (jl_value_t*)oldVector;
241 args[23] = (jl_value_t*)refValue;
242 args[24] = (jl_value_t*)refGrad;
243 args[25] = (jl_value_t*)rhsPtr;
244 args[26] = jl_box_float64(dt);
245 if (useGPU)
246 {
247 args[27] = jl_box_int32(nInternalFaces);
248 args[28] = jl_box_int32(nTotalFaces);
249 }
250 auto funcName = useGPU ? "cellBased_gpu" : "cellBased_";
251 jl_function_t* func = jl_get_function(mod, funcName);
252 jl_call(func, args, static_cast<uint32_t>(nbinputs));
253 detail::checkJuliaException();
254 return;
255 }
256
257 const auto JUowner = mesh.faceOwners().juliaPtr();
258 const auto JUneighbour = mesh.faceNeighbors().juliaPtr();
259 const auto JUdiagOffs = ls.faceToMatrixAddress()->diagOffset().juliaPtr();
260 const auto JUownOffs = ls.faceToMatrixAddress()->ownerOffset().juliaPtr();
261 const auto JUneiOffs = ls.faceToMatrixAddress()->neighbourOffset().juliaPtr();
262 const auto JUrowOffs = ls.matrix().sparsity()->rowOffs().juliaPtr();
263 auto valPtr = ls.matrix().values().juliaPtr();
264 auto bValues = ls.boundaryMatrix().values().juliaPtr();
265
266 size_t nbinputs = useGPU ? 28 : 26;
267 jl_value_t* args[28];
268 args[0] = jl_box_int32(nInternalFaces);
269 args[1] = (jl_value_t*)JUowner;
270 args[2] = (jl_value_t*)JUneighbour;
271 args[3] = (jl_value_t*)JUdiagOffs;
272 args[4] = (jl_value_t*)JUownOffs;
273 args[5] = (jl_value_t*)JUneiOffs;
274 args[6] = (jl_value_t*)JUrowOffs;
275 args[7] = (jl_value_t*)valPtr;
276 args[8] = jl_cstr_to_string(fusedOPString.c_str());
277 args[9] = (jl_value_t*)ifaceFluxV;
278 args[10] = (jl_value_t*)bfaceFluxV;
279 args[11] = (jl_value_t*)iGamma;
280 args[12] = (jl_value_t*)bGamma;
281 args[13] = (jl_value_t*)ideltaCoeffs;
282 args[14] = (jl_value_t*)bdeltaCoeffs;
283 args[15] = (jl_value_t*)magFaceAreas;
284 args[16] = (jl_value_t*)valueFraction;
285 args[17] = (jl_value_t*)refValue;
286 args[18] = (jl_value_t*)refGrad;
287 args[19] = (jl_value_t*)rhsPtr;
288 args[20] = (jl_value_t*)boundaryFaceOwners;
289 args[21] = (jl_value_t*)bValues;
290 args[22] = (jl_value_t*)bRhs;
291 args[23] = (jl_value_t*)vol;
292 args[24] = (jl_value_t*)oldVector;
293 args[25] = jl_box_float64(dt);
294 if (useGPU)
295 {
296 args[26] = jl_box_int32(nCells);
297 args[27] = jl_box_int32(nTotalFaces);
298 }
299 const bool global = strategy == Strategy::GlobalFace;
300 auto funcName = useGPU ? (global ? "globalassemble_gpu" : "assemble_gpu")
301 : (global ? "globalassemble" : "assemble");
302 jl_function_t* func = jl_get_function(mod, funcName);
303 jl_call(func, args, static_cast<uint32_t>(nbinputs));
304 detail::checkJuliaException();
305}

307/* @brief compile the JuNe face-based kernels for the operators of expr ahead of timing */
308template<typename ValueType, typename IndexType>
309void warmup(const Expression<ValueType, IndexType>& expr)
310{
311 jl_module_t* mod = (jl_module_t*)jl_eval_string("JuNe");
