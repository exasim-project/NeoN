186 template<typename OtherRHSValueType, typename OtherBoundaryMatrixType>
187 [[nodiscard]] bool matrixEquals(
188 const LinearSystem<
189 MatrixValueType,
190 OtherRHSValueType,
191 SystemMatrixType,
192 OtherBoundaryMatrixType>& other,
193 scalar tolerance = static_cast<scalar>(1e-12)
194 ) const
195 {
196 const auto lhsDim = matrix_.sparsity()->dimension();
197 const auto rhsDim = other.matrix().sparsity()->dimension();
198 if (lhsDim.rows != rhsDim.rows || lhsDim.cols != rhsDim.cols)
199 {
200 return false;
201 }
202
203 if (!vectorsEqual(matrix_.colIdxs(), other.matrix().colIdxs()))
204 {
205 return false;
206 }
207
208 if (!vectorsEqual(matrix_.rowOffs(), other.matrix().rowOffs()))
209 {
210 return false;
211 }
212
213 return matrixValuesEqual(matrix_.values(), other.matrix().values(), tolerance);
214 }

345 template<typename ValueType>
346 static bool vectorsEqual(const Vector<ValueType>& lhs, const Vector<ValueType>& rhs)
347 {
348 if (lhs.size() != rhs.size())
349 {
350 return false;
351 }
352
353 auto lhsHost = lhs.copyToHost();
354 auto rhsHost = rhs.copyToHost();
355 const auto lhsView = lhsHost.view();
356 const auto rhsView = rhsHost.view();
357 for (localIdx i = 0; i < lhsView.size(); ++i)
358 {
359 if (lhsView[i] != rhsView[i])
360 {
361 return false;
362 }
363 }
364 return true;
365 }
366
367 static bool matrixValuesEqual(
368 const Vector<MatrixValueType>& lhs,
369 const Vector<MatrixValueType>& rhs,
370 scalar tolerance
371 )
372 {
373 if (lhs.size() != rhs.size())
374 {
375 return false;
376 }
377
378 auto lhsHost = lhs.copyToHost();
379 auto rhsHost = rhs.copyToHost();
380 const auto lhsView = lhsHost.view();
381 const auto rhsView = rhsHost.view();
382 auto ret = true;
383 auto err = 0.0;
384 auto max = 0.0;
385 localIdx maxindex = 0;
386 for (localIdx i = 0; i < lhsView.size(); ++i)
387 {
388 const auto cur = abs(lhsView[i][0] - rhsView[i][0]);
389
390 if (cur > max)
391 {
392 max = cur;
393 maxindex = i;
394 }
395 err += cur;
396 if (cur > tolerance)
397 {
398 // std::cout << "wrong: " <<i<< ": " << lhsView[i] << " ------ " << rhsView[i] <<
399 // std::endl;
400 ret = false;
401 // return false;
402 }
403 }
404 std::cout << "highest error: " << max << " at " << maxindex << std::endl;
405 std::cout << "neon: " << lhsView[maxindex] << ", julia: " << rhsView[maxindex] <<std::endl;
406
407 std::cout << "total error: " << err << std::endl;
408 std::cout << "total error per nnz: " << err / lhsView.size() << std::endl;
409 return ret;
410 }

458/*@brief helper function that creates a zero initialised linear system based on a given mesh
459 */
460template<
461 typename ValueType,
462 typename RHSValueType = ValueType,
463 typename SystemMatrixType = CSRMatrix<ValueType, localIdx>,
464 typename BoundaryMatrixType = COOMatrix<ValueType, localIdx>>
465LinearSystem<ValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType> createEmptyLinearSystem(
466 const UnstructuredMesh& mesh,
467 std::shared_ptr<MeshIterationStrategy> strategy = std::make_shared<FaceBasedIterator>()
468)
469{
470 // Consume the per-mesh cached, immutable topology bundle (CSR system sparsity +
471 // FaceToMatrixAddress + boundary sparsity). These arrays depend only on mesh topology, so they
472 // are shared by every LinearSystem built on this mesh; only the per-system value/RHS vectors
473 // below are allocated fresh.
474 auto bundle = readOrCreateSparsityBundle<
475 typename SystemMatrixType::MatrixSparsityType,
476 typename BoundaryMatrixType::MatrixSparsityType>(mesh);
477 const auto& sp = bundle.systemSparsity;
478 const auto& mi = bundle.faceToMatrixAddress;
479 const auto& bSp = bundle.boundarySparsity;
480 const auto exec = sp->exec();
481 const auto nCells = toLocalIdx(mesh.nCells());
482 const auto nProcFaces = mesh.nProcBoundaryFaces();
483 using IndexType = typename BoundaryMatrixType::MatrixSparsityType::SparsityIndexType;
484
485 // Off-diagonal / proc-face sparsity stays per-system (comm-pattern-derived); not shared in v1.
486 Vector<IndexType> offDiagColIdxs(exec, nProcFaces, 0);
487 Vector<IndexType> offDiagRowIdxs(exec, nProcFaces, 0);
488
489#ifdef NF_WITH_MPI_SUPPORT
490 auto commPattern = computeCommunicationPattern(mesh);
491 if (nProcFaces > 0)
492 {
493 const auto nBoundaryFaces = toLocalIdx(mesh.nBoundaryFaces());
494 const auto faceOwnersH = mesh.boundaryMesh().faceOwners().copyToHost();
495 const auto faceOwnersV = faceOwnersH.view();
496 Vector<IndexType> rowH(SerialExecutor {}, nProcFaces, 0);
497 Vector<IndexType> colH(SerialExecutor {}, nProcFaces, 0);
498 auto rowHV = rowH.view();
499 auto colHV = colH.view();
500 for (localIdx i = 0; i < nProcFaces; ++i)
501 {
502 // Store the local row index directly. The global offset used to be added here and
503 // subtracted again on the Ginkgo side; keeping the rows local avoids that round-trip.
504 // The column index stays global (it identifies a remote cell) and is consumed by
505 // Ginkgo's distributed index_map.
506 rowHV[i] = faceOwnersV[nBoundaryFaces + i];
507 colHV[i] = static_cast<IndexType>(commPattern.recvIdx[static_cast<sizeType>(i)]);
508 }
509 // offDiagRowSortPerm[j] = proc-face index whose row/col belongs at sorted position j.
510 // Already computed (and stored) in BoundaryMesh; reuse it here to avoid re-sorting.
511 const auto& offDiagRowSortPerm = mesh.boundaryMesh().getRowSortPerm();
512 {
513 std::vector<IndexType> sortedRow(static_cast<sizeType>(nProcFaces));
514 std::vector<IndexType> sortedCol(static_cast<sizeType>(nProcFaces));
515 for (localIdx j = 0; j < nProcFaces; ++j)
516 {
517 auto src = offDiagRowSortPerm[static_cast<sizeType>(j)];
518 sortedRow[static_cast<sizeType>(j)] = rowHV[src];
519 sortedCol[static_cast<sizeType>(j)] = colHV[src];
520 }
521 offDiagRowIdxs = Vector<IndexType>(exec, std::move(sortedRow));
522 offDiagColIdxs = Vector<IndexType>(exec, std::move(sortedCol));
523 }
524 commPattern.offDiagRowSortPerm = std::move(offDiagRowSortPerm);
525 }
526#endif
527
528 auto offDiagSp = std::make_shared<const typename BoundaryMatrixType::MatrixSparsityType>(
529 std::move(offDiagColIdxs), std::move(offDiagRowIdxs), Dimensions {nCells, nCells}
530 );
531
532 LinearSystem<ValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType> ls {
533 SystemMatrixType(Vector<ValueType>(sp->exec(), sp->nnz(), zero<ValueType>()), sp, mi),
534 Vector<RHSValueType>(sp->exec(), sp->rows(), zero<RHSValueType>()),
535 BoundaryMatrixType(Vector<ValueType>(exec, nProcFaces, zero<ValueType>()), offDiagSp),
536 BoundaryMatrixType(Vector<ValueType>(bSp->exec(), bSp->nnz(), zero<ValueType>()), bSp),
537 Vector<RHSValueType>(bSp->exec(), bSp->nnz(), zero<RHSValueType>()),
538 strategy
539 };
540
541#ifdef NF_WITH_MPI_SUPPORT
542 // ls.commPattern() = std::move(commPattern);
543#endif
544
545 return ls;
546}

554template<
555 typename MatrixValueType,
556 typename RHSValueType,
557 typename SystemMatrixType,
558 typename BoundaryMatrixType>
559inline la::LinearSystem<MatrixValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType>
560removeBoundaryContributions(
561 const la::LinearSystem<MatrixValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType>&
562 lsIn
563)
564{
565 auto ls =
566 la::LinearSystem<MatrixValueType, RHSValueType, SystemMatrixType, BoundaryMatrixType>(lsIn);
567 auto lsView = ls.view();
568 auto& matrix = lsView.matrix;
569 auto& rhs = lsView.rhs;
570 auto& bMatrix = lsView.boundaryMatrix;
571 auto& bRhs = lsView.boundaryRhs;
572
573 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
574
575 parallelFor(
576 ls.exec(),
577 {0, bMatrix.values.size()},
578 NEON_LAMBDA(const localIdx facei) {
579 const auto celli = bMatrix.sparsity.rowOffs[facei]; // cell index stored in rowOffs
580 Kokkos::atomic_add(&matrix.values[ma.diagIdx(celli)], bMatrix.values[facei]);
581 Kokkos::atomic_add(&rhs[celli], bRhs[facei]);
582 },
583 "removeBoundaryContributions"
584 );
585
586 return ls;
587}
