151inline void
152crossInto(const Executor& exec, Vector<Vec3>& target, const Vector<Vec3>& other)
153{
154 auto out = target.view();
155 auto rhs = other.view();
156 parallelFor(
157 exec,
158 {0, out.size()},
159 KOKKOS_LAMBDA(const localIdx i) { out[i] = out[i] ^ rhs[i]; },
160 "vec3FieldCross"
161 );
162}
163
164} // namespace volumeFieldDetail

173inline VolumeField<Vec3> cross(const VolumeField<Vec3>& lhs, const VolumeField<Vec3>& rhs)
174{
175 VolumeField<Vec3> result(lhs);
176 volumeFieldDetail::crossInto(result.exec(), result.internalVector(), rhs.internalVector());
177 volumeFieldDetail::crossInto(result.exec(), result.boundaryData().value(), rhs.boundaryData().value());
178 return result;
179}
