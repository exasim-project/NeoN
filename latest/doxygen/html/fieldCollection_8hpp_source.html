386template<typename VectorType>
387class CreateFromExistingVector
388{
389public:
390
391 std::string name;
392 const VectorType& field;
393 std::int64_t timeIndex = std::numeric_limits<std::int64_t>::max();
394 std::int64_t iterationIndex = std::numeric_limits<std::int64_t>::max();
395 std::int64_t subCycleIndex = std::numeric_limits<std::int64_t>::max();
396
397 VectorDocument operator()(Database& db)
398 {
399 Field<typename VectorType::VectorValueType> domainVector(
400 field.mesh().exec(), field.internalVector(), field.boundaryData()
401 );
402
403 VectorType vf(
404 field.exec(),
405 name,
406 field.mesh(),
407 std::move(domainVector),
408 field.boundaryConditions(),
409 db,
410 "",
411 ""
412 );
413
414 if (field.registered())
415 {
416 const VectorCollection& fieldCollection = VectorCollection::instance(field);
417 const VectorDocument& fieldDoc = fieldCollection.fieldDoc(field.key);
418 if (timeIndex == std::numeric_limits<std::int64_t>::max())
419 {
420 timeIndex = fieldDoc.timeIndex();
421 }
422 if (iterationIndex == std::numeric_limits<std::int64_t>::max())
423 {
424 iterationIndex = fieldDoc.iterationIndex();
425 }
426 if (subCycleIndex == std::numeric_limits<std::int64_t>::max())
427 {
428 subCycleIndex = fieldDoc.subCycleIndex();
429 }
430 }
431 // The field is moved in: an initializer list would copy it twice.
432 Dictionary dict(
433 {{"name", vf.name},
434 {"timeIndex", timeIndex},
435 {"iterationIndex", iterationIndex},
436 {"subCycleIndex", subCycleIndex}}
437 );
438 dict.insert("field", std::any(std::move(vf)));
439 return NeoN::Document(std::move(dict), validateVectorDoc);
440 }
441};
