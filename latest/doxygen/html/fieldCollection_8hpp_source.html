382template<typename VectorType>
383class CreateFromExistingVector
384{
385public:
386
387 std::string name;
388 const VectorType& field;
389 std::int64_t timeIndex = std::numeric_limits<std::int64_t>::max();
390 std::int64_t iterationIndex = std::numeric_limits<std::int64_t>::max();
391 std::int64_t subCycleIndex = std::numeric_limits<std::int64_t>::max();
392
393 VectorDocument operator()(Database& db)
394 {
395 Field<typename VectorType::VectorValueType> domainVector(
396 field.mesh().exec(), field.internalVector(), field.boundaryData()
397 );
398
399 VectorType vf(
400 field.exec(),
401 name,
402 field.mesh(),
403 std::move(domainVector),
404 field.boundaryConditions(),
405 db,
406 "",
407 ""
408 );
409
410 if (field.registered())
411 {
412 const VectorCollection& fieldCollection = VectorCollection::instance(field);
413 const VectorDocument& fieldDoc = fieldCollection.fieldDoc(field.key);
414 if (timeIndex == std::numeric_limits<std::int64_t>::max())
415 {
416 timeIndex = fieldDoc.timeIndex();
417 }
418 if (iterationIndex == std::numeric_limits<std::int64_t>::max())
419 {
420 iterationIndex = fieldDoc.iterationIndex();
421 }
422 if (subCycleIndex == std::numeric_limits<std::int64_t>::max())
423 {
424 subCycleIndex = fieldDoc.subCycleIndex();
425 }
426 }
427 // The field is moved in: an initializer list would copy it twice.
428 Dictionary dict(
429 {{"name", vf.name},
430 {"timeIndex", timeIndex},
431 {"iterationIndex", iterationIndex},
432 {"subCycleIndex", subCycleIndex}}
433 );
434 dict.insert("field", std::any(std::move(vf)));
435 return NeoN::Document(std::move(dict), validateVectorDoc);
436 }
437};
