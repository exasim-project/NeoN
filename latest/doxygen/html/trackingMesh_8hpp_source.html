133KOKKOS_INLINE_FUNCTION
134bool pointInCell(const Vec3& x, const label c, const TrackingMeshView& m)
135{
136 const auto [start, end] = m.cellFaces.bounds(c);
137 for (localIdx k = start; k < end; k++)
138 {
139 const label f = m.cellFaces.values[k];
140 Vec3 normal = m.faceAreaVectors[f];
141 if (m.faceOwners[f] != c)
142 {
143 normal = scalar(-1) * normal;
144 }
145 if ((normal & (x - m.faceCenters[f])) > scalar(0))
146 {
147 return false;
148 }
149 }
150 return true;
151}

165KOKKOS_INLINE_FUNCTION
166label locateCell(Vec3& x, label hint, const TrackingMeshView& m)
167{
168 label c = hint;
169 if (pointInCell(x, c, m))
170 {
171 return c;
172 }
173
174 Vec3 d = x - m.cellCenters[c];
175 scalar nearestDistSqr = d & d;
176 // every step strictly decreases the distance to x, so no cell is visited twice and
177 // the walk ends after at most nCells steps (the -1 return below is the real exit)
178 for (localIdx step = 0; step < m.nCells; step++)
179 {
180 label nearestCell = -1;
181 Vec3 nearestX = x;
182 const auto [start, end] = m.cellFaces.bounds(c);
183 for (localIdx k = start; k < end; k++)
184 {
185 const label f = m.cellFaces.values[k];
186 label n;
187 Vec3 xn = x;
188 if (m.faceType[f] == label(TrackingFaceType::internal))
189 {
190 // internal faces have f < nInternalFaces, so faceNeighbors[f] exists
191 n = (m.faceOwners[f] == c) ? m.faceNeighbors[f] : m.faceOwners[f];
192 }
193 else if (m.faceType[f] == label(TrackingFaceType::cyclic))
194 {
195 n = m.cyclicPartnerCell[f];
196 xn = x + m.cyclicShift[f];
197 }
198 else
199 {
200 continue;
201 }
202
203 if (pointInCell(xn, n, m))
204 {
205 x = xn;
206 return n;
207 }
208 const Vec3 dn = xn - m.cellCenters[n];
209 const scalar distSqr = dn & dn;
210 if (distSqr < nearestDistSqr)
211 {
212 nearestDistSqr = distSqr;
213 nearestCell = n;
214 nearestX = xn;
215 }
216 }
217 if (nearestCell == -1)
218 {
219 return -1;
220 }
221 c = nearestCell;
222 x = nearestX;
223 }
224 return -1;
225}
