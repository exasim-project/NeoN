32inline int bufferHash(const std::string& str)
33{
34 // There is also an MPI environment value for that, but somehow it doesn't work using that
35 // Also reserve 10 tags for other uses
36 constexpr int maxTagValue = 32767 - 10;
37 sizeType tag = std::hash<std::string> {}(str);
38 tag &= 0x7FFFFFFF; // turn 'int' signed bit to 0, MPI does not like negative tags.
39 return (static_cast<int>(tag) % maxTagValue) + 10;
40}

99 template<typename valueType>
100 void setCommRankSize(std::vector<sizeType> rankCommSize)
101 {
102 NF_DEBUG_ASSERT(
103 !isCommInit(), "Communication buffer was initialised by name: " << commName_ << "."
104 );
105 NF_DEBUG_ASSERT(

215 template<typename valueType>
216 void setType()
217 {
218 NF_DEBUG_ASSERT(
219 !isCommInit(), "Communication buffer was initialised by name: " << commName_ << "."
220 );
221 if (0 == (typeSize_ - sizeof(valueType))) return;
222 updateDataSize(
223 [rankOffset = rankOffset_, typeSize = typeSize_](const sizeType rank)
224 { return (rankOffset[rank + 1] - rankOffset[rank]) / typeSize; },
225 sizeof(valueType)
226 );
227
228 typeSize_ = sizeof(valueType);
229 }

238 template<typename func>
239 void updateDataSize(func rankSize, sizeType newSize)
240 {
241 sizeType dataSize = 0;
242 for (sizeType rank = 0; rank < mpiEnviron_.sizeRank(); ++rank)
243 {
244 rankOffset_[rank] = dataSize;
245 dataSize += rankSize(rank) * newSize;
246 }
247 rankOffset_.back() = dataSize;
248 if (rankBuffer_.size() < dataSize) rankBuffer_.resize(dataSize); // we never size down.
249 }
