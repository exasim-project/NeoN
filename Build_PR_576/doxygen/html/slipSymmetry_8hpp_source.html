15// Shared implementation behind the `slip` and `symmetry` boundary conditions: they apply the same
16// operator and differ only in their registered name and in where they may be applied (slip on a
17// wall/regular patch, symmetry on a symmetry-plane patch). The operator is, per face:
18// scalar => zero-gradient
19// vector => tangential projection of the boundary value (the normal component is removed), plus
20// a deferred normal-damping surface-normal gradient -deltaCoeffs*(v·n)*n that drives
21// the normal component of the cell value towards zero via the per-component RHS.

33// --- Scalar specialization: zero-gradient ---
34template<>
35inline void setSlipSymmetryValue<NeoN::scalar>(
36 Field<NeoN::scalar>& domainVector,
37 const UnstructuredMesh& mesh,
38 std::pair<localIdx, localIdx> range
39)
40{
41 const auto internalV = domainVector.internalVector().view();
42
43 auto [refGradV, valueV, valueFractionV, refValueV, boundaryFaceOwnersV] = views(
44 domainVector.boundaryData().refGrad(),
45 domainVector.boundaryData().value(),
46 domainVector.boundaryData().valueFraction(),
47 domainVector.boundaryData().refValue(),
48 mesh.boundaryMesh().faceOwners()
49 );
50
51 NeoN::parallelFor(
52 domainVector.exec(),
53 range,
54 NEON_LAMBDA(const localIdx i) {
55 const localIdx owner = boundaryFaceOwnersV[i];
56 const auto v = internalV[owner];
57
58 refValueV[i] = v;
59 valueV[i] = v;
60 valueFractionV[i] = 0.0;
61 refGradV[i] = 0.0;
62 },
63 "setSlipSymmetryValue(scalar)"
64 );
65}

67// --- Vec3 specialization: tangential projection + deferred normal damping ---
68template<>
69inline void setSlipSymmetryValue<NeoN::Vec3>(
70 Field<NeoN::Vec3>& domainVector,
71 const UnstructuredMesh& mesh,
72 std::pair<localIdx, localIdx> range
73)
74{
75 const auto internalV = domainVector.internalVector().view();
76
77 auto
78 [refGradV,
79 valueV,
80 valueFractionV,
81 refValueV,
82 boundaryFaceOwnersV,
83 faceUnitNormalsV,
84 deltaCoeffsV] =
85 views(
86 domainVector.boundaryData().refGrad(),
87 domainVector.boundaryData().value(),
88 domainVector.boundaryData().valueFraction(),
89 domainVector.boundaryData().refValue(),
90 mesh.boundaryMesh().faceOwners(),
91 mesh.boundaryMesh().faceUnitNormals(),
92 mesh.boundaryMesh().deltaCoeffs()
93 );
94
95 NeoN::parallelFor(
96 domainVector.exec(),
97 range,
98 NEON_LAMBDA(const localIdx i) {
99 const localIdx owner = boundaryFaceOwnersV[i];
100 const auto v = internalV[owner];
101 const auto n = faceUnitNormalsV[i];
102
103 const auto un = (v & n); // normal component (scalar)
104 const auto vtan = v - n * un; // tangential projection (remove normal component)
105
106 // Explicit boundary value: used by convection / face interpolation (no penetration).
107 refValueV[i] = vtan;
108 valueV[i] = vtan;
109
110 // valueFraction = 0: keep the diagonal contribution component-isotropic so the shared
111 // scalar matrix and multi-RHS solve are preserved.
112 valueFractionV[i] = 0.0;
113
114 // Normal damping via the surface-normal gradient -deltaCoeffs*(v·n)*n enters the
115 // per-component RHS through the existing fixed-gradient assembly.
116 refGradV[i] = n * (-deltaCoeffsV[i] * un);
117 },
118 "setSlipSymmetryValue(Vec3)"
119 );
120}
