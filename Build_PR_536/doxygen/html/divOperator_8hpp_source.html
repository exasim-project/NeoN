90 // copy constructor
91 DivOperator(const DivOperator& divOp)

96 sameTypeStrategy_(divOp.sameTypeStrategy_ ? divOp.sameTypeStrategy_->clone() : nullptr),

103 const SurfaceField<scalar>& faceFlux,
104 const VolumeField<FieldValueType>& phi,

106 )
107 : dsl::OperatorMixin<VolumeField<FieldValueType>>(
108 phi.exec(), dsl::Coeff(1.0), phi, termType
109 ),
110 faceFlux_(faceFlux),
111 sameTypeStrategy_(DivOperatorFactory<FieldValueType, FieldValueType>::create(
112 phi.exec(), phi.mesh(), input
113 )),
114 scalarMtxStrategy_(nullptr)
115 {
116 if constexpr (!std::is_same_v<FieldValueType, scalar>)
117 {
118 scalarMtxStrategy_ =
119 DivOperatorFactory<FieldValueType, scalar>::create(phi.exec(), phi.mesh(), input);
120 }
121 };
122
123 DivOperator(
124 dsl::Operator::Type termType,
125 const SurfaceField<scalar>& faceFlux,
126 const VolumeField<FieldValueType>& phi
127 )
128 : dsl::OperatorMixin<VolumeField<FieldValueType>>(
129 phi.exec(), dsl::Coeff(1.0), phi, termType
130 ),
131 faceFlux_(faceFlux), sameTypeStrategy_(nullptr), scalarMtxStrategy_(nullptr) {};
132
133
134 void explicitOperation(Vector<FieldValueType>& source) const
135 {
136 NF_ASSERT(sameTypeStrategy_, "DivOperatorStrategy not initialized");
137 auto tmpsource =
138 Vector<FieldValueType>(source.exec(), source.size(), zero<FieldValueType>());
139 const auto operatorScaling = this->getCoefficient();
140 sameTypeStrategy_->div(tmpsource, faceFlux_, this->getVector(), operatorScaling);
141 source += tmpsource;
142 }
143
144 void implicitOperation(la::LinearSystem<FieldValueType, FieldValueType>& ls) const
145 {
146 NF_ASSERT(sameTypeStrategy_, "DivOperatorStrategy not initialized");
147 const auto operatorScaling = this->getCoefficient();
148 sameTypeStrategy_->div(ls, faceFlux_, this->getVector(), operatorScaling);
149 }
150
151 /* @brief Implicit assembly into a scalar-matrix / FieldValueType-rhs linear system
152 * (segregated vector-solve form). Only present when FieldValueType != scalar;
153 * for scalar fields the same-type overload above already covers this signature.
154 */
155 template<typename F = FieldValueType>
156 requires(!std::is_same_v<F, scalar>)
157 void implicitOperation(la::LinearSystem<scalar, FieldValueType>& ls) const
158 {
159 NF_ASSERT(scalarMtxStrategy_, "Scalar-matrix DivOperatorStrategy not initialized");
160 const auto operatorScaling = this->getCoefficient();
161 scalarMtxStrategy_->div(ls, faceFlux_, this->getVector(), operatorScaling);
162 }
163
164 void read(const Input& input)
165 {
166 const UnstructuredMesh& mesh = this->getVector().mesh();
167 NeoN::TokenList tokens;
168 if (std::holds_alternative<NeoN::Dictionary>(input))
169 {
170 auto dict = std::get<NeoN::Dictionary>(input);
171 std::string schemeName = "div(" + faceFlux_.name + "," + this->getVector().name + ")";
172 tokens = dict.subDict("divSchemes").get<NeoN::TokenList>(schemeName);
173 }
174 else
175 {
176 tokens = std::get<NeoN::TokenList>(input);
177 }
178 sameTypeStrategy_ =
179 DivOperatorFactory<FieldValueType, FieldValueType>::create(this->exec(), mesh, tokens);
180 if constexpr (!std::is_same_v<FieldValueType, scalar>)
181 {
182 scalarMtxStrategy_ =
183 DivOperatorFactory<FieldValueType, scalar>::create(this->exec(), mesh, tokens);
184 }
185 }
186
187 std::string getName() const { return "DivOperator"; }
188
189private:
190
191 const SurfaceField<NeoN::scalar>& faceFlux_;
192
193 std::unique_ptr<DivOperatorFactory<FieldValueType, FieldValueType>> sameTypeStrategy_;
194 // Only initialized when FieldValueType != scalar; used to assemble into a
195 // LinearSystem<scalar, FieldValueType> for segregated vector solves.
196 std::unique_ptr<DivOperatorFactory<FieldValueType, scalar>> scalarMtxStrategy_;
197};
