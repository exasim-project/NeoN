172 void communicate(CommunicationPattern& commPattern)
173 {
174 auto mpiEnv = commPattern.env;
175 int commRanks = mpiEnv.sizeRank();
176
177 // auto boundaryMatrixMap = Vector<localIdx>(exec(), commPattern.boundaryMapVector);
178 auto ftma = matrix_.faceToMatrixAddress();
179 auto nsp = ftma->nonLocalSparsityPattern();
180 auto rowToDiagonalMap = la::computeRowToDiagonalMap(nsp->rowOffs(), ftma);
181
182 // 1. copy bValues which need to be communicated into sendBuffer
183 auto commSize = commPattern.sendCounts[mpiEnv.sizeRank()];
184 auto recvBuffer = Vector<ValueType>(exec(), commSize);
185
186 // TODO compute using scan
187 auto sdispls = std::vector<int>(commRanks, 0);
188 for (int i = 1; i < sdispls.size(); i++)
189 {
190 auto prev = sdispls[i - 1];
191 sdispls[i] = commPattern.sendCounts[i - 1] + prev;
192 }
193
194 MPI_Alltoallv(
195 nonLocalMatrix_.values().data(),
196 commPattern.sendCounts.data(),
197 sdispls.data(),
198 mpi::getType<ValueType>(),
199 recvBuffer.data(),
200 commPattern.sendCounts.data(),
201 sdispls.data(),
202 mpi::getType<ValueType>(),
203 mpiEnv.comm()
204 );
205
206 std::cout << __FILE__ << ":" << __LINE__ << " rank " << mpiEnv.rank() << " recvBuffer "
207 << recvBuffer.view()[0] << " matrixValue " << matrix_.values().view()[9] << "\n";
208
209 // 3. apply received values to corresponding matrix
210 // add diagonal contributions
211 add(recvBuffer, rowToDiagonalMap, matrix_.values());
212 std::cout << __FILE__ << ":" << __LINE__ << " rank " << mpiEnv.rank() << " recvBuffer "
213 << recvBuffer.view()[0] << " matrixValue " << matrix_.values().view()[9] << "\n";
214 }

303template<typename ValueType>
304inline LinearSystem<ValueType> removeBoundaryContributions(const LinearSystem<ValueType>& lsIn)
305{
306 auto ls = LinearSystem<ValueType>(lsIn);
307 auto lsView = ls.view();
308 auto& matrix = lsView.matrix;
309 auto& rhs = lsView.rhs;
310 auto& bMatrix = lsView.boundaryMatrix;
311 auto& bRhs = lsView.boundaryRhs;
312
313 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
314
315 parallelFor(
316 ls.exec(),
317 {0, bMatrix.values.size()},
318 NEON_LAMBDA(const localIdx facei) {
319 const auto celli = bMatrix.sparsity.rowOffs[facei];
320 Kokkos::atomic_add(&matrix.values[ma.diagIdx(celli)], bMatrix.values[facei]);
321 Kokkos::atomic_add(&rhs[celli], bRhs[facei]);
322 },
323 "removeBoundaryContributions"
324 );
325
326 return ls;
327}

329inline void scaledInvDiagNegLUx(
330 const LinearSystem<Vec3>& ls,
331 const Vector<Vec3>& a,
332 const Vector<Vec3>& aBound,
333 const UnstructuredMesh& mesh,
334 Vector<scalar>& rAU,
335 Vector<Vec3>& out
336)
337{
338 auto& mtx = ls.matrix();
339 auto& vol = mesh.cellVolumes();
340 auto& bMesh = mesh.boundaryMesh();
341 NF_ASSERT(mtx.nRows() == a.size(), "Dimension mismatch");
342 NF_ASSERT(mtx.nRows() == out.size(), "Dimension mismatch");
343
344 const auto [rowOffsV, colIdxV, matrixV, rAUV, volV, aV, aBoundV, bV] = views(
345 mtx.sparsity()->rowOffs(),
346 ls.matrix().sparsity()->colIdxs(),
347 mtx.values(),
348 rAU,
349 vol,
350 a,
351 aBound,
352 ls.rhs()
353 );
354 const auto [nonLocalMtx, nonLocalRows] =
355 views(ls.nonLocalMatrix().values(), ls.nonLocalMatrix().rowOffs());
356 auto outV = out.view();
357
358 auto procFacesStart = bMesh.nBoundaryFaces();
359
360 parallelFor(
361 mtx.exec(),
362 {0, mtx.nRows()},
363 NEON_LAMBDA(const localIdx rowi) {
364 outV[rowi] = zero<Vec3>();
365 for (auto i = rowOffsV[rowi]; i < rowOffsV[rowi + 1]; i++)
366 {
367 auto colI = colIdxV[i];
368 if (rowi == colI)
369 {
370 rAUV[rowi] = volV[rowi] / matrixV[i][0];
371 }
372 else
373 {
374 outV[rowi] -= matrixV[i] * aV[colI];
375 }
376 }
377
378 // FIXME this scans all boundary values every iteration
379 localIdx curNonLocalRow = 0;
380 auto nonLocalVal = zero<Vec3>();
381 for (auto i = 0; i < nonLocalRows.size(); i++)
382 {
383 if (nonLocalRows[i] == rowi)
384 {
385 nonLocalVal = nonLocalMtx[i];
386 curNonLocalRow = i;
387 // FIXME is the sign correct?
388 outV[rowi] -= nonLocalVal * aBoundV[procFacesStart + curNonLocalRow];
389 }
390 }
391
392 outV[rowi] += bV[rowi];
393 outV[rowi] *= rAUV[rowi] / volV[rowi];
394 }
395 );
396}
