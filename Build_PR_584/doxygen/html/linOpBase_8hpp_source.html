17// CRTP base for the matrix-free Ginkgo operators here: it bundles gko::EnableLinOp and
18// gko::EnableCreateMethod and supplies the one advanced apply_impl(alpha, b, beta, x) through
19// the derived class' apply_impl(b, x). How to derive: report/blockamr-linear-algebra-notes.md
20template<class D, class V = double>
21class AmrexLinOpBase : public gko::EnableLinOp<D>, public gko::EnableCreateMethod<D>
22{
23protected:
24
25 using DenseV = gko::matrix::Dense<V>;
26
27

29 {}
30
31 AmrexLinOpBase(std::shared_ptr<const gko::Executor> exec, const gko::dim<2>& size)
32 : gko::EnableLinOp<D>(exec, size)
33 {}

35 // Supplied by the derived class; re-declared so the advanced overload can call it.
36 void apply_impl(const gko::LinOp* b, gko::LinOp* x) const override = 0;
37
38 // x = alpha * op(b) + beta * x. Every derived apply_impl(b, x) OVERWRITES all of x, so the
39 // intermediate is a reused scratch rather than a clone of x. The beta branches skip whole
40 // vector passes: beta == 1 needs no scale, beta == 0 discards x and needs no scratch.
41 void apply_impl(
42 const gko::LinOp* alpha, const gko::LinOp* b, const gko::LinOp* beta, gko::LinOp* x
43 ) const override
44 {
45 prof::Timer tAll("adv.apply");
46 // A view, so scale/add_scaled hit x's own memory; both are elementwise.
47 auto denseX = localView<V>(x);
48 const double alphaVal = hostScalar(alpha);
49 const double betaVal = hostScalar(beta);
50
51 if (betaVal == 0.0)
52 {
53 this->apply_impl(b, x);
54 if (alphaVal != 1.0)
55 {
56 prof::Timer t("adv.scale");
57 denseX->scale(alpha);
58 }
59 return;
60 }
61
62 const gko::dim<2> size = denseX->get_size();
63 if (!scratch_ || scratch_->get_size() != size)
64 {
65 prof::Timer t("adv.alloc");
66 scratch_ = DenseV::create(this->get_executor(), size);
67 }
68 this->apply_impl(b, scratch_.get());
69 if (betaVal != 1.0)
70 {
71 prof::Timer t("adv.scale");
72 denseX->scale(beta);
73 }
74 {
75 prof::Timer t("adv.addscaled");
76 denseX->add_scaled(alpha, scratch_);
77 }
78 }
79
80private:
81
82 // alpha/beta are 1x1 Dense on the solve executor; a device value is staged through the
83 // host master.
84 static double hostScalar(const gko::LinOp* s)
85 {
86 auto d = gko::as<DenseV>(s);
87 auto exec = d->get_executor();
88 if (exec->get_master().get() != exec.get())
89 {
90 return gko::clone(exec->get_master(), d)->at(0, 0);
91 }
92 return d->at(0, 0);
93 }
94
95 // shared_ptr, not unique_ptr: EnablePolymorphicAssignment gives these operators a
96 // copy-assignment, which a move-only member would delete. No state is kept across calls.
97 mutable std::shared_ptr<DenseV> scratch_;
98};
