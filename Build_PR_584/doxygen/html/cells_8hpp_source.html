15// y = a*x + y. No stencil, no ghosts: the pure bandwidth floor (2 reads, 1 write).
16template<class In, class Out>
17AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void
18axpyCell(In const& x, Out const& y, int i, int j, int k, double a)
19{
20 y(i, j, k) = a * x(i, j, k) + y(i, j, k);

23// Constant-coefficient 7-point Laplacian, 1 ghost -- the stencil shape the V-cycle's
24// matrix-free apply has: bandwidth-bound, 8 reads and 1 write per cell before caching.
25template<class In, class Out>
26AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void
27laplacianCell(In const& in, Out const& out, int i, int j, int k, double cx, double cy, double cz)
28{
29 const double s0 = in(i, j, k);
30 out(i, j, k) = cx * (in(i + 1, j, k) + in(i - 1, j, k) - 2.0 * s0)
31 + cy * (in(i, j + 1, k) + in(i, j - 1, k) - 2.0 * s0)
32 + cz * (in(i, j, k + 1) + in(i, j, k - 1) - 2.0 * s0);
33}

35// Harmonic-mean VanLeer correction, copied from stencilKernels.cpp so both stay comparable.
36AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE double vanleerCorr(double d_up, double d_down)
37{
38 const double prod = d_up * d_down;
39 return (prod > 0.0) ? 2.0 * prod / (d_up + d_down) : 0.0;
40}

42// VanLeer-limited upwind divergence, mirroring divVanLeerCell in stencilKernels.cpp. Two
43// ghost cells and three face fields -- FLOP-heavy and branchy, unlike the two above.
44template<class In, class Face, class Out>
45AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void divVanLeerCell(
46 In const& phi,
47 Face const& fx,
48 Face const& fy,
49 Face const& fz,
50 Out const& out,
51 int i,
52 int j,
53 int k,
54 double dx,
55 double dy,
56 double dz
57)
58{
59 double total = 0.0;
60 {
61 const double fl = fx(i, j, k), fr = fx(i + 1, j, k);
62 const double sm2 = phi(i - 2, j, k), sm1 = phi(i - 1, j, k), s0 = phi(i, j, k),
63 sp1 = phi(i + 1, j, k), sp2 = phi(i + 2, j, k);
64 const double dl = s0 - sm1;
65 const double pl = (fl >= 0.0) ? sm1 + 0.5 * vanleerCorr(sm1 - sm2, dl)
66 : s0 - 0.5 * vanleerCorr(sp1 - s0, dl);
67 const double dr = sp1 - s0;
68 const double pr = (fr >= 0.0) ? s0 + 0.5 * vanleerCorr(s0 - sm1, dr)
69 : sp1 - 0.5 * vanleerCorr(sp2 - sp1, dr);
70 total += (fr * pr - fl * pl) / dx;
71 }
72 {
73 const double fl = fy(i, j, k), fr = fy(i, j + 1, k);
74 const double sm2 = phi(i, j - 2, k), sm1 = phi(i, j - 1, k), s0 = phi(i, j, k),
75 sp1 = phi(i, j + 1, k), sp2 = phi(i, j + 2, k);
76 const double dl = s0 - sm1;
77 const double pl = (fl >= 0.0) ? sm1 + 0.5 * vanleerCorr(sm1 - sm2, dl)
78 : s0 - 0.5 * vanleerCorr(sp1 - s0, dl);
79 const double dr = sp1 - s0;
80 const double pr = (fr >= 0.0) ? s0 + 0.5 * vanleerCorr(s0 - sm1, dr)
81 : sp1 - 0.5 * vanleerCorr(sp2 - sp1, dr);
82 total += (fr * pr - fl * pl) / dy;
83 }
84 {
85 const double fl = fz(i, j, k), fr = fz(i, j, k + 1);
86 const double sm2 = phi(i, j, k - 2), sm1 = phi(i, j, k - 1), s0 = phi(i, j, k),
87 sp1 = phi(i, j, k + 1), sp2 = phi(i, j, k + 2);
88 const double dl = s0 - sm1;
89 const double pl = (fl >= 0.0) ? sm1 + 0.5 * vanleerCorr(sm1 - sm2, dl)
90 : s0 - 0.5 * vanleerCorr(sp1 - s0, dl);
91 const double dr = sp1 - s0;
92 const double pr = (fr >= 0.0) ? s0 + 0.5 * vanleerCorr(s0 - sm1, dr)
93 : sp1 - 0.5 * vanleerCorr(sp2 - sp1, dr);
94 total += (fr * pr - fl * pl) / dz;
95 }
96 out(i, j, k) = total;
97}
