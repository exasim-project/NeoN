46/* @brief Allocate the coefficient fields `mesh` implies. A free function rather than a base
47 * class: the format holds the fields itself, and this is only the derivation of the
48 * AMReX layouts from the mesh.
49 */
50inline void allocateCoefficients(
51 const MeshLevel& mesh,
52 Symmetry sym,
53 CellFieldLevel& alpha,
54 FaceFieldLevel& upper,
55 std::optional<FaceFieldLevel>& lower
56)
57{
58 // MultiFabs are not zero-initialised (the arena recycles memory), so a freshly built
59 // matrix is explicitly zeroed -- callers write only what their operator contributes.
60 auto zeroed = [](const amrex::BoxArray& ba, const amrex::DistributionMapping& dm)
61 {
62 auto mf = std::make_shared<amrex::MultiFab>(ba, dm, 1, 0);
63 mf->setVal(0.0);
64 return mf;
65 };
66 const bool asym = (sym == Symmetry::asymmetric);
67 alpha = CellFieldLevel {zeroed(mesh.ba, mesh.dm)};
68 FaceFieldLevel low {};
69 for (int d = 0; d < 3; ++d)
70 {
71 const auto i = static_cast<std::size_t>(d);
72 const amrex::BoxArray fba = amrex::convert(mesh.ba, amrex::IntVect::TheDimensionVector(d));
73 upper.dir[i] = zeroed(fba, mesh.dm);
74 if (asym)
75 {
76 low.dir[i] = zeroed(fba, mesh.dm);
77 }
78 }
79 if (asym)
80 {
81 lower = low;
82 }
83}
84
85inline void
86zeroCoefficients(CellFieldLevel& alpha, FaceFieldLevel& upper, std::optional<FaceFieldLevel>& lower)
87{
88 (*alpha).setVal(0.0);
89 for (int d = 0; d < 3; ++d)
90 {
91 upper[d].setVal(0.0);
92 // Nothing separate to zero when symmetric.
93 if (lower.has_value())
94 {
95 (*lower)[d].setVal(0.0);
96 }
97 }
98}

102/* @class MFFaceCoeffs
103 * @brief Matrix-free face-coefficient storage: no MATRIX is ever assembled -- la::toLinOp
104 * turns these fields into a FaceCoeffOp per solve. Build with symmetric()/
105 * asymmetric(), then write the coefficient fields directly.
106 */
107class MFFaceCoeffs
108{
109public:
110
111 // The fields below sit behind shared_ptr (MultiFab is not copyable), so a copy of the
112 // format SHARES them and a write through one is seen through the other.
113 NeoN::Executor exec {NeoN::SerialExecutor {}};
114 la::BcArray bc {};
115 // The layout the fields were ALLOCATED from, and nothing repoints them, so it is the one
116 // source for ba/dm rather than a second one competing with alpha's.
117 MeshLevel mesh;
118 // The negSumDiag seam: `alpha` is the cell-centred diagonal SOURCE (ddt/Sp/reaction), NOT
119 // the matrix diagonal alpha - (aE+aW+aN+aS+aT+aB), which an operator's += must not land on;
120 // deriving that one is the stencil's job.
121 CellFieldLevel alpha;
122 FaceFieldLevel upper;
123 // No low side to WRITE when symmetric: an operator writing both would double every
124 // coefficient. Storage still has one -- storedLower().
125 std::optional<FaceFieldLevel> lower; // nullopt when symmetric
126
127 // `bc` defaults to all-periodic (BcArray 0 == periodic).
128 static MFFaceCoeffs
129 symmetric(const NeoN::Executor& exec, MeshLevel mesh, const la::BcArray& bc = {})
130 {
131 return MFFaceCoeffs(exec, std::move(mesh), Symmetry::symmetric, bc);
132 }
133
134 static MFFaceCoeffs
135 asymmetric(const NeoN::Executor& exec, MeshLevel mesh, const la::BcArray& bc = {})
136 {
137 return MFFaceCoeffs(exec, std::move(mesh), Symmetry::asymmetric, bc);
138 }
139
140 void zero() { detail::zeroCoefficients(alpha, upper, lower); }
141
142 /* @brief The STORED low side, which always exists -- `upper` itself when symmetric, the
143 * alias FaceCoeffOp's convention wants. Deliberately NOT `lower`'s reading, where a
144 * symmetric matrix has none; the differing types keep the two from being confused.
145 */
146 FaceFieldLevel storedLower() const { return lower.value_or(upper); }
147
148 // Derived from `lower`, so no stored enum can drift from the storage.
149 Symmetry symmetry() const { return lower ? Symmetry::asymmetric : Symmetry::symmetric; }
150
151 // Rows this rank owns. localCount(), NOT numPts(): numPts() counts EVERY rank's cells.
152 std::size_t localRows() const { return la::localCount(*alpha); }
153
154private:
155
156 MFFaceCoeffs(
157 const NeoN::Executor& executor, MeshLevel meshLevel, Symmetry sym, const la::BcArray& bcs
158 )
159 : exec(executor), bc(bcs), mesh(std::move(meshLevel))
160 {
161 detail::allocateCoefficients(mesh, sym, alpha, upper, lower);
162 }
163};

165// Absence has exactly ONE spelling and only `lower` has it: with CellFieldLevel/FaceFieldLevel
166// non-nullable, "empty" is not expressible, which is why _la_matrix_probe's diag_empty/
167// upper_empty keys emit a literal false. Turn either into an optional and this fires.
168// (Was asserted in linearAlgebra/coefficientsConcepts.cpp, deleted with the concepts.)
169static_assert(std::is_same_v<decltype(MFFaceCoeffs::alpha), CellFieldLevel>);
170static_assert(std::is_same_v<decltype(MFFaceCoeffs::upper), FaceFieldLevel>);
171static_assert(std::is_same_v<decltype(MFFaceCoeffs::lower), std::optional<FaceFieldLevel>>);
172// The mesh TRAVELS with the coefficients, not optional and not a pointer: an operator cannot
173// write a face coefficient without dx, and passing it alongside made a mismatch representable
174// (ops::Laplacian used to hold its own amrex::Geometry).
175static_assert(std::is_same_v<decltype(MFFaceCoeffs::mesh), MeshLevel>);
