17// Cells this rank owns -- the length of the flat vectors below. NEVER
18// boxArray().numPts(), which counts EVERY rank's cells: that was the long-standing
19// multi-rank bug. Counted by the same MFIter walk gather/scatter use, so it agrees.
20template<class FA>
21inline std::size_t localCount(const FA& mf)
22{
23 std::size_t n = 0;
24 for (amrex::MFIter mfi(mf); mfi.isValid(); ++mfi)
25 {
26 n += static_cast<std::size_t>(mfi.validbox().numPts());
27 }
28 return n;
29}

31// Flat-vector <-> MultiFab transfer (component 0, valid cells only). INVARIANT: gather and
32// scatter MUST traverse in the identical order -- MFIter without tiling, then k,j,i over the
33// valid box. `scale` lets gather apply the SPD sign flip (-L) in one pass.
34template<class V, class FA>
35void gather(const FA& mf, V* buf, double scale)
36{
37 using T = typename FA::value_type;
38 const bool hostOk = mf.arena()->isHostAccessible();
39 amrex::Gpu::streamSynchronize();
40 std::size_t idx = 0;
41 for (amrex::MFIter mfi(mf); mfi.isValid(); ++mfi)
42 {
43 const amrex::Box& vbx = mfi.validbox();
44 const auto& fab = mf[mfi];
45 const amrex::Box& fbx = fab.box();
46 std::vector<T> stage;
47 auto arr = fab.const_array();
48 if (!hostOk)
49 {
50 // Component 0 occupies the first numPts() elements of the fab.
51 stage.resize(static_cast<std::size_t>(fbx.numPts()));
52 amrex::Gpu::dtoh_memcpy(stage.data(), fab.dataPtr(), stage.size() * sizeof(T));
53 arr = amrex::makeArray4<const T>(stage.data(), fbx, 1);
54 }
55 const auto lo = amrex::lbound(vbx);
56 const auto hi = amrex::ubound(vbx);
57 for (int k = lo.z; k <= hi.z; ++k)
58 {
59 for (int j = lo.y; j <= hi.y; ++j)
60 {
61 for (int i = lo.x; i <= hi.x; ++i)
62 {
63 buf[idx++] = static_cast<V>(scale * static_cast<double>(arr(i, j, k)));
64 }
65 }
66 }
67 }
68}

70template<class V, class FA>
71void scatter(const V* buf, FA& mf)
72{
73 using T = typename FA::value_type;
74 const bool hostOk = mf.arena()->isHostAccessible();
75 amrex::Gpu::streamSynchronize();
76 std::size_t idx = 0;
77 for (amrex::MFIter mfi(mf); mfi.isValid(); ++mfi)
78 {
79 const amrex::Box& vbx = mfi.validbox();
80 auto& fab = mf[mfi];
81 const amrex::Box& fbx = fab.box();
82 std::vector<T> stage;
83 auto arr = fab.array();
84 if (!hostOk)
85 {
86 // Round-trip the full fab so ghost values survive the update.
87 stage.resize(static_cast<std::size_t>(fbx.numPts()));
88 amrex::Gpu::dtoh_memcpy(stage.data(), fab.dataPtr(), stage.size() * sizeof(T));
89 arr = amrex::makeArray4<T>(stage.data(), fbx, 1);
90 }
91 const auto lo = amrex::lbound(vbx);
92 const auto hi = amrex::ubound(vbx);
93 for (int k = lo.z; k <= hi.z; ++k)
94 {
95 for (int j = lo.y; j <= hi.y; ++j)
96 {
97 for (int i = lo.x; i <= hi.x; ++i)
98 {
99 arr(i, j, k) = static_cast<T>(buf[idx++]);
100 }
101 }
102 }
103 if (!hostOk)
104 {
105 amrex::Gpu::htod_memcpy(fab.dataPtr(), stage.data(), stage.size() * sizeof(T));
106 }
107 }
108}
