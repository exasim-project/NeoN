10// The 7-point face-coefficient stencil at ONE cell: the negSumDiag diagonal
11// alpha - (aE+aW+aN+aS+aT+aB) and its off-diagonal partner. Every matrix format, smoother and
12// residual in the component must agree BIT-FOR-BIT on this arithmetic
13// (matrixFree/faceCoeffOp.hpp:27-33, gmgKokkos/kernels.hpp:15-17), and that holds only while the
14// association order exists in exactly one place -- so it lives here rather than in gmg/, whose
15// kernels are one of four consumers.
16
17namespace blockamr::la
18{

20// The 6 face-coefficient VALUES at one cell (a loop body holds Array4 views, not FabArrays).
21template<class T>
22struct FaceCoeffVals
23{
24 T aE, aW, aN, aS, aT, aB;
25};
26
27// The 6 face-coefficient VIEWS over one box. Trivially copyable, so it captures into a device
28// lambda like the six Array4s it replaces; it exists so a kernel taking the set states that in
29// one parameter instead of six that a caller can transpose.
30template<class TC>
31struct FaceCoeffArrays
32{
33 amrex::Array4<const TC> ux, lx, uy, ly, uz, lz;
34};
35
36// aE/aN/aT are the HIGH faces (upper, index +1), aW/aS/aB the LOW ones (lower, index 0).
37// C is the type the CELL computes in; TC the coefficient STORAGE type, narrower on a
38// reduced-precision GMG level (gmg/bf16.hpp) -- hence two parameters and not one.
39template<class C, class TC>
40AMREX_GPU_HOST_DEVICE FaceCoeffVals<C>
41loadFaceCoeffs(const FaceCoeffArrays<TC>& a, int i, int j, int k) noexcept
42{
43 return {
44 static_cast<C>(a.ux(i + 1, j, k)),
45 static_cast<C>(a.lx(i, j, k)),
46 static_cast<C>(a.uy(i, j + 1, k)),
47 static_cast<C>(a.ly(i, j, k)),
48 static_cast<C>(a.uz(i, j, k + 1)),
49 static_cast<C>(a.lz(i, j, k))
50 };
51}
52
53// Summation order aE+aW+aN+aS+aT+aB must stay bit-for-bit identical at every site.
54// `alpha` keeps its own deduced type and the result its natural one: where the diagonal SOURCE
55// is stored narrower or wider than the cell computes, the single rounding must happen at the
56// caller's own `const C diag = ...`, exactly as it does when the formula is written inline.
57template<class A, class C>
58AMREX_GPU_HOST_DEVICE auto stencilDiag(A alpha, const FaceCoeffVals<C>& c) noexcept
59{
60 return alpha - (c.aE + c.aW + c.aN + c.aS + c.aT + c.aB);
61}
62
63// Same rule for the neighbour values: P is the FIELD's type, which is independent of the
64// coefficients' -- a level may store fp64 fields against bf16 coefficients.
65template<class C, class P>
66AMREX_GPU_HOST_DEVICE auto
67stencilOffDiag(const FaceCoeffVals<C>& c, P pE, P pW, P pN, P pS, P pT, P pB) noexcept
68{
69 return c.aE * pE + c.aW * pW + c.aN * pN + c.aS * pS + c.aT * pT + c.aB * pB;
70}
