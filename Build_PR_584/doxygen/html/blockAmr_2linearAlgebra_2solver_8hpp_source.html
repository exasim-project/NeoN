27/* @class Solver
28 * @brief Solves a LinearSystem, holding only an executor and a SolverConfig by value.
29 * Deliberately NO factory: parseSolverKind decided the method once. Returns
30 * la::SolveResult, as every solve entry point here does.
31 */
32class Solver
33{
34public:
35
36 Solver(const NeoN::Executor& exec, const la::SolverConfig& cfg) : exec_(exec), cfg_(cfg) {}
37
38 // rhs comes from the system, not the caller; `sol`'s values seed the initial guess.
39 la::SolveResult solve(const LinearSystem& system, amrex::MultiFab& sol) const
40 {
41 // Refused: these three want the hierarchy as the SOLVER, a different object with a
42 // different stopping test. Checked before the refusal pays for a hierarchy.
43 if (cfg_.solverKind == la::SolverKind::gmg || cfg_.solverKind == la::SolverKind::ir
44 || cfg_.solverKind == la::SolverKind::mpir)
45 {
46 throw std::runtime_error(
47 "la::Solver: solver '" + cfg_.solver
48 + "' needs the GMG hierarchy, which is built from the coefficient fields rather "
49 "than from a LinOp; use a Krylov method"
50 );
51 }
52 // ASKED FOR, not decided here: the hierarchy comes from the coefficient FIELDS.
53 std::shared_ptr<const gko::LinOp> precond = la::makeHierarchy(system.matrix(), cfg_);
54 SystemKrylovSolver s(exec_, system, cfg_, std::move(precond));
55 // The rhs is READ, never written (gather takes a const FA&); the cast lives here,
56 // once, rather than widening ISolver's signature.
57 return s.solve(const_cast<amrex::MultiFab&>(system.rhs()), sol);
58 }
59
60private:

67 SystemKrylovSolver(

72 )
73 : la::KrylovSolver(

76 // rather than of a LinOp built for the question: toLinOp() stages pinned

85 cfg.solver,

87 std::move(precond)
88 );
89 }
