26// The 7 coefficient fabs held by shared ownership: the bottom operator is generated once and
27// applied throughout the solve, so it must keep its level's fabs alive, where FaceCoeffs<T>'s
28// raw views only have to survive one kernel call.
29template<class T>
30struct OwnedFaceCoeffs
31{
32 std::shared_ptr<GmgFab<T>> alpha, ux, lx, uy, ly, uz, lz;
33
34 FaceCoeffs<T> view() const
35 {
36 return {alpha.get(), ux.get(), lx.get(), uy.get(), ly.get(), uz.get(), lz.get()};
37 }
38};

40// One matrix row per cell of the level: the operator's GLOBAL size, summed over all ranks.
41template<class T>
42gko::size_type gmgLevelRows(const GmgFab<T>& alpha)
43{
44 return static_cast<gko::size_type>(alpha.boxArray().numPts());
45}

47// A single GMG level as a gko::LinOp: y = A x on that level's rediscretised coefficients,
48// with the level's geometry and BCs applied to x's ghosts first. The value type is the
49// HIERARCHY's T, not the outer Krylov's double — the bottom runs inside the V-cycle.
50template<class T>
51class GmgBottomOp : public AmrexLinOpBase<GmgBottomOp<T>, T>
52{
53public:
54
55 // Required by Ginkgo's polymorphic-object machinery (create_default/clear).
56 explicit GmgBottomOp(std::shared_ptr<const gko::Executor> exec)
57 : AmrexLinOpBase<GmgBottomOp<T>, T>(exec)
58 {}
59
60 // Size and layout are read off `coeffs.alpha` rather than passed alongside it, where a
61 // BoxArray from a different level would compile.
62 GmgBottomOp(
63 std::shared_ptr<const gko::Executor> exec,
64 amrex::Geometry geom,
65 OwnedFaceCoeffs<T> coeffs,
66 BcArray bc,
67 bool onDevice
68 )
69 : AmrexLinOpBase<GmgBottomOp<T>, T>(
70 exec, gko::dim<2> {gmgLevelRows(*coeffs.alpha), gmgLevelRows(*coeffs.alpha)}
71 ),
72 geom_(std::move(geom)), coeffs_(std::move(coeffs)), bc_(bc),
73 hasPhysBc_(std::any_of(bc.begin(), bc.end(), [](int b) { return b != 0; })),
74 onDevice_(onDevice)
75 {
76 const amrex::BoxArray& ba = coeffs_.alpha->boxArray();
77 const amrex::DistributionMapping& dm = coeffs_.alpha->DistributionMap();
78 // Own work fabs, not the level's sol/rhs: aliasing would corrupt the cycle.
79 const amrex::MFInfo info =
80 onDevice_ ? amrex::MFInfo() : amrex::MFInfo().SetArena(amrex::The_Pinned_Arena());
81 in_ = std::make_shared<GmgFab<T>>(ba, dm, 1, 1, info); // 1 ghost for the stencil
82 out_ = std::make_shared<GmgFab<T>>(ba, dm, 1, 0, info); // valid-only
83 in_->setVal(T(0));
84 out_->setVal(T(0));
85 }
86
87protected:
88
89 // Keeps the base's advanced apply_impl(alpha, b, beta, x) visible here.
90 using AmrexLinOpBase<GmgBottomOp<T>, T>::apply_impl;
91
92 void apply_impl(const gko::LinOp* b, gko::LinOp* x) const override
93 {
94 prof::Timer tAll("gmg.bottom.apply");
95 auto exec = this->get_executor();
96 if (onDevice_)
97 {
98 exec->synchronize(); // b written by Ginkgo on its own stream
99 scatter_device(localValues<T>(b), *in_);
100 }
101 else
102 {
103 scatter(localValues<T>(b), *in_);
104 }
105
106 in_->FillBoundary(geom_.periodicity());
107 if (!onDevice_)
108 {
109 amrex::Gpu::streamSynchronize(); // FillBoundary before host loops
110 }
111 if (hasPhysBc_)
112 {
113 if (onDevice_)
114 {
115 fillDomainBcGhostsDevice(*in_, geom_.Domain(), bc_);
116 }
117 else
118 {
119 fillDomainBcGhostsHost(*in_, geom_.Domain(), bc_);
120 }
121 }
122
123 gmgApply(*in_, *out_, coeffs_.view(), onDevice_);
124 if (onDevice_)
125 {
126 gather_device(*out_, localValues<T>(x), 1.0);
127 amrex::Gpu::streamSynchronize(); // x read by Ginkgo next
128 }
129 else
130 {
131 gather(*out_, localValues<T>(x), 1.0);
132 }
133 }
134
135private:
136
137 amrex::Geometry geom_;
138 OwnedFaceCoeffs<T> coeffs_;
139 std::shared_ptr<GmgFab<T>> in_, out_;
140 BcArray bc_ {};
141 bool hasPhysBc_ = false;
142 bool onDevice_ = false;
143};

145inline void validateBottomSolver(const std::string& kind, bool symmetric)
146{
147 static const char* kAll = "'smoother', 'cg', 'fcg', 'bicgstab', 'gmres' or 'gcr'";
148 if (kind != "smoother" && kind != "cg" && kind != "fcg" && kind != "bicgstab" && kind != "gmres"
149 && kind != "gcr")
150 {
151 throw std::runtime_error(
152 "GmgPrecond: unknown gmg_bottom_solver '" + kind + "' (expected " + kAll + ")"
153 );
154 }
155 // Refused, not warned: a CG bottom on an asymmetric operator fails silently.
156 if (!symmetric && (kind == "cg" || kind == "fcg"))
157 {
158 throw std::runtime_error(
159 "GmgPrecond: gmg_bottom_solver='" + kind
160 + "' needs a symmetric operator, but symmetric=False was set. Use "
161 "'bicgstab', 'gmres' or 'gcr' for an asymmetric bottom."
162 );
163 }
164}

166// Generate the bottom solver on `op`. A residual-tested Krylov bottom is NOT stationary,
167// which an outer CG assumes: keep rtol tight or use a flexible outer method — the DEFAULT
168// 'smoother' (fixed sweeps) is stationary (report/blockamr-gmg-notes.md#bottom).
169template<class T>
170std::shared_ptr<const gko::LinOp> makeBottomSolver(
171 const std::string& kind,
172 std::shared_ptr<const gko::Executor> exec,
173 std::shared_ptr<const gko::LinOp> op,
174 int max_iter,
175 double rtol
176)
177{
178 if (kind == "smoother")
179 {
180 return nullptr;
181 }
182 std::vector<std::shared_ptr<const gko::stop::CriterionFactory>> criteria;
183 criteria.push_back(
184 gko::stop::Iteration::build().with_max_iters(static_cast<gko::size_type>(max_iter)).on(exec)
185 );
186 criteria.push_back(gko::stop::ResidualNorm<T>::build()
187 .with_baseline(gko::stop::mode::rhs_norm)
188 .with_reduction_factor(static_cast<gko::remove_complex<T>>(rtol))
189 .on(exec));
190
191 if (kind == "cg")
192 {
193 return gko::share(gko::solver::Cg<T>::build().with_criteria(criteria).on(exec)->generate(op)
194 );
195 }
196 if (kind == "fcg")
197 {
198 return gko::share(gko::solver::Fcg<T>::build().with_criteria(criteria).on(exec)->generate(op
199 ));
200 }
201 if (kind == "bicgstab")
202 {
203 return gko::share(
204 gko::solver::Bicgstab<T>::build().with_criteria(criteria).on(exec)->generate(op)
205 );
206 }
207 if (kind == "gmres")
208 {
209 return gko::share(
210 gko::solver::Gmres<T>::build().with_criteria(criteria).on(exec)->generate(op)
211 );
212 }
213 if (kind == "gcr")
214 {
215 return gko::share(gko::solver::Gcr<T>::build().with_criteria(criteria).on(exec)->generate(op
216 ));
217 }
218 throw std::runtime_error("GmgPrecond: unknown gmg_bottom_solver '" + kind + "'");
219}
