45
46 void apply_impl(const gko::LinOp* b, gko::LinOp* x) const override
47 {
48 prof::Timer tAll("mp.apply");
49 const gko::dim<2> size {localRows(b), 1};
50 if (!b32_ || b32_->get_size() != size)
51 {
52 prof::Timer t("mp.alloc");
53 b32_ = Dense32::create(this->get_executor(), size);
54 x32_ = Dense32::create(this->get_executor(), size);
55 // What the inner Cg<float> is handed: the buffers are sized by THIS rank's
56 // rows, so a plain Dense would make its dots and norms rank-local.
57 b32Global_ = makeGlobalVec(this->get_executor(), this->get_size()[0], b32_.get());
58 x32Global_ = makeGlobalVec(this->get_executor(), this->get_size()[0], x32_.get());
59 }
60 {
61 // Ginkgo rounds per element on the device: one pass, no host round-trip.
62 prof::Timer t("mp.down");
63 localView<double>(b)->convert_to(b32_);
64 }
65 {
66 // Zero guess, not the incoming x: this operator IS the correction S(r).
67 prof::Timer t("mp.zero");
68 x32_->fill(0.0F);
69 }
70 {
71 prof::Timer t("mp.inner");
72 inner_->apply(b32Global_, x32Global_);
73 }
74 {
75 prof::Timer t("mp.up");
76 x32_->convert_to(localView<double>(x).get());
77 }
78 }
79
80private:
81
82 std::shared_ptr<const gko::LinOp> inner_;
83 // shared_ptr for the same reason as AmrexLinOpBase::scratch_ (copy-assignment).
84 mutable std::shared_ptr<Dense32> b32_, x32_;
85 // Non-owning views of the two above; rebuilt whenever they are.
86 mutable std::shared_ptr<gko::LinOp> b32Global_, x32Global_;
87};
