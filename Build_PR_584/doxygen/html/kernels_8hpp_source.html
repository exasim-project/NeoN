27// The cell arithmetic, one struct per kernel, written once so a fused launch cannot silently
28// compute something else. Array4s carry the STORAGE type T; locals are C = la::GmgComputeT<T>.
29
30// TC is the COEFFICIENT storage type, split from the field type T: rounding a coefficient
31// perturbs only the preconditioner's operator, while rounding a FIELD reaches the coarse grid
32// multiplied by ||A|| ~ 6/dx^2 -- report/blockamr-precision-measurements.md.
33template<class T, class TC = T>
34struct GmgGsCell
35{
36 using C = la::GmgComputeT<T>;
37
38 amrex::Array4<T> psi;
39 amrex::Array4<const T> b;
40 la::FaceCoeffArrays<TC> faces;
41 amrex::Array4<const TC> al;
42 C om;
43 int parity;
44
45 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void operator()(int i, int j, int k) const
46 {
47 if (((i + j + k) & 1) != parity)
48 {
49 return;
50 }
51 const auto c = la::loadFaceCoeffs<C>(faces, i, j, k);
52 const C off = la::stencilOffDiag(
53 c,
54 psi(i + 1, j, k),
55 psi(i - 1, j, k),
56 psi(i, j + 1, k),
57 psi(i, j - 1, k),
58 psi(i, j, k + 1),
59 psi(i, j, k - 1)
60 );
61 const C diag = la::stencilDiag(al(i, j, k), c);
62 if (amrex::Math::abs(diag) > la::gmgDiagFloor<C>())
63 {
64 const C gs = (b(i, j, k) - off) / diag;
65 psi(i, j, k) += om * (gs - static_cast<C>(psi(i, j, k)));
66 }
67 }
68};

70template<class T, class TC = T>
71struct GmgResidRestrictCell
72{
73 using C = la::GmgComputeT<T>;
74
75 amrex::Array4<T> cr;
76 amrex::Array4<const T> psi, b;
77 la::FaceCoeffArrays<TC> faces;
78 amrex::Array4<const TC> al;
79
80 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE void operator()(int ic, int jc, int kc) const
81 {
82 C acc = 0;
83 for (int dk = 0; dk < 2; ++dk)
84 {
85 for (int dj = 0; dj < 2; ++dj)
86 {
87 for (int di = 0; di < 2; ++di)
88 {
89 const int i = 2 * ic + di, j = 2 * jc + dj, k = 2 * kc + dk;
90 const auto c = la::loadFaceCoeffs<C>(faces, i, j, k);
91 const C off = la::stencilOffDiag(
92 c,
93 psi(i + 1, j, k),
94 psi(i - 1, j, k),
95 psi(i, j + 1, k),
96 psi(i, j - 1, k),
97 psi(i, j, k + 1),
98 psi(i, j, k - 1)
99 );
100 const C diag = la::stencilDiag(al(i, j, k), c);
101 acc += b(i, j, k) - (diag * psi(i, j, k) + off);
102 }
103 }
104 }
105 cr(ic, jc, kc) = static_cast<C>(0.125) * acc;
106 }
107};

121// One red-black over-relaxation colour pass. Twin of gmgGsColor.
122template<class T, class TC>
123void gmgGsColorKokkos(
124 la::GmgFab<T>& sol, const la::GmgFab<T>& rhs, const la::FaceCoeffs<TC>& fc, la::GsSweep sweep
125)
126{
127 const la::GmgComputeT<T> om = static_cast<la::GmgComputeT<T>>(sweep.omega);
128 for (amrex::MFIter mfi(rhs, gmgNoSync()); mfi.isValid(); ++mfi)
129 {
130 const la::FaceCoeffArrays<TC> faces {
131 fc.ux->const_array(mfi),
132 fc.lx->const_array(mfi),
133 fc.uy->const_array(mfi),
134 fc.ly->const_array(mfi),
135 fc.uz->const_array(mfi),
136 fc.lz->const_array(mfi)
137 };
138 const GmgGsCell<T, TC> cell {
139 sol.array(mfi),
140 rhs.const_array(mfi),
141 faces,
142 fc.alpha->const_array(mfi),
143 om,
144 sweep.parity
145 };
146 launchKokkosMdNamed(
147 "gmg_gs", mfi.validbox(), BLOCKAMR_LAMBDA(int i, int j, int k) { cell(i, j, k); }
148 );
149 }
150 Kokkos::fence();
151}
