16/* @brief One cell-centred field on ONE AMR level (one level of Python's CellField).
17 * The const and non-const operator* are a PAIR: shared_ptr does not propagate
18 * constness, so a lone const overload would hand out a writable MultiFab&.
19 */
20struct CellFieldLevel
21{
22 std::shared_ptr<amrex::MultiFab> mf;
23
24 amrex::MultiFab& operator*() { return *mf; }
25 const amrex::MultiFab& operator*() const { return *mf; }
26};

28/* @brief The three direction fields of one face-centred field, on one level. Exists to
29 * remove the ux/lx/uy/ly/uz/lz hazard: six adjacent parameters of identical type,
30 * where transposing two compiles and silently answers wrongly.
31 */
32struct FaceFieldLevel
33{
34 std::array<std::shared_ptr<amrex::MultiFab>, 3> dir {};
35
36 amrex::MultiFab& operator[](int d) { return *dir[static_cast<std::size_t>(d)]; }
37 const amrex::MultiFab& operator[](int d) const { return *dir[static_cast<std::size_t>(d)]; }
38};

40/* @brief A read-only view of one cell-centred field on one level.
41 * Build it with constView(); there is deliberately no way back to a writable handle.
42 */
43struct ConstCellFieldLevel

45 std::shared_ptr<const amrex::MultiFab> mf;

50/* @brief A read-only view of the three direction fields, for a consumer that promises not to
51 * write them. Exists because the mutable pair above cannot be built from a
52 * `const amrex::MultiFab&` without a const_cast, which would launder away the promise
53 * in kernels the layer requires to be bit-for-bit reproducible.
54 */
55struct ConstFaceFieldLevel
56{
57 std::array<std::shared_ptr<const amrex::MultiFab>, 3> dir {};
58
59 const amrex::MultiFab& operator[](int d) const { return *dir[static_cast<std::size_t>(d)]; }
60};

62// A handle to a field owned elsewhere: an aliasing shared_ptr with an EMPTY owner, so no
63// control block is allocated and no deleter ever runs.
64inline std::shared_ptr<amrex::MultiFab> nonOwning(amrex::MultiFab& mf)
65{
66 return std::shared_ptr<amrex::MultiFab> {std::shared_ptr<amrex::MultiFab> {}, &mf};
67}
68
69// The const overload is what lets a read-only consumer take a bundle at all: the mutable one
70// binds only to `MultiFab&`, so a `const MultiFab&` caller would otherwise need a const_cast.
71inline std::shared_ptr<const amrex::MultiFab> nonOwning(const amrex::MultiFab& mf)
72{
73 return std::shared_ptr<const amrex::MultiFab> {std::shared_ptr<const amrex::MultiFab> {}, &mf};
74}
