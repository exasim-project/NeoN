22/* @brief The stopping test one solve runs, as one argument: an iteration cap, a residual
23 * reduction by `reductionFactor` against `baseline` (rhs_norm for a relative rtol,
24 * absolute when the caller folded rtol in), an absolute `atol` when it is > 0, and the
25 * norm all of that is measured in -- "l2" (Ginkgo's) or "linf" (MLMG's, stopNormInf.hpp).
26 * `norm` carries its default here, so a caller that never names one stays a 4-field
27 * aggregate.
28 */
29struct StopSpec
30{
31 int maxIter;
32 gko::stop::mode baseline;
33 double reductionFactor;
34 double atol;
35 std::string norm = "l2";
36};

38// The (Iteration, ResidualNorm[, ResidualNorm]) chain every Krylov solve here uses.
39inline std::vector<std::shared_ptr<const gko::stop::CriterionFactory>>
40makeCriteria(std::shared_ptr<const gko::Executor> exec, const StopSpec& stop)
41{
42 const NormKind kind = parseNorm(stop.norm);
43 std::vector<std::shared_ptr<const gko::stop::CriterionFactory>> criteria;
44 criteria.push_back(gko::stop::Iteration::build()
45 .with_max_iters(static_cast<gko::size_type>(stop.maxIter))
46 .on(exec));
47 auto residual = [&](gko::stop::mode base,
48 double factor) -> std::shared_ptr<const gko::stop::CriterionFactory>
49 {
50 if (kind == NormKind::linf)
51 {
52 return ResidualNormInf::build().with_baseline(base).with_reduction_factor(factor).on(
53 exec
54 );
55 }
56 return gko::stop::ResidualNorm<double>::build()
57 .with_baseline(base)
58 .with_reduction_factor(factor)
59 .on(exec);
60 };
61 criteria.push_back(residual(stop.baseline, stop.reductionFactor));
62 if (stop.atol > 0.0)
63 {
64 criteria.push_back(residual(gko::stop::mode::absolute, stop.atol));
65 }
66 return criteria;
67}

91// Fill the SolveResult every solve entry point returns. `diagnostic` stays empty here:
92// its thresholds are calibrated for a V-cycle's roughly constant contraction.
93inline SolveResult makeSolveResult(
94 std::int64_t num_iters, double res_norm, bool converged, const std::vector<double>& res_history
95)
96{
97 SolveResult r;
98 r.num_iters = num_iters;
99 r.res_norm = res_norm;
100 r.converged = converged;
101 r.res_history = res_history;
102 // Geometric mean of the per-iteration reduction; the history holds size() - 1 of them.
103 double contraction = 0.0;
104 if (res_history.size() >= 2 && res_history.front() > 0.0)
105 {
106 contraction = std::pow(
107 res_history.back() / res_history.front(),
108 1.0 / static_cast<double>(res_history.size() - 1)
109 );
110 }
111 r.contraction = contraction;
112 r.diagnostic = "";
113 return r;
114}
