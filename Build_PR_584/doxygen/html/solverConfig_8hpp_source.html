16// Every value `solver` may spell. Which a PARTICULAR class/mode accepts is decided
17// downstream; this only rejects an outright typo, once, at parseSolverConfig.

20 cg,

24 fcg,
25 ir,
26 gmg,
27 mpir

30// Throws with buildKrylov's historical "unknown solver" wording. Parsing HERE is what makes
31// a typo in gmg/ir/mpir -- which never reach buildKrylov -- fail before any ctor work.
32inline SolverKind parseSolverKind(const std::string& solver)
33{
34 if (solver == "cg") return SolverKind::cg;
35 if (solver == "bicgstab") return SolverKind::bicgstab;
36 if (solver == "gmres") return SolverKind::gmres;
37 if (solver == "gcr") return SolverKind::gcr;
38 if (solver == "fcg") return SolverKind::fcg;
39 if (solver == "ir") return SolverKind::ir;
40 if (solver == "gmg") return SolverKind::gmg;
41 if (solver == "mpir") return SolverKind::mpir;
42 throw std::runtime_error("ginkgo: unknown solver '" + solver + "'");
43}

54// Throws with FaceCoeffSolver's historical wording; per-class legality still runs downstream
55// on the parsed enum.
56inline PrecondKind parsePrecondKind(const std::string& precond)
57{
58 if (precond == "none") return PrecondKind::none;
59 if (precond == "mlmg") return PrecondKind::mlmg;
60 if (precond == "gmg") return PrecondKind::gmg;
61 if (precond == "gmg_kokkos") return PrecondKind::gmg_kokkos;
62 throw std::runtime_error(
63 "FaceCoeffSolver: unknown precond '" + precond
64 + "' (expected 'none', 'mlmg', 'gmg' or 'gmg_kokkos')"
65 );
66}
