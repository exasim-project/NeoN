77// dst += src over dst's valid box, converting through dst's value_type.
78template<class DST, class T>
79void gmgConvertAdd(DST& dst, const GmgFab<T>& src, bool onDevice)
80{
81 using DT = typename DST::value_type;
82 amrex::Gpu::LaunchSafeGuard lsg(onDevice);
83 for (amrex::MFIter mfi(dst); mfi.isValid(); ++mfi)
84 {
85 const amrex::Box& vbx = mfi.validbox();
86 const auto d = dst.array(mfi);
87 const auto s = src.const_array(mfi);
88 amrex::HostDeviceParallelFor(
89 vbx,
90 [=] AMREX_GPU_HOST_DEVICE(int i, int j, int k) noexcept
91 { d(i, j, k) += static_cast<DT>(s(i, j, k)); }
92 );
93 }
94}

96// Fused residual + convert-scatter (r = rhs - A*sol - shift in DOUBLE, stored cast to T into
97// the L0 rhs) plus a second light kernel giving BOTH norms. Why not one kernel, and the
98// norm's precision: report/blockamr-gmg-notes.md#fusion (NeoFOAM repo).
99struct ResidNorms
100{
101 double sumsq; // sum r_i^2 — caller takes the sqrt for ||r||_2
102 double maxabs; // max |r_i| — ||r||_inf
103};
104
105// Combine the per-rank norms: ParReduce is rank-LOCAL and must use the SAME communicator
106// as the ||rhs|| gmgSolve compares against — report/blockamr-gmg-notes.md#rank-reduction.
107inline void reduceResidNorms(ResidNorms& r)
108{
109 const MPI_Comm comm = amrex::ParallelContext::CommunicatorSub();
110 amrex::ParallelAllReduce::Sum(r.sumsq, comm);
111 amrex::ParallelAllReduce::Max(r.maxabs, comm);
112}

114// The FP64 system one residual r = rhs - A*sol - shift is formed on, and the ONE argument of

123 const amrex::MultiFab* alpha;
124 const amrex::MultiFab* ux;
125 const amrex::MultiFab* lx;
126 const amrex::MultiFab* uy;
127 const amrex::MultiFab* ly;
128 const amrex::MultiFab* uz;
129 const amrex::MultiFab* lz;
130 double shift;

133// r = rhs - A*sol - shift formed in FP64 and stored cast to T over out's valid box.
134template<class T>
135void faceCoeffResidScatter(const GmgResidualInput& in, GmgFab<T>& out, bool onDevice)
136{
137 prof::Timer t("gmg.solve.residkern");
138 amrex::Gpu::LaunchSafeGuard lsg(onDevice);
139 const double shift = in.shift; // by value: a device lambda cannot read `in`
140 for (amrex::MFIter mfi(out); mfi.isValid(); ++mfi)
141 {
142 const amrex::Box& vbx = mfi.validbox();
143 const auto psi = in.sol->const_array(mfi);
144 const auto bb = in.rhs->const_array(mfi);
145 const auto o = out.array(mfi);
146 const auto al = in.alpha->const_array(mfi);
147 const FaceCoeffArrays fca {
148 in.ux->const_array(mfi),
149 in.lx->const_array(mfi),
150 in.uy->const_array(mfi),
151 in.ly->const_array(mfi),
152 in.uz->const_array(mfi),
153 in.lz->const_array(mfi)
154 };
155 amrex::HostDeviceParallelFor(
156 vbx,
157 [=] AMREX_GPU_HOST_DEVICE(int i, int j, int k) noexcept
158 {
159 const auto c = loadFaceCoeffs<double>(fca, i, j, k);
160 const double offd = stencilOffDiag(
161 c,
162 psi(i + 1, j, k),
163 psi(i - 1, j, k),
164 psi(i, j + 1, k),
165 psi(i, j - 1, k),
166 psi(i, j, k + 1),
167 psi(i, j, k - 1)
168 );
169 const double diag = stencilDiag(al(i, j, k), c);
170 const double r = bb(i, j, k) - (diag * psi(i, j, k) + offd) - shift;
171 o(i, j, k) = static_cast<T>(r);
172 }
173 );
174 }
175}

177// Both norms of the STORED residual, reduced on the device.
178template<class T>
179ResidNorms residNormsDevice(const GmgFab<T>& out)
180{
181 prof::Timer t("gmg.solve.normkern");
182 const auto o_ma = out.const_arrays();
183 const auto both = amrex::ParReduce(
184 amrex::TypeList<amrex::ReduceOpSum, amrex::ReduceOpMax> {},
185 amrex::TypeList<double, double> {},
186 out,
187 amrex::IntVect(0),
188 [=] AMREX_GPU_DEVICE(int box, int i, int j, int k) -> amrex::GpuTuple<double, double>
189 {
190 const double v = static_cast<double>(o_ma[box](i, j, k));
191 return {v * v, amrex::Math::abs(v)};
192 }
193 );
194 return {amrex::get<0>(both), amrex::get<1>(both)};
195}

197// Host twin: ParReduce picks its backend at compile time, so this is a sequential loop over the
198// same STORED values, in the same accumulation order.
199template<class T>
200ResidNorms residNormsHost(const GmgFab<T>& out)
201{
202 ResidNorms res {};
203 for (amrex::MFIter mfi(out); mfi.isValid(); ++mfi)
204 {
205 const amrex::Box& vbx = mfi.validbox();
206 const auto o = out.const_array(mfi);
207 const auto lo = amrex::lbound(vbx);
208 const auto hi = amrex::ubound(vbx);
209 for (int k = lo.z; k <= hi.z; ++k)
210 {
211 for (int j = lo.y; j <= hi.y; ++j)
212 {
213 for (int i = lo.x; i <= hi.x; ++i)
214 {
215 const double v = static_cast<double>(o(i, j, k));
216 res.sumsq += v * v;
217 res.maxabs = std::max(res.maxabs, std::abs(v));
218 }
219 }
220 }
221 }
222 return res;
223}

225template<class T>
226ResidNorms faceCoeffResidScatterNorm(const GmgResidualInput& in, GmgFab<T>& out, bool onDevice)
227{
228 faceCoeffResidScatter(in, out, onDevice);
229 ResidNorms res = onDevice ? residNormsDevice(out) : residNormsHost(out);
230 reduceResidNorms(res);
231 return res;
232}

234// ||mf||_2 over the valid region, accumulated in the fab's value_type and summed across ranks
235// (setup power iteration only). Why the cross-rank sum is required:
236// report/blockamr-gmg-notes.md#rank-reduction.
237template<class T>
238double gmgNorm2(const GmgFab<T>& mf)
239{
240 const T sq = amrex::ReduceSum(
241 mf,
242 amrex::IntVect(0),
243 [=] AMREX_GPU_HOST_DEVICE(const amrex::Box& bx, const amrex::Array4<const T>& a) -> T
244 {
245 T s = 0;
246 const auto lo = amrex::lbound(bx);
247 const auto hi = amrex::ubound(bx);
248 for (int k = lo.z; k <= hi.z; ++k)
249 {
250 for (int j = lo.y; j <= hi.y; ++j)
251 {
252 for (int i = lo.x; i <= hi.x; ++i)
253 {
254 s += a(i, j, k) * a(i, j, k);
255 }
256 }
257 }
258 return s;
259 }
260 );
261 double sumsq = static_cast<double>(sq);
262 amrex::ParallelAllReduce::Sum(sumsq, amrex::ParallelContext::CommunicatorSub());
263 return std::sqrt(sumsq);
264}

315// One Chebyshev degree's polynomial coefficients; `readOld` is false only at degree 0, where
316// there is no previous increment to fold in.
317template<class T>
318struct ChebStep
319{
320 T ca;
321 T cb;
322 bool readOld;
323};
324
325// One fused Jacobi-Chebyshev degree step: d = cb * D^{-1} r + (readOld ? ca*d : 0), r on the
326// fly (sol's ghosts filled). sol is NOT written here, so the step is race-free and CG-safe.
327template<class T>
328void gmgChebComputeD(const GmgSystem<T>& sys, GmgFab<T>& d, ChebStep<T> step, bool onDevice)
329{
330 amrex::Gpu::LaunchSafeGuard lsg(onDevice);
331 for (amrex::MFIter mfi(*sys.rhs); mfi.isValid(); ++mfi)
332 {
333 const amrex::Box& vbx = mfi.validbox();
334 const auto psi = sys.sol->const_array(mfi);
335 const auto b = sys.rhs->const_array(mfi);
336 const auto dd = d.array(mfi);
337 const auto al = sys.coeffs.alpha->const_array(mfi);
338 const FaceCoeffArrays fca {
339 sys.coeffs.ux->const_array(mfi),
340 sys.coeffs.lx->const_array(mfi),
341 sys.coeffs.uy->const_array(mfi),
342 sys.coeffs.ly->const_array(mfi),
343 sys.coeffs.uz->const_array(mfi),
344 sys.coeffs.lz->const_array(mfi)
345 };
346 amrex::HostDeviceParallelFor(
347 vbx,
348 [=] AMREX_GPU_HOST_DEVICE(int i, int j, int k) noexcept
349 {
350 const auto c = loadFaceCoeffs<T>(fca, i, j, k);
351 const T off = stencilOffDiag(
352 c,
353 psi(i + 1, j, k),
354 psi(i - 1, j, k),
355 psi(i, j + 1, k),
356 psi(i, j - 1, k),
357 psi(i, j, k + 1),
358 psi(i, j, k - 1)
359 );
360 const T diag = stencilDiag(al(i, j, k), c);
361 const T r = b(i, j, k) - (diag * psi(i, j, k) + off);
362 // Same floor gmgGsColor applies to RB-GS: a degenerate diagonal contributes no
363 // correction rather than inf/NaN, and `d` is Saxpy'd straight into `sol`.
364 const T dinv = (amrex::Math::abs(diag) > gmgDiagFloor<T>()) ? (r / diag) : T(0);
365 T dval = step.cb * dinv;
366 if (step.readOld)
367 {
368 dval += step.ca * dd(i, j, k);
369 }
370 dd(i, j, k) = dval;
371 }
372 );
373 }
374}

429 fc.ux->const_array(mfi),
430 fc.lx->const_array(mfi),
431 fc.uy->const_array(mfi),
432 fc.ly->const_array(mfi),

437 vbx,
438 [=] AMREX_GPU_HOST_DEVICE(int i, int j, int k) noexcept
439 {
440 const auto c = loadFaceCoeffs<T>(fca, i, j, k);
441 const T off = stencilOffDiag(
442 c,
443 psi(i + 1, j, k),
444 psi(i - 1, j, k),
445 psi(i, j + 1, k),
446 psi(i, j - 1, k),
447 psi(i, j, k + 1),
448 psi(i, j, k - 1)
449 );
450 const T diag = stencilDiag(al(i, j, k), c);
451 // Zero, not psi: this feeds only the lambdaMax power iteration, where a
452 // degenerate cell must add nothing to ||w||. Unguarded it is inf/NaN, and NaN
453 // passes estimateLambdaMax's `lambda <= 0` break and poisons the eigenvector.
454 o(i, j, k) = (amrex::Math::abs(diag) > gmgDiagFloor<T>())
455 ? (diag * psi(i, j, k) + off) / diag
456 : T(0);
457 }
458 );
459 }
