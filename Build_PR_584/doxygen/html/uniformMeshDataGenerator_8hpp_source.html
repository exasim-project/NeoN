20struct MeshParams
21{
22 localIdx nx, ny, nz;
23 scalar Lx, Ly, Lz;
24
25 // Derived spacings
26 scalar dx() const { return Lx / static_cast<scalar>(nx); }
27 scalar dy() const { return Ly / static_cast<scalar>(ny); }
28 scalar dz() const { return Lz / static_cast<scalar>(nz); }
29
30 // Layout of cells in memory: go through x first, then y, then z.
31 // So cell indices in the first layer are ordered as:
32 // (0,0,0), (1,0,0), ..., (nx-1,0,0),
33 // (0,1,0), (1,1,0), ..., (nx-1,1,0),
34 // ...,
35 // (0,ny-1,0), (1,ny-1,0), ..., (nx-1,ny-1,0)
36 localIdx cellIdx(localIdx i, localIdx j, localIdx k) const { return k * nx * ny + j * nx + i; }
37
38 localIdx ptIdx(localIdx i, localIdx j, localIdx k) const
39 {
40 return k * (nx + 1) * (ny + 1) + j * (nx + 1) + i;
41 }
42};
