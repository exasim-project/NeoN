17/* @class LinearSystem
18 * @brief Pure data: a matrix and a right-hand side, non-owning -- both must outlive it,
19 * and the rhs a caller passes IS the one the solve reads. No BC, Geometry or
20 * discretisation knowledge of its own; the operator reads those off the matrix.
21 */
22class LinearSystem
23{
24public:
25
26 LinearSystem(MFFaceCoeffs& matrix, amrex::MultiFab& rhs) : matrix_(&matrix), rhs_(&rhs) {}
27
28 // ACCUMULATES: a system is zeroed once, then written by however many operators
29 // contribute. A template, so the operator's SIGNATURE is the whole contract -- there is
30 // no erasure and no virtual call between `+=` and the kernels.
31 template<class Op>
32 LinearSystem& operator+=(const Op& op)
33 {
34 op.assemble(*this);
35 return *this;
36 }
37
38 // Coefficients AND rhs together: zeroing half a system is never meant.
39 void zero()
40 {
41 matrix_->zero();
42 rhs_->setVal(0.0);
43 }
44
45 // Plain accessors: nothing is derived from the coefficients on this side, so there is
46 // nothing a write has to invalidate.
47 MFFaceCoeffs& matrix() { return *matrix_; }
48
49 const MFFaceCoeffs& matrix() const { return *matrix_; }
50
51 amrex::MultiFab& rhs() { return *rhs_; }
52
53 const amrex::MultiFab& rhs() const { return *rhs_; }
54
55 // Rank-local, never boxArray().numPts(): see localCount in transfer.hpp.
56 std::size_t localRows() const { return matrix_->localRows(); }
57
58 const NeoN::Executor& executor() const { return matrix_->exec; }
59
60private:
61
62 MFFaceCoeffs* matrix_;
63 amrex::MultiFab* rhs_;
64};
