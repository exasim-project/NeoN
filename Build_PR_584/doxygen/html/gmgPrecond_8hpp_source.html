36// One multigrid level: geometry, rediscretised coefficients, work fields (sol has 1 ghost).
37template<class T>
38struct GmgLevelT
39{
40 amrex::Geometry geom;
41 std::shared_ptr<GmgFab<T>> alpha, ux, lx, uy, ly, uz, lz;
42 std::shared_ptr<GmgFab<T>> sol, rhs;
43 std::shared_ptr<GmgFab<T>> chebD; // Chebyshev increment (only when smoother="chebyshev")
44 double lambdaMax = 0.0; // estimate of lambda_max(D^{-1}A) on this level
45};

47// Abstract hook exposing a GMG V-cycle as operations on FP64 MultiFabs, so the native
48// stationary solver can drive GmgPrecondT<T> without knowing T. No Ginkgo vector involved.

51public:

55 // Fused r = in.rhs - A*in.sol - in.shift -> (cast to T) L0 rhs; L0 sol := 0. Returns BOTH
56 // FP64 norms of r, so the norm authority stays double. `in.sol`'s ghosts must already be
57 // filled. The system arrives as the GmgResidualInput the kernel reads (gmgKernels.hpp), so
58 // the nine loose MultiFabs -- six of them the transposable ux/lx/uy/ly/uz/lz -- are named
59 // once, at the caller, with designated initialisers.
60 virtual ResidNorms residScatterNorm(const GmgResidualInput& in) const = 0;
61
62 // Run nCycles_ V-cycles on the L0 rhs set by residScatterNorm, then x += the correction.
63 virtual void vcycleGather(amrex::MultiFab& x) const = 0;
64};

66// Native GMG V-cycle preconditioner on the face-coefficient operator: z = M^{-1} r via
67// n_cycles V-cycles of RB-GS smoothing, volume-average restriction and piecewise-constant
68// prolongation (laws: gmgKernels.hpp; smoother measurements: report/blockamr-gmg-notes.md).
69template<class T>
70class GmgPrecondT : public AmrexLinOpBase<GmgPrecondT<T>>, public GmgApplyMf

72public:

74 explicit GmgPrecondT(std::shared_ptr<const gko::Executor> exec)
75 : AmrexLinOpBase<GmgPrecondT<T>>(exec)
76 {}

78 GmgPrecondT(
79 std::shared_ptr<const gko::Executor> exec,
80 gko::size_type n,
81 const FaceCoeffLevel& level,
82 BcArray bc,
83 const GmgPrecondSpec& spec
84 )
85 : AmrexLinOpBase<GmgPrecondT<T>>(exec, gko::dim<2> {n, n}), bc_(bc),
86 hasPhysBc_(std::any_of(bc.begin(), bc.end(), [](int b) { return b != 0; })),
87 onDevice_(exec->get_master().get() != exec.get()), nCycles_(spec.nCycles),
88 preSweeps_(spec.gmg.preSweeps), postSweeps_(spec.gmg.postSweeps),
89 coarsestSweeps_(spec.gmg.coarsestSweeps), useCheb_(spec.gmg.smoother == "chebyshev"),
90 omega_(spec.gmg.omega), symmetric_(spec.gmg.symmetric)
91 {
92 const GmgConfig& gmg = spec.gmg;
93 validateOptions(gmg);
94 // Finest level: the coefficients are COPIED into this preconditioner's own arena, so
95 // later caller writes go unseen — a stale preconditioner only costs iterations.
96 levels_.push_back(makeLevel(level.mesh.ba, level.mesh.dm, level.mesh.geom));
97 copyCoeff(*levels_[0].alpha, *level.alpha);
98 copyCoeff(*levels_[0].ux, level.upper[0]);
99 copyCoeff(*levels_[0].lx, level.lower[0]);
100 copyCoeff(*levels_[0].uy, level.upper[1]);
101 copyCoeff(*levels_[0].ly, level.lower[1]);
102 copyCoeff(*levels_[0].uz, level.upper[2]);
103 copyCoeff(*levels_[0].lz, level.lower[2]);
104 buildCoarseLevels(level.mesh.dm, gmg.maxLevels, gmg.minBottom);
105 if (useCheb_)
106 {
107 setupChebyshev();
108 }
109 if (gmg.bottomSolver != "smoother")
110 {
111 setupBottomSolver(exec, gmg.bottomSolver, gmg.bottomMaxIter, gmg.bottomRtol);
112 }
113 }

115 // One kernel forms the FP64 residual, casts it into the T-typed L0 rhs and reduces its norm.
116 ResidNorms residScatterNorm(const GmgResidualInput& in) const override
117 {
118 const GmgLevelT<T>& L0 = levels_.front();
119 ResidNorms norms = faceCoeffResidScatterNorm<T>(in, *L0.rhs, onDevice_);
120 L0.sol->setVal(T(0)); // z0 = 0: apply M^{-1}, not a warm-started solve
121 if (!onDevice_)
122 {
123 amrex::Gpu::streamSynchronize();
124 }
125 return norms;
126 }

128 void vcycleGather(amrex::MultiFab& x) const override
129 {
130 const GmgLevelT<T>& L0 = levels_.front();
131 if (onDevice_)
132 {
133 {
134 prof::Timer t("gmg.vcycle");
135 for (int c = 0; c < nCycles_; ++c)
136 {
137 vcycle(0);
138 }
139 }
140 {
141 prof::Timer t("gmg.solve.gather");
142 gmgConvertAdd(x, *L0.sol, onDevice_); // x += (double) L0 correction
143 amrex::Gpu::streamSynchronize();
144 }
145 }
146 else
147 {
148 for (int c = 0; c < nCycles_; ++c)
149 {
150 vcycle(0);
151 }
152 gmgConvertAdd(x, *L0.sol, onDevice_);
153 amrex::Gpu::streamSynchronize();
154 }
155 }

162 void apply_impl(const gko::LinOp* b, gko::LinOp* x) const override
163 {
164 auto exec = this->get_executor();
165 const GmgLevelT<T>& L0 = levels_.front();
166 if (onDevice_)
167 {
168 prof::Timer tAll("gmg.apply");
169 {
170 prof::Timer t("gmg.sync_gko");
171 exec->synchronize(); // b written by Ginkgo
172 }
173 {
174 prof::Timer t("gmg.scatter");
175 scatter_device(localValues<double>(b), *L0.rhs);
176 L0.sol->setVal(0.0); // z0 = 0: apply M^{-1}, not a warm-started solve
177 }
178 {
179 prof::Timer t("gmg.vcycle");
180 for (int c = 0; c < nCycles_; ++c)
181 {
182 vcycle(0);
183 }
184 }
185 {
186 prof::Timer t("gmg.gather");
187 gather_device(*L0.sol, localValues<double>(x), 1.0);
188 amrex::Gpu::streamSynchronize(); // x complete before Ginkgo reads it
189 }
190 }
191 else
192 {
193 auto host = exec->get_master();
194 auto bHost = gko::clone(host, localView<double>(b));
195 scatter(bHost->get_const_values(), *L0.rhs);
196 L0.sol->setVal(0.0);
197 amrex::Gpu::streamSynchronize(); // setVal may run on the GPU stream
198 for (int c = 0; c < nCycles_; ++c)
199 {
200 vcycle(0);
201 }
202 auto xHost = Dense::create(host, gko::dim<2> {localRows(x), 1});
203 gather(*L0.sol, xHost->get_values(), 1.0);
204 localView<double>(x)->copy_from(xHost);
205 }
206 }

210 // Chebyshev smooths [lambdaMax / kChebEigRatio, lambdaMax] and leaves the lower modes to
211 // the coarse grid; a sweep put the minimum at 6 (report/blockamr-gmg-notes.md#smoother).
212 static constexpr double kChebEigRatio = 6.0;
213 static constexpr double kChebSafety = 1.05; // inflate the lambda_max estimate
214 static constexpr int kPowerIters = 15; // power iterations for lambda_max

216 // Refuses, rather than warns about, an option combination the cycle cannot honour
217 // (report/blockamr-gmg-notes.md#smoother).
218 void validateOptions(const GmgConfig& gmg) const
219 {
220 const std::string& smoother = gmg.smoother;
221 const double omega = gmg.omega;
222 const bool symmetric = gmg.symmetric;
223 if (omega <= 0.0 || omega >= 2.0)
224 {
225 throw std::runtime_error(
226 "GmgPrecond: gmg_omega must lie in (0, 2) for a convergent "
227 "relaxation (got "
228 + std::to_string(omega) + ")"
229 );
230 }
231 if (smoother != "rbgs" && smoother != "chebyshev")
232 {
233 throw std::runtime_error(
234 "GmgPrecond: unknown gmg_smoother '" + smoother
235 + "' (expected 'rbgs' or 'chebyshev')"
236 );
237 }
238 validateBottomSolver(gmg.bottomSolver, symmetric);
239 if (symmetric)
240 {
241 return;
242 }
243 if (omega != 1.0)
244 {
245 throw std::runtime_error(
246 "GmgPrecond: gmg_omega != 1.0 assumes a symmetric operator, but "
247 "symmetric=False was set (got omega = "
248 + std::to_string(omega) + "). Set gmg_omega=1.0."
249 );
250 }
251 if (useCheb_)
252 {
253 throw std::runtime_error(
254 "GmgPrecond: gmg_smoother='chebyshev' builds its polynomial on a real "
255 "eigenvalue interval and assumes a symmetric operator, but "
256 "symmetric=False was set. Use gmg_smoother='rbgs'."
257 );
258 }
259 }

261 // Coarsen by 2 while every box dimension stays divisible and the coarse domain keeps
262 // >= min_bottom cells. alpha via gmgRestrict, faces via gmgCoarsenFace(scale = 4) —
263 // two DIFFERENT laws, see gmgKernels.hpp.
264 void buildCoarseLevels(const amrex::DistributionMapping& dm, int max_levels, int min_bottom)
265 {
266 while (canCoarsen(max_levels, min_bottom))
267 {
268 const GmgLevelT<T>& f = levels_.back();
269 amrex::BoxArray cba = f.alpha->boxArray();
270 cba.coarsen(2);
271 const amrex::Geometry cgeom(
272 amrex::coarsen(f.geom.Domain(), 2),
273 f.geom.ProbDomain(),
274 f.geom.Coord(),
275 {f.geom.isPeriodic(0), f.geom.isPeriodic(1), f.geom.isPeriodic(2)}
276 );
277 levels_.push_back(makeLevel(cba, dm, cgeom));
278 GmgLevelT<T>& c = levels_.back();
279 const GmgLevelT<T>& fl = levels_[levels_.size() - 2];
280 gmgRestrict(*fl.alpha, *c.alpha, onDevice_);
281 gmgCoarsenFace(*fl.ux, *c.ux, 0, 4.0, onDevice_);
282 gmgCoarsenFace(*fl.lx, *c.lx, 0, 4.0, onDevice_);
283 gmgCoarsenFace(*fl.uy, *c.uy, 1, 4.0, onDevice_);
284 gmgCoarsenFace(*fl.ly, *c.ly, 1, 4.0, onDevice_);
285 gmgCoarsenFace(*fl.uz, *c.uz, 2, 4.0, onDevice_);
286 gmgCoarsenFace(*fl.lz, *c.lz, 2, 4.0, onDevice_);
287 }
288 amrex::Gpu::streamSynchronize();
289 }

291 // Whether one more level fits below the current coarsest one.
292 bool canCoarsen(int max_levels, int min_bottom) const
293 {
294 if (max_levels > 0 && static_cast<int>(levels_.size()) >= max_levels)
295 {
296 return false;
297 }
298 const GmgLevelT<T>& f = levels_.back();
299 if (!f.alpha->boxArray().coarsenable(2, 2))
300 {
301 return false;
302 }
303 return amrex::coarsen(f.geom.Domain(), 2).shortside() >= min_bottom;
304 }

306 // The Chebyshev increment field plus a lambda_max estimate, per level.
307 void setupChebyshev()
308 {
309 for (auto& L : levels_)
310 {
311 L.chebD = makeMf(L.alpha->boxArray(), L.alpha->DistributionMap(), 0);
312 }
313 for (std::size_t l = 0; l < levels_.size(); ++l)
314 {
315 levels_[l].lambdaMax = estimateLambdaMax(l);
316 }
317 amrex::Gpu::streamSynchronize();
318 }

320 // The Krylov bottom, built once on the coarsest level; left null for "smoother".
321 void setupBottomSolver(
322 std::shared_ptr<const gko::Executor> exec,
323 const std::string& bottom_solver,
324 int bottom_max_iter,
325 double bottom_rtol
326 )
327 {
328 const GmgLevelT<T>& B = levels_.back();
329 // GLOBAL size for the operator, LOCAL for the vectors gather/scatter fill.
330 const gko::size_type nBottom = gmgLevelRows(*B.alpha);
331 const auto nBottomLocal = static_cast<gko::size_type>(localCount(*B.alpha));
332 bottomOp_ = gko::share(GmgBottomOp<T>::create(
333 exec,
334 B.geom,
335 OwnedFaceCoeffs<T> {B.alpha, B.ux, B.lx, B.uy, B.ly, B.uz, B.lz},
336 bc_,
337 onDevice_
338 ));
339 bottomSolver_ =
340 makeBottomSolver<T>(bottom_solver, exec, bottomOp_, bottom_max_iter, bottom_rtol);
341 bottomB_ = gko::matrix::Dense<T>::create(exec, gko::dim<2> {nBottomLocal, 1});
342 bottomX_ = gko::matrix::Dense<T>::create(exec, gko::dim<2> {nBottomLocal, 1});
343 // The bottom Krylov's dots and norms need the outer solve's distributed views.
344 bottomBGlobal_ = makeGlobalVec(exec, nBottom, bottomB_.get());
345 bottomXGlobal_ = makeGlobalVec(exec, nBottom, bottomX_.get());
346 }

359 GmgLevelT<T> makeLevel(
360 const amrex::BoxArray& ba, const amrex::DistributionMapping& dm, const amrex::Geometry& geom
361 ) const
362 {
363 GmgLevelT<T> L;
364 L.geom = geom;
365 L.alpha = makeMf(ba, dm, 0);
366 const auto fba = [&ba](int d)
367 { return amrex::convert(ba, amrex::IntVect::TheDimensionVector(d)); };
368 L.ux = makeMf(fba(0), dm, 0);
369 L.lx = makeMf(fba(0), dm, 0);
370 L.uy = makeMf(fba(1), dm, 0);
371 L.ly = makeMf(fba(1), dm, 0);
372 L.uz = makeMf(fba(2), dm, 0);
373 L.lz = makeMf(fba(2), dm, 0);
374 L.sol = makeMf(ba, dm, 1);
375 L.rhs = makeMf(ba, dm, 0);
376 return L;
377 }

431 // The level's 7 rediscretised coefficients as the view every kernel takes.
432 static FaceCoeffs<T> levelCoeffs(const GmgLevelT<T>& L)
433 {
434 return {
435 L.alpha.get(), L.ux.get(), L.lx.get(), L.uy.get(), L.ly.get(), L.uz.get(), L.lz.get()
436 };
437 }

439 // RB-GS sweeps; `reversed` (black-red) is the forward sweep's adjoint, which is what keeps
440 // the V-cycle symmetric and so CG-safe.
441 void rbgsSmooth(std::size_t l, int sweeps, bool reversed) const
442 {
443 const GmgLevelT<T>& L = levels_[l];
444 const FaceCoeffs<T> fc = levelCoeffs(L);
445 for (int s = 0; s < sweeps; ++s)
446 {
447 for (int c = 0; c < 2; ++c)
448 {
449 const GsSweep sweep {(reversed ? 1 + c : c) & 1, omega_};
450 fillGhosts(L, static_cast<int>(l)); // the other colour changed — refresh ghosts
451 if (onDevice_)
452 {
453 prof::Timer t("gmg.gs", static_cast<int>(l));
454 gmgGsColor(*L.sol, *L.rhs, fc, sweep, onDevice_);
455 }
456 else
457 {
458 gmgGsColor(*L.sol, *L.rhs, fc, sweep, onDevice_);
459 }
460 }
461 }

464 // The constants of one level's Chebyshev recurrence; `rho` advances with each degree.
465 struct ChebRecurrence
466 {
467 double theta;
468 double delta;
469 double sigma;
470 double rho;
471 };
472
473 // The recurrence over [lambdaMax / kChebEigRatio, lambdaMax], the band the smoother owns.
474 static ChebRecurrence chebRecurrence(double lambdaMax)
475 {
476 const double b = lambdaMax;
477 const double a = b / kChebEigRatio;
478 const double theta = 0.5 * (b + a);
479 const double delta = 0.5 * (b - a);
480 const double sigma = theta / delta;
481 return {theta, delta, sigma, 1.0 / sigma};
482 }
483
484 // Degree m's coefficients, advancing `rec`; degree 0 has no previous increment to fold in.
485 static ChebStep<T> nextChebStep(int m, ChebRecurrence& rec)
486 {
487 if (m == 0)
488 {
489 return {T(0), static_cast<T>(1.0 / rec.theta), false}; // d = (1/theta) D^{-1} r
490 }
491 const double rhoNew = 1.0 / (2.0 * rec.sigma - rec.rho);
492 const double ca = rec.rho * rhoNew; // d = ca * d + cb * D^{-1} r
493 const double cb = 2.0 * rhoNew / rec.delta;
494 rec.rho = rhoNew;
495 return {static_cast<T>(ca), static_cast<T>(cb), true};
496 }
497
498 // Jacobi-preconditioned Chebyshev: one fused kernel and one ghost fill per degree; a fixed
499 // polynomial in a symmetric operator, so CG-safe by construction.
500 void chebyshevSmooth(std::size_t l, int degree) const
501 {
502 if (degree <= 0)
503 {
504 return;
505 }
506 const GmgLevelT<T>& L = levels_[l];
507 const GmgSystem<T> sys {L.sol.get(), L.rhs.get(), levelCoeffs(L)};
508 ChebRecurrence rec = chebRecurrence(L.lambdaMax);
509 for (int m = 0; m < degree; ++m)
510 {
511 fillGhosts(L, static_cast<int>(l));
512 const ChebStep<T> step = nextChebStep(m, rec);
513 if (onDevice_)
514 {
515 prof::Timer t("gmg.cheb", static_cast<int>(l));
516 gmgChebComputeD(sys, *L.chebD, step, onDevice_);
517 }
518 else
519 {
520 gmgChebComputeD(sys, *L.chebD, step, onDevice_);
521 amrex::Gpu::streamSynchronize();
522 }
523 GmgFab<T>::Saxpy(*L.sol, T(1), *L.chebD, 0, 0, 1, amrex::IntVect(0)); // sol += d
524 if (!onDevice_)
525 {
526 amrex::Gpu::streamSynchronize();
527 }
528 }
529 }

531 // lambda_max(D^{-1}A) by power iteration from a checkerboard seed, inflated by kChebSafety.
532 double estimateLambdaMax(std::size_t l) const
533 {
534 const GmgLevelT<T>& L = levels_[l];
535 GmgFab<T>& v = *L.sol; // scratch (1 ghost)
536 GmgFab<T>& w = *L.chebD; // scratch (0 ghost)
537 const FaceCoeffs<T> fc = levelCoeffs(L);
538 gmgFillChecker(v, onDevice_);
539 if (!onDevice_)
540 {
541 amrex::Gpu::streamSynchronize();
542 }
543 double norm = gmgNorm2(v);
544 v.mult(static_cast<T>(1.0 / norm), 0, 1, 0);
545 double lambda = 0.0;
546 for (int it = 0; it < kPowerIters; ++it)
547 {
548 fillGhosts(L, static_cast<int>(l));
549 gmgDinvApply(v, w, fc, onDevice_);
550 if (!onDevice_)
551 {
552 amrex::Gpu::streamSynchronize();
553 }
554 lambda = gmgNorm2(w); // v is unit-norm -> ||D^{-1}A v|| ~ lambda_max
555 if (lambda <= 0.0)
556 {
557 break;
558 }
559 gmgConvertCopy(v, w, onDevice_); // v <- w
560 if (!onDevice_)
561 {
562 amrex::Gpu::streamSynchronize();
563 }
564 v.mult(static_cast<T>(1.0 / lambda), 0, 1, 0);
565 }
566 v.setVal(T(0)); // leave sol clean for the V-cycle
567 amrex::Gpu::streamSynchronize();
568 return lambda * kChebSafety;
569 }

571 // Krylov solve of the coarsest system A z = rhs, replacing the fixed sweeps; x starts at
572 // zero, which keeps the bottom count reproducible (report/blockamr-gmg-notes.md#bottom).
573 void bottomSolve(const GmgLevelT<T>& L) const
574 {
575 prof::Timer t("gmg.bottom");
576 if (onDevice_)
577 {
578 gather_device(*L.rhs, bottomB_->get_values(), 1.0);
579 amrex::Gpu::streamSynchronize(); // b read by Ginkgo next
580 }
581 else
582 {
583 gather(*L.rhs, bottomB_->get_values(), 1.0);
584 }
585 bottomX_->fill(T(0));
586 bottomSolver_->apply(bottomBGlobal_, bottomXGlobal_);
587 if (onDevice_)
588 {
589 this->get_executor()->synchronize(); // x written by Ginkgo
590 scatter_device(bottomX_->get_const_values(), *L.sol);
591 }
592 else
593 {
594 scatter(bottomX_->get_const_values(), *L.sol);
595 }
596 }

598 // One V-cycle correcting levels_[l].sol in place; a warm start is allowed.
599 void vcycle(std::size_t l) const
600 {
601 const GmgLevelT<T>& L = levels_[l];
602 if (l + 1 == levels_.size())
603 {
604 if (bottomSolver_)
605 {
606 bottomSolve(L);
607 return;
608 }
609 // Forward + reversed halves keep the coarsest "solve" self-adjoint.
610 smooth(l, coarsestSweeps_ / 2, false);
611 smooth(l, coarsestSweeps_ / 2, true);
612 return;
613 }
614 smooth(l, preSweeps_, false);
615 fillGhosts(L, static_cast<int>(l));
616 const GmgLevelT<T>& C = levels_[l + 1];
