15// bfloat16 as a STORAGE type for the GMG level hierarchy: every operation below converts
16// to float and rounds once back. Rejected for the FIELDS, kept for the COEFFICIENTS —
17// report/blockamr-precision-measurements.md (NeoFOAM repo).
18struct Bf16
19{
20 std::uint16_t bits;
21
22 // Trivially default constructible: BaseFab's placementNew is then a no-op.
23 Bf16() noexcept = default;
24
25 // The only converting constructor: a double overload would make Bf16(0) ambiguous.
26 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE Bf16(float f) noexcept : bits(fromFloat(f)) {}
27
28 // Implicit, so kernel expressions over a bf16 level promote to float and compute there.
29 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE operator float() const noexcept
30 {
31 return toFloat(bits);
32 }
33
34 // A class type finds no built-in +=, and the prolongation is written that way.
35 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE Bf16& operator+=(float rhs) noexcept
36 {
37 bits = fromFloat(toFloat(bits) + rhs);
38 return *this;
39 }
40
41 // Round to nearest, ties to even; a NaN maps to a quiet NaN rather than an infinity.
42 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE static std::uint16_t fromFloat(float f) noexcept
43 {
44 Bits b;
45 b.f = f;
46 if ((b.u & 0x7F800000u) == 0x7F800000u && (b.u & 0x007FFFFFu) != 0u)
47 {
48 return 0x7FC0u;
49 }
50 const std::uint32_t rounding = 0x7FFFu + ((b.u >> 16) & 1u);
51 return static_cast<std::uint16_t>((b.u + rounding) >> 16);
52 }
53
54 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE static float toFloat(std::uint16_t h) noexcept
55 {
56 Bits b;
57 b.u = static_cast<std::uint32_t>(h) << 16;
58 return b.f;
59 }
60
61private:
62
63 // Union punning: what GCC, Clang and nvcc all support; std::bit_cast is not
64 // device-usable across our toolchains.
65 union Bits
66 {
67 float f;
68 std::uint32_t u;
69 };
70};
71
