23/* @brief What a ONE-SHOT solve takes beyond the system itself -- the union of the three
24 * `ginkgo_solve*` Python argument lists, so a binding lambda fills the fields its own
25 * surface exposes and leaves the rest at these defaults (which ARE the Python ones).
26 * Not a SolverConfig: nothing here is parsed to an enum, and `solver` must reach
27 * generateBasicSolver unvalidated so an unknown spelling still fails in its wording.
28 * Each entry point below names the fields it reads.
29 */
30struct OneshotSpec
31{
32 // "cg"/"bicgstab"/"gmres" (generateBasicSolver's subset).
33 std::string solver = "bicgstab";
34 int maxIter = 1000;
35 // Quoted against the ORIGINAL system's ||rhs||, then applied absolutely; atol > 0 adds
36 // the plain ||r_k||_2 <= atol.
37 double rtol = 1e-10;
38 double atol = 0.0;
39 // sign*L must be SPD: -1 for MLPoisson, +1 for MLABecLaplacian. MLLinOp paths only.
40 double sign = -1.0;
41 // Where the Krylov vector ops run; resolved at CALL time. MLLinOp paths only.
42 std::optional<NeoN::Executor> executor;
43};
