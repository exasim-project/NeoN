29// Threads per work block, and so the team size of the copy kernel.
30constexpr int kCopyBlock = 128;
31
32// One work block of one rectangular region copy, in GLOBAL cell indices:
33// dst[dstBox](i, j, k) = sign * src[srcBox](i + sh[0], j + sh[1], k + sh[2])
34// over cells [base, base + kCopyBlock) of [lo, lo + len), i fastest. Box indices are LOCAL;
35// `sh` is 0 for a same-region copy, the negated image shift for a wrapped ghost. Why
36// fixed-size blocks and not one team per region: notes#halo-plans.
37struct CopyTask
38{
39 int dst;
40 int src;
41 int lo[3];
42 int len[3];
43 int sh[3];
44 int base;
45 // -1 only for a reflect-odd (homogeneous Dirichlet) domain ghost: a mirror copy, negated.
46 int sign;
47};
48
49// A whole exchange as one device table, so it executes in one launch.
50struct CopyPlan
51{
52 Kokkos::View<CopyTask*> tasks;
53
54 [[nodiscard]] int size() const { return static_cast<int>(tasks.extent(0)); }
55};

60inline CopyPlan toDevice(const std::vector<CopyTask>& host, const char* name)
61{
62 CopyPlan plan;
63 // std::string, not the char*: view_alloc reads a raw pointer as memory to wrap.
64 plan.tasks = Kokkos::View<CopyTask*>(
65 Kokkos::view_alloc(std::string(name), Kokkos::WithoutInitializing), host.size()
66 );
67 auto mirror = Kokkos::create_mirror_view(plan.tasks);
68 for (std::size_t t = 0; t < host.size(); ++t)
69 {
70 mirror(t) = host[t];
71 }
72 Kokkos::deep_copy(plan.tasks, mirror);
73 return plan;
74}

76// The two LOCAL box indices one copy names.
77struct BoxPair
78{
79 int dst;
80 int src;

83inline void addTask(
84 std::vector<CopyTask>& out,
85 BoxPair boxes,
86 const amrex::Box& region,
87 const amrex::IntVect& sh,
88 int sign = 1
89)
90{
91 CopyTask t {};
92 t.dst = boxes.dst;
93 t.src = boxes.src;
94 t.sign = sign;
95 for (int d = 0; d < 3; ++d)
96 {
97 t.lo[d] = region.smallEnd(d);
98 t.len[d] = region.length(d);
99 t.sh[d] = sh[d];
100 }
101 const int npts = t.len[0] * t.len[1] * t.len[2];
102 for (int base = 0; base < npts; base += kCopyBlock)
103 {
104 t.base = base;
105 out.push_back(t);
106 }
107}

111// The ghost exchange of one FabArray: for every ghost cell, the valid cell -- of a box or of a
112// periodic image -- that FillBoundary would copy it from. Why the shell partition makes task
113// order irrelevant: notes#halo-plans.
114template<class FAB>
115CopyPlan makeHaloPlan(const amrex::FabArray<FAB>& mf, const amrex::Periodicity& period)
116{
117 const amrex::BoxArray& ba = mf.boxArray();
118 const amrex::IntVect ng = mf.nGrowVect();
119 const std::vector<amrex::IntVect> shifts = period.shiftIntVect();
120
121 std::vector<CopyTask> tasks;
122 std::vector<std::pair<int, amrex::Box>> isects;
123 for (int li = 0; li < mf.local_size(); ++li)
124 {
125 const amrex::Box valid = ba[mf.IndexArray()[li]];
126 for (const amrex::Box& shell : amrex::boxDiff(amrex::grow(valid, ng), valid))
127 {
128 for (const amrex::IntVect& img : shifts)
129 {
130 // Shift the query back into the source frame and the answer forward again.
131 ba.intersections(amrex::shift(shell, -img), isects);
132 for (const auto& is : isects)
133 {
134 detail::addTask(
135 tasks, {li, mf.localindex(is.first)}, amrex::shift(is.second, img), -img
136 );
137 }
138 }
139 }
140 }
141 return detail::toDevice(tasks, "gmg_halo_plan");
142}

144// The homogeneous domain-boundary ghost fill as a plan: per valid box on a non-periodic domain
145// face, the one-cell ghost layer and the mirror interior cell (sign -1 Dirichlet, +1 Neumann).
146// Twin of fillDomainBcGhosts* via la::bcGhostFill; face layers only, and runs AFTER the halo.
147template<class FAB>
148CopyPlan makeBcPlan(const amrex::FabArray<FAB>& mf, const amrex::Box& domain, const la::BcArray& bc)
149{
150 const amrex::BoxArray& ba = mf.boxArray();
151 std::vector<CopyTask> tasks;
152 for (int li = 0; li < mf.local_size(); ++li)
153 {
154 const amrex::Box valid = ba[mf.IndexArray()[li]];
155 for (int s = 0; s < 6; ++s)
156 {
157 la::BcGhostFill f;
158 if (!la::bcGhostFill(valid, domain, bc, s, f))
159 {
160 continue;
161 }
162 // A ghost layer comes from its own box's interior, so dst and src are one index.
163 detail::addTask(
164 tasks, {li, li}, f.gbx, amrex::IntVect(f.di, f.dj, f.dk), (f.sign < 0.0) ? -1 : 1
165 );
166 }
167 }
168 return detail::toDevice(tasks, "gmg_bc_plan");
169}

171// The valid-to-valid copy between two decompositions of one region -- what ParallelCopy does.
172// Cell-centred only: a face cell can have several sources, so the result would depend on task
173// order (notes#halo-plans).
174template<class FAB>
175CopyPlan makeCopyPlan(const amrex::FabArray<FAB>& dst, const amrex::FabArray<FAB>& src)
176{
177 std::vector<CopyTask> tasks;
178 std::vector<std::pair<int, amrex::Box>> isects;
179 for (int li = 0; li < dst.local_size(); ++li)
180 {
181 src.boxArray().intersections(dst.boxArray()[dst.IndexArray()[li]], isects);
182 for (const auto& is : isects)
183 {
184 detail::addTask(tasks, {li, src.localindex(is.first)}, is.second, amrex::IntVect(0));
185 }
186 }
187 return detail::toDevice(tasks, "gmg_copy_plan");
188}
