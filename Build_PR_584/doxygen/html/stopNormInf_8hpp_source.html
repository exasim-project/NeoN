32// Which norm the stopping criterion (and the reported residual) measures.
33enum class NormKind
34{
35 l2,
36 linf
37};

39inline NormKind parseNorm(const std::string& norm)
40{
41 if (norm == "l2")
42 {
43 return NormKind::l2;
44 }
45 if (norm == "linf")
46 {
47 return NormKind::linf;
48 }
49 throw std::runtime_error(
50 "ginkgo: unknown norm '" + norm + "' (expected 'l2' or 'linf'; 'linf' is MLMG's)"
51 );
52}

54// max |v_i| over a single-column Dense: on its own executor (hence the synchronize --
55// AMReX's stream is unordered against the one Ginkgo wrote `v` on) and Max-reduced across
56// ranks, or a rank stops on its own residual. Size and values come from the local
57// accessors: get_size() on a distributed vector is the GLOBAL row count.
58inline double normInf(const gko::LinOp* v)
59{
60 const auto n = localRows(v);
61 double m = 0.0;
62 if (n > 0)
63 {
64 const double* p = localValues<double>(v);
65 auto exec = v->get_executor();
66 if (exec->get_master().get() == exec.get())
67 {
68 for (gko::size_type i = 0; i < n; ++i)
69 {
70 m = std::max(m, std::abs(p[i]));
71 }
72 }
73 else
74 {
75 exec->synchronize();
76 // Max and Min rather than a Max over |p_i|: the pointer overloads take no
77 // device lambda, so this header stays includable anywhere.
78 const auto hi = amrex::Reduce::Max<double>(n, p);
79 const auto lo = amrex::Reduce::Min<double>(n, p);
80 m = std::max(std::abs(hi), std::abs(lo));
81 }
82 }
83 amrex::ParallelAllReduce::Max(m, amrex::ParallelContext::CommunicatorSub());
84 return m;
85}

87// ||r||_inf <= tau * baseline. Single right-hand side only: a per-column inf-norm would
88// need a per-column reduction amrex::Reduce cannot express over a flat pointer.
89class ResidualNormInf : public gko::EnablePolymorphicObject<ResidualNormInf, gko::stop::Criterion>
90{
91 friend class gko::EnablePolymorphicObject<ResidualNormInf, gko::stop::Criterion>;
92
93public:
94
95 GKO_CREATE_FACTORY_PARAMETERS(parameters, Factory)
96 {
97 // As in gko::stop::ResidualNorm: the factor the baseline is multiplied by.
98 double GKO_FACTORY_PARAMETER_SCALAR(reduction_factor, 1e-10);
99
100 // absolute | rhs_norm | initial_resnorm, as gko::stop::ResidualNorm but in the
101 // inf-norm. MLMG uses the GREATER of the two, which coincide at a zero guess.
102 gko::stop::mode GKO_FACTORY_PARAMETER_SCALAR(baseline, gko::stop::mode::rhs_norm);
103 };
104 GKO_ENABLE_CRITERION_FACTORY(ResidualNormInf, parameters, Factory);
105 GKO_ENABLE_BUILD_METHOD(Factory);
106
107protected:
108
109 // Required by Ginkgo's polymorphic-object machinery (create_default/clear).
110 explicit ResidualNormInf(std::shared_ptr<const gko::Executor> exec)
111 : gko::EnablePolymorphicObject<ResidualNormInf, gko::stop::Criterion>(std::move(exec))
112 {}
113
114 explicit ResidualNormInf(const Factory* factory, const gko::stop::CriterionArgs& args)
115 : gko::EnablePolymorphicObject<ResidualNormInf, gko::stop::Criterion>(factory->get_executor(
116 )),
117 parameters_ {factory->get_parameters()}
118 {
119 const double tau = parameters_.reduction_factor;
120 switch (parameters_.baseline)
121 {
122 case gko::stop::mode::absolute:
123 target_ = tau;
124 break;
125 case gko::stop::mode::rhs_norm:
126 target_ = tau * baselineNorm(args.b.get(), "b");
127 break;
128 case gko::stop::mode::initial_resnorm:
129 target_ = tau * baselineNorm(args.initial_residual, "initial_residual");
130 break;
131 }
132 }

134 bool check_impl(
135 gko::uint8 stoppingId,
136 bool setFinalized,
137 gko::array<gko::stopping_status>* stop_status,
138 bool* one_changed,
139 const Updater& updater
140 ) override
141 {
142 const gko::LinOp* r = updater.residual_;
143 if (r == nullptr && updater.ignore_residual_check_)
144 {
145 // Not a missing residual: Ir asks for the cheap checks first and calls again
146 // with it. Throwing here instead made ResidualNormInf unusable with ir/mpir.
147 return false;
148 }
149 if (r == nullptr)
150 {
151 // A solver publishing only the implicit squared 2-norm cannot be stopped on
152 // an inf-norm; refuse rather than silently stop on a different criterion.
153 throw gko::NotSupported(
154 __FILE__, __LINE__, __func__, "ResidualNormInf needs the residual vector"
155 );
156 }
157 last_ = normInf(r);
158 const bool converged = last_ <= target_;
159 if (converged && stop_status != nullptr)
160 {
161 auto exec = this->get_executor();
162 const auto n = stop_status->get_size();
163 gko::array<gko::stopping_status> host(exec->get_master(), *stop_status);
164 for (gko::size_type i = 0; i < n; ++i)
165 {
166 host.get_data()[i].converge(stoppingId, setFinalized);
167 }
168 // copy_from into the existing storage: the solver holds pointers into it.
169 exec->copy_from(exec->get_master(), n, host.get_const_data(), stop_status->get_data());
170 }
171 if (one_changed != nullptr)
172 {
173 *one_changed = converged;
174 }
175 return converged;
176 }
177
178private:
179
180 static double baselineNorm(const gko::LinOp* v, const char* what)
181 {
182 if (v == nullptr)
183 {
184 throw gko::NotSupported(
185 __FILE__, __LINE__, __func__, std::string("ResidualNormInf needs ") + what
186 );
187 }
188 const double nrm = normInf(v);
189 // A zero baseline would make the target 0 and the solve unstoppable.
190 return (nrm > 0.0) ? nrm : 1.0;
191 }
192
193 double target_ = 0.0;
194 double last_ = 0.0;
195};
