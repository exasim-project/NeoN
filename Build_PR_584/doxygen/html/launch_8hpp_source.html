37template<class F>
38void launchKokkosMd(const amrex::Box& bx, F const& f)
39{
40 const auto lo = bx.smallEnd();
41 const auto hi = bx.bigEnd();
42 Kokkos::parallel_for(
43 "bench_md",
44 Kokkos::MDRangePolicy<Kokkos::Rank<3>>(
45 {lo[0], lo[1], lo[2]}, {hi[0] + 1, hi[1] + 1, hi[2] + 1}
46 ),
47 f
48 );
49}

51template<class F>
52void launchKokkosFlat(const amrex::Box& bx, F const& f)
53{
54 const auto lo = bx.smallEnd();
55 const int nx = bx.length(0);
56 const int ny = bx.length(1);
57 const long n = static_cast<long>(nx) * bx.length(1) * bx.length(2);
58 const int lx = lo[0], ly = lo[1], lz = lo[2];
59 Kokkos::parallel_for(
60 "bench_flat",
61 Kokkos::RangePolicy<>(0, n),
62 KOKKOS_LAMBDA(const long t) {
63 // i fastest, matching Array4's i-contiguous layout.
64 const int i = static_cast<int>(t % nx);
65 const int j = static_cast<int>((t / nx) % ny);
66 const int k = static_cast<int>(t / (static_cast<long>(nx) * ny));
67 f(i + lx, j + ly, k + lz);
68 }
69 );
70}

72// Streams: AMReX round-robins its box loop over numGpuStreams() so short per-box kernels
73// overlap; a plain Kokkos::parallel_for issues to ONE -- the measured multi-box penalty.
74
75// Heap-owned, not a function-local static: the instances own cudaStreams and must die
76// BEFORE Kokkos::finalize and amrex::Finalize, not at static teardown afterwards.
77inline std::vector<Kokkos::DefaultExecutionSpace>*& streamPoolPtr()
78{
79 static std::vector<Kokkos::DefaultExecutionSpace>* pool = nullptr;
80 return pool;
81}
82
83inline const Kokkos::DefaultExecutionSpace& benchStream(int ibox)
84{
85 auto*& pool = streamPoolPtr();
86 if (pool == nullptr)
87 {
88 // Exactly as many streams as AMReX uses, so the comparison is of the launcher.
89 const int n = std::max(1, amrex::Gpu::Device::numGpuStreams());
90 pool = new std::vector<Kokkos::DefaultExecutionSpace>(Kokkos::Experimental::partition_space(
91 Kokkos::DefaultExecutionSpace {}, std::vector<int>(static_cast<std::size_t>(n), 1)
92 ));
93 }
94 return (*pool)[static_cast<std::size_t>(ibox) % pool->size()];
95}
96
97inline void releaseStreamPool()
98{
99 delete streamPoolPtr();
100 streamPoolPtr() = nullptr;
101}

140// The Kokkos twin of AMReX's fused path: MT, nblocks_per_box and the cached DEVICE
141// BoxIndexer table are AMReX's own, so the comparison is one portable TeamPolicy against
142// AMReX's per-backend launch. Valid cells only -- the MDRangePolicy fallback below rejects
143// a negative lower bound.
144template<class MF, class F>
145void launchKokkosTeamNamed(const char* name, const MF& mf, F const& f)
146{
147 const int nboxes = static_cast<int>(mf.IndexArray().size());
148 if (nboxes == 0)
149 {
150 return;
151 }
152 if (nboxes == 1)
153 {
154 // The same fallback AMReX makes: with one box there is nothing to fuse and the
155 // team mapping costs, so the fused column would report a handicap AMReX does not.
156 const amrex::Box bx = mf.box(mf.IndexArray()[0]);
157 launchKokkosMdNamed(
158 name, bx, BLOCKAMR_LAMBDA(int i, int j, int k) { f(0, i, j, k); }
159 );
160 return;
161 }
162
163 constexpr int MT = 128; // threads per block, == AMReX's CUDA MT
164 constexpr int VL = 32; // vector lanes = one warp, so blockDim is (32, 4, 1)
165
166 const auto& info = mf.getParForInfo(amrex::IntVect(0));
167 const int nblocksPerBox = info.getNBlocksPerBox(MT);
168 const amrex::BoxIndexer* boxes = info.getBoxes();
169
170 using Policy = Kokkos::TeamPolicy<Kokkos::DefaultExecutionSpace>;
171 Kokkos::parallel_for(
172 name,
173 Policy(nboxes * nblocksPerBox, MT / VL, VL),
174 KOKKOS_LAMBDA(const Policy::member_type& team) {
175 const int blk = team.league_rank();
176 const int ibox = blk / nblocksPerBox;
177 const auto base = static_cast<std::uint64_t>(blk - ibox * nblocksPerBox) * MT;
178 const amrex::BoxIndexer& indexer = boxes[ibox];
179 Kokkos::parallel_for(
180 Kokkos::TeamVectorRange(team, MT),
181 [&](const int t)
182 {
183 const std::uint64_t icell = base + static_cast<std::uint64_t>(t);
184 if (icell < indexer.numPts())
185 {
186 const auto ijk = indexer(icell);
187 f(ibox, ijk.x, ijk.y, ijk.z);
188 }
189 }
190 );
191 }
192 );
193}

201// Accessors: both take GLOBAL (i, j, k). amrex::Array4 already does; ViewAcc gives an
202// unmanaged Kokkos View the same convention by subtracting the fab box origin.
203
204using FabView = Kokkos::View<
205 double***,
206 Kokkos::LayoutLeft,
207 Kokkos::DefaultExecutionSpace::memory_space,
208 Kokkos::MemoryTraits<Kokkos::Unmanaged>>;
209
210struct ViewAcc
211{
212 FabView v;
213 int ox, oy, oz;
214
215 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE double& operator()(int i, int j, int k) const
216 {
217 return v(i - ox, j - oy, k - oz);
218 }
219};
220
221// Unmanaged View over the whole fab box, origin recorded so the accessor is global.
222inline ViewAcc viewAcc(amrex::FArrayBox& fab)
223{
224 const amrex::Box& b = fab.box();
225 return ViewAcc {
226 FabView(fab.dataPtr(), b.length(0), b.length(1), b.length(2)),
227 b.smallEnd(0),
228 b.smallEnd(1),
229 b.smallEnd(2)
230 };
231}
