14// Vectors are Dense on one rank and distributed::Vector on several -- unrelated types, so
15// `gko::as<Dense>` throws on the latter; these accessors are the seam. The distributed type
16// exists only so the dots/norms inside Ginkgo's Krylov solvers become allreduces.
17#if GINKGO_BUILD_MPI
18template<class V>
19using DistVec = gko::experimental::distributed::Vector<V>;
20#endif

22// Writable raw pointer to the process-local values of x.
23template<class V>
24V* localValues(gko::LinOp* x)
25{
26#if GINKGO_BUILD_MPI
27 if (auto* d = dynamic_cast<DistVec<V>*>(x))
28 {
29 return d->get_local_values();
30 }
31#endif
32 return gko::as<gko::matrix::Dense<V>>(x)->get_values();
33}

47// Process-local row count. NOT get_size()[0], which on a distributed vector is the GLOBAL
48// count -- sizing flat vectors by it was a multi-rank bug.
49inline gko::size_type localRows(const gko::LinOp* x)
50{
51#if GINKGO_BUILD_MPI
52 if (const auto* d = dynamic_cast<const DistVec<double>*>(x))
53 {
54 return d->get_local_vector()->get_size()[0];
55 }
56 if (const auto* d = dynamic_cast<const DistVec<float>*>(x))
57 {
58 return d->get_local_vector()->get_size()[0];
59 }
60#endif
61 return x->get_size()[0];
62}

64// Non-owning Dense VIEW of the process-local part. Aliases x: writes go through to x.
65template<class V>
66std::unique_ptr<gko::matrix::Dense<V>> localView(gko::LinOp* x)
67{
68 const auto n = localRows(x);
69 auto exec = x->get_executor();
70 return gko::matrix::Dense<V>::create(
71 exec, gko::dim<2> {n, 1}, gko::make_array_view(exec, n, localValues<V>(x)), 1
72 );
73}

75// const overload, for the host paths that clone b to the master executor.
76template<class V>

92inline double globalNorm2(const gko::LinOp* v)
93{
94 auto exec = v->get_executor();
95 auto nrm = gko::matrix::Dense<double>::create(exec, gko::dim<2> {1, 1});
96#if GINKGO_BUILD_MPI
97 if (const auto* d = dynamic_cast<const DistVec<double>*>(v))
98 {
99 d->compute_norm2(nrm);
100 return reduceScalar(v, nrm.get());
101 }
102#endif
103 gko::as<const gko::matrix::Dense<double>>(v)->compute_norm2(nrm);
104 return reduceScalar(v, nrm.get());
105}

107inline double globalDot(const gko::LinOp* v, const gko::LinOp* w)
108{
109 auto exec = v->get_executor();
110 auto res = gko::matrix::Dense<double>::create(exec, gko::dim<2> {1, 1});
111#if GINKGO_BUILD_MPI
112 if (const auto* d = dynamic_cast<const DistVec<double>*>(v))
113 {
114 d->compute_dot(w, res);
115 return reduceScalar(v, res.get());
116 }
117#endif
118 gko::as<const gko::matrix::Dense<double>>(v)->compute_dot(w, res);
119 return reduceScalar(v, res.get());
120}
