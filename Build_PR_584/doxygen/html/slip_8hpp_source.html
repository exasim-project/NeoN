28// --- Scalar specialization: zero-gradient ---
29template<>
30inline void setSlipValue<NeoN::scalar>(
31 Field<NeoN::scalar>& domainVector,
32 const UnstructuredMesh& mesh,
33 std::pair<localIdx, localIdx> range
34)
35{
36 const auto internalV = domainVector.internalVector().view();
37
38 auto [refGradV, valueV, valueFractionV, refValueV, boundaryFaceOwnersV] = views(
39 domainVector.boundaryData().refGrad(),
40 domainVector.boundaryData().value(),
41 domainVector.boundaryData().valueFraction(),
42 domainVector.boundaryData().refValue(),
43 mesh.boundaryMesh().faceOwners()
44 );
45
46 NeoN::parallelFor(
47 domainVector.exec(),
48 range,
49 NEON_LAMBDA(const localIdx i) {
50 const localIdx owner = boundaryFaceOwnersV[i];
51 const auto v = internalV[owner];
52
53 refValueV[i] = v;
54 valueV[i] = v;
55 valueFractionV[i] = 0.0;
56 refGradV[i] = 0.0;
57 },
58 "setSlipValue(scalar)"
59 );
60}

62// --- Vec3 specialization: tangential projection (remove normal component) ---
63template<>
64inline void setSlipValue<NeoN::Vec3>(
65 Field<NeoN::Vec3>& domainVector,
66 const UnstructuredMesh& mesh,
67 std::pair<localIdx, localIdx> range
68)
69{
70 const auto internalV = domainVector.internalVector().view();
71
72 auto [refGradV, valueV, valueFractionV, refValueV, boundaryFaceOwnersV, faceUnitNormalsV] =
73 views(
74 domainVector.boundaryData().refGrad(),
75 domainVector.boundaryData().value(),
76 domainVector.boundaryData().valueFraction(),
77 domainVector.boundaryData().refValue(),
78 mesh.boundaryMesh().faceOwners(),
79 mesh.boundaryMesh().faceUnitNormals()
80 );
81
82 NeoN::parallelFor(
83 domainVector.exec(),
84 range,
85 NEON_LAMBDA(const localIdx i) {
86 const localIdx owner = boundaryFaceOwnersV[i];
87 const auto v = internalV[owner];
88 const auto n = faceUnitNormalsV[i];
89
90 const auto vtan = v - n * (v & n);
91
92 refValueV[i] = vtan;
93 valueV[i] = vtan;
94 valueFractionV[i] = 0.0;
95 refGradV[i] = NeoN::zero<NeoN::Vec3>();
96 },
97 "setSlipValue(Vec3)"
98 );
99}
