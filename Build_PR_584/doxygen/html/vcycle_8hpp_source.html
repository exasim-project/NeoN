40// The level scalar is a template parameter, as in GmgPrecondT: the hierarchy below level 0 can
41// run narrower while the operator and the caller's fields stay fp64. bf16 is STORAGE only
42// (gmg/bf16.hpp); only kokkos_opt is instantiated narrow, so the baselines stay baselines.
43enum class Precision
44{
45 fp64,
46 fp32,
47 bf16
48};
49
50Precision parsePrecision(const std::string& p)
51{
52 if (p == "fp64")
53 {
54 return Precision::fp64;
55 }
56 if (p == "fp32")
57 {
58 return Precision::fp32;
59 }
60 if (p == "bf16")
61 {
62 return Precision::bf16;
63 }
64 // Not "benchGmgVcycle": apply.cpp reaches this from the production factory, where a
65 // bench-flavoured prefix names a function the caller never invoked.
66 throw std::runtime_error(
67 "blockAmr GMG: unknown precision '" + p + "' (expected 'fp64', 'fp32' or 'bf16')"
68 );
69}
70
71// Only used to reject a coefficient type WIDER than the field type: a configuration mistake.
72int precisionBytes(Precision p)
73{
74 switch (p)
75 {
76 case Precision::fp64:
77 return 8;
78 case Precision::fp32:
79 return 4;
80 case Precision::bf16:
81 return 2;
82 }
83 return 8;
84}
85
86// The coefficient precision defaults to the field precision.
87Precision parseCoeffPrecision(const std::string& coeff, const std::string& field)
88{
89 const Precision f = parsePrecision(field);
90 if (coeff.empty())
91 {
92 return f;
93 }
94 const Precision c = parsePrecision(coeff);
95 if (precisionBytes(c) > precisionBytes(f))
96 {
97 throw std::runtime_error(
98 "blockAmr GMG: coeff_precision '" + coeff + "' is wider than precision '" + field
99 + "'; narrow the fields first"
100 );
101 }
102 return c;
103}

139// Are these two fields the same numbers everywhere? ux == lx pointwise IS symmetry of the
140// operator, and so the exact condition for one array to stand in for both. Bitwise on purpose
141// (report/blockamr-gmg-notes.md#share-coeffs); setup only, so the reduction is never timed.
142bool sameField(const amrex::MultiFab& a, const amrex::MultiFab& b)
143{
144 if (&a == &b)
145 {
146 return true;
147 }
148 if (a.boxArray() != b.boxArray() || a.DistributionMap() != b.DistributionMap())
149 {
150 return false;
151 }
152 const auto aa = a.const_arrays();
153 const auto bb = b.const_arrays();
154 double diff = amrex::ParReduce(
155 amrex::TypeList<amrex::ReduceOpMax> {},
156 amrex::TypeList<double> {},
157 a,
158 amrex::IntVect(0),
159 [=] AMREX_GPU_DEVICE(int box, int i, int j, int k) -> amrex::GpuTuple<double>
160 { return {amrex::Math::abs(aa[box](i, j, k) - bb[box](i, j, k))}; }
161 );
162 // ParReduce is rank-local, and ranks that disagreed would build different hierarchies.
163 amrex::ParallelAllReduce::Max(diff, amrex::ParallelContext::CommunicatorSub());
164 return diff == 0.0;
165}

177 const amrex::MultiFab* lz;

180// One multigrid level, as in GmgLevelT: geometry, coefficients, work fields (sol has 1 ghost).
181template<class T, class TC>
182struct LevelT
183{
184 // Production's fab type, so the production kernels apply unchanged; fields T, coeffs TC.
185 using Fab = la::GmgFab<T>;
186 using CoeffFab = la::GmgFab<TC>;
187
188 amrex::Geometry geom;
189 std::unique_ptr<CoeffFab> alpha, ux, lx, uy, ly, uz, lz;
190 std::unique_ptr<Fab> sol, rhs;
191
192 // lx/ly/lz are NULL on a level sharing one face coefficient per direction (kokkos_opt only).
193 // Always read the lower coefficients through these accessors (notes#share-coeffs).
194 [[nodiscard]] const CoeffFab& lxf() const { return lx ? *lx : *ux; }
195 [[nodiscard]] const CoeffFab& lyf() const { return ly ? *ly : *uy; }
196 [[nodiscard]] const CoeffFab& lzf() const { return lz ? *lz : *uz; }
197 [[nodiscard]] bool shared() const { return lx == nullptr; }
198
199 // Agglomerated levels only: the restriction output and prolongation input on coarsen(fine
200 // BoxArray, 2) with the FINE DistributionMapping, since the inter-level kernels address
201 // fine and coarse at the SAME local box index (notes#agglomeration).
202 std::unique_ptr<Fab> xferRhs, xferSol;
203 bool agglomerated = false;
204
205 // kokkos_opt only, empty otherwise: this level's data movements resolved to device tables.
206 CopyPlan halo, bc, xferIn, xferOut;
207};

218 // The coefficient fields are read at SETUP only (copyCallerCoeffs); the rhs is not read here
219 // at all, so it stays a per-call argument of reset/applyFlat.
220 Vcycle(const amrex::Geometry& geom, const CallerCoeffs& c, const KokkosGmgOpts& opts)
221 : preSweeps_(opts.preSweeps), postSweeps_(opts.postSweeps),
222 coarsestSweeps_(opts.coarsestSweeps), omega_(opts.omega), bc_(opts.bc),
223 hasPhysBc_(std::any_of(opts.bc.begin(), opts.bc.end(), [](int b) { return b != 0; })),
224 amrexFree_(Backend::amrexFreeCycle && amrex::ParallelContext::NProcsSub() == 1)
225 {
226 // One face coefficient per direction instead of a pair -- only when asked, only if the
227 // backend allows it, and only if the operator really is symmetric.
228 shared_ = Backend::canShareCoeffs && opts.shareCoeffs && sameField(*c.ux, *c.lx)
229 && sameField(*c.uy, *c.ly) && sameField(*c.uz, *c.lz);
230
231 buildLevel0(geom, c, opts);
232 coarsenHierarchy(opts);
233 if (amrexFree_)
234 {
235 resolvePlans();
236 }
237 amrex::Gpu::streamSynchronize();
238 }
239
240 // L0 rhs := the caller's rhs, L0 sol := 0 — z0 = 0 applies M^{-1}, not a warm start.
241 void reset(const amrex::MultiFab& rhs)
242 {
243 if (aggL0_)
244 {
245 la::gmgConvertCopy(*iface_, rhs, /*onDevice=*/true);
246 Backend::afterAmrexWrite();
247 gmgCopyKokkos<T>(*levels_[0].rhs, *iface_, ifaceIn_);
248 gmgZeroKokkos<T>(*levels_[0].sol);
249 Kokkos::fence();
250 return;
251 }
252 Backend::beforeAmrexRead(); // a previous cycle's kernels may still be in flight
253 la::gmgConvertCopy(*levels_[0].rhs, rhs, /*onDevice=*/true);
254 levels_[0].sol->setVal(T(0));
255 amrex::Gpu::streamSynchronize();
256 }

258 void cycles(int n)
259 {
260 for (int c = 0; c < n; ++c)

262 vcycle(0);
263 }

266 // Preconditioner apply on flat device vectors: L0 rhs <- r, L0 sol <- 0, n V-cycles,
267 // z <- L0 sol. Same sequence as GmgPrecondT::apply_impl, so a comparison measures the cycle.
268 template<class V>
269 void applyFlat(const V* r, V* z, int nCycles)
270 {
271 Level& L0 = levels_.front();
272 // The flat vector's cell order is the CALLER's, so an agglomerated level 0 needs a
273 // staging fab and a plan copy each way -- once per APPLY, which more cycles amortise.
274 if (aggL0_)
275 {
276 la::scatter_device(r, *iface_);
277 Backend::afterAmrexWrite();
278 gmgCopyKokkos<T>(*L0.rhs, *iface_, ifaceIn_);
279 gmgZeroKokkos<T>(*L0.sol); // z0 = 0: apply M^{-1}, not a warm-started solve
280 cycles(nCycles);
281 gmgCopyKokkos<T>(*iface_, *L0.sol, ifaceOut_);
282 Backend::beforeAmrexRead();
283 la::gather_device(*iface_, z, 1.0);
284 amrex::Gpu::streamSynchronize();
285 return;
286 }
287 Backend::beforeAmrexRead(); // a previous apply's kernels may still be in flight
288 la::scatter_device(r, *L0.rhs);
289 L0.sol->setVal(T(0)); // z0 = 0: apply M^{-1}, not a warm-started solve
290 Backend::afterAmrexWrite();
291 cycles(nCycles);
292 Backend::beforeAmrexRead();
293 la::gather_device(*L0.sol, z, 1.0);
294 amrex::Gpu::streamSynchronize(); // z complete before the caller reads it
295 }

297 // sum((rhs - A sol)^2) on the finest level. Reporting only -- the gate that says the timed
298 // cycle did the work -- so it stays an AMReX reduction and is never timed.
299 double residSumSq()
300 {
301 Level& L = levels_.front();
302 fillGhosts(L);
303 Backend::beforeAmrexRead();
304 const auto psi = L.sol->const_arrays();
305 const auto b = L.rhs->const_arrays();
306 const auto ax = L.ux->const_arrays();
307 const auto lxa = L.lxf().const_arrays();
308 const auto ay = L.uy->const_arrays();
309 const auto lya = L.lyf().const_arrays();
310 const auto az = L.uz->const_arrays();
311 const auto lza = L.lzf().const_arrays();
312 const auto al = L.alpha->const_arrays();
313 double sum = amrex::ParReduce(
314 amrex::TypeList<amrex::ReduceOpSum> {},
315 amrex::TypeList<double> {},
316 *L.rhs,
317 amrex::IntVect(0),
318 [=] AMREX_GPU_DEVICE(int box, int i, int j, int k) -> amrex::GpuTuple<double>
319 {
320 const la::FaceCoeffArrays<TC> faces {
321 ax[box], lxa[box], ay[box], lya[box], az[box], lza[box]
322 };
323 const auto c = la::loadFaceCoeffs<double>(faces, i, j, k);
324 const double off = la::stencilOffDiag(
325 c,
326 psi[box](i + 1, j, k),
327 psi[box](i - 1, j, k),
328 psi[box](i, j + 1, k),
329 psi[box](i, j - 1, k),
330 psi[box](i, j, k + 1),
331 psi[box](i, j, k - 1)
332 );
333 const double diag = la::stencilDiag(al[box](i, j, k), c);
334 const double r = b[box](i, j, k) - (diag * psi[box](i, j, k) + off);
335 return {r * r};
336 }
337 );
338 // This rank's boxes only, like every ParReduce; the caller prints its root.
339 amrex::ParallelAllReduce::Sum(sum, amrex::ParallelContext::CommunicatorSub());
340 return sum;
341 }

396 static Level makeLevel(
397 const amrex::BoxArray& ba,
398 const amrex::DistributionMapping& dm,
399 const amrex::Geometry& geom,
400 bool shared
401 )
402 {
403 Level L;
404 L.geom = geom;
405 L.alpha = makeCoeffMf(ba, dm, 0);
406 const auto fba = [&ba](int d)
407 { return amrex::convert(ba, amrex::IntVect::TheDimensionVector(d)); };
408 L.ux = makeCoeffMf(fba(0), dm, 0);
409 L.uy = makeCoeffMf(fba(1), dm, 0);
410 L.uz = makeCoeffMf(fba(2), dm, 0);
411 if (!shared)
412 {
413 L.lx = makeCoeffMf(fba(0), dm, 0);
414 L.ly = makeCoeffMf(fba(1), dm, 0);
415 L.lz = makeCoeffMf(fba(2), dm, 0);
416 }
417 L.sol = makeMf(ba, dm, 1);
418 L.rhs = makeMf(ba, dm, 0);
419 return L;
420 }

445 // The finest level, on the caller's decomposition or -- when asked, and only where the plans
446 // are -- on its own: bigger boxes cost an interface but cut halo traffic on the level holding
447 // 7/8 of the cells (notes#agglomeration).
448 void buildLevel0(const amrex::Geometry& geom, const CallerCoeffs& c, const KokkosGmgOpts& opts)
449 {
450 const amrex::BoxArray& ba = c.alpha->boxArray();
451 const amrex::DistributionMapping& dm = c.alpha->DistributionMap();
452
453 amrex::BoxArray l0ba = ba;
454 amrex::DistributionMapping l0dm = dm;
455 // On >1 rank there are no plans, so the caller's layout is kept.
456 if (amrexFree_ && opts.aggLevel0Size > 0)
457 {
458 amrex::BoxArray tba(geom.Domain());
459 tba.maxSize(opts.aggLevel0Size);
460 // Strictly fewer boxes, or the interface is pure cost.
461 if (tba.size() < ba.size())
462 {
463 l0ba = tba;
464 l0dm = amrex::DistributionMapping(tba);
465 aggL0_ = true;
466 }
467 }
468
469 levels_.push_back(makeLevel(l0ba, l0dm, geom, shared_));
470 if (!aggL0_)
471 {
472 copyCallerCoeffs(c, levels_[0]);
473 return;
474 }
475 // The convert-copies need matching layouts, so rediscretise there and copy across.
476 Level t = makeLevel(ba, dm, geom, shared_);
477 copyCallerCoeffs(c, t);
478 copyCoeffs(t, levels_[0]);
479 iface_ = makeMf(ba, dm, 0);
480 }

493 // Coarsen while the BoxArray stays coarsenable and the coarse domain keeps >= minBottom
494 // cells. Without agglomeration only the box SIZE shrinks — hence a launch-bound bottom.
495 void coarsenHierarchy(const KokkosGmgOpts& opts)
496 {
497 while (opts.maxLevels <= 0 || static_cast<int>(levels_.size()) < opts.maxLevels)
498 {
499 // Copies, not references: addCoarseLevel's push_back moves the Level structs.
500 const amrex::BoxArray fba = levels_.back().alpha->boxArray();
501 const amrex::DistributionMapping fdm = levels_.back().alpha->DistributionMap();
502 const amrex::Geometry fgeom = levels_.back().geom;
503 if (!fba.coarsenable(2, 2))
504 {
505 return;
506 }
507 const amrex::Box cdom = amrex::coarsen(fgeom.Domain(), 2);
508 if (cdom.shortside() < opts.minBottom)
509 {
510 return;
511 }
512 amrex::BoxArray cba = fba;
513 cba.coarsen(2);
514 addCoarseLevel(coarseGeom(fgeom, cdom), cba, fdm, opts);
515 }
516 }

518 // Append the level below the current bottom and rediscretise onto it. Agglomeration: a fresh
519 // aggGridSize-capped decomposition of the coarse domain, taken only when it has strictly
520 // fewer boxes (notes#agglomeration).
521 void addCoarseLevel(
522 const amrex::Geometry& cgeom,
523 const amrex::BoxArray& cba,
524 const amrex::DistributionMapping& fdm,
525 const KokkosGmgOpts& opts
526 )
527 {
528 amrex::BoxArray aba = cba;
529 amrex::DistributionMapping adm = fdm;
530 bool agg = false;
531 if (opts.agglomerate)
532 {
533 amrex::BoxArray tba(cgeom.Domain());
534 tba.maxSize(opts.aggGridSize);
535 if (tba.size() < cba.size())
536 {
537 aba = tba;
538 adm = amrex::DistributionMapping(tba);
539 agg = true;
540 }
541 }
542
543 levels_.push_back(makeLevel(aba, adm, cgeom, shared_));
544 const Level& fl = levels_[levels_.size() - 2];
545 Level& c = levels_.back();
546 if (!agg)
547 {
548 restrictCoeffs(fl, c);
549 return;
550 }
551 c.agglomerated = true;
552 c.xferRhs = makeMf(cba, fdm, 0);
553 c.xferSol = makeMf(cba, fdm, 0);
554 // The restriction kernels only speak the fine layout: rediscretise there, copy.
555 Level t = makeLevel(cba, fdm, cgeom, shared_);
556 restrictCoeffs(fl, t);
557 copyCoeffs(t, c);
558 }

560 // Resolve the data movements once the hierarchy is final: a device table per movement.
561 void resolvePlans()
562 {
563 if (aggL0_)
564 {
565 ifaceIn_ = makeCopyPlan(*levels_[0].rhs, *iface_);
566 ifaceOut_ = makeCopyPlan(*iface_, *levels_[0].sol);
567 }
568 for (Level& L : levels_)
569 {
570 L.halo = makeHaloPlan(*L.sol, L.geom.periodicity());
571 if (hasPhysBc_)
572 {
573 // Coarse domains are the fine one coarsened, so one bc spec applies.
574 L.bc = makeBcPlan(*L.sol, L.geom.Domain(), bc_);
575 }
576 if (L.agglomerated)
577 {
578 L.xferIn = makeCopyPlan(*L.rhs, *L.xferRhs);
579 L.xferOut = makeCopyPlan(*L.xferSol, *L.sol);
580 }
581 }
582 }

584 // Rediscretise on the coarse level: faces via gmgCoarsenFace (4 fine faces averaged, then
585 // rescaled for the doubled dx), alpha via gmgRestrict's plain 8-child average -- correct
586 // only because alpha is dx-INDEPENDENT. A shared level does three faces, not six.
587 static void restrictCoeffs(const Level& f, Level& c)
588 {
589 la::gmgRestrict<TC>(*f.alpha, *c.alpha, /*onDevice=*/true);
590 la::gmgCoarsenFace<TC>(*f.ux, *c.ux, 0, 4.0, /*onDevice=*/true);
591 la::gmgCoarsenFace<TC>(*f.uy, *c.uy, 1, 4.0, /*onDevice=*/true);
592 la::gmgCoarsenFace<TC>(*f.uz, *c.uz, 2, 4.0, /*onDevice=*/true);
593 if (!c.shared())
594 {
595 la::gmgCoarsenFace<TC>(f.lxf(), *c.lx, 0, 4.0, /*onDevice=*/true);
596 la::gmgCoarsenFace<TC>(f.lyf(), *c.ly, 1, 4.0, /*onDevice=*/true);
597 la::gmgCoarsenFace<TC>(f.lzf(), *c.lz, 2, 4.0, /*onDevice=*/true);
598 }
599 }

601 // Move the rediscretised coefficients onto another decomposition of the same region. A
602 // shared internal face carries one value, so which source box wins does not matter.
603 static void copyCoeffs(const Level& src, Level& dst)
604 {
605 dst.alpha->ParallelCopy(*src.alpha, 0, 0, 1);

617 // Periodic/internal ghosts, then the homogeneous physical-BC reflection — production's order.
618 void fillGhosts(Level& L) const
619 {
620 if (amrexFree_)
621 {
622 gmgFillBoundaryKokkos<T>(*L.sol, L.halo);
623 if (hasPhysBc_)
624 {
625 gmgFillDomainBcKokkos<T>(*L.sol, L.bc);
626 }
627 return;
628 }
629 // AMReX is about to READ sol, which the last colour sweep wrote.
630 Backend::beforeAmrexRead();
631 L.sol->FillBoundary(L.geom.periodicity());

636 Backend::afterAmrexWrite();

639 // RB colour sweeps; `reversed` is the forward sweep's adjoint, so the V-cycle stays symmetric.
640 void smooth(std::size_t l, int sweeps, bool reversed)
641 {
642 Level& L = levels_[l];
643 for (int s = 0; s < sweeps; ++s)
644 {
645 for (int c = 0; c < 2; ++c)
646 {
647 const int parity = (reversed ? 1 + c : c) & 1;
648 fillGhosts(L); // the other colour changed — refresh ghosts
649 Backend::gsColor(*L.sol, *L.rhs, levelCoeffs(L), la::GsSweep {parity, omega_});
650 }
651 }
652 }

654 void vcycle(std::size_t l)
655 {
656 Level& L = levels_[l];
657 if (l + 1 == levels_.size())
658 {
659 // Forward + reversed halves keep the coarsest "solve" self-adjoint.
660 smooth(l, coarsestSweeps_ / 2, false);
661 smooth(l, coarsestSweeps_ / 2, true);
662 return;
663 }
664 smooth(l, preSweeps_, false);
665 fillGhosts(L);
666 Level& C = levels_[l + 1];
667 // On an agglomerated level the kernels use the transfer fab, and a copy bridges across.
668 Backend::residRestrict(
669 *L.sol, *L.rhs, C.agglomerated ? *C.xferRhs : *C.rhs, levelCoeffs(L)
670 );
671 seedCoarse(C);
672 vcycle(l + 1);
673 gatherCoarseCorrection(C);
674 Backend::prolongAdd(C.agglomerated ? *C.xferSol : *C.sol, *L.sol);
675 smooth(l, postSweeps_, true);
676 }

678 // The coarse right-hand side in place and its solution zeroed, ready for the coarse cycle.
679 void seedCoarse(Level& C)
680 {
681 if (amrexFree_)
682 {
683 if (C.agglomerated)
684 {
685 gmgCopyKokkos<T>(*C.rhs, *C.xferRhs, C.xferIn);
686 }
687 gmgZeroKokkos<T>(*C.sol);
688 return;
689 }
690 Backend::beforeAmrexRead(); // residRestrict just wrote xferRhs
691 if (C.agglomerated)
692 {
693 C.rhs->ParallelCopy(*C.xferRhs, 0, 0, 1);
694 }
695 C.sol->setVal(T(0));
696 Backend::afterAmrexWrite();
697 }

699 // The coarse correction back on the fine layout; nothing to do unless agglomerated.
700 void gatherCoarseCorrection(Level& C)
701 {
702 if (!C.agglomerated)
703 {
704 return;
705 }
706 if (amrexFree_)
707 {
708 gmgCopyKokkos<T>(*C.xferSol, *C.sol, C.xferOut);
709 return;
710 }
711 Backend::beforeAmrexRead(); // the coarse cycle just wrote C.sol
712 C.xferSol->ParallelCopy(*C.sol, 0, 0, 1);
713 Backend::afterAmrexWrite();
714 }
