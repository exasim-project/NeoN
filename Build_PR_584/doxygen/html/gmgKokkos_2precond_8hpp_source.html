18// The optimised Kokkos V-cycle as a Ginkgo preconditioner: a gko::LinOp whose apply hands two
19// device pointers to blockamr::KokkosGmgApply. Why it sits beside GmgPrecondT, and what it does
20// not carry: report/blockamr-gmg-notes.md#kokkos-handle.
21
22namespace blockamr::la
23{
24
25// V is the value type of the Krylov vectors -- double normally, float inside the
26// mixed-precision refinement. Independent of the HIERARCHY's type; KokkosGmgApply converts.
27template<class V>
28class GmgKokkosPrecondT : public AmrexLinOpBase<GmgKokkosPrecondT<V>, V>
29{
30public:
31
32 // Required by Ginkgo's polymorphic-object machinery (create_default / clear).
33 explicit GmgKokkosPrecondT(std::shared_ptr<const gko::Executor> exec)
34 : AmrexLinOpBase<GmgKokkosPrecondT<V>, V>(exec)
35 {}
36
37 GmgKokkosPrecondT(
38 std::shared_ptr<const gko::Executor> exec,
39 gko::size_type n,
40 std::shared_ptr<blockamr::KokkosGmgApply> vcycle
41 )
42 : AmrexLinOpBase<GmgKokkosPrecondT<V>, V>(exec, gko::dim<2> {n, n}),
43 vcycle_(std::move(vcycle))
44 {
45 if (exec->get_master().get() == exec.get())
46 {
47 throw std::runtime_error(
48 "GmgKokkosPrecond: the Kokkos V-cycle is a device path; use executor='cuda'"
49 );
50 }
51 }
52
53 [[nodiscard]] int nlevels() const { return vcycle_ ? vcycle_->nlevels() : 0; }
54
55protected:
56
57 // Keeps the base's advanced apply_impl(alpha, b, beta, x) visible in this scope.
58 using AmrexLinOpBase<GmgKokkosPrecondT<V>, V>::apply_impl;
59
60 void apply_impl(const gko::LinOp* b, gko::LinOp* x) const override
61 {
62 prof::Timer tAll("gmgk.apply");
63 {
64 prof::Timer t("gmgk.sync_gko");
65 this->get_executor()->synchronize(); // b written by Ginkgo
66 }
67 prof::Timer t("gmgk.vcycle");
68 // shared_ptr<T> in a const method still yields a non-const T*, so no cast is needed.
69 vcycle_->apply(localValues<V>(b), localValues<V>(x));
70 }
71
72private:
73
74 // shared_ptr like AmrexLinOpBase::scratch_: Ginkgo copy-assigns these operators.
75 std::shared_ptr<blockamr::KokkosGmgApply> vcycle_;
76};
