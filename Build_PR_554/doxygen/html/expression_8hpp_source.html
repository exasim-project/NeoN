65template<typename ValueType, typename IndexType = localIdx>
66class SetReference : public PostAssemblyBase<ValueType, IndexType>
67{
68public:
69
70 SetReference(localIdx refCell, ValueType refValue) : refCell_(refCell), refValue_(refValue) {}
71
72 void operator()(la::LinearSystem<ValueType, ValueType, la::CSRMatrix<ValueType, IndexType>>& ls
73 ) const override
74 {
75#ifdef NF_WITH_MPI_SUPPORT
76 // For distributed systems, only the rank owning refCell applies the constraint.
77 // For non-distributed systems (each rank holds a full copy), every rank applies it.
78 if (!ls.commPattern().sendCounts.empty())
79 {
80 mpi::Environment mpiEnv;
81 if (mpiEnv.isInitialized() && mpiEnv.rank() != 0) return;
82 }
83#endif
84 auto lsView = ls.view();
85 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
86 auto refVal = refValue_;
87 auto refCell = refCell_;
88 parallelFor(
89 ls.exec(),
90 {refCell, refCell + 1},
91 NEON_LAMBDA(const localIdx celli) {
92 auto dIdx = ma.diagIdx(celli);
93 auto diagVal = lsView.matrix.values[dIdx];
94 lsView.rhs[celli] += diagVal * refVal;
95 lsView.matrix.values[dIdx] += diagVal;
96 },
97 "SetReference"
98 );
99 }

257 /*@brief subtract explicit source terms from the linear system rhs, scaled by cell volumes */
258 template<typename AssemblyType = ValueType>
259 void assembleExplicitSource(
260 la::LinearSystem<AssemblyType, ValueType>& ls, const UnstructuredMesh& mesh
261 ) const
262 {
263 auto expTmp = explicitOperation(static_cast<localIdx>(mesh.nCells()));
264 auto [vol, expSource, rhs] = views(mesh.cellVolumes(), expTmp, ls.rhs());
265 parallelFor(
266 ls.exec(),
267 {0, static_cast<localIdx>(rhs.size())},
268 NEON_LAMBDA(const localIdx i) { rhs[i] -= expSource[i] * vol[i]; }
269 );
270 }

307 /* @brief assemble into a given linear system (implicit operators only, no explicit sources)
308 *
309 * @param ps post-assembly functors applied to the system after assembly
310 */
311 template<typename AssemblyType = ValueType>
312 void assemble(
313 scalar t,
314 scalar dt,
315 la::LinearSystem<AssemblyType, ValueType>& ls,
316 std::vector<const PostAssemblyBase<ValueType, IndexType>*> ps = {}
317 ) const
318 {
319 assembleSpatialOperator(ls); // add spatial operator
320 assembleTemporalOperator(ls, t, dt); // add temporal operators
321
322 // Post-assembly functors apply on the same-type form via operator(); the segregated
323 // scalar-matrix / ValueType-rhs form dispatches to applyScalarMatrix instead.
324 if constexpr (std::is_same_v<AssemblyType, ValueType>)
325 {
326 for (const auto* p : ps)
327 {
328 (*p)(ls);
329 }
330 }
331 else if constexpr (std::is_same_v<AssemblyType, scalar>)
332 {
333 for (const auto* p : ps)
334 {
335 p->applyScalarMatrix(ls);
336 }
337 }
338 }

360 template<typename OperatorType, Operator::Type Type>
361 bool hasOperatorOfType(const std::string& name) const
362 {
363 auto opType = Type;
364 auto matchNameAndType = [name, opType](const auto& op)
365 { return op.getName() == name && op.getType() == opType; };
366 if constexpr (std::is_same_v<OperatorType, SpatialOperator<ValueType>>)
367 {
368 return std::ranges::any_of(spatialOperators_, matchNameAndType);
369 }
370 else if constexpr (std::is_same_v<OperatorType, TemporalOperator<ValueType>>)
371 {
372 return std::ranges::any_of(temporalOperators_, matchNameAndType);
373 }
374 return false;
375 }
376
378 template<Operator::Type Type>
379 bool hasOperator(const std::string& name) const
380 {
381 return hasOperatorOfType<SpatialOperator<ValueType>, Type>(name)
382 || hasOperatorOfType<TemporalOperator<ValueType>, Type>(name);
383 }

386 template<typename OperatorType, Operator::Type Type>
387 OperatorType& getOperator(const std::string& name)
388 {
389 if (!hasOperatorOfType<OperatorType, Type>(name))
390 {
391 throw std::runtime_error {"No operator with given name and type found"};
392 }
393 auto opType = Type;
394 auto matchNameAndType = [name, opType](const auto& op)
395 { return op.getName() == name && op.getType() == opType; };
396 if constexpr (std::is_same_v<OperatorType, SpatialOperator<ValueType>>)
397 {
398 return *std::ranges::find_if(spatialOperators_, matchNameAndType);
399 }
400 else if constexpr (std::is_same_v<OperatorType, TemporalOperator<ValueType>>)
401 {
402 return *std::ranges::find_if(temporalOperators_, matchNameAndType);
403 }
404 throw std::runtime_error {"Unknown operator type"};
405 // should never be reached, shut up compiler warning
406 return spatialOperators_[0];
407 }

410 template<Operator::Type Type>
411 void dropOperator(const std::string& name)
412 {
413 if (!hasOperator<Type>(name))
414 {
415 throw std::runtime_error {"No operator with given name and type found"};
416 }
417 auto opType = Type;
418 auto matchNameAndType = [name, opType](const auto& op)
419 { return op.getName() == name && op.getType() == opType; };
420 if (hasOperatorOfType<SpatialOperator<ValueType>, Type>(name))
421 {
422 std::erase_if(spatialOperators_, matchNameAndType);
423 }
424 else
425 {
426 std::erase_if(temporalOperators_, matchNameAndType);
427 }
428 }
