191// A boolean switch read from an OpenFOAM dictionary is tokenized as a word and stored as a
192// std::string (or, for a numeric form, an int), not a bool -- reading it directly as bool throws
193// bad_any_cast. Accept any of the representations the dict parser may produce.
194inline bool readSwitch(const Dictionary& cfg, const std::string& key, bool defaultValue)
195{
196 if (!cfg.contains(key)) return defaultValue;
197 if (cfg.isType<bool>(key)) return cfg.get<bool>(key);
198 if (cfg.isType<int>(key)) return cfg.get<int>(key) != 0;
199 if (cfg.isType<std::string>(key))
200 {
201 const std::string v = cfg.get<std::string>(key);
202 return (v == "true" || v == "yes" || v == "on" || v == "1");
203 }
204 return defaultValue;
205}

207inline std::optional<L1ResidualControl> readL1ResidualControl(const Dictionary& cfg)
208{
209 const std::string flag = "l1ScaledResidual";
210 if (!cfg.contains(flag))
211 {
212 return std::nullopt;
213 }
214 // A boolean read from a dictionary file is stored as a word/string, not a bool;
215 // accept the common representations rather than assuming a single type.
216 bool enabled = false;
217 if (cfg.isType<bool>(flag))
218 {
219 enabled = cfg.get<bool>(flag);
220 }
221 else if (cfg.isType<int>(flag))
222 {
223 enabled = cfg.get<int>(flag) != 0;
224 }
225 else if (cfg.isType<std::string>(flag))
226 {
227 const std::string v = cfg.get<std::string>(flag);
228 enabled = (v == "true" || v == "yes" || v == "on" || v == "1");
229 }
230 if (!enabled)
231 {
232 return std::nullopt;
233 }
234
235 L1ResidualControl control {0.0, 0.0, 1000, 0, 1};
236
237 // criteria entries may be stored as int, label or scalar depending on source
238 auto readScalar = [](const Dictionary& d, const std::string& key, scalar fallback)
239 {
240 if (!d.contains(key)) return fallback;
241 if (d.isType<int>(key)) return scalar(d.get<int>(key));
242 if (d.isType<label>(key)) return scalar(d.get<label>(key));
243 return d.get<scalar>(key);
244 };
245 auto readInt = [](const Dictionary& d, const std::string& key, localIdx fallback)
246 {
247 if (!d.contains(key)) return fallback;
248 if (d.isType<int>(key)) return localIdx(d.get<int>(key));
249 if (d.isType<scalar>(key)) return localIdx(d.get<scalar>(key));
250 return d.get<localIdx>(key);
251 };
252
253 // OpenFOAM-style top-level keys (the configFile solver path keeps them: there is no
254 // mapped "criteria" subdict there). Read first so they seed the control...
255 control.tolerance = readScalar(cfg, "tolerance", control.tolerance);
256 control.relTol = readScalar(cfg, "relTol", control.relTol);
257 control.maxIter = readInt(cfg, "maxIter", control.maxIter);
258
259 // ...then let a mapped "criteria" subdict (Ginkgo-style keys) override when present,
260 // preserving the existing behaviour for fvSolution-mapped solvers.
261 if (cfg.contains("criteria"))
262 {
263 const Dictionary& criteria = cfg.subDict("criteria");
264 control.tolerance = readScalar(criteria, "absolute_residual_norm", control.tolerance);
265 control.relTol = readScalar(criteria, "initial_residual_norm", control.relTol);
266 control.maxIter = readInt(criteria, "iteration", control.maxIter);
267 }
268 control.minIter = readInt(cfg, "minIter", control.minIter);
269 control.checkFrequency = readInt(cfg, "checkFrequency", control.checkFrequency);
270
271 return control;
272}

278inline bool pnodeReferencesString(const gko::config::pnode& node, const std::string& name)
279{
280 using tag = gko::config::pnode::tag_t;
281 switch (node.get_tag())
282 {
283 case tag::string:
284 return node.get_string() == name;
285 case tag::array:
286 for (const auto& e : node.get_array())
287 if (pnodeReferencesString(e, name)) return true;
288 return false;
289 case tag::map:
290 for (const auto& kv : node.get_map())
291 if (pnodeReferencesString(kv.second, name)) return true;
292 return false;
293 default:
294 return false;
295 }

305 GinkgoSolver(Executor exec, const Dictionary& solverConfig)
306 : Base(exec), gkoExec_(getGkoExecutor(exec)),
307 coupled_(readSwitch(solverConfig, "coupled", false)),
308 l1Control_(readL1ResidualControl(solverConfig)), config_(parse(solverConfig)),
309 localMatrixFormat_(solverConfig.get("localMatrixFormat", std::string("Csr"))),
310 cacheSolver_(readSwitch(solverConfig, "cacheSolver", false)),
311 preconditionerRebuildInterval_(
312 static_cast<localIdx>(solverConfig.get("preconditionerRebuildInterval", 0))
313 )
314 {
315 // Register NeoN's L1-scaled residual criterion in the Ginkgo config registry so a
316 // configFile can name it (l1CriterionKey) in its "criteria" array. Only needed when
317 // the L1 stop is requested (l1ScaledResidual); the factory carries the tolerances and
318 // a report sink. If the parsed config actually references it, the criterion lives
319 // INSIDE the built solver (l1InConfig_) and reports via l1Report_; otherwise the
320 // existing post-hoc attach in solve() handles the flag-only case unchanged.
321 gko::config::registry reg;
322 if (l1Control_)
323 {
324 l1CritFactory_ = makeL1CriterionFactory(gkoExec_, *l1Control_, &l1Report_);
325 reg.emplace(std::string(l1CriterionKey), l1CritFactory_);
326 l1InConfig_ = pnodeReferencesString(config_, l1CriterionKey);
327 }
328
329 // Register NeoN's MergedPgm coarseners (mergeLevels-style: merge k Pgm steps into one level
330 // via SpGEMM-composed prolongations). A configFile's `mg_level` can then name them, e.g.
331 // "mg_level": ["neon::pgmMerge2"]. Named-registry pattern, same as the L1 criterion above.
332 // pgmMerge1 = merge_levels 1 = plain Pgm (single coarsening step), but via the SAME
333 // MergedPgm code path as 2/3 -- so the cache (update_matrix_value) and distributed branches
334 // behave identically across the merge-levels sweep, unlike swapping in native gko
335 // multigrid::Pgm.
336 reg.emplace("neon::pgmMerge1", makeMergedPgmFactory<scalar>(gkoExec_, 1));
337 reg.emplace("neon::pgmMerge2", makeMergedPgmFactory<scalar>(gkoExec_, 2));
338 reg.emplace("neon::pgmMerge3", makeMergedPgmFactory<scalar>(gkoExec_, 3));
339 reg.emplace("neon::pgmMerge4", makeMergedPgmFactory<scalar>(gkoExec_, 4));
340
341 // Mixed-precision: parse the user's solver config in lower precision and wrap it in an
342 // outer fp64 Ir (Iterative Refinement) solver. The outer IR maintains the solution and
343 // residual in fp64; the inner correction solve runs in the reduced precision, which is
344 // faster on GPU. irIterations controls how many outer refinement steps are taken (1 is
345 // sufficient for most CFD use cases where per-step accuracy requirements are modest).
346 // Incompatible with l1ScaledResidual-in-config because l1 criterion is fp64-typed and
347 // cannot be embedded in the lower-precision inner factory.
348 const auto innerPrecision = solverConfig.get("innerPrecision", std::string(""));
349 if (!innerPrecision.empty())
350 {
351 if (l1InConfig_)
352 NF_ERROR_EXIT("innerPrecision cannot be combined with l1ScaledResidual in config");
353 const auto irIterations =
354 static_cast<gko::size_type>(solverConfig.get("irIterations", 1));
355 auto buildIR = [&](gko::config::type_descriptor innerTypeDesc)
356 {
357 auto innerFactory = gko::config::parse(config_, reg, innerTypeDesc).on(gkoExec_);
358 factory_ =
359 gko::solver::Ir<scalar>::build()
360 .with_solver(std::move(innerFactory))
361 .with_criteria(gko::stop::Iteration::build().with_max_iters(irIterations))
362 .on(gkoExec_);
363 };
364 if (innerPrecision == "float") buildIR(gko::config::make_type_descriptor<float>());
365#if GINKGO_ENABLE_BFLOAT16
366 else if (innerPrecision == "bfloat16")
367 buildIR(gko::config::make_type_descriptor<gko::bfloat16>());
368#endif
369 else
370 {
371 const char* validOptions = "\"float\""
372#if GINKGO_ENABLE_BFLOAT16
373 " or \"bfloat16\""
374#endif
375 ;
376 NF_ERROR_EXIT(
377 "Unknown innerPrecision '" << innerPrecision << "': use " << validOptions
378 );
379 }
380 }
381 else
382 {
383 factory_ = gko::config::parse(config_, reg, gko::config::make_type_descriptor<scalar>())
384 .on(gkoExec_);
385 }
386 }

447 // docs/plans/ginkgo-solver-reuse-and-shared-allocator.md in NeoFOAM). When enabled the
448 // generated solver is cached and, on later solves with unchanged matrix structure, refreshed in
449 // place via gko::UpdateMatrixValue (the multigrid Pgm aggregation + smoother setup are reused)
450 // instead of regenerating the whole hierarchy. Indexed by cache slot: [0] for scalar /
451 // coupled-Vec3-rhs systems. Structure key = {nRows, localNnz, nonLocalNnz}; a mismatch drops
452 // the cache and regenerates. Wired for the scalar distributed (pressure) solveDist only; the

456 // cacheSolver_: opt-in via the "cacheSolver" dict entry (default off -> regenerate every solve,

458 // dict entry; when > 0 the cached solver is fully regenerated (preconditioner rebuilt from
459 // scratch) every Nth solve instead of updated in place, so Pgm aggregation drift is bounded;
460 // 0 updates in place indefinitely. cachedSolveCount_: solves served by the current cached
461 // solver per slot, reset on each (re)generate.

464 mutable std::array<std::shared_ptr<gko::LinOp>, 3> cachedSolver_;
465 mutable std::array<std::array<gko::size_type, 3>, 3> cachedSolverStructure_ {};
466 mutable std::array<localIdx, 3> cachedSolveCount_ {};

470 cachedImap_;
471 mutable std::shared_ptr<gko::matrix::Coo<scalar, localIdx>> cachedNonLocalMtx_;
472#endif
