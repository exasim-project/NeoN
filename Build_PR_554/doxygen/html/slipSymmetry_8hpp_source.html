15// Shared implementation behind the `slip` and `symmetry` boundary conditions: they apply the same
16// operator and differ only in their registered name and in where they may be applied (slip on a
17// wall/regular patch, symmetry on a symmetry-plane patch). The operator is, per face:
18// scalar => zero-gradient
19// vector => tangential projection of the boundary value (the normal component is removed), plus a
20// normal-damping surface-normal gradient that drives the normal component of the cell
21// value towards zero.
22//
23// The normal damping can be realised two ways, selected by NormalDamping:
24// - Deferred: written into refGrad as -deltaCoeffs*(v·n)*n, so it enters the per-component RHS
25// through the existing fixed-gradient assembly. Keeps the shared scalar matrix + multi-RHS
26// solve, at the cost of lagging the normal coupling by one outer iteration.
27// - Implicit: refGrad is left zero here; the BoundaryAttributes::transformImplicit flag signals
28// the Laplacian assembly to add a per-component diagonal correction (γ|S|·deltaCoeffs·|n_c|)
29// instead, which the solver applies column-by-column. Fully implicit, single matrix, but the
30// solve runs segregated rather than multi-RHS.
31//
32// NOTE: a future Tensor specialization (e.g. for a transported Reynolds-stress field) must use the
33// full reflective transform (T + H·T·H)/2 with H = I - 2 n⊗n, NOT a per-component projection.
34namespace NeoN::finiteVolume::cellCentred::volumeBoundary::detail

40 Implicit

49inline bool readTransformImplicit(const Dictionary& dict)
50{
51 const std::string key = "implicit";
52 if (!dict.contains(key)) return false;
53 if (dict.isType<bool>(key)) return dict.get<bool>(key);
54 if (dict.isType<int>(key)) return dict.get<int>(key) != 0;
55 if (dict.isType<std::string>(key))
56 {
57 const std::string v = dict.get<std::string>(key);
58 return (v == "true" || v == "yes" || v == "on" || v == "1");
59 }
60 return false;
61}

78// --- Scalar specialization: zero-gradient (mode is irrelevant; no normal component) ---
79template<>
80inline void setSlipSymmetryValue<NeoN::scalar>(
81 Field<NeoN::scalar>& domainVector,
82 const UnstructuredMesh& mesh,
83 std::pair<localIdx, localIdx> range,
84 [[maybe_unused]] NormalDamping mode
85)
86{
87 const auto internalV = domainVector.internalVector().view();
88
89 auto [refGradV, valueV, valueFractionV, refValueV, boundaryFaceOwnersV] = views(
90 domainVector.boundaryData().refGrad(),
91 domainVector.boundaryData().value(),
92 domainVector.boundaryData().valueFraction(),
93 domainVector.boundaryData().refValue(),
94 mesh.boundaryMesh().faceOwners()
95 );
96
97 NeoN::parallelFor(
98 domainVector.exec(),
99 range,
100 NEON_LAMBDA(const localIdx i) {
101 const localIdx owner = boundaryFaceOwnersV[i];
102 const auto v = internalV[owner];
103
104 refValueV[i] = v;
105 valueV[i] = v;
106 valueFractionV[i] = 0.0;
107 refGradV[i] = 0.0;
108 },
109 "setSlipSymmetryValue(scalar)"
110 );
111}

113// --- Vec3 specialization: tangential projection + normal damping ---
114template<>
115inline void setSlipSymmetryValue<NeoN::Vec3>(
116 Field<NeoN::Vec3>& domainVector,
117 const UnstructuredMesh& mesh,
118 std::pair<localIdx, localIdx> range,
119 NormalDamping mode
120)
121{
122 const auto internalV = domainVector.internalVector().view();
123 const bool deferred = (mode == NormalDamping::Deferred);
124
125 auto
126 [refGradV,
127 valueV,
128 valueFractionV,
129 refValueV,
130 boundaryFaceOwnersV,
131 faceUnitNormalsV,
132 deltaCoeffsV] =
133 views(
134 domainVector.boundaryData().refGrad(),
135 domainVector.boundaryData().value(),
136 domainVector.boundaryData().valueFraction(),
137 domainVector.boundaryData().refValue(),
138 mesh.boundaryMesh().faceOwners(),
139 mesh.boundaryMesh().faceUnitNormals(),
140 mesh.boundaryMesh().deltaCoeffs()
141 );
142
143 NeoN::parallelFor(
144 domainVector.exec(),
145 range,
146 NEON_LAMBDA(const localIdx i) {
147 const localIdx owner = boundaryFaceOwnersV[i];
148 const auto v = internalV[owner];
149 const auto n = faceUnitNormalsV[i];
150
151 const auto un = (v & n); // normal component (scalar)
152 const auto vtan = v - n * un; // tangential projection (remove normal component)
153
154 // Explicit boundary value: used by convection / face interpolation (no penetration).
155 refValueV[i] = vtan;
156 valueV[i] = vtan;
157
158 // Keep the diagonal contribution component-isotropic (zero here) so the shared scalar
159 // matrix and its multi-RHS solve are preserved.
160 valueFractionV[i] = 0.0;
161
162 // Deferred mode: normal damping as the surface-normal gradient -deltaCoeffs*(v·n)*n
163 // (purely normal, so the tangential components stay zero-gradient). Implicit mode:
164 // leave refGrad zero — the damping is applied as a per-component diagonal correction
165 // during assembly/solve.
166 refGradV[i] = deferred ? n * (-deltaCoeffsV[i] * un) : NeoN::zero<NeoN::Vec3>();
167 },
168 "setSlipSymmetryValue(Vec3)"
169 );
170}
