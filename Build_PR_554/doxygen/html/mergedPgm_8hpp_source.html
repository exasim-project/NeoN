84 void update_matrix_value(std::shared_ptr<const gko::LinOp> new_matrix) override
85 {
86 system_matrix_ = new_matrix;
87 // Value-only refresh with FROZEN structure, cuSPARSE-free. The merged prolongation
88 // (prolong_/restrict_) is structural and stays fixed; only the coarse operator's values
89 // change. Rather than recompute A_merged = R A P by SpGEMM (cuSPARSE SpGEMM aborts with
90 // CUSPARSE_STATUS_INSUFFICIENT_RESOURCES on these large injection operands), walk the
91 // retained Pgm chain: each Pgm::update_matrix_value scatters the new fine values into its
92 // frozen coarse structure (no SpGEMM) and refreshes get_coarse_op() in place; that refreshed
93 // coarse feeds the next level's fine op. After the last level, get_coarse_op() is the
94 // refreshed A_merged. This keeps the cached-solver (update_matrix_value) reuse engaged.
95 const bool isDist = is_distributed(new_matrix);
96 // local path converts to Csr; distributed path feeds the distributed op through unchanged
97 // (Pgm::update_matrix_value has its own distributed branch).
98 std::shared_ptr<const gko::LinOp> fineOp = isDist ? new_matrix : as_csr(new_matrix);
99 for (auto& lvl : levels_)
100 {
101 gko::as<gko::UpdateMatrixValue>(lvl.get())->update_matrix_value(fineOp);
102 fineOp = gko::as<gko::multigrid::MultigridLevel>(lvl.get())->get_coarse_op();
103 }
104 std::shared_ptr<const gko::LinOp> coarse =
105 gko::as<gko::multigrid::MultigridLevel>(levels_.back().get())->get_coarse_op();
106 if (!isDist) coarse = as_csr(coarse);
107 this->set_multigrid_level(prolong_, coarse, restrict_);
108 }

148 // ---- localized / Schwarz path: fine op is a plain Csr on this rank -----------------------------
149 void generateLocal()
150 {
151 auto exec = this->get_executor();
152 auto A = as_csr(system_matrix_);
153 const auto fineRows = A->get_size()[0];
154
155 // Run Pgm `merge_levels` times, composing the per-level aggregations BY INDEX. Each inner
156 // Pgm coarsens `coarse` and yields a piecewise-constant prolong (a RowGatherer: fine row ->
157 // aggregate) plus the next coarse operator. Because the prolongs are injections, the merged
158 // prolongation is the injection of the composed aggregate map `mergedAgg[i] =
159 // aggThis[mergedAgg[i]]`
160 // -- no Csr*Csr SpGEMM (cuSPARSE aborts with INSUFFICIENT_RESOURCES on large operands). The
161 // Pgm levels are retained so update_matrix_value can refresh the coarse values in place.
162 levels_.clear();
163 std::shared_ptr<const csr> coarse = A;
164 std::vector<IndexType> mergedAgg; // fine row -> current coarse index (host)
165 gko::size_type coarseRows = 0;
166 for (unsigned i = 0; i < parameters_.merge_levels; ++i)
167 {
168 auto level = makePgm()->generate(coarse);
169 auto aggThis = gather_agg(level.get()); // host, length coarse->rows, in [0, coarseRows)
170 coarseRows = level->get_prolong_op()->get_size()[1];
171 composeAgg(mergedAgg, aggThis, fineRows, i);
172 coarse = as_csr(level->get_coarse_op());
173 levels_.push_back(gko::share(std::move(level)));
174 }
175
176 auto pCsr = make_injection(mergedAgg, fineRows, coarseRows); // Csr, fine x coarse
177 prolong_ = pCsr;
178 restrict_ = gko::share(gko::as<csr>(pCsr->transpose())); // Csr, coarse x fine
179 // coarse == P_merged^T A P_merged already (last Pgm level's coarse op); use it directly.
180 this->set_multigrid_level(prolong_, coarse, restrict_);
181 }

183#ifdef NF_WITH_MPI_SUPPORT
184 // ---- global path: fine op is a distributed::Matrix ---------------------------------------------
185 void generateDistributed()
186 {
187 auto exec = this->get_executor();
188 auto distFine = gko::as<const dist_mtx>(system_matrix_);
189 auto comm = gko::as<const gko::experimental::distributed::DistributedBase>(system_matrix_)
190 ->get_communicator();
191 const auto fineGlobalRows = system_matrix_->get_size()[0];
192 // The prolong composition is rank-local because the distributed Pgm prolong is
193 // block-diagonal; all sizes below are LOCAL (this rank's diag block).
194 const auto localFineRows = distFine->get_diag_matrix()->get_size()[0];
195
196 levels_.clear();
197 std::shared_ptr<const gko::LinOp> coarse = system_matrix_; // stays distributed
198 std::vector<IndexType> mergedAgg; // LOCAL fine row -> LOCAL coarse idx
199 gko::size_type localCoarseRows = 0;
200 for (unsigned i = 0; i < parameters_.merge_levels; ++i)
201 {
202 auto level = makePgm()->generate(coarse); // distributed Pgm level
203 // Aggregation lives in the block-diagonal prolong's DIAG block (a RowGatherer).
204 auto prolongDiag = gko::as<const dist_mtx>(level->get_prolong_op())->get_diag_matrix();
205 auto aggThis = agg_from_rowgatherer(prolongDiag.get()); // host, length = local fine rows
206 // local coarse rows == this level's coarse DIAG block rows
207 localCoarseRows =
208 gko::as<const dist_mtx>(level->get_coarse_op())->get_diag_matrix()->get_size()[0];
209 composeAgg(mergedAgg, aggThis, localFineRows, i);
210 coarse = level->get_coarse_op(); // distributed coarse op -> next Pgm fine op
211 levels_.push_back(gko::share(std::move(level)));
212 }
213 const auto coarseGlobalRows = coarse->get_size()[0];
214
215 // Build the merged local injection Csr, then re-wrap prolong/restrict as block-diagonal
216 // distributed matrices (diag block only, empty off-diagonal) -- mirrors pgm.cpp's
217 // distributed_setup, which builds prolong/restrict from the local block alone.
218 std::shared_ptr<csr> pLocal = make_injection(mergedAgg, localFineRows, localCoarseRows);
219 std::shared_ptr<gko::LinOp> rLocal = gko::share(gko::as<csr>(pLocal->transpose()));
220 prolong_ = gko::share(
221 dist_mtx::create(exec, comm, gko::dim<2> {fineGlobalRows, coarseGlobalRows}, pLocal)
222 );
223 restrict_ = gko::share(
224 dist_mtx::create(exec, comm, gko::dim<2> {coarseGlobalRows, fineGlobalRows}, rLocal)
225 );
226 // coarse == the last inner Pgm's distributed coarse op == A_merged; use it directly.
227 this->set_multigrid_level(prolong_, coarse, restrict_);
228 }
229#else
230 void generateDistributed() { GKO_NOT_IMPLEMENTED; }
231#endif

246 std::unique_ptr<typename pgm::Factory> makePgm() const
247 {
248 return pgm::build()
249 .with_max_iterations(parameters_.max_iterations)
250 .with_max_unassigned_ratio(parameters_.max_unassigned_ratio)
251 .with_deterministic(parameters_.deterministic)
252 .with_skip_sorting(parameters_.skip_sorting)
253 .on(this->get_executor());
254 }
255
256 // mergedAgg[k] <- aggThis[mergedAgg[k]] (or aggThis directly on the first level). Identical for
257 // the local and distributed paths -- the distributed prolong is block-diagonal so its diag-block
258 // aggregation composes rank-locally, exactly like a local Csr aggregation.
259 static void composeAgg(
260 std::vector<IndexType>& mergedAgg,
261 const gko::array<IndexType>& aggThis,
262 gko::size_type fineRows,
263 unsigned i
264 )
265 {
266 if (i == 0)
267 {
268 mergedAgg.assign(aggThis.get_const_data(), aggThis.get_const_data() + fineRows);
269 }
270 else
271 {
272 for (gko::size_type k = 0; k < fineRows; ++k)
273 {
274 mergedAgg[k] = aggThis.get_const_data()[mergedAgg[k]];
275 }
276 }
277 }

279 static std::shared_ptr<const csr> as_csr(std::shared_ptr<const gko::LinOp> op)
280 {
281 if (auto c = std::dynamic_pointer_cast<const csr>(op)) return c;
282 auto exec = op->get_executor();
283 auto out = csr::create(exec);
284 gko::as<const gko::ConvertibleTo<csr>>(op.get())->convert_to(out);
285 return std::move(out);
286 }

288 // Pull a RowGatherer's per-row aggregate map (fine row i -> its aggregate) to the host. Length ==
289 // the gatherer's row count; values in [0, coarseRows). One entry per row.
290 gko::array<IndexType> agg_from_rowgatherer(const gko::LinOp* rgOp) const
291 {
292 auto exec = this->get_executor();
293 auto host = exec->get_master();
294 auto rg = gko::as<const row_gatherer>(rgOp);
295 auto fineRows = rg->get_size()[0];
296 gko::array<IndexType> agg(host, fineRows);
297 host->copy_from(exec, fineRows, rg->get_const_row_idxs(), agg.get_data());
298 return agg;
299 }

301 // Localized-path convenience: the level's prolong IS a RowGatherer.
302 gko::array<IndexType> gather_agg(const gko::multigrid::MultigridLevel* level) const
303 {
304 return agg_from_rowgatherer(level->get_prolong_op().get());
305 }
306
307 // Build the merged injection Csr P (P[i, mergedAgg[i]] = 1) from the composed aggregate map on
308 // the host, then clone to the device executor. row_ptrs = iota (one entry per row); values =
309 // ones. One entry per row => naturally sorted. Returned non-const so it can serve as the
310 // (mutable) diag block of a distributed::Matrix.
311 std::shared_ptr<csr> make_injection(
312 const std::vector<IndexType>& mergedAgg, gko::size_type fineRows, gko::size_type coarseRows
313 ) const
314 {
315 auto exec = this->get_executor();
316 auto host = exec->get_master();
317 gko::array<IndexType> cols(host, fineRows);
318 std::copy(mergedAgg.begin(), mergedAgg.end(), cols.get_data());
319 gko::array<IndexType> rowPtrs(host, fineRows + 1);
320 std::iota(rowPtrs.get_data(), rowPtrs.get_data() + fineRows + 1, IndexType {0});
321 gko::array<ValueType> vals(host, fineRows);
322 std::fill_n(vals.get_data(), fineRows, gko::one<ValueType>());
323
324 auto pHost = csr::create(
325 host,
326 gko::dim<2> {fineRows, coarseRows},
327 std::move(vals),
328 std::move(cols),
329 std::move(rowPtrs)
330 );
331 return gko::share(gko::clone(exec, pHost));
332 }

334 // NB: `parameters_` is provided by GKO_ENABLE_LIN_OP_FACTORY; do not redeclare it.
335 std::shared_ptr<const gko::LinOp> system_matrix_;
336 // Merged prolong/restrict: a Csr in the local path, a block-diagonal distributed::Matrix in the
337 // global path. Structural (composed once); reused across update_matrix_value refreshes.
338 std::shared_ptr<const gko::LinOp> prolong_;
339 std::shared_ptr<const gko::LinOp> restrict_;
340 // Retained inner Pgm chain (each a MultigridLevel + UpdateMatrixValue): lets
341 // update_matrix_value refresh the merged coarse operator's values in place with frozen
342 // aggregation, cuSPARSE-free.
343 std::vector<std::shared_ptr<gko::LinOp>> levels_;

355inline std::shared_ptr<typename MergedPgm<ValueType, IndexType>::Factory> makeMergedPgmFactory(

361 .with_deterministic(deterministic)
362 .on(std::move(exec));
