29// Shared kernel behind FlowRateInletVelocity, mirroring OpenFOAM's
30// flowRateInletVelocityFvPatchVectorField::updateValues(rho). With n the unit outward normal of a
31// face, magSf its area and rho the density of the face (1 for a volumetric flow rate):
32//
33// uniform profile (default):
34// avgU = -flowRate / sum(rho * magSf) (sum over the whole patch, all ranks)
35// value = avgU * n
36//
37// extrapolateProfile:
38// Up = ownerValue; nUp = n & Up; Up -= nUp * n (keep the tangential part)
39// nUp = min(nUp, 0) (remove reverse flow)
40// estimated = -sum(rho * magSf * nUp)
41// if (estimated > 0.5 * flowRate) nUp *= |flowRate| / |estimated|
42// else nUp -= (flowRate - estimated) / sum(rho * magSf)
43// value = Up + nUp * n
44//
45// The patch is a Dirichlet patch (valueFraction = 1, refGrad = 0). When the density the flow rate
46// needs is not available (see FlowRateInletVelocity) the patch degrades to zero-gradient, the same
47// fallback every context-aware NeoN BC uses.
48inline void setFlowRateInletVelocity(
49 Field<NeoN::Vec3>& domainVector,
50 const UnstructuredMesh& mesh,
51 std::pair<localIdx, localIdx> range,
52 NeoN::scalar flowRate,
53 View<const NeoN::scalar> boundaryRho,
54 NeoN::scalar constantRho,
55 bool useBoundaryRho,
56 bool extrapolateProfile
57)
58{
59 const auto internalV = domainVector.internalVector().view();
60 auto [faceNormalsV, faceAreasV, boundaryFaceOwnersV] = views(
61 mesh.boundaryMesh().faceNormals(),
62 mesh.boundaryMesh().faceAreas(),
63 mesh.boundaryMesh().faceOwners()
64 );
65
66 // density of face i: (useBoundaryRho ? boundaryRho[i] : constantRho). Written out in every
67 // kernel instead of a shared lambda: device lambdas must not be nested.
68 const NeoN::scalar rhoAreaSum = patchSum(
69 mesh,
70 range,
71 NEON_LAMBDA(const localIdx i, NeoN::scalar& sum) {
72 sum += (useBoundaryRho ? boundaryRho[i] : constantRho) * faceAreasV[i];
73 }
74 );
75
76 auto [refGradV, valueV, valueFractionV, refValueV] = views(
77 domainVector.boundaryData().refGrad(),
78 domainVector.boundaryData().value(),
79 domainVector.boundaryData().valueFraction(),
80 domainVector.boundaryData().refValue()
81 );
82
83 if (!extrapolateProfile)
84 {
85 const NeoN::scalar avgU = -flowRate / rhoAreaSum;
86 NeoN::parallelFor(
87 domainVector.exec(),
88 range,
89 NEON_LAMBDA(const localIdx i) {
90 const NeoN::Vec3 n = faceNormalsV[i] / faceAreasV[i];
91 const NeoN::Vec3 newValue = avgU * n;
92 refGradV[i] = NeoN::zero<NeoN::Vec3>();
93 refValueV[i] = newValue;
94 valueFractionV[i] = NeoN::scalar(1);
95 valueV[i] = newValue;
96 },
97 "setFlowRateInletVelocity"
98 );
99 return;
100 }
101
102 // flow rate carried by the inflow part of the patch normal velocity of the adjacent cells
103 const NeoN::scalar estimatedFlowRate = -patchSum(
104 mesh,
105 range,
106 NEON_LAMBDA(const localIdx i, NeoN::scalar& sum) {
107 const NeoN::Vec3 n = faceNormalsV[i] / faceAreasV[i];
108 const NeoN::scalar nUp = n & internalV[boundaryFaceOwnersV[i]];
109 const NeoN::scalar inflowNUp = nUp < NeoN::scalar(0) ? nUp : NeoN::scalar(0);
110 sum += (useBoundaryRho ? boundaryRho[i] : constantRho) * faceAreasV[i] * inflowNUp;
111 }
112 );
113
114 const bool scale = estimatedFlowRate > 0.5 * flowRate;
115 const NeoN::scalar factor =
116 scale ? Kokkos::abs(flowRate) / Kokkos::abs(estimatedFlowRate) : NeoN::scalar(1);
117 const NeoN::scalar shift =
118 scale ? NeoN::scalar(0) : (flowRate - estimatedFlowRate) / rhoAreaSum;
119
120 NeoN::parallelFor(
121 domainVector.exec(),
122 range,
123 NEON_LAMBDA(const localIdx i) {
124 const NeoN::Vec3 n = faceNormalsV[i] / faceAreasV[i];
125 const NeoN::Vec3 ownerValue = internalV[boundaryFaceOwnersV[i]];
126 const NeoN::scalar ownerNUp = n & ownerValue;
127 const NeoN::Vec3 tangential = ownerValue - n * ownerNUp;
128 const NeoN::scalar inflowNUp = ownerNUp < NeoN::scalar(0) ? ownerNUp : NeoN::scalar(0);
129 const NeoN::scalar nUp = inflowNUp * factor - shift;
130 const NeoN::Vec3 newValue = tangential + nUp * n;
131 refGradV[i] = NeoN::zero<NeoN::Vec3>();
132 refValueV[i] = newValue;
133 valueFractionV[i] = NeoN::scalar(1);
134 valueV[i] = newValue;
135 },
136 "setFlowRateInletVelocityExtrapolated"
137 );
138}

141inline void setFlowRateInletVelocityZeroGradient(

144 std::pair<localIdx, localIdx> range

146{

154 );
155

157 domainVector.exec(),
158 range,

163 valueFractionV[i] = NeoN::scalar(0);
164 valueV[i] = ownerValue;
165 },
166 "setFlowRateInletVelocityZeroGradient"

172// FlowRateInletVelocity: inlet velocity that carries a prescribed flow rate through the patch,
173// mirroring OpenFOAM's flowRateInletVelocity. See detail::setFlowRateInletVelocity for the exact
174// formulas.
175//
176// Dictionary entries (flow rates are plain scalars; OpenFOAM's Function1 form
177// `massFlowRate constant X` has to be flattened by the case reader):
178// volumetricFlowRate volumetric flow rate in m^3/s (takes precedence), or
179// massFlowRate mass flow rate in kg/s, which needs the density, see below
180// rho name of the density field (default "rho"; "none" means density 1)
181// rhoInlet constant density, used when the density field is not available
182// extrapolateProfile keep the tangential velocity and the shape of the inflow of the adjacent
183// cells instead of a uniform profile (default false)
184//
185// For a mass flow rate the density is read from the boundary values of the field `rho` in the
186// BoundaryContext. Without such a field the constant `rhoInlet` is used. If neither exists the
187// context-aware overload throws, as OpenFOAM stops with an error; the plain overload (no context
188// at all, e.g. while the field is created) degrades to zero-gradient.
189class FlowRateInletVelocity :
190 public VolumeBoundaryFactory<NeoN::Vec3>::template Register<FlowRateInletVelocity>
191{
192 using Base = VolumeBoundaryFactory<NeoN::Vec3>::template Register<FlowRateInletVelocity>;
193
194public:
195
196 using Base::correctBoundaryCondition;
197
198 // A Dirichlet patch, like OpenFOAM's (it derives from fixedValue): assignable = false, so a
199 // pressure-velocity coupling that constrains HbyA on non-assignable velocity patches
200 // (constrainHbyA) feeds the prescribed inlet velocity into the flux, and fixesValue = true.
201 // The value is still recomputed from the flow rate on every call.
202 FlowRateInletVelocity(const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID)
203 : Base(mesh, dict, patchID, {.assignable = false, .fixesValue = true}), mesh_(mesh),
204 flowRate_(0), volumetric_(dict.contains("volumetricFlowRate")),
205 rhoName_(dict.contains("rho") ? dict.get<std::string>("rho") : std::string("rho")),
206 rhoInlet_(
207 dict.contains("rhoInlet") ? dict.get<NeoN::scalar>("rhoInlet")
208 : -std::numeric_limits<NeoN::scalar>::max()
209 ),
210 extrapolateProfile_(
211 dict.contains("extrapolateProfile") ? dict.get<bool>("extrapolateProfile") : false
212 )
213 {
214 if (volumetric_)
215 {
216 flowRate_ = dict.get<NeoN::scalar>("volumetricFlowRate");
217 }
218 else if (dict.contains("massFlowRate"))
219 {
220 flowRate_ = dict.get<NeoN::scalar>("massFlowRate");
221 }
222 else
223 {
224 NF_THROW(
225 "flowRateInletVelocity boundary condition on patch " + std::to_string(patchID)
226 + " requires either 'volumetricFlowRate' or 'massFlowRate'"
227 );
228 }
229 }
230
231 // No context => no density field: fine for a volumetric flow rate, rho "none" or a constant
232 // rhoInlet, otherwise zero-gradient.
233 virtual void correctBoundaryCondition(Field<NeoN::Vec3>& domainVector) final
234 {
235 if (volumetric_ || rhoName_ == "none")
236 {
237 applyConstantDensity(domainVector, NeoN::scalar(1));
238 }
239 else if (rhoInlet_ >= NeoN::scalar(0))
240 {
241 applyConstantDensity(domainVector, rhoInlet_);
242 }
243 else
244 {
245 detail::setFlowRateInletVelocityZeroGradient(domainVector, mesh_, this->range());
246 }
247 }
248
249 // Context-aware, in OpenFOAM's order: density 1 for a volumetric flow rate or rho "none";
250 // else the boundary values of the field `rho`, if the context has it; else the constant
251 // rhoInlet; else an error.
252 virtual void
253 correctBoundaryCondition(Field<NeoN::Vec3>& domainVector, const BoundaryContext& ctx) final
254 {
255 if (volumetric_ || rhoName_ == "none")
256 {
257 applyConstantDensity(domainVector, NeoN::scalar(1));
258 }
259 else if (ctx.hasScalar(rhoName_))
260 {
261 detail::setFlowRateInletVelocity(
262 domainVector,
263 mesh_,
264 this->range(),
265 flowRate_,
266 ctx.scalarFieldPtr(rhoName_).boundaryData().value().view(),
267 NeoN::scalar(1),
268 true,
269 extrapolateProfile_
270 );
271 }
272 else if (rhoInlet_ >= NeoN::scalar(0))
273 {
274 applyConstantDensity(domainVector, rhoInlet_);
275 }
276 else
277 {
278 NF_THROW(
279 "flowRateInletVelocity boundary condition on patch "
280 + std::to_string(this->patchID()) + ": a mass flow rate needs the density field '"
281 + rhoName_ + "' in the boundary context or a constant 'rhoInlet'"
282 );
283 }
284 }
