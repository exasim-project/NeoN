24// Shared kernel behind FixedMean, mirroring OpenFOAM's fixedMeanFvPatchField::updateCoeffs()
25// for a scalar field. The patch value is the adjacent cell value, corrected so that the
26// area-weighted mean over the whole patch (all ranks) is the prescribed meanValue:
27// meanPsi = sum(magSf * ownerValue) / sum(magSf) (gWeightedAverage)
28// if (|meanValue| > SMALL && |meanPsi| > 0.5 |meanValue|)
29// value = ownerValue * |meanValue| / |meanPsi| (scale)
30// else
31// value = ownerValue + (meanValue - meanPsi) (shift)
32// with SMALL = 1e-15, OpenFOAM's double-precision SMALL. The scaled branch uses the magnitudes
33// like OpenFOAM does, so the sign of the owner values is kept. The result is a Dirichlet patch
34// (valueFraction = 1, refGrad = 0) recomputed from the current owner values on every call, as
35// OpenFOAM does in updateCoeffs().
36inline void setFixedMeanValue(
37 Field<NeoN::scalar>& domainVector,
38 const UnstructuredMesh& mesh,
39 std::pair<localIdx, localIdx> range,
40 NeoN::scalar meanValue
41)
42{
43 constexpr NeoN::scalar small = 1.0e-15;
44
45 const auto internalV = domainVector.internalVector().view();
46 auto [faceAreasV, boundaryFaceOwnersV] =
47 views(mesh.boundaryMesh().faceAreas(), mesh.boundaryMesh().faceOwners());
48
49 const NeoN::scalar areaSum = patchSum(
50 mesh, range, NEON_LAMBDA(const localIdx i, NeoN::scalar& sum) { sum += faceAreasV[i]; }
51 );
52 const NeoN::scalar weightedSum = patchSum(
53 mesh,
54 range,
55 NEON_LAMBDA(const localIdx i, NeoN::scalar& sum) {
56 sum += faceAreasV[i] * internalV[boundaryFaceOwnersV[i]];
57 }
58 );
59
60 // gWeightedAverage: no weight, no average (a rank-less or empty patch keeps its owner values)
61 const NeoN::scalar meanPsi = areaSum > NeoN::scalar(0) ? weightedSum / areaSum : meanValue;
62
63 const bool scale =
64 Kokkos::abs(meanValue) > small && Kokkos::abs(meanPsi) > 0.5 * Kokkos::abs(meanValue);
65 const NeoN::scalar factor =
66 scale ? Kokkos::abs(meanValue) / Kokkos::abs(meanPsi) : NeoN::scalar(1);
67 const NeoN::scalar shift = scale ? NeoN::scalar(0) : meanValue - meanPsi;
68
69 auto [refGradV, valueV, valueFractionV, refValueV] = views(
70 domainVector.boundaryData().refGrad(),
71 domainVector.boundaryData().value(),
72 domainVector.boundaryData().valueFraction(),
73 domainVector.boundaryData().refValue()
74 );
75
76 NeoN::parallelFor(
77 domainVector.exec(),
78 range,
79 NEON_LAMBDA(const localIdx i) {
80 const NeoN::scalar newValue = internalV[boundaryFaceOwnersV[i]] * factor + shift;
81 refGradV[i] = NeoN::scalar(0);
82 refValueV[i] = newValue;
83 valueFractionV[i] = NeoN::scalar(1);
84 valueV[i] = newValue;
85 },
86 "setFixedMeanValue"
87 );
88}

90} // namespace detail

92// FixedMean: a Dirichlet patch whose values follow the adjacent cells, shifted or scaled so that
93// the area-weighted patch mean is meanValue (OpenFOAM's fixedMean). It is used on a pressure
94// outlet where the mean, not the profile, is prescribed. See detail::setFixedMeanValue for the
95// exact formula.
96//
97// Scalar fields only. meanValue is a plain scalar; OpenFOAM's Function1 form `meanValue constant X`
98// has to be flattened by the case reader.
99class FixedMean : public VolumeBoundaryFactory<NeoN::scalar>::template Register<FixedMean>
100{
101 using Base = VolumeBoundaryFactory<NeoN::scalar>::template Register<FixedMean>;
102
103public:
104
105 using Base::correctBoundaryCondition;
106
107 // A Dirichlet patch, like OpenFOAM's (it derives from fixedValue): assignable = false and
108 // fixesValue = true. The values are still recomputed from the owner cells on every call.
109 FixedMean(const UnstructuredMesh& mesh, const Dictionary& dict, localIdx patchID)
110 : Base(mesh, dict, patchID, {.assignable = false, .fixesValue = true}), mesh_(mesh),
111 meanValue_(0)
112 {
113 if (!dict.contains("meanValue"))
114 {
115 NF_THROW(
116 "fixedMean boundary condition on patch " + std::to_string(patchID)
117 + " requires a 'meanValue' entry"
118 );
119 }
120 meanValue_ = dict.get<NeoN::scalar>("meanValue");
121 }
122
123 virtual void correctBoundaryCondition(Field<NeoN::scalar>& domainVector) final
124 {
125 detail::setFixedMeanValue(domainVector, mesh_, this->range(), meanValue_);
126 }
127
128 static std::string name() { return "fixedMean"; }
129
130 std::string getName() const override { return name(); }
131
132 static std::string doc()
133 {
134 return "Dirichlet patch following the adjacent cell values, shifted or scaled so that the "
135 "area-weighted mean over the patch is meanValue (OpenFOAM's fixedMean).";
136 }
137
138 static std::string schema() { return "none"; }
139
140 virtual std::unique_ptr<VolumeBoundaryFactory<NeoN::scalar>> clone() const final
141 {
142 return std::make_unique<FixedMean>(*this);
143 }
144
145private:
146
147 const UnstructuredMesh& mesh_;
148 NeoN::scalar meanValue_;
149};
