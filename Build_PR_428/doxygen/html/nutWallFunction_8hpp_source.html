19static constexpr label maxIter = 10;
20static constexpr scalar tolerance = 1e-9;
21static constexpr scalar kappa = 0.41;
22static constexpr scalar E = 9.8;

24KOKKOS_INLINE_FUNCTION
25scalar computeUTau(
26 const scalar magGradU,
27 const scalar magUp,
28 const scalar y,
29 const scalar nuw,
30 const scalar nutw,
31 scalar& err
32)
33{
34 err = 0.0;
35
36 scalar ut = Kokkos::sqrt((nutw + nuw) * magGradU);
37 if (ut <= ROOTVSMALL)
38 {
39 return 0.0;
40 }
41
42 int iter = 0;
43 do
44 {
45 const scalar kUu = Kokkos::min(kappa * magUp / ut, scalar(50));
46 const scalar fkUu = Kokkos::exp(kUu) - 1.0 - kUu * (1.0 + 0.5 * kUu);
47
48 const scalar f =
49 -ut * y / nuw + magUp / ut + (1.0 / E) * (fkUu - (1.0 / 6.0) * kUu * kUu * kUu);
50
51 const scalar df = y / nuw + magUp / (ut * ut) + (1.0 / E) * kUu * fkUu / ut;
52
53 const scalar uTauNew = ut + f / df;
54 err = NeoN::mag((ut - uTauNew) / ut);
55 ut = uTauNew;
56 }
57 while (ut > ROOTVSMALL && err > tolerance && ++iter < maxIter);
58
59 return ut > 0.0 ? ut : 0.0;
60}

62inline void setNutUSpaldingWallFunction(
63 Field<scalar>& domainVector,
64 const fvcc::VolumeField<Vec3>& U,
65 const fvcc::VolumeField<scalar>& nu,
66 const UnstructuredMesh& mesh,
67 std::pair<localIdx, localIdx> range
68)
69{
70 const auto uInternal = U.internalVector().view();
71 const auto uBoundary = U.boundaryData().value().view();
72 const auto nuBoundary = nu.boundaryData().value().view();
73
74 auto [refGradient, value, valueFraction, refValue, faceCells, deltaCoeffs, delta] = views(
75 domainVector.boundaryData().refGrad(),
76 domainVector.boundaryData().value(),
77 domainVector.boundaryData().valueFraction(),
78 domainVector.boundaryData().refValue(),
79 mesh.boundaryMesh().faceCells(),
80 mesh.boundaryMesh().deltaCoeffs(),
81 mesh.boundaryMesh().delta()
82 );
83 // NeoN::Logging::info("Before Loop {}",uInternal[0][0]);
84 NeoN::parallelFor(
85 domainVector.exec(),
86 range,
87 NEON_LAMBDA(const localIdx i) {
88 const localIdx owner = faceCells[i];
89
90 const Vec3 uInt = uInternal[owner];
91 const Vec3 uWall = uBoundary[i];
92 const Vec3 diff = uWall - uInt;
93
94 const scalar magUp = NeoN::mag(diff);
95 const scalar magGradU = NeoN::mag(diff * deltaCoeffs[i]);
96 const scalar y = NeoN::mag(delta[i]);
97 const scalar nuw = nuBoundary[i];
98
99 const scalar currentNut = value[i];
100
101 scalar err = 0.0;
102 const scalar uTau = computeUTau(magGradU, magUp, y, nuw, currentNut, err);
103
104 const scalar nutCandidate = (uTau * uTau) / (magGradU + ROOTVSMALL) - nuw;
105
106 const scalar nutw = nutCandidate > 0.0 ? nutCandidate : 0.0;
107
108 refValue[i] = nutw;
109 value[i] = nutw;
110 valueFraction[i] = 1.0;
111 refGradient[i] = 0.0;
112 },
113 "setNutUSpaldingWallFunction"
114 );
115}
