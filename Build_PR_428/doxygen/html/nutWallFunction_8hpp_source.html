20static constexpr label maxIter = 10;
21static constexpr scalar tolerance = 1e-9;
22static constexpr scalar kappa = 0.41;
23static constexpr scalar E = 9.8;
24
25KOKKOS_INLINE_FUNCTION
26scalar computeUTau(
27 const scalar magGradU,
28 const scalar magUp,
29 const scalar y,
30 const scalar nuw,
31 const scalar nutw,
32 scalar& err
33)
34{
35 err = 0.0;
36
37 scalar ut = Kokkos::sqrt((nutw + nuw) * magGradU);
38 if (ut <= ROOTVSMALL)
39 {
40 return 0.0;
41 }
42
43 int iter = 0;
44 do
45 {
46 const scalar kUu = Kokkos::min(kappa * magUp / ut, scalar(50));
47 const scalar fkUu = Kokkos::exp(kUu) - 1.0 - kUu * (1.0 + 0.5 * kUu);
48
49 const scalar f =
50 -ut * y / nuw + magUp / ut + (1.0 / E) * (fkUu - (1.0 / 6.0) * kUu * kUu * kUu);
51
52 const scalar df = y / nuw + magUp / (ut * ut) + (1.0 / E) * kUu * fkUu / ut;
53
54 const scalar uTauNew = ut + f / df;
55 err = NeoN::mag((ut - uTauNew) / ut);
56 ut = uTauNew;
57 }
58 while (ut > ROOTVSMALL && err > tolerance && ++iter < maxIter);
59
60 return ut > 0.0 ? ut : 0.0;
61}

94inline void setNutUSpaldingWallFunction(
95 Field<scalar>& domainVector,
96 const fvcc::VolumeField<Vec3>& U,
97 const fvcc::VolumeField<scalar>& nu,
98 const UnstructuredMesh& mesh,
99 std::pair<localIdx, localIdx> range
100)
101{
102 const auto uInternal = U.internalVector().view();
103 const auto uBoundary = U.boundaryData().value().view();
104 const auto nuBoundary = nu.boundaryData().value().view();
105
106 auto [refGradient, value, valueFraction, refValue, faceCells, deltaCoeffs, delta] = views(
107 domainVector.boundaryData().refGrad(),
108 domainVector.boundaryData().value(),
109 domainVector.boundaryData().valueFraction(),
110 domainVector.boundaryData().refValue(),
111 mesh.boundaryMesh().faceCells(),
112 mesh.boundaryMesh().deltaCoeffs(),
113 mesh.boundaryMesh().delta()
114 );
115
116 NeoN::parallelFor(
117 domainVector.exec(),
118 range,
119 NEON_LAMBDA(const localIdx i) {
120 const localIdx owner = faceCells[i];
121
122 const Vec3 uInt = uInternal[owner];
123 const Vec3 uWall = uBoundary[i];
124 const Vec3 diff = uWall - uInt;
125
126 const scalar magUp = NeoN::mag(diff);
127 const scalar magGradU = NeoN::mag(diff * deltaCoeffs[i]);
128 const scalar y = NeoN::mag(delta[i]);
129 const scalar nuw = nuBoundary[i];
130
131 const scalar currentNut = value[i];
132
133 scalar err = 0.0;
134 const scalar uTau = computeUTau(magGradU, magUp, y, nuw, currentNut, err);
135
136 const scalar nutCandidate = (uTau * uTau) / (magGradU + ROOTVSMALL) - nuw;
137
138 const scalar nutw = nutCandidate > 0.0 ? nutCandidate : 0.0;
139
140 refValue[i] = nutw;
141 value[i] = nutw;
142 valueFraction[i] = 1.0;
143 refGradient[i] = 0.0;
144 },
145 "setNutUSpaldingWallFunction"
146 );
147}
