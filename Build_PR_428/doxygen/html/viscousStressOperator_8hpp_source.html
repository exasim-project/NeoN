176 void explicitOperation(Vector<Vec3>& source) const
177 {
178 NF_ASSERT(viscousOp_, "ViscousStressOperatorStrategy not initialized");
179 Vector<Vec3> tmpsource(source.exec(), source.size(), zero<Vec3>());
180 const auto operatorScaling = this->getCoefficient();
181 viscousOp_->explicitOp(
182 tmpsource,
183 nuF_,
184 nutF_,
185 nuTildeF_,
186 this->field_,
187 gradUx_,
188 gradUy_,
189 gradUz_,
190 operatorScaling
191 );
192 source += tmpsource;
193 }

197 void read(const Input& input)
198 {
199 const UnstructuredMesh& mesh = this->field_.mesh();
200 if (std::holds_alternative<NeoN::Dictionary>(input))
201 {
202 auto dict = std::get<NeoN::Dictionary>(input);
203 auto tokens =
204 dict.subDict("viscousStressSchemes").get<NeoN::TokenList>("viscousStress");
205 viscousOp_ = ViscousStressOperatorFactory::create(this->exec(), mesh, tokens);
206 }
207 else
208 {
209 auto tokens = std::get<NeoN::TokenList>(input);
210 viscousOp_ = ViscousStressOperatorFactory::create(this->exec(), mesh, tokens);
211 }
212 }
