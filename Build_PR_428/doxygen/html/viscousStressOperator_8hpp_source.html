54class ViscousStressOperatorFactory :
55 public NeoN::RuntimeSelectionFactory<
56 ViscousStressOperatorFactory,
57 Parameters<const Executor&, const UnstructuredMesh&, const Input&>>
58{
59public:
60
61 static std::unique_ptr<ViscousStressOperatorFactory>
62 create(const Executor& exec, const UnstructuredMesh& mesh, const Input& inputs)
63 {
64 std::string key = (std::holds_alternative<NeoN::Dictionary>(inputs))
65 ? std::get<NeoN::Dictionary>(inputs).get<std::string>(
66 "div((nuEff*dev2(T(grad(U)))))"
67 )
68 : std::get<NeoN::TokenList>(inputs).next<std::string>();
69
70 ViscousStressOperatorFactory::keyExistsOrError(key);
71 return ViscousStressOperatorFactory::table().at(key)(exec, mesh, inputs);
72 }
73
74 static std::string name() { return "ViscousStressOperatorFactory"; }
75
76 ViscousStressOperatorFactory(const Executor& exec, const UnstructuredMesh& mesh)
77 : exec_(exec), mesh_(mesh)

125 const VolumeField<scalar>& nut,
126 const TensorVecField& gradU,

130 nu_(nu), nut_(nut), gradU_(gradU),

133
134 ViscousStressOperator(
135 dsl::Operator::Type termType,
136 const VolumeField<scalar>& nu,
137 const VolumeField<scalar>& nut,
138 const TensorVecField& gradU,
139 std::unique_ptr<ViscousStressOperatorFactory> viscousOp
140 )
141 : dsl::OperatorMixin<VolumeField<Vec3>>(nut.exec(), dsl::Coeff(1.0), gradU.Tx, termType),
142 nu_(nu), nut_(nut), gradU_(gradU), viscousOp_(std::move(viscousOp))
143 {}
144
145 ViscousStressOperator(
146 dsl::Operator::Type termType,
147 const VolumeField<scalar>& nu,
148 const VolumeField<scalar>& nut,
149 const TensorVecField& gradU
150 )
151 : dsl::OperatorMixin<VolumeField<Vec3>>(nut.exec(), dsl::Coeff(1.0), gradU.Tx, termType),
152 nu_(nu), nut_(nut), gradU_(gradU), viscousOp_(nullptr)
153 {}
154
155 void explicitOperation(Vector<Vec3>& source) const
156 {
157 NF_ASSERT(viscousOp_, "ViscousStressOperatorStrategy not initialized");
158 Vector<Vec3> tmpsource(source.exec(), source.size(), zero<Vec3>());
159 const auto operatorScaling = this->getCoefficient();
160 viscousOp_->explicitOp(tmpsource, nu_, nut_, gradU_, operatorScaling);
161 source += tmpsource;
162 }
163
164 void implicitOperation(la::LinearSystem<Vec3>& ls) const;
165
166 void read(const Input& input)
167 {
168 const UnstructuredMesh& mesh = this->field_.mesh();
169 if (std::holds_alternative<NeoN::Dictionary>(input))
170 {
171 auto dict = std::get<NeoN::Dictionary>(input);
172 auto tokens =
173 dict.subDict("divSchemes").get<NeoN::TokenList>("div((nuEff*dev2(T(grad(U)))))");
174 viscousOp_ = ViscousStressOperatorFactory::create(this->exec(), mesh, tokens);
175 }
176 else
177 {
178 auto tokens = std::get<NeoN::TokenList>(input);
179 viscousOp_ = ViscousStressOperatorFactory::create(this->exec(), mesh, tokens);
180 }
181 }
182
183 std::string getName() const { return "ViscousStressOperator"; }
184
185private:
186
187 const VolumeField<scalar>& nu_;
188 const VolumeField<scalar>& nut_;
189 const TensorVecField& gradU_;
190 std::unique_ptr<ViscousStressOperatorFactory> viscousOp_;
191};
