164 template<typename T = ValueType>
165 requires std::is_same_v<T, Vec3>
166 [[nodiscard]] VolumeField<scalar>
167 componentField(const size_t componentIndex, const std::string& suffix) const
168 {
169 if (componentIndex >= 3)
170 {
171 NF_ERROR_EXIT("VolumeField<Vec3>::component index out of range.");
172 }
173
174 VolumeField<scalar> result(
175 this->exec(),
176 this->name + "." + suffix,
177 this->mesh(),
178 createCalculatedBCs<VolumeBoundary<scalar>>(this->mesh())
179 );
180
181 auto sourceInternal = this->internalVector().view();
182 parallelFor(
183 result.internalVector(),
184 NEON_LAMBDA(const localIdx i) { return sourceInternal[i][componentIndex]; }
185 );
186
187 auto sourceValue = this->boundaryData().value().view();
188 parallelFor(
189 result.boundaryData().value(),
190 NEON_LAMBDA(const localIdx i) { return sourceValue[i][componentIndex]; }
191 );
192
193 auto sourceRefValue = this->boundaryData().refValue().view();
194 parallelFor(
195 result.boundaryData().refValue(),
196 NEON_LAMBDA(const localIdx i) { return sourceRefValue[i][componentIndex]; }
197 );
198
199 auto sourceRefGrad = this->boundaryData().refGrad().view();
200 parallelFor(
201 result.boundaryData().refGrad(),
202 NEON_LAMBDA(const localIdx i) { return sourceRefGrad[i][componentIndex]; }
203 );
204
205 result.boundaryData().valueFraction() = this->boundaryData().valueFraction();
206
207 return result;
208 }
