37 static void setupUmpirePool(MemorySpace memSpace, size_t size)
38 {
39 auto& rm = umpire::ResourceManager::getInstance();
40
41 // do nothing if pool already exists
42 if (hasPool(memSpace)) return;
43
44 if (memSpace == MemorySpace::GPU)
45 {
46 auto allocator = rm.getAllocator("DEVICE");
47 auto pooledAllocator =
48 rm.makeAllocator<umpire::strategy::QuickPool>("DEVICE_POOL", allocator, size);
49 }
50 else if (memSpace == MemorySpace::CPU)
51 {
52 auto allocator = rm.getAllocator("HOST");
53 auto pooledAllocator =
54 rm.makeAllocator<umpire::strategy::QuickPool>("HOST_POOL", allocator, size);
55 }
56 }

58 static auto getUmpirePool(MemorySpace memSpace)
59 {
60 auto& rm = umpire::ResourceManager::getInstance();
61 if (memSpace == MemorySpace::GPU)
62 {
