17// CRTP base for the matrix-free Ginkgo operators here: it bundles gko::LinOp and
18// gko::EnableCreateMethod and supplies the one advanced apply_impl(alpha, b, beta, x) through
19// the derived class' apply_impl(b, x). How to derive: report/blockamr-linear-algebra-notes.md
20template<class D, class V = double>
21class AmrexLinOpBase : public gko::LinOp, public gko::EnableCreateMethod<D>
22{
23protected:
24
25 using DenseV = gko::matrix::Dense<V>;
26

29
30 AmrexLinOpBase(std::shared_ptr<const gko::Executor> exec, const gko::dim<2>& size)
31 : gko::LinOp(exec, size)
32 {}

34 // Supplied by the derived class; re-declared so the advanced overload can call it.
35 void apply_impl(const gko::LinOp* b, gko::LinOp* x) const override = 0;
36
37 // x = alpha * op(b) + beta * x. Every derived apply_impl(b, x) OVERWRITES all of x, so the
38 // intermediate is a reused scratch rather than a clone of x. The beta branches skip whole
39 // vector passes: beta == 1 needs no scale, beta == 0 discards x and needs no scratch.
40 void apply_impl(
41 const gko::LinOp* alpha, const gko::LinOp* b, const gko::LinOp* beta, gko::LinOp* x
42 ) const override
43 {
44 prof::Timer tAll("adv.apply");
45 // A view, so scale/add_scaled hit x's own memory; both are elementwise.
46 auto denseX = localView<V>(x);
47 const double alphaVal = hostScalar(alpha);
48 const double betaVal = hostScalar(beta);
49
50 if (betaVal == 0.0)
51 {
52 this->apply_impl(b, x);
53 if (alphaVal != 1.0)
54 {
55 prof::Timer t("adv.scale");
56 denseX->scale(alpha);
57 }
58 return;
59 }
60
61 const gko::dim<2> size = denseX->get_size();
62 if (!scratch_ || scratch_->get_size() != size)
63 {
64 prof::Timer t("adv.alloc");
65 scratch_ = DenseV::create(this->get_executor(), size);
66 }
67 this->apply_impl(b, scratch_.get());
68 if (betaVal != 1.0)
69 {
70 prof::Timer t("adv.scale");
71 denseX->scale(beta);
72 }
73 {
74 prof::Timer t("adv.addscaled");
75 denseX->add_scaled(alpha, scratch_);
76 }
77 }
78
79private:
80
81 // alpha/beta are 1x1 Dense on the solve executor; a device value is staged through the
82 // host master.
83 static double hostScalar(const gko::LinOp* s)
84 {
85 auto d = gko::as<DenseV>(s);
86 auto exec = d->get_executor();
87 if (exec->get_master().get() != exec.get())
88 {
89 return gko::clone(exec->get_master(), d)->at(0, 0);
90 }
91 return d->at(0, 0);
92 }
93
94 // shared_ptr, not unique_ptr: these operators stay copy-assignable like gko::LinOp, which a
95 // move-only member would prevent. No state is kept across calls.
96 mutable std::shared_ptr<DenseV> scratch_;
97};
