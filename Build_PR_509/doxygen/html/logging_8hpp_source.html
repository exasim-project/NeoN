74 location.line(),

139/*@brief convenience function to call log on logger with std::format */
140inline void log(std::shared_ptr<BaseLogger> logger, LogEvent& event, std::string_view delim = ",")
141{
142 if (logger != nullptr)
143 {
144 // console output
145 auto formattedMessage =
146 logger->getTarget() == Target::Console ? event.message : event.toJson(delim);

148 logger->log(formattedMessage);
