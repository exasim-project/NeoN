164 {

166 if (ws == nullptr || ws->get_size() != like->get_size())

174 template<typename VecType>
175 void scaled(
176 const gko::LinOp* alpha,
177 const gko::LinOp* e,
178 const gko::LinOp* beta,
179 VecType* x,
180 const VecType* r
181 ) const
182 {
183 auto exec = this->get_executor();
184 const auto ncols = x->get_size()[1];
185 auto* d = workspace(delta_, x);
186 auto* ad = workspace(aDelta_, x);
187 auto* t = workspace(tmp_, x);
188
189 prolong_->apply(e, d); // d = P e
190 fine_->apply(d, ad); // ad = A d
191
192 auto num = dense::create(exec, gko::dim<2> {1, ncols});
193 auto den = dense::create(exec, gko::dim<2> {1, ncols});
194 d->compute_dot(r, num);
195 d->compute_dot(ad, den);
196 auto host = exec->get_master();
197 auto numHost = gko::clone(host, num);
198 auto denHost = gko::clone(host, den);
199 auto sfHost = dense::create(host, gko::dim<2> {1, ncols});
200 for (gko::size_type c = 0; c < ncols; ++c)
201 {
202 const auto dn = denHost->at(0, c);
203 sfHost->at(0, c) =
204 dn != gko::zero<ValueType>() ? numHost->at(0, c) / dn : gko::one<ValueType>();
205 }
206 auto sf = gko::clone(exec, sfHost);
207
208 auto one = gko::initialize<dense>({gko::one<ValueType>()}, exec);
209 ad->scale(sf); // ad = sf A d
210 t->copy_from(r);
211 t->sub_scaled(one, ad); // t = r - sf A d
212 d->scale(sf); // d = sf d
213 invDiag_->apply(one, local(static_cast<const VecType*>(t)), one, local(d)); // d += D^-1 t
214
215 x->scale(beta);
216 x->add_scaled(alpha, d);
217 }
218
219 std::shared_ptr<const gko::LinOp> prolong_;
220 std::shared_ptr<const gko::LinOp> fine_;
221 std::shared_ptr<const diagonal> invDiag_;
222 std::shared_ptr<ScaleCorrectionState> state_;
223 mutable std::unique_ptr<gko::LinOp> delta_;
224 mutable std::unique_ptr<gko::LinOp> aDelta_;
225 mutable std::unique_ptr<gko::LinOp> tmp_;
226};

229template<typename ValueType, typename IndexType>
230std::shared_ptr<const gko::matrix::Diagonal<ValueType>>
231invertedDiagonal(const gko::matrix::Csr<ValueType, IndexType>* mtx)
232{
233 auto exec = mtx->get_executor();
234 auto diag = gko::clone(exec->get_master(), mtx->extract_diagonal());
235 auto* v = diag->get_values();
236 for (gko::size_type i = 0; i < diag->get_size()[0]; ++i)
237 {
238 v[i] =
239 v[i] != gko::zero<ValueType>() ? gko::one<ValueType>() / v[i] : gko::zero<ValueType>();
240 }
241 return gko::share(gko::clone(exec, diag));
242}

395 // ---- localized / Schwarz path: fine op is a plain Csr on this rank ----
396 void generateLocal(reuse_data_type* reuseData)
397 {
398 auto A = as_csr(system_matrix_);
399 const auto fineRows = A->get_size()[0];
400 const bool reuseMerged = reuseData && reuseData->initialized_;
401
402 // Run Pgm `merge_levels` times, composing the per-level aggregations BY INDEX. Each inner
403 // Pgm coarsens `coarse` and yields a piecewise-constant prolong (a RowGatherer: fine row ->
404 // aggregate) plus the next coarse operator. Because the prolongs are injections, the merged
405 // prolongation is the injection of the composed aggregate map `mergedAgg[i] =
406 // aggThis[mergedAgg[i]]`
407 // -- no Csr*Csr SpGEMM (cuSPARSE aborts with INSUFFICIENT_RESOURCES on large operands).
408 // When reusing, the inner aggregates are frozen, so the recorded merged injection is reused
409 // and only the coarse operator is refreshed.
410 auto pgmFactory = makePgm();
411 std::vector<std::unique_ptr<gko::LinOpFactory::ReuseData>> newSlots;
412 std::shared_ptr<const csr> coarse = A;
413 std::vector<IndexType> mergedAgg; // fine row -> current coarse index (host)
414 gko::size_type coarseRows = 0;
415 for (unsigned i = 0; i < parameters_.merge_levels; ++i)
416 {
417 auto level = generateLevel(pgmFactory.get(), coarse, i, reuseData, newSlots);
418 if (!reuseMerged)
419 {
420 auto aggThis = gather_agg(level.get()); // host, length coarse->rows
421 coarseRows = level->get_prolong_op()->get_size()[1];
422 composeAgg(mergedAgg, aggThis, fineRows, i);
423 }
424 coarse = as_csr(level->get_coarse_op());
425 }
426
427 if (reuseMerged)
428 {
429 prolong_ = reuseData->prolong_;
430 restrict_ = reuseData->restrict_;
431 }
432 else
433 {
434 auto pCsr = make_injection(mergedAgg, fineRows, coarseRows); // Csr, fine x coarse
435 prolong_ = pCsr;
436 restrict_ = gko::share(gko::as<csr>(pCsr->transpose())); // Csr, coarse x fine
437 }
438 // coarse == P_merged^T A P_merged already (last Pgm level's coarse op); use it directly.
439 setLevel(A, A.get(), coarse);
440 recordReuse(reuseData, std::move(newSlots), false);
441 }

443#ifdef NF_WITH_MPI_SUPPORT
444 // ---- global path: fine op is a distributed::Matrix ----
445 void generateDistributed(reuse_data_type* reuseData)
446 {
447 auto exec = this->get_executor();
448 auto distFine = gko::as<const dist_mtx>(system_matrix_);
449 auto comm = gko::as<const gko::experimental::distributed::DistributedBase>(system_matrix_)
450 ->get_communicator();
451 const auto fineGlobalRows = system_matrix_->get_size()[0];
452 // The prolong composition is rank-local because the distributed Pgm prolong is
453 // block-diagonal; all sizes below are LOCAL (this rank's diag block).
454 const auto localFineRows = distFine->get_diag_matrix()->get_size()[0];
455 const bool reuseMerged = reuseData && reuseData->initialized_;
456
457 auto pgmFactory = makePgm();
458 std::vector<std::unique_ptr<gko::LinOpFactory::ReuseData>> newSlots;
459 std::shared_ptr<const gko::LinOp> coarse = system_matrix_; // stays distributed
460 std::vector<IndexType> mergedAgg; // LOCAL fine row -> LOCAL coarse idx
461 gko::size_type localCoarseRows = 0;
462 for (unsigned i = 0; i < parameters_.merge_levels; ++i)
463 {
464 auto level = generateLevel(pgmFactory.get(), coarse, i, reuseData, newSlots);
465 if (!reuseMerged)
466 {
467 // Aggregation lives in the block-diagonal prolong's DIAG block (a RowGatherer).
468 auto prolongDiag =
469 gko::as<const dist_mtx>(level->get_prolong_op())->get_diag_matrix();
470 auto aggThis =
471 agg_from_rowgatherer(prolongDiag.get()); // host, length = local fine rows
472 // local coarse rows == this level's coarse DIAG block rows
473 localCoarseRows = gko::as<const dist_mtx>(level->get_coarse_op())
474 ->get_diag_matrix()
475 ->get_size()[0];
476 composeAgg(mergedAgg, aggThis, localFineRows, i);
477 }
478 coarse = level->get_coarse_op(); // distributed coarse op -> next Pgm fine op
479 }
480
481 if (reuseMerged)
482 {
483 prolong_ = reuseData->prolong_;
484 restrict_ = reuseData->restrict_;
485 }
486 else
487 {
488 const auto coarseGlobalRows = coarse->get_size()[0];
489 // Build the merged local injection Csr, then re-wrap prolong/restrict as
490 // block-diagonal distributed matrices (diag block only, empty off-diagonal) -- mirrors
491 // pgm.cpp's distributed_setup, which builds prolong/restrict from the local block
492 // alone.
493 std::shared_ptr<csr> pLocal = make_injection(mergedAgg, localFineRows, localCoarseRows);
494 std::shared_ptr<gko::LinOp> rLocal = gko::share(gko::as<csr>(pLocal->transpose()));
495 prolong_ = gko::share(
496 dist_mtx::create(exec, comm, gko::dim<2> {fineGlobalRows, coarseGlobalRows}, pLocal)
497 );
498 restrict_ = gko::share(
499 dist_mtx::create(exec, comm, gko::dim<2> {coarseGlobalRows, fineGlobalRows}, rLocal)
500 );
501 }
502 // coarse == the last inner Pgm's distributed coarse op == A_merged; use it directly.
503 setLevel(system_matrix_, gko::as<csr>(distFine->get_diag_matrix().get()), coarse);
504 recordReuse(reuseData, std::move(newSlots), true);
505 }
506#else
507 void generateDistributed(reuse_data_type*) { GKO_NOT_IMPLEMENTED; }
508#endif

536 // Generate inner Pgm level `i`: plainly without reuse data, through the recorded slot once the
537 // reuse data is initialized, and otherwise through a fresh slot that recordReuse() commits.
538 static std::unique_ptr<pgm> generateLevel(
539 const typename pgm::Factory* pgmFactory,
540 std::shared_ptr<const gko::LinOp> fine,
541 unsigned i,
542 reuse_data_type* reuseData,
543 std::vector<std::unique_ptr<gko::LinOpFactory::ReuseData>>& newSlots
544 )
545 {
546 if (!reuseData) return pgmFactory->generate(std::move(fine));
547 if (reuseData->initialized_)
548 {
549 return pgmFactory->generate_reuse(std::move(fine), *reuseData->levels_.at(i));
550 }
551 newSlots.push_back(pgmFactory->create_empty_reuse_data());
552 return pgmFactory->generate_reuse(std::move(fine), *newSlots.back());
553 }

592 // mergedAgg[k] <- aggThis[mergedAgg[k]] (or aggThis directly on the first level). Identical for
593 // the local and distributed paths -- the distributed prolong is block-diagonal so its
594 // diag-block aggregation composes rank-locally, exactly like a local Csr aggregation.
595 static void composeAgg(
596 std::vector<IndexType>& mergedAgg,
597 const gko::array<IndexType>& aggThis,
598 gko::size_type fineRows,
599 unsigned i
600 )
601 {
602 if (i == 0)
603 {
604 mergedAgg.assign(aggThis.get_const_data(), aggThis.get_const_data() + fineRows);
605 }
606 else
607 {
608 for (gko::size_type k = 0; k < fineRows; ++k)
609 {
610 mergedAgg[k] = aggThis.get_const_data()[mergedAgg[k]];
611 }
612 }
613 }

615 static std::shared_ptr<const csr> as_csr(std::shared_ptr<const gko::LinOp> op)
616 {
617 if (auto c = std::dynamic_pointer_cast<const csr>(op)) return c;
618 auto exec = op->get_executor();
619 auto out = csr::create(exec);
620 gko::as<const gko::ConvertibleTo<csr>>(op.get())->convert_to(out);
621 return std::move(out);
622 }

624 // Pull a RowGatherer's per-row aggregate map (fine row i -> its aggregate) to the host. Length
625 // == the gatherer's row count; values in [0, coarseRows). One entry per row.
626 gko::array<IndexType> agg_from_rowgatherer(const gko::LinOp* rgOp) const
627 {
628 auto exec = this->get_executor();
629 auto host = exec->get_master();
630 auto rg = gko::as<const row_gatherer>(rgOp);
631 auto fineRows = rg->get_size()[0];
632 gko::array<IndexType> agg(host, fineRows);
633 host->copy_from(exec, fineRows, rg->get_const_row_idxs(), agg.get_data());
634 return agg;
635 }

637 // Localized-path convenience: the level's prolong IS a RowGatherer.
638 gko::array<IndexType> gather_agg(const gko::multigrid::MultigridLevel* level) const
639 {
640 return agg_from_rowgatherer(level->get_prolong_op().get());
641 }
642
643 // Build the merged injection Csr P (P[i, mergedAgg[i]] = 1) from the composed aggregate map on
644 // the host, then clone to the device executor. row_ptrs = iota (one entry per row); values =
645 // ones. One entry per row => naturally sorted. Returned non-const so it can serve as the
646 // (mutable) diag block of a distributed::Matrix.
647 std::shared_ptr<csr> make_injection(
648 const std::vector<IndexType>& mergedAgg, gko::size_type fineRows, gko::size_type coarseRows
649 ) const
650 {
651 auto exec = this->get_executor();
652 auto host = exec->get_master();
653 gko::array<IndexType> cols(host, fineRows);
654 std::copy(mergedAgg.begin(), mergedAgg.end(), cols.get_data());
655 gko::array<IndexType> rowPtrs(host, fineRows + 1);
656 std::iota(rowPtrs.get_data(), rowPtrs.get_data() + fineRows + 1, IndexType {0});
657 gko::array<ValueType> vals(host, fineRows);
658 std::fill_n(vals.get_data(), fineRows, gko::one<ValueType>());
659
660 auto pHost = csr::create(
661 host,
662 gko::dim<2> {fineRows, coarseRows},
663 std::move(vals),
664 std::move(cols),
665 std::move(rowPtrs)
666 );
667 return gko::share(gko::clone(exec, pHost));
668 }

683// NB: return the CONCRETE factory type, not the abstract gko::LinOpFactory. gko::config::registry
684// keys entries by base_type<T>::type, which is defined for concrete factories (-> LinOpFactory) but
685// not for the abstract base — emplacing a shared_ptr<const LinOpFactory> fails to compile.
686template<typename ValueType = gko::default_precision, typename IndexType = gko::int32>
687inline std::shared_ptr<typename MergedPgm<ValueType, IndexType>::Factory> makeMergedPgmFactory(
688 std::shared_ptr<const gko::Executor> exec,
689 unsigned mergeLevels,
690 bool deterministic = true,
691 bool scaleCorrection = false
692)
693{
694 return MergedPgm<ValueType, IndexType>::build()
695 .with_merge_levels(mergeLevels)
696 .with_deterministic(deterministic)
697 .with_scale_correction(scaleCorrection)
698 .on(std::move(exec));
699}

707inline gko::config::configuration_map mergedPgmConfigMap()
708{
709 return {
710 {"neon::MergedPgm",
711 [](const gko::config::pnode& config,
712 const gko::config::registry&,
713 gko::config::type_descriptor) -> gko::deferred_factory_parameter<gko::LinOpFactory>
714 {
715 auto params = MergedPgm<ValueType, IndexType>::build();
716 auto toUnsigned = [](const gko::config::pnode& node, const std::string& key)
717 {
718 const auto v = node.get_integer();
719 if (v < 0)
720 {
721 throw std::invalid_argument("neon::MergedPgm: " + key + " must be >= 0");
722 }
723 return static_cast<unsigned>(v);
724 };
725 for (const auto& [key, value] : config.get_map())
726 {
727 if (key == "type" || key == "value_type") continue;
728 if (key == "merge_levels") params.with_merge_levels(toUnsigned(value, key));
729 else if (key == "max_iterations")
730 params.with_max_iterations(toUnsigned(value, key));
731 else if (key == "max_unassigned_ratio")
732 params.with_max_unassigned_ratio(value.get_real());
733 else if (key == "deterministic")
734 params.with_deterministic(value.get_boolean());
735 else if (key == "skip_sorting")
736 params.with_skip_sorting(value.get_boolean());
737 else if (key == "scale_correction")
738 params.with_scale_correction(value.get_boolean());
739 else
740 throw std::invalid_argument("neon::MergedPgm: unknown key '" + key + "'");
741 }
742 return params;
743 }}
