95 void update_matrix_value(std::shared_ptr<const gko::LinOp> new_matrix) override
96 {
97 system_matrix_ = new_matrix;
98 // Value-only refresh with FROZEN structure, cuSPARSE-free. The merged prolongation
99 // (prolong_/restrict_) is structural and stays fixed; only the coarse operator's values
100 // change. Rather than recompute A_merged = R A P by SpGEMM (cuSPARSE SpGEMM aborts with
101 // CUSPARSE_STATUS_INSUFFICIENT_RESOURCES on these large injection operands), walk the
102 // retained Pgm chain: each Pgm::update_matrix_value scatters the new fine values into its
103 // frozen coarse structure (no SpGEMM) and refreshes get_coarse_op() in place; that
104 // refreshed coarse feeds the next level's fine op. After the last level, get_coarse_op() is
105 // the refreshed A_merged. This keeps the cached-solver (update_matrix_value) reuse engaged.
106 const bool isDist = is_distributed(new_matrix);
107 // local path converts to Csr; distributed path feeds the distributed op through unchanged
108 // (Pgm::update_matrix_value has its own distributed branch).
109 std::shared_ptr<const gko::LinOp> fineOp = isDist ? new_matrix : as_csr(new_matrix);
110 for (auto& lvl : levels_)
111 {
112 gko::as<gko::UpdateMatrixValue>(lvl.get())->update_matrix_value(fineOp);
113 fineOp = gko::as<gko::multigrid::MultigridLevel>(lvl.get())->get_coarse_op();
114 }
115 std::shared_ptr<const gko::LinOp> coarse =
116 gko::as<gko::multigrid::MultigridLevel>(levels_.back().get())->get_coarse_op();
117 if (!isDist) coarse = as_csr(coarse);
118 this->set_multigrid_level(prolong_, coarse, restrict_);
119 }

159 // ---- localized / Schwarz path: fine op is a plain Csr on this rank ----
160 void generateLocal()
161 {
162 auto exec = this->get_executor();
163 auto A = as_csr(system_matrix_);
164 const auto fineRows = A->get_size()[0];
165
166 // Run Pgm `merge_levels` times, composing the per-level aggregations BY INDEX. Each inner
167 // Pgm coarsens `coarse` and yields a piecewise-constant prolong (a RowGatherer: fine row ->
168 // aggregate) plus the next coarse operator. Because the prolongs are injections, the merged
169 // prolongation is the injection of the composed aggregate map `mergedAgg[i] =
170 // aggThis[mergedAgg[i]]`
171 // -- no Csr*Csr SpGEMM (cuSPARSE aborts with INSUFFICIENT_RESOURCES on large operands). The
172 // Pgm levels are retained so update_matrix_value can refresh the coarse values in place.
173 levels_.clear();
174 std::shared_ptr<const csr> coarse = A;
175 std::vector<IndexType> mergedAgg; // fine row -> current coarse index (host)
176 gko::size_type coarseRows = 0;
177 for (unsigned i = 0; i < parameters_.merge_levels; ++i)
178 {
179 auto level = makePgm()->generate(coarse);
180 auto aggThis = gather_agg(level.get()); // host, length coarse->rows, in [0, coarseRows)
181 coarseRows = level->get_prolong_op()->get_size()[1];
182 composeAgg(mergedAgg, aggThis, fineRows, i);
183 coarse = as_csr(level->get_coarse_op());
184 levels_.push_back(gko::share(std::move(level)));
185 }
186
187 auto pCsr = make_injection(mergedAgg, fineRows, coarseRows); // Csr, fine x coarse
188 prolong_ = pCsr;
189 restrict_ = gko::share(gko::as<csr>(pCsr->transpose())); // Csr, coarse x fine
190 // coarse == P_merged^T A P_merged already (last Pgm level's coarse op); use it directly.
191 this->set_multigrid_level(prolong_, coarse, restrict_);
192 }

194#ifdef NF_WITH_MPI_SUPPORT
195 // ---- global path: fine op is a distributed::Matrix ----
196 void generateDistributed()
197 {
198 auto exec = this->get_executor();
199 auto distFine = gko::as<const dist_mtx>(system_matrix_);
200 auto comm = gko::as<const gko::experimental::distributed::DistributedBase>(system_matrix_)
201 ->get_communicator();
202 const auto fineGlobalRows = system_matrix_->get_size()[0];
203 // The prolong composition is rank-local because the distributed Pgm prolong is
204 // block-diagonal; all sizes below are LOCAL (this rank's diag block).
205 const auto localFineRows = distFine->get_diag_matrix()->get_size()[0];
206
207 levels_.clear();
208 std::shared_ptr<const gko::LinOp> coarse = system_matrix_; // stays distributed
209 std::vector<IndexType> mergedAgg; // LOCAL fine row -> LOCAL coarse idx
210 gko::size_type localCoarseRows = 0;
211 for (unsigned i = 0; i < parameters_.merge_levels; ++i)
212 {
213 auto level = makePgm()->generate(coarse); // distributed Pgm level
214 // Aggregation lives in the block-diagonal prolong's DIAG block (a RowGatherer).
215 auto prolongDiag = gko::as<const dist_mtx>(level->get_prolong_op())->get_diag_matrix();
216 auto aggThis =
217 agg_from_rowgatherer(prolongDiag.get()); // host, length = local fine rows
218 // local coarse rows == this level's coarse DIAG block rows
219 localCoarseRows =
220 gko::as<const dist_mtx>(level->get_coarse_op())->get_diag_matrix()->get_size()[0];
221 composeAgg(mergedAgg, aggThis, localFineRows, i);
222 coarse = level->get_coarse_op(); // distributed coarse op -> next Pgm fine op
223 levels_.push_back(gko::share(std::move(level)));
224 }
225 const auto coarseGlobalRows = coarse->get_size()[0];
226
227 // Build the merged local injection Csr, then re-wrap prolong/restrict as block-diagonal
228 // distributed matrices (diag block only, empty off-diagonal) -- mirrors pgm.cpp's
229 // distributed_setup, which builds prolong/restrict from the local block alone.
230 std::shared_ptr<csr> pLocal = make_injection(mergedAgg, localFineRows, localCoarseRows);
231 std::shared_ptr<gko::LinOp> rLocal = gko::share(gko::as<csr>(pLocal->transpose()));
232 prolong_ = gko::share(
233 dist_mtx::create(exec, comm, gko::dim<2> {fineGlobalRows, coarseGlobalRows}, pLocal)
234 );
235 restrict_ = gko::share(
236 dist_mtx::create(exec, comm, gko::dim<2> {coarseGlobalRows, fineGlobalRows}, rLocal)
237 );
238 // coarse == the last inner Pgm's distributed coarse op == A_merged; use it directly.
239 this->set_multigrid_level(prolong_, coarse, restrict_);
240 }
241#else
242 void generateDistributed() { GKO_NOT_IMPLEMENTED; }
243#endif

258 std::unique_ptr<typename pgm::Factory> makePgm() const
259 {
260 return pgm::build()
261 .with_max_iterations(parameters_.max_iterations)
262 .with_max_unassigned_ratio(parameters_.max_unassigned_ratio)
263 .with_deterministic(parameters_.deterministic)
264 .with_skip_sorting(parameters_.skip_sorting)
265 .on(this->get_executor());
266 }

268 // mergedAgg[k] <- aggThis[mergedAgg[k]] (or aggThis directly on the first level). Identical for
269 // the local and distributed paths -- the distributed prolong is block-diagonal so its
270 // diag-block aggregation composes rank-locally, exactly like a local Csr aggregation.
271 static void composeAgg(
272 std::vector<IndexType>& mergedAgg,
273 const gko::array<IndexType>& aggThis,
274 gko::size_type fineRows,
275 unsigned i
276 )
277 {
278 if (i == 0)
279 {
280 mergedAgg.assign(aggThis.get_const_data(), aggThis.get_const_data() + fineRows);
281 }
282 else
283 {
284 for (gko::size_type k = 0; k < fineRows; ++k)
285 {
286 mergedAgg[k] = aggThis.get_const_data()[mergedAgg[k]];
287 }
288 }
289 }

291 static std::shared_ptr<const csr> as_csr(std::shared_ptr<const gko::LinOp> op)
292 {
293 if (auto c = std::dynamic_pointer_cast<const csr>(op)) return c;
294 auto exec = op->get_executor();
295 auto out = csr::create(exec);
296 gko::as<const gko::ConvertibleTo<csr>>(op.get())->convert_to(out);
297 return std::move(out);
298 }

300 // Pull a RowGatherer's per-row aggregate map (fine row i -> its aggregate) to the host. Length
301 // == the gatherer's row count; values in [0, coarseRows). One entry per row.
302 gko::array<IndexType> agg_from_rowgatherer(const gko::LinOp* rgOp) const
303 {
304 auto exec = this->get_executor();
305 auto host = exec->get_master();
306 auto rg = gko::as<const row_gatherer>(rgOp);
307 auto fineRows = rg->get_size()[0];
308 gko::array<IndexType> agg(host, fineRows);
309 host->copy_from(exec, fineRows, rg->get_const_row_idxs(), agg.get_data());
310 return agg;
311 }
312
313 // Localized-path convenience: the level's prolong IS a RowGatherer.
314 gko::array<IndexType> gather_agg(const gko::multigrid::MultigridLevel* level) const
315 {
316 return agg_from_rowgatherer(level->get_prolong_op().get());
317 }
318
319 // Build the merged injection Csr P (P[i, mergedAgg[i]] = 1) from the composed aggregate map on
320 // the host, then clone to the device executor. row_ptrs = iota (one entry per row); values =
321 // ones. One entry per row => naturally sorted. Returned non-const so it can serve as the
322 // (mutable) diag block of a distributed::Matrix.
323 std::shared_ptr<csr> make_injection(
324 const std::vector<IndexType>& mergedAgg, gko::size_type fineRows, gko::size_type coarseRows
325 ) const
326 {
327 auto exec = this->get_executor();
328 auto host = exec->get_master();
329 gko::array<IndexType> cols(host, fineRows);
330 std::copy(mergedAgg.begin(), mergedAgg.end(), cols.get_data());
331 gko::array<IndexType> rowPtrs(host, fineRows + 1);
332 std::iota(rowPtrs.get_data(), rowPtrs.get_data() + fineRows + 1, IndexType {0});
333 gko::array<ValueType> vals(host, fineRows);
334 std::fill_n(vals.get_data(), fineRows, gko::one<ValueType>());
335
336 auto pHost = csr::create(
337 host,
338 gko::dim<2> {fineRows, coarseRows},
339 std::move(vals),
340 std::move(cols),
341 std::move(rowPtrs)
342 );
343 return gko::share(gko::clone(exec, pHost));
344 }

346 // NB: `parameters_` is provided by GKO_ENABLE_LIN_OP_FACTORY; do not redeclare it.
347 std::shared_ptr<const gko::LinOp> system_matrix_;
348 // Merged prolong/restrict: a Csr in the local path, a block-diagonal distributed::Matrix in the
349 // global path. Structural (composed once); reused across update_matrix_value refreshes.
350 std::shared_ptr<const gko::LinOp> prolong_;
351 std::shared_ptr<const gko::LinOp> restrict_;
352 // Retained inner Pgm chain (each a MultigridLevel + UpdateMatrixValue): lets
353 // update_matrix_value refresh the merged coarse operator's values in place with frozen
354 // aggregation, cuSPARSE-free.
355 std::vector<std::shared_ptr<gko::LinOp>> levels_;
356};

363// NB: return the CONCRETE factory type, not the abstract gko::LinOpFactory. gko::config::registry
364// keys entries by base_type<T>::type, which is defined for concrete factories (-> LinOpFactory) but
365// not for the abstract base — emplacing a shared_ptr<const LinOpFactory> fails to compile.
366template<typename ValueType = gko::default_precision, typename IndexType = gko::int32>
367inline std::shared_ptr<typename MergedPgm<ValueType, IndexType>::Factory> makeMergedPgmFactory(
368 std::shared_ptr<const gko::Executor> exec, unsigned mergeLevels, bool deterministic = true
369)
370{
371 return MergedPgm<ValueType, IndexType>::build()
372 .with_merge_levels(mergeLevels)
373 .with_deterministic(deterministic)
374 .on(std::move(exec));
375}
