20 /* @brief Returns the flat values-array index of the diagonal entry for cell celli.
21 * diagIdx(celli) = rowOffs[celli] + diagOffset[celli]
22 */
23 KOKKOS_INLINE_FUNCTION localIdx diagIdx(localIdx celli) const
24 {
25 return rowOffs[celli] + diagOffset[celli];
26 }
27
28 /* @brief Returns the flat values-array index of the upper-triangular entry A[own, nei].
29 *
30 * By construction own < nei for every internal face, so the column index nei is greater
31 * than the row index own — this entry lies in the upper triangle.
32 *
33 * upperIdx(own, f) = rowOffs[own] + ownerOffset[f]
34 */
35 KOKKOS_INLINE_FUNCTION localIdx upperIdx(localIdx own, localIdx faceIdx) const
36 {
37 return rowOffs[own] + ownerOffset[faceIdx];
38 }
39
40 /* @brief Returns the flat values-array index of the lower-triangular entry A[nei, own].
41 *
42 * By construction nei > own for every internal face, so the column index own is less
43 * than the row index nei — this entry lies in the lower triangle.
44 *
45 * lowerIdx(nei, f) = rowOffs[nei] + neighbourOffset[f]
46 */
47 KOKKOS_INLINE_FUNCTION localIdx lowerIdx(localIdx nei, localIdx faceIdx) const
48 {
49 return rowOffs[nei] + neighbourOffset[faceIdx];
50 }

55 View<const uint8_t> neighbourOffset;

63/* @class FaceToMatrixAddress
64 * @brief Stores the mapping between mesh faces and matrix row offsets.
65 *
66 * Based on a given computational mesh this class stores a mapping for a consistent iteration
67 * procedure for matrices which share the same sparsity pattern.
68 *
69 * This class implements the finite volume 3/5/7 pt stencil specific generation
70 * of face-to-matrix offset mappings. The sparsity pattern itself is owned by
71 * the Matrix, not by this class; this class only borrows a view of the row offsets.
72 *
73 */
74class FaceToMatrixAddress : public NeoN::SupportsCopyTo<FaceToMatrixAddress>
75{
76 void validate() const;
77
78 // clang-format off
79 // NOTE The following data members store a simple mapping from face ids to offsets within
80 // the corresponding row of the CSR values array.
81 //
82 // For an internal face f with owner P and neighbour N (P < N by construction):
83 // ownerOffset[f] = offset within row P for column N → A[P, N] (upper triangular)
84 // neighbourOffset[f] = offset within row N for column P → A[N, P] (lower triangular)
85 //
86 // Example row layout for a cell with two lower entries and one upper entry:
87 // Row P: [ A[P,j0] A[P,j1] | A[P,P] | A[P,N] ]
88 // lower (j<P) diag upper (N>P)
89 // ownerOffset[f] = 3 (position of A[P,N] within row P)
90 // neighbourOffset[f] = 0 (position of A[N,P] within row N)
91 // clang-format on
92 Array<uint8_t>
93 ownerOffset_;
94
95 Array<uint8_t> neighbourOffset_;
97
98 Array<uint8_t> diagOffset_;
99
100public:

102 /* @brief constructor
103 *
104 * @param ownerOffset face-to-upper-offset mapping for the owner row
