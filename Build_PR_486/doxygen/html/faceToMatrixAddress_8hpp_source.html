16/* @class FaceToMatrixAddress
17 * @brief Stores the mapping between mesh faces and matrix row offsets.
18 *
19 * Based on a given computational mesh this class stores a mapping for a consistent iteration
20 * procedure for matrices which share the same sparsity pattern.
21 *
22 * This class implements the finite volume 3/5/7 pt stencil specific generation
23 * of face-to-matrix offset mappings. The sparsity pattern itself is owned by
24 * the Matrix, not by this class; this class only borrows a view of the row offsets.
25 *
26 */
27template<typename IndexType = localIdx, typename MeshType = UnstructuredMesh>
28class FaceToMatrixAddress
29{
30
31 // clang-format off
32 // NOTE The following data members store a simple mapping from face ids to offsets within
33 // the corresponding row of the CSR values array.
34 //
35 // For an internal face f with owner P and neighbour N (P < N by construction):
36 // ownerOffset[f] = offset within row P for column N → A[P, N] (upper triangular)
37 // neighbourOffset[f] = offset within row N for column P → A[N, P] (lower triangular)
38 //
39 // Example row layout for a cell with two lower entries and one upper entry:
40 // Row P: [ A[P,j0] A[P,j1] | A[P,P] | A[P,N] ]
41 // lower (j<P) diag upper (N>P)
42 // ownerOffset[f] = 3 (position of A[P,N] within row P)
43 // neighbourOffset[f] = 0 (position of A[N,P] within row N)
44 // clang-format on
45 Array<uint8_t>
46 ownerOffset_;
47
48 Array<uint8_t> neighbourOffset_;
50
51 Array<uint8_t> diagOffset_;
52
53 // corresponding views
54 View<uint8_t> ownerOffsetV_;
55
56 View<uint8_t> neighbourOffsetV_;
57
58 View<uint8_t> diagOffsetV_;
59
60 // row offsets borrowed from the owning Matrix's sparsity pattern
61 View<const IndexType> rowOffsV_;
62
63public:
64
65 /* @brief constructor
66 *
67 * @param ownerOffset face-to-lower-offset mapping
68 * @param neighbourOffset face-to-upper-offset mapping
69 * @param diagOffset cell-to-diagonal-offset mapping
70 * @param rowOffsView view of row offsets from the owning Matrix's CsrSparsityPattern
71 */
72 FaceToMatrixAddress(

101 /* @brief Returns the flat values-array index of the diagonal entry for cell celli.
102 * diagIdx(celli) = rowOffs[celli] + diagOffset[celli]
103 */
104 KOKKOS_INLINE_FUNCTION localIdx diagIdx(localIdx celli) const
105 {
106 return rowOffsV_[celli] + diagOffsetV_[celli];
107 }
108
109 /* @brief Returns the flat values-array index of the upper-triangular entry A[own, nei].
110 *
111 * By construction own < nei for every internal face, so the column index nei is greater
112 * than the row index own — this entry lies in the upper triangle.
113 *
114 * upperIdx(own, f) = rowOffs[own] + ownerOffset[f]
115 */
116 KOKKOS_INLINE_FUNCTION localIdx upperIdx(localIdx own, localIdx faceIdx) const
117 {
118 return rowOffsV_[own] + ownerOffsetV_[faceIdx];
119 }
120
121 /* @brief Returns the flat values-array index of the lower-triangular entry A[nei, own].
122 *
123 * By construction nei > own for every internal face, so the column index own is less
124 * than the row index nei — this entry lies in the lower triangle.
125 *
126 * lowerIdx(nei, f) = rowOffs[nei] + neighbourOffset[f]
127 */
128 KOKKOS_INLINE_FUNCTION localIdx lowerIdx(localIdx nei, localIdx faceIdx) const
129 {
130 return rowOffsV_[nei] + neighbourOffsetV_[faceIdx];
131 }
132};
