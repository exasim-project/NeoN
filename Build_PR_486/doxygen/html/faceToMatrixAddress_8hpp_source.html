16/* @class FaceToMatrixAddress
17 * @brief Stores the mapping between mesh faces and matrix row offsets.
18 *
19 * Based on a given computational mesh this class stores a mapping for a consistent iteration
20 * procedure for matrices which share the same sparsity pattern.
21 *
22 * This class implements the finite volume 3/5/7 pt stencil specific generation
23 * of face-to-matrix offset mappings. The sparsity pattern itself is owned by
24 * the Matrix, not by this class; this class only borrows a view of the row offsets.
25 *
26 */
27template<typename IndexType = localIdx, typename MeshType = UnstructuredMesh>
28class FaceToMatrixAddress
29{
30
31 // clang-format off
32 // NOTE The following data member store a simple mapping from face ids to offsets in the
33 // corresponding rows.
34 // I.e. Assume the following row [ . a_18 . a_20 d a_18 . a_20 . ]
35 // where a_i corresponds to a value on a face i
36 // this yields:
37 // ownerOffset[18] = 0
38 // ownerOffset[20] = 1
39 // and
40 // neighbourOffset[18] = 4
41 // neighbourOffset[20] = 5
42 // clang-format on
43 Array<uint8_t> ownerOffset_;
44
45 Array<uint8_t> neighbourOffset_;
46
47 Array<uint8_t> diagOffset_;
48
49 // corresponding views
50 View<uint8_t> ownerOffsetV_;
51
52 View<uint8_t> neighbourOffsetV_;
53
54 View<uint8_t> diagOffsetV_;
55
56 // row offsets borrowed from the owning Matrix's sparsity pattern
57 View<const IndexType> rowOffsV_;
58
59public:
60
61 /* @brief constructor
62 *
63 * @param ownerOffset face-to-lower-offset mapping
64 * @param neighbourOffset face-to-upper-offset mapping
65 * @param diagOffset cell-to-diagonal-offset mapping
66 * @param rowOffsView view of row offsets from the owning Matrix's CsrSparsityPattern
67 */
68 FaceToMatrixAddress(

96 /* @brief given a cell ID, the function returns index of the diagonal element */
97 KOKKOS_INLINE_FUNCTION localIdx diagIdx(localIdx celli) const
98 {
99 return rowOffsV_[celli] + diagOffsetV_[celli];
100 }
101
102 /* @brief given a cell ID and a face ID, returns index of the element in the upper triangular */
103 KOKKOS_INLINE_FUNCTION localIdx upperIdx(localIdx celli, localIdx faceIdx) const
104 {
105 return rowOffsV_[celli] + neighbourOffsetV_[faceIdx];
106 }
107
108 /* @brief given a cell ID and a face ID, returns index of the element in the lower triangular */
109 KOKKOS_INLINE_FUNCTION localIdx lowerIdx(localIdx celli, localIdx faceIdx) const
110 {
111 return rowOffsV_[celli] + ownerOffsetV_[faceIdx];
112 }
113};
