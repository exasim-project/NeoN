19 /* @brief Returns the flat values-array index of the diagonal entry for cell celli.
20 * diagIdx(celli) = rowOffs[celli] + diagOffset[celli]
21 */
22 KOKKOS_INLINE_FUNCTION localIdx diagIdx(localIdx celli) const
23 {
24 return rowOffs[celli] + diagOffset[celli];
25 }
26
27 /* @brief Returns the flat values-array index of the upper-triangular entry A[own, nei].
28 *
29 * By construction own < nei for every internal face, so the column index nei is greater
30 * than the row index own — this entry lies in the upper triangle.
31 *
32 * upperIdx(own, f) = rowOffs[own] + ownerOffset[f]
33 */
34 KOKKOS_INLINE_FUNCTION localIdx upperIdx(localIdx own, localIdx faceIdx) const
35 {
36 return rowOffs[own] + ownerOffset[faceIdx];
37 }
38
39 /* @brief Returns the flat values-array index of the lower-triangular entry A[nei, own].
40 *
41 * By construction nei > own for every internal face, so the column index own is less
42 * than the row index nei — this entry lies in the lower triangle.
43 *
44 * lowerIdx(nei, f) = rowOffs[nei] + neighbourOffset[f]
45 */
46 KOKKOS_INLINE_FUNCTION localIdx lowerIdx(localIdx nei, localIdx faceIdx) const
47 {
48 return rowOffs[nei] + neighbourOffset[faceIdx];
49 }
50
51 // corresponding views
52 View<const uint8_t> ownerOffset;
53
54 View<const uint8_t> neighbourOffset;
55
56 View<const uint8_t> diagOffset;
57
58 // row offsets borrowed from the owning Matrix's sparsity pattern
59 View<const localIdx> rowOffs;
60};

62/* @class FaceToMatrixAddress
63 * @brief Stores the mapping between mesh faces and matrix row offsets.
64 *
65 * Based on a given computational mesh this class stores a mapping for a consistent iteration
66 * procedure for matrices which share the same sparsity pattern.
67 *
68 * This class implements the finite volume 3/5/7 pt stencil specific generation
69 * of face-to-matrix offset mappings. The sparsity pattern itself is owned by
70 * the Matrix, not by this class; this class only borrows a view of the row offsets.
71 *
72 */
73template<typename IndexType = localIdx, typename MeshType = UnstructuredMesh>
74class FaceToMatrixAddress
75{
76
77 // clang-format off
78 // NOTE The following data members store a simple mapping from face ids to offsets within
79 // the corresponding row of the CSR values array.
80 //
81 // For an internal face f with owner P and neighbour N (P < N by construction):
82 // ownerOffset[f] = offset within row P for column N → A[P, N] (upper triangular)
83 // neighbourOffset[f] = offset within row N for column P → A[N, P] (lower triangular)
84 //
85 // Example row layout for a cell with two lower entries and one upper entry:
86 // Row P: [ A[P,j0] A[P,j1] | A[P,P] | A[P,N] ]
87 // lower (j<P) diag upper (N>P)
88 // ownerOffset[f] = 3 (position of A[P,N] within row P)
89 // neighbourOffset[f] = 0 (position of A[N,P] within row N)
90 // clang-format on
91 Array<uint8_t>
92 ownerOffset_;
93
94 Array<uint8_t> neighbourOffset_;
96
97 Array<uint8_t> diagOffset_;
98
99public:
100
101 /* @brief constructor
102 *
103 * @param ownerOffset face-to-lower-offset mapping
104 * @param neighbourOffset face-to-upper-offset mapping
105 * @param diagOffset cell-to-diagonal-offset mapping
106 * @param rowOffsView view of row offsets from the owning Matrix's CsrSparsityPattern
107 */
108 FaceToMatrixAddress(
109 Array<uint8_t> ownerOffset, Array<uint8_t> neighbourOffset, Array<uint8_t> diagOffset
110 );
111
112 /* @brief copy constructor */
113 FaceToMatrixAddress(const FaceToMatrixAddress& mi);
114
119 [[nodiscard]] FaceToMatrixView view(View<const localIdx> rowOffsView) const
