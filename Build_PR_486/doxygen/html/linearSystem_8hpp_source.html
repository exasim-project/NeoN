185 typename SystemMatrixType = CSRMatrix<ValueType, localIdx>,
186 typename BoundaryMatrixType = COOMatrix<ValueType, localIdx>>
187LinearSystem<ValueType, SystemMatrixType, BoundaryMatrixType>
188createEmptyLinearSystem(const UnstructuredMesh& mesh)
189{
190 auto [csrSp, ftma] =
191 createSparsityPatternFaceToMatrixAddress<typename SystemMatrixType::MatrixSparsityType>(mesh
192 );
193 auto bSp =
194 createBoundarySparsityPattern<typename BoundaryMatrixType::MatrixSparsityType>(mesh, *ftma);
195 return {
196 SystemMatrixType(
197 Vector<ValueType>(csrSp->exec(), csrSp->nnz(), zero<ValueType>()), csrSp, ftma
198 ),
199 Vector<ValueType>(csrSp->exec(), csrSp->rows(), zero<ValueType>()),
200 BoundaryMatrixType(Vector<ValueType>(bSp->exec(), bSp->nnz(), zero<ValueType>()), bSp),
201 Vector<ValueType>(bSp->exec(), bSp->nnz(), zero<ValueType>())
202 };
203}

207template<typename ValueType>
208inline la::LinearSystem<ValueType>
209removeBoundaryContributions(const la::LinearSystem<ValueType>& lsIn)
210{
211 auto ls = la::LinearSystem<ValueType>(lsIn);
212 auto lsView = ls.view();
213 auto& matrix = lsView.matrix;
214 auto& rhs = lsView.rhs;
215 auto& bMatrix = lsView.boundaryMatrix;
216 auto& bRhs = lsView.boundaryRhs;
217
218 const auto ma = ls.faceToMatrixAddress()->view(ls.matrix().sparsity()->rowOffs().view());
219
220 parallelFor(
221 ls.exec(),
222 {0, bMatrix.values.size()},
223 NEON_LAMBDA(const localIdx facei) {
224 const auto celli = bMatrix.sparsity.rowOffs[facei]; // cell index stored in rowOffs
225 Kokkos::atomic_add(&matrix.values[ma.diagIdx(celli)], bMatrix.values[facei]);
226 Kokkos::atomic_add(&rhs[celli], bRhs[facei]);
227 },
228 "removeBoundaryContributions"
229 );
230
231 return ls;
232}
